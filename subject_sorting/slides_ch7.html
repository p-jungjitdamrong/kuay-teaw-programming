Big O คืออะไร?

Big O วัด อัตราการเติบโต ของเวลา (หรือ memory) เมื่อ input n ใหญ่ขึ้น — ไม่สนใจ hardware หรือ constant เล็กน้อย

ตัวอย่างเวลาจริง
/* n = จำนวนธุรกรรม */
n = 1,000
  O(n)     → ~1,000 ops   ✅
  O(n²)    → ~1,000,000   ⚠️
  O(n³)    → ~1,000,000,000 ❌

n = 1,000,000
  O(n)     → 1M ops       ✅
  O(n log n)→ ~20M ops    ✅
  O(n²)    → 1,000,000M   💀
นิยามทางคณิตศาสตร์
/* f(n) = O(g(n)) หมายความว่า:
   มี c > 0 และ n₀ ที่ทำให้
   f(n) ≤ c·g(n) สำหรับทุก n ≥ n₀ */

/* ภาษาคน:
   g(n) คือ "กรอบบน" ของ f(n)
   เมื่อ n ใหญ่พอ — constant ไม่สำคัญ */

3n² + 100n + 500  = O(n²)
  ↑ ทิ้ง 100n และ 500
  ↑ ทิ้ง constant 3
  → เหลือ n²
Big O บอก worst case "กรอบบน" — ไม่ใช่เวลาจริง แต่ช่วยเปรียบเทียบ algorithm ได้โดยไม่สน hardware

   ครอบงำ term อื่นทั้งหมด */

O(1) Constant และ O(log n) Logarithmic

O(1) — ไม่ว่า n จะใหญ่แค่ไหน เร็วเท่าเดิม
int get_first(int arr[], int n) {
    return arr[0];  /* 1 operation */
}

void push(Stack *s, int val) {
    s->data[++s->top] = val; /* O(1) */
}

/* ตัวอย่าง O(1):
   - array access arr[i]
   - hash table lookup
   - push/pop stack
   - เพิ่ม/ลบ head linked list */
O(log n) — ทุก step ลด n ลงครึ่ง
int binary_search(int arr[], int n, int target) {
    int lo = 0, hi = n - 1;
    while (lo <= hi) {   /* วน log₂(n) ครั้ง */
        int mid = (lo+hi)/2;
        if (arr[mid] == target) return mid;
        else if (arr[mid] < target) lo = mid+1;
        else                        hi = mid-1;
    }
    return -1;
}
/* n=1,000,000 → แค่ 20 steps!
   log₂(1,000,000) ≈ 20 */
O(log n) เกิดเมื่อ: แต่ละ step ตัด input ออกด้วยปัจจัยคงที่ (ครึ่ง, หนึ่งสาม ฯลฯ) — Binary Search, Heap operations, Balanced BST

O(n) Linear และ O(n log n) Linearithmic

O(n) — วนครบทุก element ครั้งเดียว
int find_max(int arr[], int n) {
    int max = arr[0];
    for (int i=1; i<n; i++) { /* n-1 ops */
        if (arr[i] > max) max = arr[i];
    }
    return max;
}
/* ตัวอย่าง O(n):
   - Linear Search
   - Counting Sort inner loop
   - Copy array
   - Sum all elements */
O(n log n) — loop + recursive half
/* Merge Sort: */
merge_sort(arr, 0, n-1)
  ↓ แบ่งครึ่ง log n ครั้ง
  ↓ แต่ละ level ทำ merge O(n)
  → O(n) × O(log n) = O(n log n)

/* Quick Sort average: */
partition O(n) × recursion depth O(log n)
  → O(n log n)

/* O(n log n) คือ theoretical minimum
   ของ comparison-based sort! */
ไม่มี comparison-based sort ที่เร็วกว่า O(n log n) ได้ — ใครบอกว่า sort ได้ O(n) แสดงว่าใช้ข้อมูลพิเศษ (Counting/Radix)

O(n²) Quadratic และ O(2ⁿ) Exponential

O(n²) — nested loop ครบทุกคู่
/* Bubble Sort inner + outer */
for (i = 0; i < n; i++) {       /* n */
    for (j = 0; j < n; j++) {   /* n */
        /* n × n = n² ops */
    }
}

/* ตัวอย่าง O(n²):
   - Bubble/Selection/Insertion Sort
   - Matrix multiplication ปกติ (n²)
   - Printing all pairs */

/* n=1000 → 1,000,000 ops
   n=10,000 → 100,000,000 ops ⚠️ */
O(2ⁿ) และ O(n!) — หลีกเลี่ยง!
/* O(2ⁿ): Fibonacci แบบ naive */
int fib(int n) {
    if (n <= 1) return n;
    return fib(n-1) + fib(n-2);
}
/* n=50 → 2^50 ≈ 1 quadrillion ops 💀 */

/* O(n!): Permutation / Brute-force TSP */
/* n=20 → 2,432,902,008,176,640,000 ops */

/* แก้ O(2ⁿ) ด้วย Memoization/DP
   → O(n) time, O(n) space */
บทที่ 7 · อ่านจากโค้ด

กฎ: อ่าน Big O จาก Loop

ดูที่ จำนวนครั้งที่ loop body ทำงาน เมื่อ n เปลี่ยน — นั่นคือ complexity

Pattern ที่เจอบ่อย
/* 1. วน n ครั้ง → O(n) */
for (i = 0; i < n; i++) { ... }

/* 2. วน n/2 ครั้ง → O(n) */
for (i = 0; i < n; i += 2) { ... }

/* 3. หาร 2 ทุกรอบ → O(log n) */
for (i = 1; i < n; i *= 2) { ... }
for (i = n; i > 0; i /= 2) { ... }

/* 4. Nested n×n → O(n²) */
for (i = 0; i < n; i++)
    for (j = 0; j < n; j++) { ... }

/* 5. Nested n×log n → O(n log n) */
for (i = 0; i < n; i++)
    for (j = 1; j < n; j *= 2) { ... }
ทดสอบความเข้าใจ
/* กี่ ops? */
for (i = 0; i < n; i++)        /* n */
    for (j = 0; j < i; j++) { } /* ? */

/* นับ: i=0→0, i=1→1, i=2→2 ... i=n-1→n-1
   รวม: 0+1+2+...+(n-1) = n(n-1)/2
   → O(n²) แม้ไม่ใช่ n×n ตรงๆ */

/* อีกตัวอย่าง: */
for (i = 0; i < n; i++) { }   /* n */
for (j = 0; j < n; j++) { }   /* n */
/* สอง loop ติดกัน (ไม่ nest):
   n + n = 2n → O(n) ไม่ใช่ O(n²)! */
Loop ติดกัน (sequential) = บวก · Loop ซ้อนกัน (nested) = คูณ
บทที่ 7 · อ่านจากโค้ด

วิเคราะห์ Recursion ด้วย Recurrence

เขียน T(n) = work per call + recursive calls → แก้หา Big O

Binary Search: T(n) = T(n/2) + O(1)
int binary_search(int arr[], int lo, int hi, int x) {
    if (lo > hi) return -1;   /* O(1) */
    int mid = (lo+hi)/2;      /* O(1) */
    if (arr[mid] == x) return mid;
    if (arr[mid] < x)            /* call 1 ครั้ง */
        return binary_search(arr, mid+1, hi, x);  /* n/2 */
    return binary_search(arr, lo, mid-1, x);     /* n/2 */
}
/* T(n) = T(n/2) + 1
   → T(n) = O(log n) */
Merge Sort: T(n) = 2T(n/2) + O(n)
void merge_sort(int arr[], int l, int r) {
    if (l >= r) return;
    int m = l + (r-l)/2;
    merge_sort(arr, l, m);      /* T(n/2) */
    merge_sort(arr, m+1, r);   /* T(n/2) */
    merge(arr, l, m, r);        /* O(n) */
}
/* T(n) = 2T(n/2) + n
   → Master Theorem Case 2
   → T(n) = O(n log n) */
Master Theorem: T(n) = aT(n/b) + f(n) ถ้า f(n) = O(n^log_b(a)) → T(n) = O(n^log_b(a) · log n)
บทที่ 7 · Workshop

อ่าน Big O จากโค้ด — ฝึกทีละตัว

ดูโค้ดแล้ว วิเคราะห์ทีละบรรทัด — ใช้กฎ loop บวก/คูณ

/* ① ฟังก์ชันนี้ O(?) */
void f1(int n) {
    int sum = 0;
    for (int i = 0; i < n; i++)         /* n */
        for (int j = 0; j < 10; j++)     /* 10 (constant!) */
            sum++;
}  /* n × 10 = 10n → O(n) ← constant ทิ้งได้! */

/* ② O(?) */
void f2(int n) {
    for (int i = 0; i < n; i++)         /* n */
        for (int j = i; j < n; j++)      /* n-i */
            printf("%d", i*j);
} /* 0+1+...+(n-1) ≈ n²/2 → O(n²) */

/* ③ O(?) */
void f3(int n) {
    for (int i = n; i > 0; i /= 3)     /* หาร 3 ทุกรอบ */
        printf("%d\n", i);
} /* log₃(n) ≈ log(n) → O(log n) */

/* ④ O(?) */
void f4(int n) {
    for (int i = 0; i < n; i++)         /* n */
        for (int j = 1; j < n; j *= 2)  /* log n */
            printf("%d", i+j);
} /* n × log n → O(n log n) */
บทที่ 7 · Cases

Best / Average / Worst Case

Big O ปกติหมาย worst case แต่ algorithm หนึ่งตัวมี 3 case ต้องแยกออกจากกัน

Linear Search — 3 cases
int linear_search(int arr[], int n, int x) {
    for (int i = 0; i < n; i++) {
        if (arr[i] == x)
            return i;
    }
    return -1;
}

Best:    O(1)   → x อยู่ index แรก
Average: O(n/2) → O(n)
Worst:   O(n)   → x ไม่มี หรืออยู่สุดท้าย
Notation อื่นๆ
/* Big O   = กรอบบน (worst case)
   Big Ω   = กรอบล่าง (best case)
   Big Θ   = กรอบทั้งสอง (tight bound)

   ถ้า Best = Worst = O(n log n)
   → Θ(n log n) — Merge Sort! */

/* Amortized Analysis:
   Dynamic array (realloc):
   - push ปกติ: O(1)
   - push ที่ต้อง resize: O(n)
   - Amortized per push: O(1)
     เพราะ resize ไม่บ่อย */
Quick Sort: Best/Avg O(n log n) แต่ Worst O(n²) — ถ้าบอกแค่ "O(n log n)" แสดงว่าพูดถึง average case

Trade-off: Time vs Space
/* Fibonacci */
/* Naive O(2ⁿ) time, O(n) stack */
int fib_naive(int n) {
    if(n<=1) return n;
    return fib_naive(n-1) + fib_naive(n-2);
}

/* Memoized: O(n) time, O(n) space */
int memo[1000] = {-1};
int fib_memo(int n) {
    if(n<=1) return n;
    if(memo[n]!=-1) return memo[n];
    return memo[n] = fib_memo(n-1) + fib_memo(n-2);
}
/* ใช้ O(n) space แลกกับ O(n) time */
