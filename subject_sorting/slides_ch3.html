Insertion Sort — เหมือนเรียงไพ่

หยิบไพ่ใบใหม่ (key) แล้วเลื่อนไพ่ที่ใหญ่กว่าออก จนหาช่องที่เหมาะสมเจอ — sorted portion ขยายทีละ 1

ภาพจำลอง: เรียงไพ่ {5,2,4,6,1}
มือเริ่มต้น: [5]

หยิบ 2: เลื่อน 5 ออก → ใส่ 2
  → [2, 5]

หยิบ 4: เลื่อน 5 ออก → ใส่ 4
  → [2, 4, 5]

หยิบ 6: 6 > 5 → ใส่หลัง 5
  → [2, 4, 5, 6]

หยิบ 1: เลื่อน 6,5,4,2 ออก → ใส่ 1
  → [1, 2, 4, 5, 6] ✅
ข้อสังเกต
/* ไม่ swap — ใช้ shift แทน */
/* shift เร็วกว่า swap (1 write) */

arr[j+1] = arr[j];  /* shift right */
j--;

/* เมื่อหาช่องเจอ: */
arr[j+1] = key;     /* วาง key */

/* Nearly sorted → เกือบไม่ shift
   → O(n) best case */
Insertion Sort ดีที่สุดสำหรับข้อมูล nearly sorted — แต่ละ key ใช้ shift น้อยมาก
บทที่ 3 · โค้ด

Insertion Sort — โค้ด C

#include <stdio.h>

void insertion_sort(int arr[], int n) {
    int i, j, key;

    for (i = 1; i < n; i++) {          /* เริ่มจาก index 1 */
        key = arr[i];                    /* หยิบ key ออกมา */
        j = i - 1;

        /* เลื่อนตัวที่ > key ไปทางขวา */
        while (j >= 0 && arr[j] > key) {
            arr[j + 1] = arr[j];        /* shift right */
            j--;
        }
        arr[j + 1] = key;               /* วาง key ลงช่องที่ถูกต้อง */
    }
}

int main() {
    int arr[] = {12, 11, 13, 5, 6};
    int n = sizeof(arr) / sizeof(arr[0]);
    insertion_sort(arr, n);
    for (int i=0; i<n; i++) printf("%d ", arr[i]);
    /* 5 6 11 12 13 */
    return 0;
}
ใช้ shift (arr[j+1]=arr[j]) แทน swap — ลด write operation ลงครึ่งหนึ่ง เร็วกว่า Bubble ในทางปฏิบัติ

บทที่ 3 · Complexity

Best O(n) เมื่อไหร่?

Best Case: Already Sorted
/* {1, 2, 3, 4, 5} */
i=1: key=2, arr[0]=1 < 2 → ไม่ shift
i=2: key=3, arr[1]=2 < 3 → ไม่ shift
i=3: key=4, arr[2]=3 < 4 → ไม่ shift
i=4: key=5, arr[3]=4 < 5 → ไม่ shift

Shifts: 0 → O(n) comparisons เท่านั้น
/* เร็วกว่า Selection/Bubble มาก! */
Worst Case: Reverse Sorted
/* {5, 4, 3, 2, 1} */
i=1: key=4, shift 1 ครั้ง
i=2: key=3, shift 2 ครั้ง
i=3: key=2, shift 3 ครั้ง
i=4: key=1, shift 4 ครั้ง

Total shifts: 1+2+3+4 = n(n-1)/2
= O(n²) shifts

Best:  O(n)   ← sorted!
Avg:   O(n²)
Worst: O(n²)  ← reverse sorted
Insertion Sort ดีที่สุดเมื่อ: 1) ข้อมูลเกือบเรียงแล้ว 2) n เล็ก (<50) 3) Online sort (รับข้อมูลทีละตัว)
บทที่ 3 · Online Sort

Online Sort — รับข้อมูลทีละตัว

Insertion Sort เป็น Online Algorithm — รับ element ใหม่มาแล้วแทรกเข้า sorted array ได้ทันที ไม่ต้องรอข้อมูลครบ

/* รับ transaction แบบ realtime แล้ว maintain sorted order */
void insert_txn(double amounts[], int *size, double new_amount) {
    int j = *size - 1;

    /* เลื่อนตัวที่ > new_amount ไปขวา */
    while (j >= 0 && amounts[j] > new_amount) {
        amounts[j + 1] = amounts[j];
        j--;
    }
    amounts[j + 1] = new_amount;
    (*size)++;
}

int main() {
    double sorted[100]; int size = 0;

    /* รับ transaction realtime */
    insert_txn(sorted, &size, 500.0);  /* → {500} */
    insert_txn(sorted, &size, 200.0);  /* → {200, 500} */
    insert_txn(sorted, &size, 350.0);  /* → {200, 350, 500} */
    /* array เรียงอยู่เสมอ ไม่ต้อง sort ใหม่ */
}
บทที่ 3 · Optimization

Binary Insertion Sort

ใช้ Binary Search หาตำแหน่งแทน linear scan — ลด comparisons จาก O(n) เป็น O(log n) ต่อ key แต่ยังต้อง shift O(n)

/* Binary Search หาตำแหน่งที่จะแทรก */
int binary_search(int arr[], int lo, int hi, int key) {
    while (lo < hi) {
        int mid = (lo + hi) / 2;
        if (arr[mid] < key) lo = mid + 1;
        else                hi = mid;
    }
    return lo;
}

void binary_insertion_sort(int arr[], int n) {
    int i, j, key, pos;
    for (i = 1; i < n; i++) {
        key = arr[i];
        pos = binary_search(arr, 0, i, key);  /* O(log i) */
        j = i - 1;
        while (j >= pos) { arr[j+1] = arr[j]; j--; }  /* shift O(n) */
        arr[pos] = key;
    }
}
/* Total: O(n log n) comparisons, O(n²) shifts
   → ยังเป็น O(n²) overall แต่ comparison ลดลง */
บทที่ 3 · Banking Use Case

🏦 Realtime Transaction Leaderboard

ระบบ anti-fraud ต้องการ Top-10 transaction ใหญ่สุดแบบ realtime — ใช้ Insertion Sort maintain sorted array ขนาดเล็ก

#define TOP_K 10

typedef struct { char id[10]; double amount; } Txn;

/* รักษา top-K sorted array */
void update_top_k(Txn top[], int *size, Txn new_txn) {
    if (*size < TOP_K || new_txn.amount > top[*size-1].amount) {
        int pos = (*size < TOP_K) ? *size : TOP_K - 1;
        int j = pos - 1;

        /* หาตำแหน่งแทรก (descending) */
        while (j >= 0 && top[j].amount < new_txn.amount) {
            top[j+1] = top[j]; j--;
        }
        top[j+1] = new_txn;
        if (*size < TOP_K) (*size)++;
    }
}
/* n=10 เสมอ → O(10) = O(1) per insert
   เร็วมาก เหมาะกับ realtime feed */
