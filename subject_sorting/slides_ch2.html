บทที่ 2 · หลักการ

Selection Sort คิดยังไง?

แบ่ง array เป็น 2 ส่วน: sorted (ซ้าย) และ unsorted (ขวา) — ทุก pass หา min ใน unsorted แล้ว swap ไปต่อท้าย sorted

แต่ละ pass ทำอะไร
/* arr = {29,10,14,37,13} */

Pass 0: unsorted=[0..4]
  min อยู่ที่ index 4 (13)
  swap arr[0]↔arr[4]
  → {13,10,14,37,29}

Pass 1: unsorted=[1..4]
  min อยู่ที่ index 1 (10)
  swap arr[1]↔arr[1] (ตัวเอง)
  → {13,10,14,37,29}

Pass 2: unsorted=[2..4]
  min อยู่ที่ index 4 (29)
  swap arr[2]↔arr[4] ...
ข้อสังเกต
/* ทุก pass: swap แค่ 1 ครั้ง! */
/* n-1 passes → n-1 swaps max */

/* เทียบกับ Bubble: */
Bubble:    O(n²) swaps worst
Selection: O(n)  swaps เสมอ

/* ถ้า swap cost สูง (เขียน disk,
   เขียน flash memory) → Selection
   ดีกว่า Bubble มาก */
Selection Sort = หยิบตัวเล็กสุดมาวางซ้ายสุดของ unsorted — ทำจนหมด n-1 รอบ
บทที่ 2 · โค้ด

Selection Sort — โค้ด C

#include <stdio.h>

void swap(int *a, int *b) {
    int temp = *a; *a = *b; *b = temp;
}

void selection_sort(int arr[], int n) {
    int i, j, min_idx;

    for (i = 0; i < n - 1; i++) {     /* ตำแหน่งที่จะวาง min */
        min_idx = i;                      /* สมมติว่า i คือ min */

        for (j = i + 1; j < n; j++) {   /* หา min จริงๆ */
            if (arr[j] < arr[min_idx]) {
                min_idx = j;              /* อัปเดต index ของ min */
            }
        }

        if (min_idx != i) {               /* swap ก็ต่อเมื่อ min ไม่ใช่ตำแหน่ง i */
            swap(&arr[min_idx], &arr[i]);
        }
    }
}

int main() {
    int arr[] = {64, 25, 12, 22, 11};
    int n = sizeof(arr) / sizeof(arr[0]);
    selection_sort(arr, n);
    for (int i=0; i<n; i++) printf("%d ", arr[i]);
    /* 11 12 22 25 64 */
}

บทที่ 2 · ทำไม Unstable

ทำไม Selection Sort ถึง Unstable?

เพราะ swap ข้ามระยะไกล อาจสลับตำแหน่ง relative ของค่าเท่ากัน

ตัวอย่าง Unstable
/* arr = {3a, 3b, 1} */
/* 3a, 3b = ค่า 3 ทั้งคู่
   แต่ 3a มาก่อน 3b ตั้งต้น */

Pass 0 (i=0):
  min = 1 at idx 2
  swap arr[0]↔arr[2]
  → {1, 3b, 3a}
      ↑  ⚠️ 3a, 3b สลับแล้ว!

/* Stable sort จะได้ {1, 3a, 3b} */
Stable version (ใช้ shift แทน swap)
/* ทำให้ Stable ได้ แต่ cost สูงขึ้น */
void stable_selection(int arr[], int n) {
  for (int i=0; i<n-1; i++) {
    int min_idx = i;
    for (int j=i+1; j<n; j++)
      if(arr[j]<arr[min_idx]) min_idx=j;
    int key = arr[min_idx];
    /* shift แทน swap */
    while(min_idx > i) {
      arr[min_idx] = arr[min_idx-1];
      min_idx--;
    }
    arr[i] = key;
  }
} /* O(n²) shifts — ช้ากว่าเดิม */
ในทางปฏิบัติ: ถ้าต้องการ Stable sort ใช้ Insertion Sort หรือ Merge Sort แทน
บทที่ 2 · Complexity

Complexity วิเคราะห์

Comparisons
Pass 0: n-1 comparisons
Pass 1: n-2 comparisons
...
Pass n-2: 1 comparison

Total: (n-1)+(n-2)+...+1
     = n(n-1)/2 = O(n²)

/* ทุก case = O(n²) */
/* ไม่มี early termination */
/* แม้ข้อมูลเรียงแล้วก็ยังวน */
Summary
Best Case:   O(n²)  ← เหมือน worst!
Average:     O(n²)
Worst Case:  O(n²)

Swaps:       O(n)   ← จุดเด่น!
Space:       O(1)   in-place
Stable:      ❌ No
In-place:    ✅ Yes

/* เลือกใช้เมื่อ: */
/* swap cost สูงมาก */
/* (flash mem, SSD write) */
Selection Sort ดีกว่า Bubble เพราะ swap น้อยกว่ามาก แต่แย่กว่าตรงที่ทุก case = O(n²) ไม่มี best case ดีกว่า
บทที่ 2 · Variant

หา Max แทน Min — เรียง Descending

เปลี่ยนเล็กน้อย: หา max แทน min วางทางขวา → ได้ descending order

/* Descending sort */
void selection_sort_desc(int arr[], int n) {
    int i, j, max_idx;
    for (i = 0; i < n - 1; i++) {
        max_idx = i;                        /* หา max แทน min */
        for (j = i + 1; j < n; j++) {
            if (arr[j] > arr[max_idx])      /* เปลี่ยน < เป็น > */
                max_idx = j;
        }
        if (max_idx != i)
            swap(&arr[max_idx], &arr[i]);
    }
}

/* {64,25,12,22,11} → {64,25,22,12,11} */

/* Banking: เรียง transaction จาก amount มากไปน้อย
   เพื่อแสดง top transactions ก่อน */
void sort_txn_desc(Transaction arr[], int n) {
    int i, j, max_idx;
    Transaction temp;
    for (i = 0; i < n-1; i++) {
        max_idx = i;
        for (j = i+1; j < n; j++)
            if (arr[j].amount > arr[max_idx].amount) max_idx = j;
        if (max_idx != i) { temp=arr[i]; arr[i]=arr[max_idx]; arr[max_idx]=temp; }
    }
}
บทที่ 2 · Banking Use Case

🏦 เลือก Algorithm ตาม Constraint

ธนาคารต้อง sort ข้อมูล ATM logs ที่เก็บใน flash memory — swap cost สูง เพราะ write ทุกครั้ง = wear

Scenario: ATM Flash Memory
n = 200 records
Write cost: 1ms per swap
Read cost:  0.01ms per compare

Bubble Sort:
  Swaps worst: ~20,000
  Write time: 20 seconds ⚠️

Selection Sort:
  Swaps: 199 max
  Write time: 0.2 seconds ✅
Code สำหรับ ATM logs
typedef struct {
    int  atm_id;
    long timestamp;
    double amount;
} ATMLog;

/* sort by timestamp, minimize writes */
void sort_atm_logs(ATMLog logs[], int n) {
    int i, j, min_idx;
    for (i=0; i<n-1; i++) {
        min_idx = i;
        for (j=i+1; j<n; j++)
          if(logs[j].timestamp < logs[min_idx].timestamp)
            min_idx = j;
        if(min_idx!=i){
          ATMLog t=logs[i]; logs[i]=logs[min_idx];
          logs[min_idx]=t;
        }
    }
}
