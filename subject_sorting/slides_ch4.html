Divide & Conquer

Divide: แบ่ง array ครึ่งซ้ำๆ จนเหลือ 1 ตัว (base case) · Conquer: merge สอง sorted half กลับ

[38, 27, 43, 3, 9, 82, 10] ↙ ↘ [38, 27, 43] [3, 9, 82, 10] ↙ ↘ ↙ ↘ [38] [27, 43] [3, 9] [82, 10] ↙ ↘ ↙ ↘ ↙ ↘ [27][43] [3] [9] [82] [10] ↘ ↙ ↘ ↙ ↘ ↙ [27,43] [3,9] [10,82] ↘ ↙ [3, 9, 27, 43] [10, 82] (ตัวอย่างย่อ) ↘ ↙ [3, 9, 10, 27, 43, 82]
ความลึกของ recursion = log₂(n) · แต่ละ level ทำ merge รวม O(n) → Total O(n log n)
บทที่ 4 · Merge

Merge Function — หัวใจของ Merge Sort

void merge(int arr[], int l, int m, int r) {
    int n1 = m - l + 1, n2 = r - m;
    int L[n1], R[n2];           /* temp arrays — O(n) space */

    for (int i=0; i<n1; i++) L[i] = arr[l+i];
    for (int j=0; j<n2; j++) R[j] = arr[m+1+j];

    int i=0, j=0, k=l;
    while (i < n1 && j < n2) {
        if (L[i] <= R[j]) {        /* <= คือ Stable! ค่าเท่ากัน L มาก่อน */
            arr[k++] = L[i++];
        } else {
            arr[k++] = R[j++];
        }
    }
    while (i < n1) arr[k++] = L[i++];  /* เหลือใน L */
    while (j < n2) arr[k++] = R[j++];  /* เหลือใน R */
}
ใช้ <= แทน < ตรงนี้สำคัญมาก — ทำให้ Merge Sort เป็น Stable sort
บทที่ 4 · โค้ดเต็ม

Merge Sort — โค้ด C เต็ม

#include <stdio.h>

void merge(int arr[], int l, int m, int r); /* จาก slide ก่อน */

void merge_sort(int arr[], int l, int r) {
    if (l >= r) return;           /* base case: 1 ตัว */

    int m = l + (r - l) / 2;     /* หาจุดกึ่งกลาง */

    merge_sort(arr, l, m);         /* sort ซ้าย */
    merge_sort(arr, m + 1, r);    /* sort ขวา */
    merge(arr, l, m, r);           /* merge สองฝั่ง */
}

int main() {
    int arr[] = {38, 27, 43, 3, 9, 82, 10};
    int n = sizeof(arr) / sizeof(arr[0]);
    merge_sort(arr, 0, n - 1);
    for (int i=0; i<n; i++) printf("%d ", arr[i]);
    /* 3 9 10 27 38 43 82 */
    return 0;
}
หมายเหตุ: m = l + (r-l)/2 ป้องกัน integer overflow ดีกว่า (l+r)/2
บทที่ 4 · Trace

Trace: Merge [1,3,5] และ [2,4,6]

ดูว่า merge function ทำงานยังไงเมื่อรับ 2 sorted array

L = [1, 3, 5]   R = [2, 4, 6]
i=0, j=0, k=0

Step 1: L[0]=1 ≤ R[0]=2 → arr[0]=1, i=1
Step 2: L[1]=3 > R[0]=2 → arr[1]=2, j=1
Step 3: L[1]=3 ≤ R[1]=4 → arr[2]=3, i=2
Step 4: L[2]=5 > R[1]=4 → arr[3]=4, j=2
Step 5: L[2]=5 ≤ R[2]=6 → arr[4]=5, i=3
Step 6: i=3=n1 → exit while
        copy R[2]=6 → arr[5]=6

ผลลัพธ์: [1, 2, 3, 4, 5, 6] ✅
merge ใช้ O(n) time และ O(n) space (temp arrays L, R) — นี่คือข้อเสียเดียวของ Merge Sort
บทที่ 4 · Complexity

ทำไม O(n log n)?

วิเคราะห์ตาม level
n = 8

Level 0: 1 call  → merge O(n)
Level 1: 2 calls → merge O(n/2) × 2 = O(n)
Level 2: 4 calls → merge O(n/4) × 4 = O(n)
Level 3: 8 calls → base case

จำนวน level = log₂(8) = 3
แต่ละ level รวม O(n)
→ Total = O(n × log n)
Summary
Best Case:  O(n log n) ✅
Average:    O(n log n) ✅
Worst Case: O(n log n) ✅ ← guaranteed!

Space:  O(n) ← จุดเสีย
Stable: ✅ Yes
In-place: ❌ No

บทที่ 4 · Stable

ทำไม Merge Sort ถึง Stable?

เพราะ merge function ใช้ L[i] <= R[j] — ค่าเท่ากัน ฝั่ง L (ซ้าย = เดิมมาก่อน) จะถูกหยิบก่อนเสมอ

ตัวอย่าง: ค่าเท่ากัน
/* L = [3a, 5]  R = [3b, 7]
   3a มาจากฝั่งซ้าย (index ก่อน)
   3b มาจากฝั่งขวา */

Merge:
  L[0]=3a ≤ R[0]=3b → arr[0]=3a
  L[1]=5  > R[0]=3b → arr[1]=3b
  L[1]=5  ≤ R[1]=7  → arr[2]=5
  arr[3]=7

Result: [3a, 3b, 5, 7]
✅ 3a ยังมาก่อน 3b
ประโยชน์ใน Multi-key Sort
/* ธนาคาร: sort ธุรกรรม 2 รอบ */

Step 1: sort ตาม amount
Step 2: sort ตาม date
        (ใช้ Stable sort)

ผลลัพธ์: เรียงตาม date
  แต่ในวันเดียวกัน → ยังเรียง
  ตาม amount จาก step 1 ✅

/* ถ้า sort step 2 ด้วย Unstable

บทที่ 4 · Iterative

Bottom-up Merge Sort (ไม่ใช้ Recursion)

เริ่มจาก subarray ขนาด 1 แล้วเพิ่มทีละ 2× — ใช้ loop แทน recursion ไม่มี stack overflow

void merge_sort_iterative(int arr[], int n) {
    int curr_size, left_start;

    for (curr_size = 1; curr_size < n; curr_size *= 2) {
        for (left_start = 0; left_start < n-1; left_start += 2*curr_size) {
            int mid = left_start + curr_size - 1;
            int right_end = left_start + 2*curr_size - 1;
            if (right_end >= n) right_end = n - 1;
            if (mid < right_end)
                merge(arr, left_start, mid, right_end);
        }
    }
}
/* n=8:
   Round 1: merge pairs  [0,1][2,3][4,5][6,7]
   Round 2: merge quads  [0,3][4,7]
   Round 3: merge all    [0,7]
*/
บทที่ 4 · Banking Use Case

🏦 Sort Statement แบบ Multi-key

#include <string.h>

typedef struct {
    char date[11];   /* "2024-01-15" */
    double amount;
    char type[10];   /* "DEBIT"/"CREDIT" */
} Statement;

int cmp_date_then_amount(Statement a, Statement b) {
    int d = strcmp(a.date, b.date);
    if (d != 0) return d;          /* primary: date */
    return (a.amount > b.amount) -   /* secondary: amount desc */
           (a.amount < b.amount);
}

void merge_stmt(Statement arr[], int l, int m, int r) {
    int n1=m-l+1, n2=r-m;
    Statement L[n1], R[n2];
    for(int i=0;i<n1;i++) L[i]=arr[l+i];
    for(int j=0;j<n2;j++) R[j]=arr[m+1+j];
    int i=0,j=0,k=l;
    while(i<n1&&j<n2) {
        if(cmp_date_then_amount(L[i],R[j])<=0) arr[k++]=L[i++];
        else arr[k++]=R[j++];
    }
    while(i<n1) arr[k++]=L[i++];
    while(j<n2) arr[k++]=R[j++];
}
