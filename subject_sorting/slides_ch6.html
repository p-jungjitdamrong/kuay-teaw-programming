Max Heap คืออะไร?

Max Heap: Binary Tree ที่ parent ≥ children เสมอ → root คือค่าสูงสุด เก็บใน array โดย children ของ i อยู่ที่ 2i+1, 2i+2

90
/ \
75 82
/ \ / \
55 60 38 45
/ \
10 20
เก็บใน Array
idx: 0   1   2   3   4   5   6   7   8
val: 90  75  82  55  60  38  45  10  20

parent(i) = (i-1)/2
left(i)   = 2*i + 1
right(i)  = 2*i + 2

idx 1 (75): left=3(55), right=4(60)
idx 2 (82): left=5(38), right=6(45)

/* parent ≥ children ทุกตัว */
Heap Sort ใช้ Max Heap เก็บใน array เดิม — In-place O(1) extra space
บทที่ 6 · โค้ด

Heapify และ Heap Sort

void heapify(int arr[], int n, int i) {
    int largest = i;
    int l = 2*i + 1, r = 2*i + 2;

    if (l < n && arr[l] > arr[largest]) largest = l;
    if (r < n && arr[r] > arr[largest]) largest = r;

    if (largest != i) {
        swap(&arr[i], &arr[largest]);
        heapify(arr, n, largest);  /* fix ลงไปข้างล่าง */
    }
}

void heap_sort(int arr[], int n) {
    /* Phase 1: Build Max Heap — O(n) */
    for (int i = n/2 - 1; i >= 0; i--)
        heapify(arr, n, i);

    /* Phase 2: ดึง max ออกทีละตัว — O(n log n) */
    for (int i = n-1; i > 0; i--) {
        swap(&arr[0], &arr[i]);   /* root (max) ไปท้าย */
        heapify(arr, i, 0);      /* heapify ส่วนที่เหลือ */
    }
}
Build heap = O(n) · Phase 2 = n-1 ครั้ง × O(log n) = O(n log n) · Total O(n log n) ทุก case
บทที่ 6 · Non-Comparison

Counting Sort — O(n+k)

นับจำนวนของแต่ละค่า ไม่ต้องเปรียบเทียบ — เร็วกว่า O(n log n) แต่ใช้ได้กับ integer ที่รู้ขอบเขต k เท่านั้น

void counting_sort(int arr[], int n, int max_val) {
    int count[max_val + 1];
    int output[n];

    /* Phase 1: นับ */
    for (int i=0; i<=max_val; i++) count[i] = 0;
    for (int i=0; i<n; i++) count[arr[i]]++;

    /* Phase 2: cumulative — หาตำแหน่ง */
    for (int i=1; i<=max_val; i++) count[i] += count[i-1];

    /* Phase 3: วาง output (วนจากหลังเพื่อ Stable) */
    for (int i=n-1; i>=0; i--) {
        output[--count[arr[i]]] = arr[i];
    }
    for (int i=0; i<n; i++) arr[i] = output[i];
}
/* {4,2,2,8,3,3,1} max=8
   count[1]=1,count[2]=2,count[3]=2,count[4]=1,count[8]=1
   → {1,2,2,3,3,4,8} */
บทที่ 6 · Radix

Radix Sort — sort ทีละหลัก

ใช้ Counting Sort เป็น subroutine sort ทีละหลัก (ones → tens → hundreds) — O(d × n) โดย d = จำนวนหลัก

void counting_sort_digit(int arr[], int n, int exp) {
    int output[n], count[10] = {0};
    for (int i=0; i<n; i++) count[(arr[i]/exp)%10]++;
    for (int i=1; i<10; i++) count[i]+=count[i-1];
    for (int i=n-1; i>=0; i--) { output[--count[(arr[i]/exp)%10]]=arr[i]; }
    for (int i=0; i<n; i++) arr[i]=output[i];
}

void radix_sort(int arr[], int n) {
    int max_val = arr[0];
    for (int i=1; i<n; i++) if(arr[i]>max_val) max_val=arr[i];

    for (int exp=1; max_val/exp>0; exp*=10)
        counting_sort_digit(arr, n, exp);
}
/* {170,45,75,90,802,24,2,66}
   ones:  {170,90,802,2,24,45,75,66}
   tens:  {802,2,24,45,66,170,75,90}
   hundreds: {2,24,45,66,75,90,170,802} ✅ */

บทที่ 6 · Trace

Trace Heap Sort: {4, 10, 3, 5, 1}

/* Phase 1: Build Max Heap
   เริ่มจาก i = n/2-1 = 1 ถึง 0 */

i=1: heapify({4,10,3,5,1}, 5, 1)
  left=3(5), right=4(1), largest=3 → swap(arr[1],arr[3])
  → {4,5,3,10,1} → wait, 10>5 so largest=idx-of-10
  arr[1]=10, left=arr[3]=5, right=arr[4]=1 → largest=1 (10) no swap
  → {4,10,3,5,1}

i=0: heapify({4,10,3,5,1}, 5, 0)
  left=arr[1]=10, right=arr[2]=3 → largest=1(10) → swap(arr[0],arr[1])
  → {10,4,3,5,1} → heapify(idx1): left=arr[3]=5>arr[1]=4 → swap
  → {10,5,3,4,1}  ← Max Heap สร้างแล้ว!

/* Phase 2: ดึง max ทีละตัว */
i=4: swap(arr[0],arr[4])={1,5,3,4,10}, heapify(4,0)→{5,4,3,1,10}
i=3: swap(arr[0],arr[3])={1,4,3,5,10}, heapify(3,0)→{4,1,3,5,10}
i=2: swap(arr[0],arr[2])={3,1,4,5,10}, heapify(2,0)→{3,1,4,5,10}
i=1: swap(arr[0],arr[1])={1,3,4,5,10}
→ {1,3,4,5,10} ✅
