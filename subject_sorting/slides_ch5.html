บทที่ 5 · หลักการ

Quick Sort — Pivot และ Partition

เลือก pivot → จัด element: ตัวที่ < pivot ไปซ้าย, ตัวที่ > pivot ไปขวา → pivot อยู่ตำแหน่งถูกต้อง → recurse ซ้ายขวา

แนวคิด partition
/* arr = {3,6,8,10,1,2,1}
   pivot = arr[r] = 1 (ตัวสุดท้าย) */

หลัง partition:
{1, 1, | 8, 10, 6, 2, 3}
         ↑
      pivot อยู่ตำแหน่งนี้แล้ว

ซ้ายของ pivot: ทุกตัว ≤ 1
ขวาของ pivot: ทุกตัว > 1

→ recurse ซ้าย {1}
→ recurse ขวา {8,10,6,2,3}
ทำไมเร็ว
/* In-place: ไม่ต้องสร้าง array ใหม่ */
/* Cache friendly: เข้าถึงหน่วยความจำ
   ตามลำดับ → cache hit rate สูง */

/* constant factor ต่ำกว่า Merge Sort
   ในทางปฏิบัติเร็วกว่า 2-3 เท่า */

/* แม้ worst O(n²) แต่ถ้าใช้ random
   pivot โอกาสเกิด worst น้อยมาก */
Quick Sort ไม่ stable เพราะ partition swap ข้ามระยะไกล — ถ้าต้องการ stable ใช้ Merge Sort
บทที่ 5 · Partition

Partition Function (Lomuto Scheme)

int partition(int arr[], int lo, int hi) {
    int pivot = arr[hi];  /* เลือกตัวสุดท้ายเป็น pivot */
    int i = lo - 1;       /* i = ขอบของ "zone น้อยกว่า pivot" */

    for (int j = lo; j < hi; j++) {
        if (arr[j] <= pivot) {    /* ถ้า j น้อยกว่าหรือเท่า pivot */
            i++;                      /* ขยาย zone น้อยกว่า */
            swap(&arr[i], &arr[j]); /* ดึงเข้า zone */
        }
    }
    swap(&arr[i + 1], &arr[hi]);  /* วาง pivot ที่ตำแหน่งถูกต้อง */
    return i + 1;                 /* return ตำแหน่ง pivot */
}

void quick_sort(int arr[], int lo, int hi) {
    if (lo < hi) {
        int pi = partition(arr, lo, hi);
        quick_sort(arr, lo, pi - 1);   /* ซ้ายของ pivot */
        quick_sort(arr, pi + 1, hi);  /* ขวาของ pivot */
    }
}

บทที่ 5 · Hoare

Hoare Partition Scheme — swap น้อยกว่า

Hoare ใช้ 2 pointer จากสองข้าง — swap น้อยกว่า Lomuto ประมาณ 3 เท่า แต่โค้ดซับซ้อนกว่า

int partition_hoare(int arr[], int lo, int hi) {
    int pivot = arr[lo + (hi-lo)/2];  /* pivot ตรงกลาง */
    int i = lo - 1, j = hi + 1;

    while (1) {
        do { i++; } while (arr[i] < pivot);   /* หา i ที่ arr[i] >= pivot */
        do { j--; } while (arr[j] > pivot);   /* หา j ที่ arr[j] <= pivot */

        if (i >= j) return j;               /* pointer ชนกัน → เสร็จ */
        swap(&arr[i], &arr[j]);
    }
}

void quick_sort_hoare(int arr[], int lo, int hi) {
    if (lo < hi) {
        int p = partition_hoare(arr, lo, hi);
        quick_sort_hoare(arr, lo, p);       /* ไม่ใช่ p-1 ! */
        quick_sort_hoare(arr, p + 1, hi);
    }
}
บทที่ 5 · Complexity

Complexity Analysis

Best / Average: O(n log n)
/* pivot แบ่งครึ่งพอดีทุกครั้ง */
Level 0: partition n elements
Level 1: partition n/2 each × 2
Level 2: partition n/4 each × 4
...
log n levels × O(n) per level
= O(n log n)

/* Random pivot → average O(n log n)
   ด้วย high probability */
Worst: O(n²)
/* pivot ไม่สมดุล: 1 ตัว vs n-1 */
Level 0: n comparisons
Level 1: n-1 comparisons
...
Level n-1: 1 comparison
Total: n(n-1)/2 = O(n²)

Summary:
Best:   O(n log n)
Avg:    O(n log n)
Worst:  O(n²) ← ถ้า pivot แย่
Space:  O(log n) call stack
Stable: ❌ No
In-place: ✅ Yes
บทที่ 5 · 3-way

3-way Partition — จัดการค่าซ้ำ

ถ้ามีค่าซ้ำเยอะ Quick Sort ปกติยังช้า — 3-way partition แยก: <pivot | ==pivot | >pivot ทำให้ดีขึ้นมาก

void quick_sort_3way(int arr[], int lo, int hi) {
    if (lo >= hi) return;
    int lt = lo, gt = hi, i = lo + 1;
    int pivot = arr[lo];

    while (i <= gt) {
        if (arr[i] < pivot) {
            swap(&arr[lt++], &arr[i++]);  /* น้อยกว่า → ซ้าย */
        } else if (arr[i] > pivot) {
            swap(&arr[i], &arr[gt--]);    /* มากกว่า → ขวา */
        } else {
            i++;                           /* เท่ากัน → ข้ามไป */
        }
    }
    /* arr[lo..lt-1] < pivot
       arr[lt..gt]   == pivot  (ไม่ต้อง recurse!)
       arr[gt+1..hi] > pivot */
    quick_sort_3way(arr, lo, lt - 1);
    quick_sort_3way(arr, gt + 1, hi);
}
/* ข้อมูลที่มีค่าซ้ำเยอะ → O(n) !! */
บทที่ 5 · Banking Use Case

🏦 Credit Score Ranking

#include <stdlib.h>
#include <time.h>

typedef struct { char id[12]; int score; } Customer;

void swap_cust(Customer *a, Customer *b) { Customer t=*a; *a=*b; *b=t; }

int partition_cust(Customer arr[], int lo, int hi) {
    /* random pivot เพื่อป้องกัน worst case */
    int r = lo + rand() % (hi - lo + 1);
    swap_cust(&arr[r], &arr[hi]);

    int pivot = arr[hi].score, i = lo - 1;
    for (int j=lo; j<hi; j++) {
        if (arr[j].score >= pivot)  /* descending: score สูงมาก่อน */
            swap_cust(&arr[++i], &arr[j]);
    }
    swap_cust(&arr[i+1], &arr[hi]);
    return i + 1;
}

void rank_customers(Customer arr[], int lo, int hi) {
    if (lo < hi) {
        int p = partition_cust(arr, lo, hi);
        rank_customers(arr, lo, p-1);
        rank_customers(arr, p+1, hi);
    }
}
/* n=1,000,000 customers → sort ใน ~20M comparisons */
