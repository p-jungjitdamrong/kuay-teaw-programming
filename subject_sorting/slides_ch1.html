Bubble Sort คิดยังไง?

เปรียบเทียบ คู่ข้างๆ กัน ทีละคู่ ถ้าซ้าย > ขวา → swap — ทำซ้ำจนค่าใหญ่สุด "ลอย" ไปตำแหน่งสุดท้าย

หนึ่ง pass เต็มๆ
/* arr = {5, 3, 8, 1} */

Pass 1:
  [5,3] → swap → {3,5,8,1}
  [5,8] → ไม่ swap
  [8,1] → swap → {3,5,1,8}
                              ↑ 8 ลอยถึงจุดหมาย!

Pass 2: จบที่ตำแหน่ง n-2
Pass 3: จบที่ตำแหน่ง n-3
...
ภาพรวม: ทำไมเรียก Bubble?
/* ค่าใหญ่ไหลขึ้นทางขวา
   เหมือนฟองอากาศลอยขึ้น */

Pass 1: ค่าใหญ่สุดอยู่ที่ arr[n-1]
Pass 2: ค่าใหญ่ที่ 2 อยู่ที่ arr[n-2]
...
Pass n-1: เรียงครบแล้ว!

/* ทำ n-1 pass → O(n²) comparisons
   n=1000 → ≈500,000 comparisons */
กฎง่ายๆ: ถ้า arr[j] > arr[j+1] → swap — วนซ้ำจนไม่มี swap เกิดขึ้น = เรียงแล้ว
บทที่ 1 · โค้ด

Bubble Sort — โค้ด C

#include <stdio.h>

void swap(int *a, int *b) {
    int temp = *a; *a = *b; *b = temp;
}

void bubble_sort(int arr[], int n) {
    int i, j;
    for (i = 0; i < n - 1; i++) {          /* n-1 passes */
        for (j = 0; j < n - 1 - i; j++) {  /* ลดขอบขวาทุก pass */
            if (arr[j] > arr[j + 1]) {
                swap(&arr[j], &arr[j + 1]);
            }
        }
    }
}

int main() {
    int arr[] = {64, 34, 25, 12, 22, 11, 90};
    int n = sizeof(arr) / sizeof(arr[0]);
    bubble_sort(arr, n);
    for (int i = 0; i < n; i++)
        printf("%d ", arr[i]);  /* 11 12 22 25 34 64 90 */
    return 0;
}
inner loop วน n-1-i รอบ เพราะ i ตัวหลังสุดได้ที่ถูกต้องแล้ว ไม่ต้องตรวจซ้ำ

บทที่ 1 · Optimization

Early Termination — O(n) Best Case

ถ้า pass ไหนไม่มี swap เกิดขึ้นเลย แสดงว่าเรียงแล้ว — หยุดได้ทันที ทำให้ best case กลายเป็น O(n)

void bubble_sort_optimized(int arr[], int n) {
    int i, j;
    int swapped;  /* flag: มี swap ใน pass นี้ไหม */

    for (i = 0; i < n - 1; i++) {
        swapped = 0;  /* reset ทุก pass */

        for (j = 0; j < n - 1 - i; j++) {
            if (arr[j] > arr[j + 1]) {
                swap(&arr[j], &arr[j + 1]);
                swapped = 1;  /* มี swap เกิดขึ้น */
            }
        }

        if (!swapped) break;  /* ไม่มี swap = เรียงแล้ว! */
    }
}

/* Best case: already sorted → O(n) เพราะ pass แรกไม่มี swap */
/* Worst case: reverse sorted → O(n²) ยังเหมือนเดิม */
ข้อมูลธนาคารที่ sort ไว้แล้ว แค่มีธุรกรรมใหม่ 1-2 ตัวต่อท้าย → Insertion/Bubble + early term เร็วมาก

บทที่ 1 · เปรียบเทียบ

Bubble vs Selection vs Insertion

Bubble — swap บ่อย
{5,1,4,2,8}
→ swap ทุกครั้งที่ไม่ลำดับ
→ 5 ต้องเลื่อนไปทีละช่อง
→ swap มาก กว่า Selection

Swap count: O(n²) worst
Selection — swap น้อยที่สุด
{5,1,4,2,8}
→ หา min ก่อน แล้ว swap
   ครั้งเดียวต่อ pass
→ swap แค่ n-1 ครั้ง

Swap count: O(n) เสมอ
Insertion — เหมาะกับ nearly sorted
{1,2,3,10,4}
→ เลื่อน 10 ออก แล้ว ใส่ 4
→ กรณี nearly sorted: O(n)
→ Comparison-wise ดีกว่า Bubble
เลือกใช้
n < 50, random order:
  → ใช้อะไรก็ได้ ต่างกันน้อย

น้อย swap (เขียน disk):
  → Selection Sort

Nearly sorted:
  → Insertion Sort
บทที่ 1 · Banking Use Case

🏦 Sort ยอดธุรกรรมด้วย Bubble Sort

#include <stdio.h>
#include <string.h>

typedef struct {
    char txn_id[10];
    double amount;
} Transaction;

void bubble_sort_txn(Transaction arr[], int n) {
    int i, j; Transaction temp; int swapped;
    for (i = 0; i < n - 1; i++) {
        swapped = 0;
        for (j = 0; j < n - 1 - i; j++) {
            if (arr[j].amount > arr[j+1].amount) {
                temp = arr[j]; arr[j] = arr[j+1]; arr[j+1] = temp;
                swapped = 1;
            }
        }
        if (!swapped) break;
    }
}

int main() {
    Transaction txns[] = {{"T001",500.0},{"T002",150.0},{"T003",300.0}};
    bubble_sort_txn(txns, 3);
    for(int i=0;i<3;i++) printf("%s %.0f\n",txns[i].txn_id,txns[i].amount);
    /* T002 150 / T003 300 / T001 500 */
}
