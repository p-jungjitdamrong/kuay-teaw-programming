Lab 6 · super

super() — ต่อสายถึง constructor แม่

กติกา
public class Undergrad extends Student {
    private String year;

    public Undergrad(int id, String name,
                     double gpa, String year) {
        super(id, name, gpa); // เรียก constructor แม่
        this.year = year;     // แล้วค่อยจัดการของตัวเอง
    }
}
กฎเหล็ก 3 ข้อ
// ① super(...) ต้องเป็น
//    บรรทัดแรกของ constructor

// ② ไม่เขียน super() เอง
//    → Java แอบเรียก super() ว่างให้
//    → ถ้าแม่ไม่มี constructor ว่าง
//      = compile error!

// ③ super.method() เรียก method
//    ของแม่ได้ด้วย เช่น
//    super.toString() ใน Override
แม่สร้างเสร็จก่อน ลูกค่อยเติมของตัวเอง — เหมือนสร้างบ้าน ต้องมีฐานรากก่อนต่อเติมชั้นสอง

Lab 6 · ข้อ 1 เฉลย (1/2)

ข้อ 1: class Student

public class Student {
    protected String name;                // protected ให้ลูกเข้าถึงได้
    protected int id;
    protected double gpa;

    public Student(int id, String name, double gpa) {
        this.id = id;
        this.name = name;
        this.gpa = gpa;
    }
    public Student(int id, double gpa) {  // overload: ไม่มีชื่อ (s2 ในโจทย์)
        this(id, "", gpa);                 // this(...) เรียก constructor ตัวบน!
    }

    public String getName() { return name; }
    public double getGPA()  { return gpa; }
    public int getId()      { return id; }
    public void setName(String name) { this.name = name; }

    public String toString() {             // Override จาก Object
        return "Student:\nID: " + id + "\nName: " + name + "\nGPA: " + gpa;
    }
}
/* System.out.println(s1) →
   Student:
   ID: 97000
   Name: Sameer
   GPA: 3.51   ✓ ตรงโจทย์ */
โบนัส: this(id, "", gpa) — constructor เรียก constructor กันเองได้ ลด code ซ้ำ (เหมือน super แต่ในคลาสเดียวกัน)
Lab 6 · ข้อ 1 เฉลย (2/2)

ข้อ 1: Undergrad และ Graduate

public class Undergrad extends Student {
    private String year;

    public Undergrad(int id, String name, double gpa, String year) {
        super(id, name, gpa);              // ให้แม่จัดการ 3 field แรก
        this.year = year;
    }
    public void setYear(String year) { this.year = year; }
    public String getYear() { return year; }

    public String toString() {             // Override อีกชั้น
        return "Undergraduate " + super.toString()   // reuse ของแม่!
             + "\nYear: " + year;
    }
}

public class Graduate extends Student {
    private String thesisTitle;

    public Graduate(int id, String name, double gpa, String thesisTitle) {
        super(id, name, gpa);
        this.thesisTitle = thesisTitle;
    }
    public void setThesisTitle(String t) { thesisTitle = t; }
    public String getThesisTitle() { return thesisTitle; }

    public String toString() {
        return "Graduate " + super.toString() + "\nThesis: " + thesisTitle;
    }
}
/* Undergraduate Student: / ID: 99000 / Name: Shahid / GPA: 2.91 / Year: Junior ✓ */

Lab 6 · abstract

abstract — แม่แบบที่บังคับให้ลูกเขียนเอง

โจทย์ให้ Employee มาแบบนี้
public abstract class Employee {
    protected String firstName;
    protected String lastName;
    protected String securityNumber;

    public Employee(String first,
        String last, String ssn) { ... }

    public String toString() { ... }

    public abstract double earnings();
    // ↑ ไม่มี body! ลูกต้องเขียนเอง
}
ความหมาย
// abstract class:
// - new ตรงๆ ไม่ได้!
//   new Employee(...) ← error
// - มีไว้เป็นแม่แบบให้สืบทอด

// abstract method:
// - ประกาศไว้แต่ไม่เขียน body
// - ลูก "ทุกตัว" ต้อง override
//   ไม่งั้น compile error

// ทำไมดี? → บังคับสัญญา:
// "พนักงานทุกประเภทต้องคำนวณ
//  เงินเดือนได้ แต่วิธีใครวิธีมัน"
Fixed = เงินเดือนตายตัว · Hourly = รายชั่วโมง · Commission = % จากยอดขาย — สูตรต่างกันหมด แต่ทุกตัวต้องมี earnings()
Lab 6 · ข้อ 3 เฉลย

ข้อ 3: FixedSalary / HourlySalary / CommissionPay

public class FixedSalary extends Employee {
    private double monthlySalary;
    public FixedSalary(String first, String last, String ssn, double salary) {
        super(first, last, ssn);
        monthlySalary = salary;
    }
    public double earnings() { return monthlySalary; }          // ① ตายตัว
    public String toString() {
        return String.format("Fixed salary employee: %s\nmonthly salary: $%.2f",
                             super.toString(), monthlySalary);
    }
}

public class HourlySalary extends Employee {
    private double wage, hours;
    public HourlySalary(String first, String last, String ssn,
                        double wage, double hours) {
        super(first, last, ssn);
        this.wage = wage; this.hours = hours;
    }
    public double earnings() { return wage * hours; }          // ② 20×165 = 3300
    public String toString() {
        return String.format("hourly employee: %s\nhourly pay: $%.2f; hours worked: %.2f",
                             super.toString(), wage, hours);
    }
}

public class CommissionPay extends Employee {
    private double grossSales, rate;
    public CommissionPay(String first, String last, String ssn,
                         double sales, double rate) {
        super(first, last, ssn);
        grossSales = sales; this.rate = rate;
    }
    public double earnings() { return grossSales * rate; }     // ③ 95000×0.06 = 5700
    public String toString() {
        return String.format("commission employee: %s\ngross sales: $%.2f; commission rate: %.2f",
                             super.toString(), grossSales, rate);
    }
}
Lab 6 · ข้อ 3 main

ArrayList<Employee> — เก็บลูกทุกชนิดใน list เดียว

import java.util.ArrayList;

public static void main(String[] args) {
    Employee fixedSalary   = new FixedSalary("Clark", "Kent",
                                 "555-999-5555", 15000.00);
    Employee hourlySalary  = new HourlySalary("Bruce", "Wayne",
                                 "555-777-1111", 20.00, 165);
    Employee commissionPay = new CommissionPay("Peter", "Parker",
                                 "555-111-5555", 95000, 0.06);
    // ↑ ตัวแปรชนิด Employee (แม่) ชี้ object ลูกได้ทุกชนิด

    ArrayList<Employee> employees = new ArrayList<>();   // แก้ [ ผิดในโจทย์เป็น ()
    employees.add(fixedSalary);
    employees.add(hourlySalary);
    employees.add(commissionPay);

    for (Employee emp : employees) {                     // for-each ทีละคน
        System.out.println(emp);                         // เรียก toString ของลูกแต่ละตัว!
        System.out.printf("earned $%.2f%n%n", emp.earnings());
    }
}
/* Fixed salary employee: Clark Kent ... earned $15000.00
   hourly employee: Bruce Wayne ...     earned $3300.00
   commission employee: Peter Parker ...earned $5700.00 ✓

   บรรทัด emp.earnings() เรียกสูตรของใครของมันเอง
   ทั้งที่ตัวแปรเป็นชนิด Employee เหมือนกันหมด
   ← นี่คือ Polymorphism! (Lab 7 เจาะลึกต่อ) */
