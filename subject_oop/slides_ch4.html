ปัญหาของ public — ใครก็พังข้อมูลได้

แบบ Lab 3 (public หมด)
public class Date {
    public int day;
    public int month;
    public int year;
}

// ใครก็ยัดค่าพังๆ ได้:
Date d = new Date();
d.day = 99;      // วันที่ 99?!
d.month = -5;    // เดือน -5?!

// compile ผ่าน รันผ่าน
// แต่ข้อมูลผิดเพี้ยนทั้งระบบ 💀
แบบ Encapsulation (Lab 4)
public class Date {
    private int day;    // ← ซ่อน!
    private int month;
    private int year;

    public void setDay(int d) {
        if (d >= 1 && d <= 31)
            day = d;   // รับเฉพาะค่าดี
        // ค่าเพี้ยน → ไม่รับ เก็บค่าเดิม
    }
}

d.day = 99;      // ❌ compile error!
d.setDay(99);    // ✓ แต่ถูกปฏิเสธ
Encapsulation = ซ่อน field เป็น private แล้วเปิด "ประตูที่มียาม" (getter/setter) — ยามคือ validation ที่คัดค่าเพี้ยนออก
Lab 4 · Pattern

Getter / Setter — สูตรมาตรฐาน

โครงสร้างพื้นฐาน
private int day;

// getter: อ่านค่า
public int getDay() {
    return day;
}

// setter: เขียนค่า (+ validation)
public void setDay(int day) {
    if (day >= 1 && day <= 31)
        this.day = day;
}
getter ไม่จำเป็นต้องคืนค่าดิบ
// โจทย์ Date: "ปีต้องส่งกลับเป็น พ.ศ."
// เก็บ ค.ศ. ข้างใน แต่ getter แปลงให้:

private int year;      // เก็บ ค.ศ.

public int getYear() {
    return year + 543;  // คืน พ.ศ.!
}

// คนใช้ไม่ต้องรู้ว่าข้างในเก็บยังไง
// นี่คือพลังของ encapsulation:
// เปลี่ยนข้างในได้โดยไม่กระทบคนใช้
สูตรจำ: getX() คืนค่า · setX(v) ตรวจก่อนเซ็ต — IDE (IntelliJ/Eclipse) generate ให้ได้ แต่ validation ต้องเขียนเอง
Lab 4 · ข้อ 1 เฉลย

ข้อ 1: class Date ครบทุกเงื่อนไข

public class Date {
    private int day;
    private int month;
    private int year;                        // เก็บเป็น ค.ศ.

    public Date(int day, int month, int year) {   // constructor ตามโจทย์
        this.day = day;
        this.month = month;
        this.year = year;
    }

    public int getDay() { return day; }
    public void setDay(int day) {
        if (day >= 1 && day <= 31)          // เงื่อนไข: 1-31 เท่านั้น
            this.day = day;                  // นอกช่วง → เก็บค่าเดิม (ไม่ทำอะไร)
    }

    public int getMonth() { return month; }
    public void setMonth(int month) {
        if (month >= 1 && month <= 12)      // เงื่อนไข: 1-12 เท่านั้น
            this.month = month;
    }

    public int getYear() {
        return year + 543;                   // เงื่อนไข: คืนเป็น พ.ศ.
    }
    public void setYear(int year) { this.year = year; }

    public String getInfo() {
        return day + "/" + month + "/" + getYear();
    }
}

โจทย์ต้องการ
// field:
//   name (ชื่อบัญชี)
//   balance (เงินคงเหลือ)
//   openDate (ใช้ Date จากข้อ 1!)

// constructor 2 ตัว (overload):
//   BankAccount(name)
//     → เงิน 0, วันนี้
//   BankAccount(name, amount, date)

// methods:
//   deposit(amount)      ฝาก
//   withdraw(amount)     ถอน → boolean
//   transfer(other, amt) โอน
//   getBalance()         ดูเงิน
//   getInfo()            ชื่อ+วันเปิด
จุดคิดสำคัญ
// ① withdraw ต้องเช็คเงินพอ
//    ถอนเกิน → คืน false ไม่หักเงิน

// ② transfer = withdraw ตัวเอง
//              + deposit ให้ other
//    → reuse method ที่มีอยู่!
//    ถ้า withdraw สำเร็จค่อย deposit

// ③ balance ต้อง private
//    ห้ามใครมาแก้เงินตรงๆ:
//    acc.balance = 999999; ← ห้าม!

// ④ Date จากข้อ 1 มาเป็น field
//    = object ซ้อน object
//    (composition — เจอเต็มๆ Lab 5)
transfer อย่าเขียน logic ซ้ำ — เรียก withdraw + deposit ที่เขียนแล้ว ทั้งสั้นกว่าและแก้ bug ที่เดียว
Lab 4 · ข้อ 2 เฉลย

ข้อ 2: โค้ด BankAccount

public class BankAccount {
    private String name;
    private double balance;
    private Date openDate;

    public BankAccount(String name) {                  // เปิดบัญชีเงิน 0 วันนี้
        this.name = name;
        this.balance = 0;
        this.openDate = new Date(19, 7, 2026);     // วันปัจจุบัน
    }
    public BankAccount(String name, double amount, Date date) {
        this.name = name;
        this.balance = amount;
        this.openDate = date;
    }

    public void deposit(double amount) {
        if (amount > 0) balance += amount;             // กันฝากติดลบด้วย
    }

    public boolean withdraw(double amount) {
        if (amount <= balance) {                       // เช็คเงินพอตามโจทย์
            balance -= amount;
            return true;                               // คืนสถานะการถอน
        }
        return false;
    }

    public void transfer(BankAccount other, double amount) {
        if (withdraw(amount))                          // reuse! ถอนสำเร็จ...
            other.deposit(amount);                     // ...ค่อยฝากให้อีกบัญชี
    }

    public double getBalance() { return balance; }   // property ดึงเงินคงเหลือ
    public String getInfo() {
        return name + " (opened " + openDate.getInfo() + ")";
    }
}
Lab 4 · ข้อ 2 ทดสอบ

ทดสอบตามโจทย์ + ไล่ผลลัพธ์

public class TestBank {
    public static void main(String[] args) {
        BankAccount a1 = new BankAccount("Somchai");                    // เงิน 0, วันนี้
        BankAccount a2 = new BankAccount("Somsri", 500,
                             new Date(1, 11, 2015));                // 500 บาท, 1 พ.ย. 2015

        a1.deposit(1000);        // a1: 0 → 1000
        a2.transfer(a1, 300);    // a2: 500 → 200, a1: 1000 → 1300
        a2.withdraw(300);        // a2 มี 200 < 300 → false ไม่หัก!

        System.out.println(a1.getInfo() + " balance: " + a1.getBalance());
        System.out.println(a2.getInfo() + " balance: " + a2.getBalance());
    }
}
/* ── ไล่ทีละบรรทัด (ตอบ "จงบอกผลลัพธ์") ──
   a1 เปิด 0 บาท → deposit 1000 → 1000
   a2 เปิด 500 → โอนให้ a1 300 → a2=200, a1=1300
   a2 ถอน 300 → มีแค่ 200 → withdraw คืน false เงินไม่เปลี่ยน

   Somchai (opened 19/7/2569) balance: 1300.0
   Somsri (opened 1/11/2558) balance: 200.0
   ← ปีแสดงเป็น พ.ศ. เพราะ getYear() บวก 543! */
จุดหลอกของโจทย์: ถอน 300 จากบัญชีที่เหลือ 200 — ต้องไม่หัก! ถ้าโค้ดใครได้ −100 แสดงว่าลืมเช็คใน withdraw

ข้อ 3.1-3.3: Product + กำหนดสินค้า

public class Product {                             // 3.1 นิยาม Product
    private int code;
    private String name;
    private double price;

    public Product(int code, String name, double price) {
        this.code = code;
        this.name = name;
        this.price = price;
    }
    public int getCode() { return code; }
    public String getName() { return name; }
    public double getPrice() { return price; }
    public void setName(String name) { this.name = name; }
    public void setPrice(double price) { this.price = price; }
}

// ── ใน main ──
Scanner sc = new Scanner(System.in);
Product[] products = new Product[5];               // 3.2 array 5 ช่อง

products[0] = new Product(0,  "Mama", 5.5);      // 3.3 รหัส 0, 10 fix ค่า
products[1] = new Product(10, "Lays", 10.5);

System.out.println("======== Enter Products detail========");
for (int i = 2; i < 5; i++) {                    // รหัส 20,30,40 รับจากคีย์บอร์ด
    int code = i * 10;                             // index → รหัส
    System.out.println("Product code " + code);
    System.out.print("name:");
    String name = sc.next();
    System.out.print("price:");
    double price = sc.nextDouble();
    products[i] = new Product(code, name, price);
}

ข้อ 3.4-3.5: แสดงรายการ + รับซื้อจนใส่ −1

// 3.4 วนแสดงสินค้าทั้งหมด
System.out.println("========== List of Products ===========");
for (Product p : products) {
    System.out.println("Product code " + p.getCode());
    System.out.println("Name:" + p.getName() + " , Price: " + p.getPrice());
}

// 3.5 วนรับซื้อจนกว่าจะใส่ -1 (sentinel loop)
System.out.println("===== What do you want to buy? ======");
System.out.println("Enter product code (press -1 to exit)");

double total = 0;
while (true) {
    System.out.print("Product Code:");
    int code = sc.nextInt();

    if (code == -1) break;                       // เงื่อนไขออกจาก loop

    Product p = products[code / 10];              // รหัส → index ทันที O(1)
    System.out.print("Amount of " + p.getName() + ":");
    int amount = sc.nextInt();

    total += p.getPrice() * amount;              // สะสมราคารวม
}
System.out.println("You have to pay " + total + " Bath");

/* Product Code:20 → Amount of Magnum:2   (40×2 = 80)
   Product Code:0  → Amount of Mama:2     (5.5×2 = 11)
   Product Code:-1
   You have to pay 91.0 Bath  ✓ ตรงโจทย์เป๊ะ */
Pattern "วนจนเจอค่าพิเศษ" (sentinel loop): while(true) + if(x == -1) break — ใช้บ่อยมากในโจทย์รับ input ไม่จำกัดจำนวน
