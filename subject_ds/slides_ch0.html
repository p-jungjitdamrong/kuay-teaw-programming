Array — เก็บข้อมูลชุดเดียวกันต่อกัน
Array คือช่องความจำต่อกันเป็นแถว เข้าถึงด้วย index — ใน Data Structures เราจะใช้ Array เป็นทางเลือกแข่งกับ Linked List เสมอ
Array 1D และ 2D
int score[5] = {80,75,90,60,88}; printf("%d", score[2]); /* 90 */ /* 2D = ตาราง แถว x คอลัมน์ */ int grid[2][3] = {{1,2,3},{4,5,6}}; printf("%d", grid[1][2]); /* 6 */
⚠️ ข้อจำกัดของ Array
ขนาดต้องกำหนดตั้งแต่แรก (fixed size)
แทรก/ลบกลางแถว ต้องขยับสมาชิกทุกตัว → ช้า O(n)

นี่คือเหตุผลที่ Linked List เกิดขึ้น — แทรก/ลบได้เร็วกว่า

Pointer — ตัวแปรที่เก็บ Address
Data Structures ทุกตัวเชื่อมกันด้วย Pointer — โหนดหนึ่งชี้ไปยังโหนดถัดไป ถ้า & และ * ไม่แม่น จะงงตอนเรียน Linked List ทันที
& และ * ทบทวนไว ๆ
int x = 10; int *p = &x; /* p เก็บ address ของ x */ printf("%d", *p); /* 10 — dereference อ่านค่า */ *p = 99; /* แก้ค่า x ผ่าน pointer */ printf("%d", x); /* 99 */

บทที่ 0 · Struct
Struct — รวมหลาย Field ไว้ในชื่อเดียว
Array เก็บข้อมูล "ชนิดเดียวกัน" เท่านั้น แต่ข้อมูลจริงมีหลาย field ต่าง type — Struct คือคำตอบ และเป็นหัวใจของทุก Data Structure
นิยาม Struct
struct Student { char name[30]; int age; float gpa; }; struct Student s1; strcpy(s1.name, "Som"); s1.age = 20; s1.gpa = 3.5; printf("%s age %d", s1.name, s1.age);
. กับ -> ต่างกันยังไง
ตัวแปร struct ปกติ → ใช้ .
s1.age

ตัวแปรเป็น pointer ไปที่ struct → ใช้ ->
struct Student *p = &s1;
p->age เท่ากับ (*p).age
เข้าถึงผ่าน Pointer — ใช้บ่อยใน Data Structure
struct Student s1 = {"Som", 20, 3.5}; struct Student *p = &s1; printf("%s", p->name); /* Som */ p->age = 21; /* แก้ผ่าน pointer */ printf("%d", s1.age); /* 21 */
💡 ทำไมบทนี้สำคัญ
Linked List node = struct ที่มี field "ค่า" + field "pointer ชี้โหนดถัดไป" — ถ้า struct + pointer ไม่แม่น จะอ่าน code Linked List ไม่ออกเลย
บทที่ 0 · Dynamic Memory
malloc & free — ขอ/คืน Memory ตอนรันโปรแกรม
Array ต้องกำหนดขนาดตั้งแต่ compile แต่ Data Structures ต้องขยายขนาดได้ตอนรันจริง — malloc คือคำตอบ
malloc สร้าง 1 ช่อง
#include <stdlib.h> int *p = (int *)malloc(sizeof(int)); if (p == NULL) { /* memory เต็ม */ printf("malloc failed"); return 1; } *p = 42; printf("%d", *p); /* 42 */ free(p); /* คืน memory ทุกครั้งที่เลิกใช้ */ p = NULL; /* ป้องกัน dangling pointer */
malloc สร้าง struct — Pattern ที่ใช้ตลอดวิชานี้
struct Student *p = (struct Student *)malloc(sizeof(struct Student)); p->age = 20; strcpy(p->name, "Som"); /* ใช้งานเสร็จแล้ว ต้อง free เสมอ */ free(p);
❌ Memory Leak
malloc แล้วไม่ free = memory leak
สร้าง node ของ Linked List/Tree ทุกครั้งต้องคิดคู่กับการ free ตอนลบ — มิฉะนั้นโปรแกรมกิน memory เพิ่มเรื่อยๆ จนพัง
บทที่ 0 · ภาพรวม
Struct + Pointer + malloc = "Node"
ทุก Data Structure ที่จะเรียนต่อไป (Linked List, Tree, Graph) สร้างจาก pattern เดียวกันนี้ — จำ pattern นี้ให้ขึ้นใจ
นิยาม Node ของ Linked List (พรีวิวบทที่ 1)
struct Node { int data; /* เก็บค่า */ struct Node *next; /* ชี้โหนดถัดไป */ }; struct Node *n1 = (struct Node *)malloc(sizeof(struct Node)); n1->data = 10; n1->next = NULL; /* ยังไม่มีโหนดถัดไป */
💡 สังเกต pattern
struct → ห่อค่า + pointer ไว้ด้วยกัน
malloc → ขอที่เก็บ node ตอนรันจริง (ไม่รู้ล่วงหน้าว่าจะมีกี่โหนด)
pointer (next) → เชื่อมโหนดนี้กับโหนดถัดไป

สามอย่างนี้รวมกันแก้ข้อจำกัดของ Array ได้ทั้งหมด — ขนาดไม่ fixed และแทรก/ลบไม่ต้องขยับข้อมูล
บทที่ 0 · Use Case
🏦 Use Case จริง — บันทึกธุรกรรมลูกค้า
ธนาคารต้องเก็บประวัติธุรกรรมที่ "ไม่รู้ล่วงหน้าว่าจะมีกี่รายการ" — Array fixed size ใช้ไม่ได้ ต้องใช้ struct + malloc
โจทย์ระบบธนาคาร
สร้างรายการธุรกรรมทีละรายการ
ลูกค้าฝาก/ถอนกี่ครั้งก็ได้ในหนึ่งวัน — โปรแกรมต้องขอ memory ใหม่ทุกครั้งที่มีธุรกรรมเกิดขึ้น ไม่ใช่จองไว้ล่วงหน้าแบบ Array
ทำไมไม่ใช้ Array
Array ต้องบอกขนาดล่วงหน้า เช่น txn[100]
ถ้าลูกค้าทำธุรกรรมเกิน 100 ครั้ง → array ล้น
ถ้าทำแค่ 2 ครั้ง → เปลือง memory 98 ช่อง
สร้าง Transaction ด้วย struct + malloc
struct Transaction { char type[10]; /* "deposit"/"withdraw" */ double amount; }; struct Transaction *newTxn(char *t, double amt) { struct Transaction *tx = (struct Transaction *) malloc(sizeof(struct Transaction)); strcpy(tx->type, t); tx->amount = amt; return tx; } struct Transaction *t1 = newTxn("deposit", 500.0); printf("%s %.2f", t1->type, t1->amount); free(t1);
บทที่ 0 · Think!
คิดก่อน เขียน
ถ้าตอบข้อเหล่านี้ได้ครบ แปลว่าพร้อมเรียน Linked List แล้ว
1. struct Node *p; p->data = 5; ทำได้ไหมถ้า p ยังไม่ malloc?
💡 ไม่ได้ — p ยังไม่ชี้ที่ไหน (ค่าสุ่ม) dereference จะ crash ต้อง malloc ก่อนเสมอ
2. ทำไม Linked List ใช้ struct + pointer แทน Array?
💡 เพราะไม่รู้ล่วงหน้าว่าจะมีกี่โหนด และต้องแทรก/ลบกลางลิสต์ได้เร็วโดยไม่ขยับข้อมูล
3. p->next เท่ากับนิพจน์ไหนถ้าเขียนแบบไม่ใช้ ->?
💡 (*p).next — -> คือ shortcut ของ (*p).field
4. ถ้า malloc node แล้วลืม free ทุกครั้งที่ลบโหนด จะเกิดอะไร?
💡 Memory leak — โปรแกรมที่รันนาน ๆ จะกิน memory เพิ่มขึ้นเรื่อย ๆ จนเครื่องช้าหรือพัง
5. struct Node *head = NULL; หมายความว่าอะไร?
💡 ลิสต์ยังว่างเปล่า — ยังไม่มีโหนดแรก นี่คือจุดเริ่มต้นของ Linked List ทุกอัน
