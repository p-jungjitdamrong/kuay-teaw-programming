Binary Tree — โครงสร้างแบบไม่เชิงเส้น
Linked List/Stack/Queue เดินทางเดียวเป็นเส้นตรง แต่ Tree แตกกิ่งได้ — แต่ละโหนดมีลูกได้สูงสุด 2 ตัว เรียก left และ right

struct TreeNode { int data; struct TreeNode *left; struct TreeNode *right; }; struct TreeNode *root = NULL; /* จุดเริ่มต้นของต้นไม้ */

บทที่ 5 · Traversal
Inorder — left → node → right
Traversal คือการเดินเยี่ยมทุกโหนดในต้นไม้ — Inorder ของ BST จะได้ค่าเรียงจากน้อยไปมากเสมอ ใช้ recursion เป็นหัวใจหลัก
inorder — recursive
void inorder(struct TreeNode *node) { if (node == NULL) return; /* base case */ inorder(node->left); /* 1. ซ้ายก่อน */ printf("%d ", node->data); /* 2. ตัวเอง */ inorder(node->right); /* 3. ขวาสุดท้าย */ } /* tree: 50 มีลูก 30,70 ; 30 มีลูก 20,40 ; 70 มีลูก 80 */ inorder(root); // 20 30 40 50 70 80 — เรียงน้อยไปมาก!
💡 ทำไม Inorder ของ BST เรียงเสมอ
เพราะ BST การันตีว่า left < node < right ทุกโหนด — เดิน left-node-right จึงได้ลำดับน้อยไปมากโดยอัตโนมัติ
⚠️ Recursion คือหัวใจของ Tree
ทุก function ของ Tree เกือบทั้งหมดเขียนแบบ recursive:
1. เช็ค base case (node == NULL) ก่อนเสมอ
2. เรียกตัวเองกับ left/right
3. ทำงานกับ node ปัจจุบัน
❌ ลืม base case = Stack Overflow
ถ้าไม่เช็ค node == NULL ก่อน function จะเรียกตัวเองไม่จบ (infinite recursion) จนโปรแกรม crash
บทที่ 5 · Traversal
Preorder & Postorder — ลำดับอื่นที่ใช้บ่อย
ต่างกันแค่ "เมื่อไหร่จะพิมพ์ตัวเอง" เทียบกับการเดิน left/right — จำง่ายจากชื่อ: Pre=ก่อนลูก, Post=หลังลูก
preorder — node → left → right
void preorder(struct TreeNode *node) { if (node == NULL) return; printf("%d ", node->data); /* 1. ตัวเองก่อน */ preorder(node->left); preorder(node->right); } // tree เดิม: 50 30 20 40 70 80
ใช้ทำอะไร
Preorder ใช้ copy ต้นไม้ทั้งใบ (สร้างโครงสร้างใหม่ตามลำดับเดิม) เพราะพิมพ์ root ก่อนลูกเสมอ
postorder — left → right → node
void postorder(struct TreeNode *node) { if (node == NULL) return; postorder(node->left); postorder(node->right); printf("%d ", node->data); /* 3. ตัวเองสุดท้าย */ } // tree เดิม: 20 40 30 80 70 50
ใช้ทำอะไร
Postorder ใช้ตอนลบต้นไม้ทั้งใบ (free ทุกโหนด) เพราะต้อง free ลูกก่อนแม่เสมอ ไม่งั้นจะเข้าถึงลูกไม่ได้
บทที่ 5 · BST
Binary Search Tree — Tree ที่ค้นหาเร็ว
กฎทอง BST: ทุกโหนด — ค่าทาง left ต้องน้อยกว่า, ค่าทาง right ต้องมากกว่า กฎนี้ทำให้ search/insert เร็วกว่า Linked List มาก
insert — เปรียบเทียบแล้วเลือกทาง
struct TreeNode *insert(struct TreeNode *node, int val) { if (node == NULL) { /* เจอที่ว่าง สร้างโหนดใหม่ */ struct TreeNode *n = (struct TreeNode *)malloc(sizeof(struct TreeNode)); n->data = val; n->left = n->right = NULL; return n; } if (val < node->data) node->left = insert(node->left, val); else node->right = insert(node->right, val); return node; } /* เรียกใช้: root = insert(root, 50); */
insert(50,30,70,20,40,80) ทีละค่า
insert(50) → root ว่าง → 50 เป็น root
insert(30) → 30<50 → ไปทาง left ของ 50
insert(70) → 70>50 → ไปทาง right ของ 50
insert(20) → 20<50→left, 20<30→left ของ 30
insert(40) → 40<50→left, 40>30→right ของ 30
insert(80) → 80>50→right, 80>70→right ของ 70
💡 BST = Binary Search ในรูป Tree
เทียบกับ Binary Search ใน Array — แต่ละจุดเทียบแล้วตัดครึ่งซ้าย/ขวา ทำให้ search เร็ว O(log n) ถ้า tree สมดุล
บทที่ 5 · BST
Search ใน BST — เร็วกว่า Linked List
ใช้กฎเดียวกับ insert — เทียบค่าแล้วเลือกว่าจะไปทาง left หรือ right ไม่ต้องเช็คทุกโหนดแบบ Linked List
search — recursive
struct TreeNode *search(struct TreeNode *node, int target) { if (node == NULL || node->data == target) return node; /* เจอ หรือ ไม่เจอ (NULL) */ if (target < node->data) return search(node->left, target); else return search(node->right, target); } /* ใช้งาน */ struct TreeNode *found = search(root, 40); if (found != NULL) printf("เจอ %d\n", found->data);
Linked ListBST (สมดุล)
SearchO(n) เช็คทุกโหนดO(log n) ตัดครึ่งทุกครั้ง
InsertO(1) ที่หัวO(log n) ต้องหาที่วาง
⚠️ "สมดุล" คือเงื่อนไขสำคัญ
ถ้า insert ค่าเรียงๆ ติดกัน (เช่น 1,2,3,4,5) BST จะเอียงเป็นเส้นตรงเหมือน Linked List ทำให้ search กลับมาเป็น O(n) — ปัญหานี้แก้ด้วย Balanced Tree (AVL, Red-Black) ซึ่งเป็นเรื่องขั้นสูงกว่านี้
บทที่ 5 · Use Case
🏦 Use Case จริง — ค้นหาบัญชีลูกค้าด้วยเลขบัญชี
ธนาคารมีลูกค้าเป็นล้านคน ค้นหาด้วย Linked List (O(n)) จะช้ามาก — เก็บเป็น BST ตามเลขบัญชี ทำให้ search เร็วขึ้นมาก
โจทย์ระบบธนาคาร
ค้นหาบัญชีด้วย BST ตามเลขบัญชี
เก็บบัญชีลูกค้าเป็น BST โดยใช้เลขบัญชีเป็นเกณฑ์เปรียบเทียบ — ค้นหาบัญชีจากล้านรายการได้เร็วกว่าไล่ทีละตัว
ทำไมไม่ใช้ Linked List
ลูกค้า 1 ล้านคน, Linked List ต้องไล่เฉลี่ย 500,000 ครั้งถึงจะเจอ แต่ BST ที่สมดุลใช้แค่ประมาณ 20 ครั้ง (log₂ 1,000,000 ≈ 20)
ระบบค้นหาบัญชีด้วย BST
struct Account { int accNo; double balance; struct Account *left, *right; }; struct Account *root = NULL; struct Account *findAccount(struct Account *node, int accNo) { if (node == NULL || node->accNo == accNo) return node; if (accNo < node->accNo) return findAccount(node->left, accNo); return findAccount(node->right, accNo); } struct Account *acc = findAccount(root, 10025); if (acc) printf("ยอดเงิน: %.2f\n", acc->balance);
บทที่ 5 · Think!
คิดก่อน เขียน
Tree ต้องคิดเป็น recursion — ลองนึกภาพต้นไม้ในหัวเวลา trace
1. tree: root=50, left=30, right=70 — inorder จะพิมพ์อะไร?
💡 left ก่อน (30), node (50), right (70) → "30 50 70"
2. ทำไม inorder ของ BST ได้ค่าเรียงน้อยไปมากเสมอ แต่ของ Binary Tree ทั่วไปไม่ใช่?
💡 เพราะ BST การันตี left<node<right ทุกโหนด ส่วน Binary Tree ทั่วไปไม่มีกฎนี้ ลำดับค่าจึงไม่แน่นอน
3. insert(60) เข้าไปใน tree (50,30,70,20,40,80) จะไปอยู่ตำแหน่งไหน?
💡 60>50→right(70), 60<70→left ของ 70 — กลายเป็นลูกซ้ายของโหนด 70
4. ทำไมต้อง free ลูกก่อนแม่ (postorder) ตอนลบต้นไม้ทั้งใบ ทำสลับลำดับได้ไหม?
💡 ไม่ได้ — ถ้า free แม่ก่อน จะไม่มี pointer ไปหาลูกอีกแล้ว (memory leak) ต้อง free ลูกให้ครบก่อนเสมอ
5. ถ้า insert ค่า 10,20,30,40,50 เรียงกันลงไป tree จะมีรูปร่างหน้าตายังไง?
💡 จะเอียงไปทาง right ทั้งหมด กลายเป็นเส้นตรงเหมือน Linked List — search กลับไปเป็น O(n) ไม่เร็วเหมือน BST สมดุล
