บทที่ 1 · พื้นฐาน
Node และ head pointer
เริ่มจากนิยาม struct Node แล้วประกาศ head เป็นจุดเริ่มต้นของลิสต์ — ทั้งลิสต์เข้าถึงได้จาก head ตัวเดียว
นิยาม struct Node
struct Node { int data; struct Node *next; }; /* head ชี้โหนดแรกของลิสต์ */ struct Node *head = NULL; /* เริ่มต้นลิสต์ว่าง */
head ไม่ใช่ "ลิสต์" head คือ pointer ตัวแรก
head เก็บ address ของโหนดแรกเท่านั้น ตัวลิสต์จริงๆ คือโหนดที่เชื่อมกันด้วย next — ถ้าทำ head หาย จะเข้าถึงลิสต์ไม่ได้เลย (memory leak ทั้งลิสต์)
สร้างโหนดแรกด้วยมือ (ก่อนเขียน function)
struct Node *n1 = (struct Node *)malloc(sizeof(struct Node)); n1->data = 10; n1->next = NULL; /* ยังไม่มีโหนดถัดไป */ head = n1; /* head ชี้โหนดแรก */ printf("%d", head->data); /* 10 */
⚠️ ลำดับสำคัญ
ต้อง malloc ก่อนใส่ค่า n1->data เสมอ — ถ้า n1 ยังไม่ malloc แล้ว dereference จะ crash
บทที่ 1 · Traverse
เดินลิสต์ทั้งสาย — Traverse
ใช้ pointer ตัวช่วย (เช่น cur) เดินจาก head ไปทีละโหนดจนชน NULL — pattern นี้ใช้ซ้ำในทุก operation ของ Linked List
Pattern Traverse — ท่องจำให้ขึ้นใจ
void printList(struct Node *head) { struct Node *cur = head; /* ตัวช่วยเดิน ไม่แก้ head */ while (cur != NULL) { printf("%d -> ", cur->data); cur = cur->next; /* เลื่อนไปโหนดถัดไป */ } printf("NULL\n"); }
💡 ทำไมต้องใช้ cur ไม่ใช้ head ตรงๆ
ถ้าเดินด้วย head = head->next โดยตรง จะทำ head หายไปเรื่อยๆ จนสุดท้ายเข้าถึงลิสต์ไม่ได้อีก ต้อง copy ไปไว้ในตัวแปรใหม่ (cur) เสมอ

บทที่ 1 · Insert
แทรกที่หัวลิสต์ — Insert at Head
แทรกหัวลิสต์เร็วที่สุด O(1) — สร้างโหนดใหม่ ให้ชี้ไปที่โหนดเดิม แล้วขยับ head มาที่โหนดใหม่
insertHead — 3 ขั้นตอนคงที่
void insertHead(struct Node **head, int val) { struct Node *n = (struct Node *)malloc(sizeof(struct Node)); n->data = val; /* 1. ใส่ค่า */ n->next = *head; /* 2. ชี้ไปที่หัวเดิม */ *head = n; /* 3. หัวใหม่คือ n */ } /* เรียกใช้ */ insertHead(&head, 5);
⚠️ ทำไมต้องใช้ struct Node **head
เราต้องการ "แก้ค่า head ของตัวเรียก" ไม่ใช่แค่ copy — ต้องส่ง pointer ไปที่ head (pointer to pointer) เหมือนหลักการ pass by reference ในบทที่ 0

บทที่ 1 · Insert
แทรกท้ายลิสต์และกลางลิสต์
แทรกท้าย/กลาง ต้อง traverse หาตำแหน่งก่อนเสมอ — O(n) เพราะต้องเดินไปหาโหนดก่อนจะเชื่อม pointer
insertTail — เดินจนสุดแล้วต่อท้าย
void insertTail(struct Node **head, int val) { struct Node *n = (struct Node *)malloc(sizeof(struct Node)); n->data = val; n->next = NULL; if (*head == NULL) { *head = n; return; } struct Node *cur = *head; while (cur->next != NULL) /* เดินจนถึงโหนดสุดท้าย */ cur = cur->next; cur->next = n; /* ต่อท้าย */ }
insertAfter — แทรกหลังโหนดที่มีค่า target
void insertAfter(struct Node *head, int target, int val) { struct Node *cur = head; while (cur != NULL && cur->data != target) cur = cur->next; if (cur == NULL) return; /* ไม่เจอ target */ struct Node *n = (struct Node *)malloc(sizeof(struct Node)); n->data = val; n->next = cur->next; /* 1. ชี้ของใหม่ไปที่ตัวถัดไปเดิม */ cur->next = n; /* 2. แล้วค่อยให้ cur ชี้มาที่ใหม่ */ }
❌ สลับลำดับ 2 ขั้นตอนนี้ไม่ได้
ถ้าทำ cur->next = n ก่อน จะ"หลุด"ไม่รู้ว่าตัวถัดไปเดิมคือใคร เพราะ cur->next ถูกเขียนทับไปแล้ว ต้อง n->next = cur->next ก่อนเสมอ
บทที่ 1 · Delete
ลบโหนด — Delete + free
ลบโหนดต้องหาโหนด "ก่อนหน้า" ก่อน เพื่อข้ามโหนดที่จะลบ แล้ว free โหนดนั้นทิ้งเสมอ ไม่งั้น memory leak
deleteValue — ลบโหนดแรกที่มีค่า target
void deleteValue(struct Node **head, int target) { struct Node *cur = *head, *prev = NULL; while (cur != NULL && cur->data != target) { prev = cur; cur = cur->next; } if (cur == NULL) return; /* ไม่เจอ */ if (prev == NULL) /* ลบโหนดแรก */ *head = cur->next; else /* ลบกลาง/ท้าย */ prev->next = cur->next; free(cur); /* คืน memory เสมอ */ }

ถ้า prev เป็น NULL แปลว่ากำลังลบโหนดแรก ต้องขยับ *head ไปที่ cur->next เอง เพราะไม่มี prev ให้ข้าม
บทที่ 1 · Operations อื่นๆ
Search และ Count — operation ที่ใช้บ่อย
ทั้งสอง function ใช้ pattern traverse เดิม — ต่างกันแค่เงื่อนไขที่ทำในแต่ละโหนด
search — หาว่ามีค่านี้ในลิสต์ไหม
int search(struct Node *head, int target) { struct Node *cur = head; int pos = 0; while (cur != NULL) { if (cur->data == target) return pos; cur = cur->next; pos++; } return -1; /* ไม่เจอ */ }
countNodes — นับจำนวนโหนดทั้งหมด
int countNodes(struct Node *head) { int count = 0; struct Node *cur = head; while (cur != NULL) { count++; cur = cur->next; } return count; }
💡 Pattern ทุก function ของ Linked List
เกือบทุก function = "เริ่มที่ cur = head" + "while (cur != NULL)" + "ทำอะไรบางอย่าง" + "cur = cur->next" — จำ pattern นี้แล้วเขียน Linked List ได้ทุกแบบ
บทที่ 1 · Use Case
🏦 Use Case จริง — ประวัติธุรกรรมลูกค้า
ลูกค้าฝาก/ถอนกี่ครั้งก็ได้ — แต่ละธุรกรรมคือโหนดใหม่ที่ต่อเข้าไปในลิสต์ ไม่ต้องรู้จำนวนล่วงหน้า
โจทย์ระบบธนาคาร
บันทึกประวัติธุรกรรมแบบ Linked List
ทุกครั้งที่ลูกค้าฝาก/ถอน ระบบสร้างโหนดใหม่ต่อท้ายประวัติ — แสดงรายการทั้งหมดด้วย traverse
ทำไมใช้ Linked List ไม่ใช่ Array
ไม่รู้ว่าลูกค้าจะทำธุรกรรมกี่ครั้งในชีวิต — Array ต้องเดากำหนดขนาด แต่ Linked List โตได้ไม่จำกัด (จนกว่า memory เต็ม)
ระบบบันทึกธุรกรรมแบบ Linked List
struct Txn { char type[10]; double amount; struct Txn *next; }; struct Txn *history = NULL; void addTxn(char *t, double amt) { struct Txn *n = (struct Txn *)malloc(sizeof(struct Txn)); strcpy(n->type, t); n->amount = amt; n->next = history; /* insert head — ล่าสุดอยู่บนสุด */ history = n; } void printHistory() { struct Txn *cur = history; while (cur != NULL) { printf("%s %.2f\n", cur->type, cur->amount); cur = cur->next; } }
บทที่ 1 · Think!
คิดก่อน เขียน
Linked List ต้อง trace ทีละ pointer ในหัว — ฝึกตรงนี้ให้คล่อง
1. ทำไม insertHead ใช้ struct Node **head แต่ printList ใช้แค่ struct Node *head?
💡 insertHead ต้องแก้ค่า head จริงของผู้เรียก ส่วน printList แค่อ่านอย่างเดียวไม่ต้องแก้
2. ถ้าลืม n->next = NULL; ตอนสร้างโหนดสุดท้าย จะเกิดอะไร?
💡 next จะเป็นค่าสุ่ม (garbage) — traverse จะวิ่งต่อไปเรื่อยๆ จนเข้าถึง memory ที่ไม่ใช่ของเรา → crash
3. delete โหนดแล้วไม่ free จะเกิดผลอะไรในระยะยาว?
💡 Memory leak — โหนดยังอยู่ใน memory แต่ไม่มีใครเข้าถึงได้ ลบไปเรื่อยๆ memory จะเต็ม
4. head→10→20→30→NULL ทำ head = head->next สองครั้ง head จะชี้โหนดไหน?
💡 ครั้งแรกชี้ 20 ครั้งที่สองชี้ 30 — แต่ระวัง โหนด 10 จะ "หลุด" ไม่มีใครเข้าถึงได้แล้ว (ไม่ได้ free)
5. insertAfter ทำไมต้องเขียน n->next = cur->next ก่อน cur->next = n?
💡 ถ้าสลับลำดับ cur->next จะถูกเขียนทับเป็น n ก่อน ทำให้ไม่รู้ว่าโหนดถัดไปเดิมคือใครอีกแล้ว

บทที่ 1 · สรุป
เก่ง Linked List แล้ว!
สร้าง traverse insert delete ได้ครบ — ขั้นต่อไปคือ Linked List แบบสองทิศทาง
1
Node + head — struct เก็บค่า, head ชี้โหนดแรกลิสต์ว่าง = head เป็น NULL
2
Traverse — cur = head, while(cur!=NULL), cur=cur->nextpattern หลักของทุก operation
3
Insert — head O(1), tail/middle O(n)ต้องระวังลำดับการเชื่อม pointer
4
Delete — หา prev ก่อน ข้าม cur แล้ว freeลืม free = memory leak
