บทที่ 2 · พื้นฐาน
นิยาม Node — เพิ่ม field prev
ต่างจากบทที่ 1 แค่เพิ่ม pointer หนึ่งตัว — ทุก operation ต้องดูแล 2 pointer พร้อมกันเสมอ
นิยาม struct DNode
struct DNode { int data; struct DNode *prev; /* ชี้โหนดก่อนหน้า */ struct DNode *next; /* ชี้โหนดถัดไป */ }; struct DNode *head = NULL; struct DNode *tail = NULL; /* เก็บ tail ไว้เดินถอยหลังเร็ว */
⚠️ ทำไมต้องเก็บ tail ด้วย
ถ้าไม่เก็บ tail ต้อง traverse จาก head ไปจนสุดทุกครั้งที่จะ insertTail หรือเดินถอยหลังจากท้ายลิสต์ — เก็บ tail ไว้ทำให้ insertTail เป็น O(1)
สร้างโหนดแรกด้วยมือ
struct DNode *n1 = (struct DNode *)malloc(sizeof(struct DNode)); n1->data = 10; n1->prev = NULL; /* โหนดแรก ไม่มีก่อนหน้า */ n1->next = NULL; /* ยังไม่มีโหนดถัดไป */ head = n1; tail = n1; /* มีโหนดเดียว head กับ tail คือตัวเดียวกัน */
บทที่ 2 · Insert
Insert at Tail — O(1) เพราะมี tail
มี tail แล้ว ไม่ต้อง traverse — ต่อโหนดใหม่ที่ tail ตรงๆ แล้วขยับ tail มาที่โหนดใหม่
insertTail — เชื่อม 2 ทิศทางให้ครบ
void insertTail(int val) { struct DNode *n = (struct DNode *)malloc(sizeof(struct DNode)); n->data = val; n->next = NULL; if (tail == NULL) { /* ลิสต์ว่าง */ n->prev = NULL; head = tail = n; return; } n->prev = tail; /* 1. ใหม่ชี้กลับไปที่ tail เดิม */ tail->next = n; /* 2. tail เดิมชี้ไปข้างหน้าที่ใหม่ */ tail = n; /* 3. ขยับ tail */ }

บทที่ 2 · Traverse & Delete
เดินถอยหลัง และ Delete
เดินถอยหลังเริ่มจาก tail ใช้ prev แทน next — ส่วน Delete สะดวกขึ้นมาก เพราะรู้ prev ของโหนดทันทีไม่ต้องเดินหา
printReverse — เดินจาก tail ถอยไป head
void printReverse() { struct DNode *cur = tail; while (cur != NULL) { printf("%d ", cur->data); cur = cur->prev; /* ใช้ prev แทน next */ } }
💡 สมมาตรกับ Singly List
เดินหน้า: cur = head, ใช้ next
เดินถอยหลัง: cur = tail, ใช้ prev
โครงสร้างเหมือนกันทุกอย่าง สลับแค่ทิศทาง
deleteNode — ไม่ต้องเดินหา prev เอง
void deleteNode(struct DNode *n) { if (n->prev != NULL) n->prev->next = n->next; else head = n->next; /* n คือ head */ if (n->next != NULL) n->next->prev = n->prev; else tail = n->prev; /* n คือ tail */ free(n); }
บทที่ 2 · Circular List
Circular Linked List — ลิสต์ที่ไม่มีจุดสิ้นสุด
โหนดสุดท้ายไม่ชี้ NULL แต่ชี้กลับไปที่โหนดแรก — เดินวนได้ไม่จบ ใช้ทำ playlist เพลงที่เล่นวนซ้ำ หรือ round-robin scheduling
insertCircular — ต่อท้ายแบบวงกลม
struct Node *last = NULL; /* เก็บโหนดสุดท้าย */ void insertCircular(int val) { struct Node *n = (struct Node *)malloc(sizeof(struct Node)); n->data = val; if (last == NULL) { n->next = n; /* โหนดแรก ชี้ตัวเอง */ last = n; return; } n->next = last->next; /* ชี้ไปที่หัวเดิม (last->next) */ last->next = n; last = n; }

❌ Infinite Loop ถ้าเช็คเงื่อนไขผิด
while (cur != NULL) จะวนไม่จบเพราะไม่มี NULL เลย ใน Circular List ต้องเช็คแบบ:
do { ... cur = cur->next; } while (cur != head);
บทที่ 2 · Use Case
🏦 Use Case จริง — Undo ธุรกรรมล่าสุด
ระบบธนาคารต้อง undo รายการล่าสุดได้ (ยกเลิกการโอนผิด) — Doubly List ทำให้เดินถอยหลังไปดูรายการก่อนหน้าได้ทันที
โจทย์ระบบธนาคาร
เลื่อนดูประวัติย้อนกลับและไปข้างหน้า
พนักงานธนาคารเลื่อนดูรายการธุรกรรมไปข้างหน้า-ถอยหลังได้ เหมือนปุ่ม back/forward ในเบราว์เซอร์
ทำไมต้องเป็น Doubly
Singly List เดินถอยหลังไม่ได้โดยตรง ต้องวน loop จาก head ใหม่ทุกครั้ง — ช้าและไม่สะดวก ส่วน Doubly เดินถอยหลังได้ทันทีด้วย prev
Cursor เลื่อนดูประวัติ
struct DNode *cursor = tail; /* เริ่มที่รายการล่าสุด */ void viewPrevious() { if (cursor->prev != NULL) { cursor = cursor->prev; printf("ดูรายการ: %d\n", cursor->data); } else { printf("ไม่มีรายการก่อนหน้า\n"); } } void viewNext() { if (cursor->next != NULL) { cursor = cursor->next; printf("ดูรายการ: %d\n", cursor->data); } }
บทที่ 2 · Think!
คิดก่อน เขียน
Doubly List ต้องดูแล 2 pointer พร้อมกันเสมอ — ฝึก trace ให้คล่อง
1. ทำไม insertTail ของ Doubly List เป็น O(1) แต่ของ Singly List (ไม่เก็บ tail) เป็น O(n)?
💡 Doubly List เก็บ tail pointer ไว้เลย ไม่ต้อง traverse หาโหนดสุดท้าย ส่วน Singly ต้องเดินจาก head จนสุดทุกครั้ง
2. ลบโหนดที่เป็น head ของ Doubly List ต้องอัปเดตอะไรเป็นพิเศษ?
💡 ต้องเช็คว่า n->prev == NULL แล้วขยับ head = n->next และตั้ง head ใหม่ prev = NULL
3. Circular List เช็คจบลิสต์ด้วย NULL ได้ไหม?
💡 ไม่ได้ — ไม่มี NULL เลยในวงกลม ต้องเช็คว่า cur กลับมาที่ head/last หรือยัง
4. ถ้า insertTail แล้วลืม n->prev = tail จะส่งผลยังไงตอนเดินถอยหลัง?
💡 n->prev จะเป็นค่าสุ่ม — printReverse จะ crash หรือพิมพ์ค่าผิด เพราะ dereference pointer ที่ไม่ได้ตั้งค่า
5. ทำไมต้องเก็บตัวแปร last ใน Circular List แทนการเก็บ head?
💡 last->next คือ head เสมอ — เก็บ last ทำให้ insertTail แบบวงกลมเป็น O(1) ได้เหมือน Doubly List
