ต้นไม้ทวิภาคที่สร้างด้วยอาเรย์

โจทย์: จากส่วน main และการประกาศตัวแปรภายนอกที่กำหนดให้ จงเติมฟังก์ชันที่เกี่ยวข้องกับการจัดการต้นไม้ทวิภาคที่สร้างด้วยอาเรย์ ทั้ง 7 ตัว: isLeft · isRight · LeftNode · RightNode · MotherNode · SetLeft · SetRight แล้วแสดงผลลัพธ์ทางจอภาพของโปรแกรมนี้
C · ส่วนที่โจทย์ให้มา 
#include <stdio.h>
#include <stdbool.h>
#define MAXTREE  15

char T[MAXTREE] = {'C'};   // ราก = 'C' ช่องอื่นเป็น '\0'

int main() {
   SetLeft(0,'P');   SetRight(0,'E');
   SetLeft(1,'A');   SetRight(1,'T');
   SetLeft(2,'K');   SetRight(2,'P');
   SetLeft(3,'S');
   SetRight(5,'X');
   SetLeft(6,'Y');

   printf("%d\n",MotherNode(1));
   printf("%d\n",MotherNode(2));
   printf("%d\n",MotherNode(3));
   printf("%c is mother node of %c\n",T[MotherNode(5)],T[5]);
   printf("%c is left node of %c\n",T[LeftNode(2)],T[2]);
   printf("%c is right node of %c\n",T[RightNode(5)],T[5]);

   printf("%d\n",isLeft(5));
   printf("%d\n",isRight(4));

  for(int i=0;i<MAXTREE;i++)
      if(T[i]=='\0')  printf(" ");
      else          printf("%c",T[i]);
  return 0;
}
⚠️ จุดที่ต้องสังเกตก่อนเติมโค้ด
① T[MAXTREE] = {'C'} — ภาษา C จะเติม '\0' ให้ช่องที่เหลืออัตโนมัติ ดังนั้น "ช่องว่าง" = '\0'
② โจทย์เรียก SetRight(5,'X') — ลูกขวาของ 5 คือช่อง 12 ซึ่งยังอยู่ในอาเรย์ (< 15) จึงใส่ได้
③ ฟังก์ชันคืนค่าเป็น bool ได้เพราะ include stdbool.h มาแล้ว
โครงต้นไม้ที่ main สร้างขึ้น
C[0] ╱ ╲ P[1] E[2] ╱ ╲ ╱ ╲ A[3] T[4] K[5] P[6] ╱ ╲ ╱ S[7] X[12] Y[13]
⏸️ ลองเองก่อน — ปิดสไลด์ถัดไป แล้วเขียนทั้ง 7 ฟังก์ชันลงกระดาษ จับเวลา 15 นาที · ทำได้กี่ตัวก็ได้ แล้วค่อยเปิดเฉลยทีละตัว

🔑 ทำไมชนิดข้อมูลเป็น int ทั้งหมด? เพราะทั้งสามตัวคืนค่าเป็นตำแหน่ง (index) ในอาเรย์ ไม่ใช่ตัวอักษรในโหนด — ถ้าตอบ char จะผิดทันที เพราะ main เอาค่าที่คืนไปใช้เป็น T[...]
⚠️ กับดักของ MotherNode — ต้องเป็น (k−1)/2 ไม่ใช่ k/2
ลองแทน k = 5: ถูกคือ (5−1)/2 = 2 · ถ้าใช้ k/2 = 2 บังเอิญถูก
แต่แทน k = 6: ถูกคือ (6−1)/2 = 2 · ถ้าใช้ k/2 = 3 ผิด → ต้องใช้ (k−1)/2 เสมอ
ตรวจด้วยมือจากต้นไม้ในโจทย์
เรียกคำนวณได้
LeftNode(2)2(2)+15 → T[5] = 'K'
RightNode(5)2(5)+212 → T[12] = 'X'
MotherNode(1)(1−1)/20
MotherNode(2)(2−1)/20 (1/2 ปัดลง)
MotherNode(3)(3−1)/21
💡 int หาร int ปัดทิ้งเศษ — จุดนี้ทำให้ (2−1)/2 ได้ 0 ซึ่งถูกต้อง เพราะโหนด 1 และ 2 มีแม่เป็นโหนด 0 เหมือนกัน

⚠️ เงื่อนไข k > 0 ที่คนลืมกันทั้งห้อง
ถ้าไม่ใส่ แล้วเรียก isRight(0) จะได้ true — แต่โหนด 0 คือราก ไม่ใช่ลูกของใคร ทั้งซ้ายและขวา จึงต้องได้ false ทั้งคู่
ตรวจด้วยมือ
เรียกเหตุผลได้

เฉลย 
void SetLeft(int k, char data)
{
    int index;
    index = 2 * k + 1;            // ตำแหน่งลูกซ้าย

    if(index < MAXTREE)          // ยังอยู่ในอาเรย์ไหม
    {
       if(T[index] == '\0')        // ช่องนั้นว่างไหม
           T[index] = data;
       else
         printf("Node Inused\n");
    }
    else  printf("Tree Overflow\n");
}

void SetRight(int k, char data)
{
    int index;
    index = 2 * k + 2;            // เปลี่ยนแค่บรรทัดนี้

    if(index < MAXTREE)
    {
       if(T[index] == '\0')
           T[index] = data;
       else
         printf("Node Inused\n");
    }
    else  printf("Tree Overflow\n");
}
🔑 โครง if ซ้อนที่โจทย์ให้มา บอกคำตอบอยู่แล้ว — มี 2 ชั้นและมีข้อความ error 2 แบบ จึงต้องเดาได้ว่า:
ชั้นนอก = ล้นอาเรย์หรือยัง (Tree Overflow)
ชั้นใน = ช่องนั้นมีคนอยู่หรือยัง (Node Inused)
เทคนิคนี้ใช้ได้ทุกข้อสอบเติมโค้ด: ดูข้อความ error ที่โจทย์ให้ แล้วย้อนกลับไปหาเงื่อนไข
⚠️ ต้องเป็น index < MAXTREE ไม่ใช่ <=
อาเรย์ขนาด 15 มีช่อง 0 ถึง 14 — ถ้าใช้ <= จะเขียนลงช่อง 15 ซึ่งอยู่นอกอาเรย์
💡 ชนิดข้อมูลของพารามิเตอร์: k เป็น int (ตำแหน่ง) · data เป็น char (ตัวอักษรในโหนด) · ฟังก์ชันไม่คืนค่าอะไร จึงเป็น void
เติมค่า: ในฟังก์ชัน SetRight บรรทัด index = ? (เขียนในรูป 2*k+... )
ถ้าเรียก SetLeft(7,'Z') กับอาเรย์ขนาด 15 จะเกิดอะไรขึ้น (ตอบข้อความที่พิมพ์ออกมา)

3 ฟังก์ชันพื้นฐานของลิงค์ลิสต์ — createNode / SetLeft / SetRight

โจทย์ข้อ 2: จงเขียนโปรแกรมและฟังก์ชันสำหรับสร้างต้นไม้นิพจน์ด้วยลิงค์ลิสต์ แล้วแสดงนิพจน์แบบ Prefix และ Postfix
โครงสร้างและ main ที่โจทย์ให้มาแล้ว: 
struct Treenode {
 char data;
 struct Treenode *leftChild;
 struct Treenode *rightChild;
};
struct Treenode *Root;

// ฟังก์ชันที่ต้องเขียนเอง
struct Treenode* createNode(char);
void SetRight(struct Treenode*,char);
void SetLeft(struct Treenode*,char);
void preorder(struct Treenode *);
void postorder(struct Treenode *);
โจทย์ เขียน 3 ฟังก์ชันนี้เองก่อนดูเฉลย
เฉลย 
struct Treenode* createNode(char data) {
    struct Treenode *n = malloc(sizeof(struct Treenode));
    n->data = data;
    n->leftChild = NULL;      // ห้ามลืม 2 บรรทัดนี้
    n->rightChild = NULL;
    return n;
}

void SetLeft(struct Treenode *pn, char data) {
    pn->leftChild = createNode(data);
}

void SetRight(struct Treenode *pn, char data) {
    pn->rightChild = createNode(data);
}
⚠️ บั๊กอันดับ 1 ของข้อนี้: ลืมตั้ง leftChild กับ rightChild เป็น NULL ใน createNode
malloc ไม่ล้างหน่วยความจำให้ ค่าที่อยู่ข้างในจึงเป็นขยะ — พอ preorder เดินไปเจอขยะก็จะวิ่งต่อไปเรื่อยๆ แล้ว segmentation fault
อาการ: โปรแกรมคอมไพล์ผ่าน แต่พอรันแล้วเด้ง หาสาเหตุไม่เจอ
💡 ทำไม SetLeft ต้องรับ pointer (struct Treenode*) ไม่ใช่ตัวเลขเหมือนข้อ 1?
ข้อ 1 ต้นไม้อยู่ในอาเรย์ ตำแหน่งจึงเป็นตัวเลข index
ข้อ 2 ต้นไม้เป็นโหนดกระจายอยู่ในหน่วยความจำ ต้องอ้างถึงด้วยที่อยู่ของโหนด — นี่คือความต่างหลักของสองวิธีเก็บต้นไม้
ต้นไม้ที่ main สร้าง (รูปที่ 1)
+ ╱ ╲ * d ╱ ╲ a + ╱ ╲ b c นิพจน์เดิม: a * (b + c) + d
ถ้า createNode ลืมตั้ง leftChild/rightChild เป็น NULL ผลที่ตามมาคืออะไร
Aคอมไพล์ไม่ผ่าน
Bpreorder พิมพ์ครบปกติ
Cรันแล้วพัง (segmentation fault)

2 ฟังก์ชันท่องต้นไม้ — preorder / postorder

โจทย์ 
// เขียน preorder และ postorder แบบเวียนเกิด (recursive)
// แล้วเรียกใช้ใน main ตรงคอมเมนต์ //call preorder and postorder
เฉลย 
void preorder(struct Treenode *r) {
    if (r != NULL) {
        printf("%c ", r->data);   // ราก
        preorder(r->leftChild);       // ซ้าย
        preorder(r->rightChild);      // ขวา
    }
}

void postorder(struct Treenode *r) {
    if (r != NULL) {
        postorder(r->leftChild);      // ซ้าย
        postorder(r->rightChild);     // ขวา
        printf("%c ", r->data);   // ราก
    }
}

C 
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct Treenode {
    char data;
    struct Treenode *leftChild, *rightChild;
};

struct Treenode* createNode(char data) {
    struct Treenode *n = malloc(sizeof(struct Treenode));
    n->data = data;
    n->leftChild = n->rightChild = NULL;
    return n;
}

int isOperator(char c) {
    return c=='+'||c=='-'||c=='*'||c=='/';
}

// พิมพ์ infix พร้อมวงเล็บ
void infix(struct Treenode *r) {
    if (r == NULL) return;
    if (!isOperator(r->data)) {      // เป็นเลข
        printf("%c", r->data);
        return;
    }
    printf("(");                     // เปิดวงเล็บ
    infix(r->leftChild);
    printf("%c", r->data);
    infix(r->rightChild);
    printf(")");                     // ปิดวงเล็บ
}

// คำนวณค่าของต้นไม้
int evaluate(struct Treenode *r) {
    if (!isOperator(r->data))
        return r->data - '0';         // ตัวอักษร → ตัวเลข
    int a = evaluate(r->leftChild);
    int b = evaluate(r->rightChild);
    switch (r->data) {
        case '+': return a + b;
        case '-': return a - b;
        case '*': return a * b;
        default : return a / b;
    }
}

int main() {
    char s[51];
    struct Treenode *st[51];   // สแตคของโหนด
    int top = 0;

    scanf("%s", s);
    // อ่านจากหลังมาหน้า
    for (int i = strlen(s) - 1; i >= 0; i--) {
        struct Treenode *n = createNode(s[i]);
        if (isOperator(s[i])) {
            n->leftChild  = st[--top];   // pop ตัวแรก = ซ้าย
            n->rightChild = st[--top];   // pop ตัวสอง = ขวา
        }
        st[top++] = n;                   // push
    }

    struct Treenode *Root = st[--top];
    infix(Root);
    printf("=%d\n", evaluate(Root));
    return 0;
}
