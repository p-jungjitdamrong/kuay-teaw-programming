      - เข้าถึง/ประมวลผลเร็ว

Algorithm (ตอบตามนี้)
"ขั้นตอนวิธีในการแก้ปัญหา
ที่ชัดเจน เป็นลำดับ
มีจุดเริ่มต้นและจุดสิ้นสุด"

/* คุณสมบัติที่ต้องเลือกถูกในปรนัย:
   ① Correctness ถูกต้องทุก instance
   ② Efficiency มี 2 มุม:
      - Time Complexity (เวลา)
      - Space Complexity (memory)

   → ที่มาของบทที่ 2 ทั้งบท */
แนวข้อสอบ: "โครงสร้างข้อมูลที่ดีช่วยเรื่องใด" → ตอบ 2 อย่างเสมอ: เข้าถึงข้อมูลเร็ว + ประหยัดหน่วยความจำ — ตอบอย่างเดียวได้คะแนนครึ่งเดียว
ติวบทที่ 1 · จุดหลอกปรนัย

Array ทำอะไร "ไม่ได้" — จุดหลอกที่ออกทุกปี

สไลด์อาจารย์เน้นเรื่อง "(Lack of) Aggregate Array Operations" — ทำกับทั้งก้อนไม่ได้ ต้องวนลูปทีละตัว

❌ ผิดทั้งหมด (ทำกับทั้งก้อน)
int x[10], y[10];

x = 5;         /* ❌ assign ทั้งก้อน */
x = x * 2;     /* ❌ คูณทั้งก้อน */
x = x + y;     /* ❌ บวกสอง array */
if (x < y)     /* ❌ เปรียบเทียบทั้งก้อน */
printf("%d",x); /* ❌ พิมพ์ทั้งก้อน */
scanf("%d",x);  /* ❌ รับทั้งก้อน */
✅ ถูก (ระบุ index ทีละตัว)
x[2] = 5;              /* ✅ ทีละช่อง */

for (i=0; i<10; i++)   /* ✅ วนลูป */
    x[i] = x[i] * 2;

int x[5] = {1,2,3,4,5}; /* ✅ ตอนประกาศ
                            ทำได้ครั้งเดียว! */

/* ข้อยกเว้นเดียว: C string
   printf("%s", z); ✅ พิมพ์ทั้ง
   string ได้เพราะมี \0 ปิดท้าย */
วิธีจำเข้าห้องสอบ: Array = ต้องระบุ index เสมอ ยกเว้น 2 อย่าง — ตอนประกาศพร้อมค่าเริ่มต้น กับ C string ใน printf/scanf · เจอตัวเลือกไหนทำกับทั้งก้อน = ผิดทันที
ติวบทที่ 1 · Insert

InsertItem — แทรกต้อง "เลื่อนขวา จากท้ายมา"

โค้ดนี้อยู่ในสไลด์อาจารย์ตรงๆ และเป็นฐานของการเทียบกับ Linked List ในบท 4 — ต้องอ่านออกและเขียนได้

โค้ดจากสไลด์อาจารย์
void InsertItem(int A[], int k, int x)
{
    int i;
    for (i = Last; i >= k; i--)
        A[i+1] = A[i];   /* เลื่อนขวา */
    A[k] = x;            /* วางค่าใหม่ */
    Last++;              /* ขยับตัวชี้ท้าย */
}
Trace: InsertItem(A, 2, 10)
A = {3, 4, 5, 9}   Last = 3 (index)

i=3: A[4]=A[3] → {3,4,5,9,9}
i=2: A[3]=A[2] → {3,4,5,5,9}
i หยุด (i=1 < k=2)

A[2]=10        → {3,4,10,5,9}
Last=4 ✓

/* ⚠️ ทำไมต้องวนจาก "ท้าย" มา "k"?
   ถ้าวนจาก k ไปท้าย ค่าจะถูกทับ!
   A[3]=A[2] ก่อน A[4]=A[3]
   → 9 หายไป กลายเป็น 5 ซ้ำ
   ← จุดที่ข้อสอบชอบหลอกที่สุด */
ท่องเลย: แทรก = เลื่อนขวา วนจากท้ายมาหา k · ลบ = เลื่อนซ้าย วนจาก k ไปท้าย — ทิศการวนตรงข้ามกัน สลับเมื่อไหร่ข้อมูลพังเมื่อนั้น
ติวบทที่ 1 · Remove (Exercise ในสไลด์)

RemoveItem — โจทย์ที่อาจารย์เว้นว่างให้ทำเอง

สไลด์อาจารย์ให้ Exercise: "เขียนฟังก์ชันลบค่าออกจากอาเรย์ตำแหน่ง k" — นี่คือเฉลยพร้อมเหตุผล

เฉลย
void RemoveItem(int A[], int k)
{
    int i;
    for (i = k; i < Last; i++)
        A[i] = A[i+1];   /* เลื่อนซ้าย */
    Last--;              /* หดตัวชี้ท้าย */
}
Trace: RemoveItem(A, 1)
A = {3, 4, 10, 5, 9}   Last = 4

i=1: A[1]=A[2] → {3,10,10,5,9}
i=2: A[2]=A[3] → {3,10,5,5,9}
i=3: A[3]=A[4] → {3,10,5,9,9}
Last=3 → มอง 9 ตัวท้ายว่า "ไม่มีแล้ว"

ผล: {3, 10, 5, 9} ✓

/* สังเกต: ลบวนจาก k ไปท้าย
   (ตรงข้ามกับ insert เป๊ะ) */
ประเด็นออกสอบ: ทั้ง Insert และ Remove ของ array ใช้เวลา O(n) เพราะต้องเลื่อนข้อมูล — ประโยคนี้แหละคือ "เหตุผลที่ Linked List เกิดมา" (บท 4) ข้อสอบถามเชื่อมสองบทนี้บ่อยมาก
ติวบทที่ 1 · Exercise จากสไลด์

ทำ Exercise Y = mX + c ให้ดูจบๆ

โจทย์ในสไลด์: X = {1,2,3,4,5}, Y = 5X + 3, เทียบกับเฉลย Z = {8,13,18,23,28}

#include <stdio.h>
int main() {
    int X[5] = {1, 2, 3, 4, 5};        /* 1. ประกาศพร้อมค่า — ทำได้ */
    int Y[5];
    int Z[5] = {8, 13, 18, 23, 28};   /* 4. ผลเฉลยที่ถูกต้อง */
    int m = 5, c = 3, i, same = 1;

    for (i = 0; i < 5; i++)
        Y[i] = m * X[i] + c;              /* 2. Y=mX+c ต้องวนลูป! (Y=m*X+c ทั้งก้อน ❌) */

    for (i = 0; i < 5; i++)
        printf("%d ", Y[i]);              /* 3. แสดงผล — วนลูปเช่นกัน */

    for (i = 0; i < 5; i++)              /* 5. เปรียบเทียบ — ก็ต้องวนลูป! */
        if (Y[i] != Z[i]) same = 0;

    printf(same ? "\nY==Z" : "\nY!=Z");   /* Y==Z ✓ */
    return 0;
}
สังเกตไหม: ทุก step ของโจทย์นี้จงใจบังคับให้ "วนลูปทีละตัว" — กำหนดค่า, คำนวณ, แสดงผล, เปรียบเทียบ ครบทั้ง 4 operations ที่ทำกับทั้งก้อนไม่ได้ นี่คือวิธีอาจารย์สอนผ่านโจทย์
ติวบทที่ 1 · Pointer

Pointer 4 บรรทัดที่ต้องอ่านออก (ใช้ต่อบท 4)

อ่านให้ออกแค่นี้พอ
int a = 5;
int *p;      /* p = ตัวชี้ address */

p = &a;      /* & = "address ของ a" */
*p = 10;     /* * = "ค่าที่ p ชี้อยู่"
                → a กลายเป็น 10 */

printf("%d", a);    /* 10 */
printf("%d", *p);   /* 10 เหมือนกัน */
ทำไมต้องรู้ตอนนี้
/* บท 4 Linked List ทั้งบท
   คือ pointer ล้วนๆ:  */

struct node {
    int data;
    struct node *next;  /* ← ชี้โหนดถัดไป */
};

p->next          /* = (*p).next
   "ช่อง next ของโหนดที่ p ชี้"
   สัญลักษณ์ -> เจอทุกบรรทัดในบท 4
   อ่านไม่ออก = บท 4 จบเห่ */
จำสามตัว: & = เอา address · * = เอาค่าที่ชี้ · -> = เข้า field ของ struct ที่ pointer ชี้ — สามตัวนี้พอสำหรับสอบ midterm ทั้งฉบับ
ติวบทที่ 1 · Lab ทวน

แนวโจทย์ Lab บท 1 — ทำซ้ำให้มือจำ

Pattern ที่ lab ให้ฝึก
/* 1. อ่านค่าลง array แล้วประมวลผล */
int A[100], n, i, sum = 0;
scanf("%d", &n);
for (i = 0; i < n; i++)
    scanf("%d", &A[i]);

/* 2. หาผลรวม / max / min */
int max = A[0];
for (i = 0; i < n; i++) {
    sum += A[i];
    if (A[i] > max) max = A[i];
}
จุดที่คนพลาดใน lab
/* ❌ ลืม & ใน scanf ตัวแปรเดี่ยว */
scanf("%d", n);     /* พัง! ต้อง &n */

/* ❌ max เริ่มที่ 0
   ถ้าข้อมูลติดลบหมด → ผิด!
   ✅ เริ่มที่ A[0] เสมอ */

/* ❌ วน i <= n (เกิน 1 ตัว)
   ✅ วน i < n
   index สุดท้ายคือ n-1 */
ก่อนไปบท 2 ต้องทำได้: อ่านค่าลง array → วนประมวลผล → insert/remove พร้อมเลื่อนข้อมูล — ถ้าสามท่านี้คล่อง บท 2-3 จะลื่นทันทีเพราะใช้ loop แบบเดียวกันวิเคราะห์
