Array vs Linked List — ตารางจากสไลด์อาจารย์

1D-Array(Singly) Linked List
ขนาดคงที่ (Fixed size) — จองก่อนใช้เพิ่ม/ลดได้ (Dynamic size)
Insert / Removeต้องเลื่อนข้อมูล → O(n)ไม่ต้องเลื่อน — แก้ pointer → O(1) ถ้ารู้ตำแหน่ง
เข้าถึงสมาชิกระบุ index ได้ทันที → O(1)ต้องเดินจาก head → O(n)
หน่วยความจำติดกันเป็นแถบกระจายได้ เชื่อมด้วย pointer
พื้นที่ต่อสมาชิกเก็บแค่ datadata + pointer (เปลืองกว่าต่อตัว)
วิธีจำที่ไม่มีวันลืม: สองโครงสร้างนี้ "สลับจุดแข็งกัน" — Array เข้าถึงเร็วแต่แทรกช้า · LL แทรกเร็วแต่เข้าถึงช้า · ข้อสอบถามได้ทั้ง "ข้อดีของ LL" และ "ข้อเสียของ LL" ต้องตอบได้สองทาง
ติวบทที่ 4 · โครงสร้าง

struct node + กติกาวาดรูป

โค้ดพื้นฐาน (จาก Lab template)
struct node {
    int data;
    struct node *next;
};
typedef struct node Node;

Node *head = NULL;   /* ลิสต์ว่าง */

/* สร้างโหนดใหม่ */
Node* createNode(int x) {
    Node *n = (Node*)malloc(sizeof(Node));
    n->data = x;
    n->next = NULL;
    return n;
}
วาดรูปแบบนี้เสมอ
head │ ▼ ┌────┬───┐ ┌────┬───┐ ┌────┬───┐ │ 25 │ ●─┼──▶│ 31 │ ●─┼──▶│ 42 │ ╳ │ └────┴───┘ └────┴───┘ └────┴───┘ ╳ = NULL
/* กติกาห้องสอบ:
   ① ทุกข้อ LL — วาดรูปก่อนเขียนโค้ด
   ② เขียนเลขลำดับบนลูกศรที่แก้
   ③ โหนดสุดท้าย next = NULL เสมอ
     (ยกเว้น Circular!) */
ท่อง 3 สัญลักษณ์จากบท 1: -> เข้า field ของโหนดที่ pointer ชี้ · p = p->next เดินไปโหนดถัดไป · p->next = q เปลี่ยนลูกศร — ทั้งบทมีแค่ 3 ท่านี้ผสมกัน
ติวบทที่ 4 · Insert (1/2)

Insert หัวลิสต์ — ลำดับบรรทัดคือคะแนน

insertFront — 2 บรรทัดที่ห้ามสลับ
void insertFront(Node **head, int x) {
    Node *n = createNode(x);
    n->next = *head;   /* ① ชี้ตัวแรกเดิม */
    *head = n;         /* ② ย้าย head มาที่เรา */
}

/* ⚠️ ถ้าสลับ ①② :
   *head = n; ก่อน
   → ตัวแรกเดิมหายไปเลย
   ลิสต์ที่เหลือหลุดหมด (หลุดทั้งเส้น!)
   ← จุดหักคะแนนอันดับ 1 ของบทนี้ */
วาดตาม: insertFront 10
ก่อน: head ─▶ [25]─▶[31]─▶ ╳ ① n->next = head เดิม [10]─▶[25]─▶[31]─▶ ╳ ▲ n ② head = n head ─▶ [10]─▶[25]─▶[31]─▶ ╳ ✓
/* กรณีลิสต์ว่าง (head=NULL)?
   n->next = NULL (ค่า head เดิม)
   head = n
   → ทำงานถูกโดยไม่ต้อง if พิเศษ ✓ */
หลักจำลำดับ: "ต่อของใหม่ก่อน แล้วค่อยปล่อยมือจากของเก่า" — เหมือนโหนผาต้องคว้าเชือกใหม่ก่อนปล่อยเชือกเดิม ใช้ได้กับ insert ทุกกรณีในบทนี้
ติวบทที่ 4 · Insert (2/2)

Insert กลาง (หลังโหนด p) และ ท้ายลิสต์

แทรกหลังโหนด p
void insertAfter(Node *p, int x) {
    Node *n = createNode(x);
    n->next = p->next;   /* ① */
    p->next = n;         /* ② */
}
[p]─▶[q] แทรก n ระหว่าง: ① n->next = p->next (ชี้ q) ② p->next = n [p]─▶[n]─▶[q] ✓ (ห้ามสลับ ①②)
แทรกท้าย — ต้องเดินไปหาท้ายก่อน
void insertLast(Node **head, int x) {
    Node *n = createNode(x);
    if (*head == NULL) {   /* เคสลิสต์ว่าง! */
        *head = n;
        return;
    }
    Node *p = *head;
    while (p->next != NULL)  /* เดินหาท้าย */
        p = p->next;
    p->next = n;             /* ต่อท้าย */
}
/* เดินหาท้าย O(n) — จุดที่ Circular LL
   (ใช้ tail) จะมาแก้ทีหลัง */
เช็คลิสต์ insert ครบ 3 กรณี: หัว (แก้ head) · กลาง (แก้ p->next) · ท้าย (เดินก่อน + ระวังลิสต์ว่าง) — ข้อสอบให้เขียนกรณีใดกรณีหนึ่ง หรือให้เรียงบรรทัดที่สลับมา
ติวบทที่ 4 · Delete

Delete — ต้องหยุดที่ "ตัวก่อนหน้า" เสมอ

ลบตัวแรก
void deleteFront(Node **head) {
    if (*head == NULL) return;
    Node *tmp = *head;      /* ① จำไว้ก่อน */
    *head = (*head)->next;  /* ② ข้ามตัวแรก */
    free(tmp);              /* ③ คืนหน่วยความจำ */
}

/* ⚠️ free ก่อนย้าย head = พัง!
   ต้อง "จำ→ข้าม→ค่อย free" */
ลบโหนดที่มีค่า x
void deleteValue(Node **head, int x) {
    if (*head == NULL) return;
    if ((*head)->data == x) {      /* เคสตัวแรก */
        deleteFront(head); return;
    }
    Node *p = *head;
    while (p->next != NULL &&
           p->next->data != x)     /* หยุดที่ตัว "ก่อน" x */
        p = p->next;
    if (p->next == NULL) return;   /* ไม่เจอ */
    Node *tmp = p->next;            /* ตัวที่จะลบ */
    p->next = tmp->next;            /* ข้ามมัน */
    free(tmp);
}
ลบ 31: [25]─▶[31]─▶[42] p ต้องหยุดที่ [25] (ตัวก่อน!) p->next = p->next->next → [25]────▶[42] แล้ว free(31)
ทำไมต้องหยุดตัวก่อนหน้า: Singly LL เดินย้อนไม่ได้ — ถ้าเดินไปถึงตัวที่จะลบแล้ว จะไม่มีใครชี้ข้ามมันได้ · เงื่อนไข while ใช้ p->next->data ไม่ใช่ p->data เพราะเหตุนี้

Node* search(Node *head, int x) {
    Node *p = head;
    while (p != NULL) {
        if (p->data == x) return p;
        p = p->next;
    }
    return NULL;      /* ไม่เจอ */

ข้อสอบเขียนโค้ดสายนี้ให้คะแนนตาม 3 จุด: ① เริ่ม p = head ② เงื่อนไข p != NULL ③ อย่าลืม p = p->next (ลืมแล้ว infinite loop = ศูนย์คะแนนข้อนั้น)
ติวบทที่ 4 · Doubly

Doubly Linked List — เพิ่ม prev เดินได้สองทาง

โครงสร้าง
struct dnode {
    struct dnode *prev;   /* ← เพิ่ม */
    int data;
    struct dnode *next;
};
╳◀─[25]◀──▶[31]◀──▶[42]─▶╳ ▲ head (2 ลูกศรต่อคู่)
แทรก n หลัง p — แก้ 4 ลูกศร!
n->prev = p;            /* ① */
n->next = p->next;      /* ② */
if (p->next != NULL)
    p->next->prev = n;  /* ③ */
p->next = n;            /* ④ */

/* จุดสอบ: ครบ 4 เส้นไหม?
   ลืม ③ บ่อยสุด (prev ของตัวขวา)
   วาดรูปแล้วนับลูกศร: โหนดใหม่
   มี 2 เส้นออก + 2 เส้นเข้า */
เทียบข้อดี-เสีย (ปรนัย): Doubly เดินย้อนได้ + ลบโหนดที่รู้ตำแหน่งได้ไม่ต้องหาตัวก่อนหน้า · แลกกับ pointer เพิ่ม 1 ตัวต่อโหนด (เปลืองที่) และแก้ลูกศรเยอะขึ้นตอน insert/delete
ติวบทที่ 4 · Circular

Circular Linked List — ท้ายชี้กลับหัว ไม่มี NULL

จุดต่างจาก Singly (จากสไลด์อาจารย์)
① โหนดสุดท้าย next ชี้กลับต้นลิสต์
   → "ไม่มีพอยน์เตอร์ใดมีค่า NULL"

② นิยมใช้ pointer ชี้ "โหนดสุดท้าย"
   (tail) แทน head เพราะ:
   tail       = ท้ายลิสต์  O(1)
   tail->next = หัวลิสต์!  O(1)
   → เข้าถึงหัวและท้ายได้ทันทีทั้งคู่

③ เดินครบรอบ = กลับมาเจอจุดเริ่ม
   while (p != จุดเริ่ม) ← ไม่ใช่ NULL!
ใช้กับงานอะไร (ออกปรนัย)
/* งานที่ต้อง "วนซ้ำเป็นรอบ":
   Round-robin scheduling
   ของระบบปฏิบัติการ
   — CPU แจกเวลาให้โปรแกรม
     วนไปเรื่อยๆ ทีละตัว

   ← ตัวอย่างนี้อยู่ในสไลด์อาจารย์
     ตรงๆ ให้ตอบคำนี้เลย */

/* เดินพิมพ์ทุกตัว (เริ่มจากหัว): */
Node *p = tail->next;      /* หัว */
do {
    printf("%d ", p->data);
    p = p->next;
} while (p != tail->next); /* ครบรอบ */
จุดหลอกปรนัย: "Circular LL โหนดใดมีค่า next เป็น NULL?" → ไม่มีเลย · และการวนลูปต้องใช้ do-while (เพราะเช็คก่อนจะไม่ได้เข้าลูปเลย — p เริ่มที่จุดเช็คพอดี)
