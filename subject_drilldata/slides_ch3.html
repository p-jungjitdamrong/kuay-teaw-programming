สร้าง BST แล้วเดิน 3 แบบ

โจทย์: รับตัวเลขมาเรื่อยๆ จนเจอ −1 ให้ใส่ลง Binary Search Tree (ค่าน้อยกว่าไปซ้าย มากกว่าไปขวา ค่าซ้ำไม่ต้องใส่) แล้วแสดงผลการเดินต้นไม้ทั้ง 3 แบบ: preorder, inorder, postorder
ตัวอย่างการทำงาน
ป้อน: 50 30 70 20 40 60 80 -1 ต้นไม้ที่ได้: 50 ╱ ╲ 30 70 ╱ ╲ ╱ ╲ 20 40 60 80 preorder : 50 30 20 40 70 60 80 inorder : 20 30 40 50 60 70 80 postorder: 20 40 30 60 80 70 50
🔑 สังเกต inorder ให้ดี — ได้ผลลัพธ์เรียงจากน้อยไปมากเสมอ นี่คือคุณสมบัติสำคัญที่สุดของ BST และเป็นคำถามที่ออกสอบบ่อยที่สุด
ความต่างของ 3 แบบ — แค่ย้ายที่พิมพ์
แบบลำดับใช้ทำอะไร
preorderตัวเอง → ซ้าย → ขวาคัดลอกต้นไม้
inorderซ้าย → ตัวเอง → ขวาเรียงข้อมูล
postorderซ้าย → ขวา → ตัวเองลบต้นไม้ทิ้ง
วิธีจำ: คำว่า pre / in / post บอกว่า"ตัวเอง"อยู่ตรงไหน — pre=ก่อน, in=กลาง, post=หลัง ส่วนซ้ายมาก่อนขวาเสมอทั้ง 3 แบบ
การใส่ค่าลง BST — ตามหาที่ว่าง
ใส่ 40 ลงต้นไม้ที่มี 50, 30: ที่ 50: 40 < 50 → ไปซ้าย ที่ 30: 40 > 30 → ไปขวา ขวาของ 30 ว่าง → วางตรงนี้ 50 ╱ 30 ╲ 40
⏸️ หยุดลองเอง — ลองวาดต้นไม้จากลำดับ 10 20 30 40 ดู แล้วดูว่าเกิดอะไรขึ้น (เฉลย: กลายเป็นเส้นตรงเอียงขวา = ต้นไม้เสียสมดุล ค้นหากลายเป็น O(n))

โค้ดเต็ม — C และ Python

C 
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left, *right;
};

struct Node* newNode(int v) {
    struct Node *p = malloc(sizeof(struct Node));
    p->data = v;
    p->left = p->right = NULL;
    return p;
}

// ใส่ค่า — เขียนแบบเวียนบังเกิด สั้นที่สุด
struct Node* insert(struct Node *root, int v) {
    if (root == NULL) return newNode(v);   // เจอที่ว่าง
    if (v < root->data)
        root->left  = insert(root->left, v);
    else if (v > root->data)
        root->right = insert(root->right, v);
    // v == data → ไม่ทำอะไร (ห้ามค่าซ้ำ)
    return root;
}

void preorder(struct Node *r) {
    if (r == NULL) return;
    printf("%d ", r->data);   // ← ตัวเองก่อน
    preorder(r->left);
    preorder(r->right);
}

void inorder(struct Node *r) {
    if (r == NULL) return;
    inorder(r->left);
    printf("%d ", r->data);   // ← ตัวเองตรงกลาง
    inorder(r->right);
}

void postorder(struct Node *r) {
    if (r == NULL) return;
    postorder(r->left);
    postorder(r->right);
    printf("%d ", r->data);   // ← ตัวเองทีหลัง
}

int main() {
    struct Node *root = NULL;
    int v;
    while (scanf("%d", &v) == 1 && v != -1)
        root = insert(root, v);

    printf("preorder : "); preorder(root);  printf("\n");
    printf("inorder  : "); inorder(root);   printf("\n");
    printf("postorder: "); postorder(root); printf("\n");
    return 0;
}

หาความสูง + นับใบของต้นไม้

โจทย์: จากต้นไม้ที่สร้างไว้ ให้หา
① ความสูง (height) = จำนวนระดับที่ลึกที่สุด   ② จำนวนใบ (leaf) = โหนดที่ไม่มีลูกเลย
③ จำนวนโหนดทั้งหมด
ตัวอย่างการทำงาน
ต้นไม้: 50 ← ระดับ 1 ╱ ╲ 30 70 ← ระดับ 2 ╱ ╲ ╲ 20 40 80 ← ระดับ 3 ความสูง = 3 จำนวนใบ = 3 (20, 40, 80) จำนวนโหนด = 6
ทำไมข้อนี้ถึงต้องคิดแบบเวียนบังเกิด?
เพราะ "ความสูงของต้นไม้" นิยามด้วยตัวมันเอง:
ความสูง = 1 + ความสูงของกิ่งที่สูงกว่า
พอนิยามเป็นแบบนี้ โค้ดก็เขียนตามนิยามได้ตรงๆ เลย
แปลนิยามเป็นโค้ด — ทั้ง 3 ข้อ
height(r): ถ้า r ว่าง → 0 ไม่งั้น → 1 + max(height(ซ้าย), height(ขวา)) countLeaf(r): ถ้า r ว่าง → 0 ถ้าไม่มีลูกทั้งคู่ → 1 ไม่งั้น → countLeaf(ซ้าย) + countLeaf(ขวา) countNode(r): ถ้า r ว่าง → 0 ไม่งั้น → 1 + countNode(ซ้าย) + countNode(ขวา)
💡 สังเกตว่าทั้ง 3 ตัวมีโครงเดียวกันเป๊ะ: เช็คว่าง → คืน 0 · แล้วรวมผลของซ้ายกับขวา ต่างกันแค่จะบวก 1 ตอนไหน — จำโครงนี้ได้ก็เขียนโจทย์ตระกูลนี้ได้ทั้งหมด
⏸️ หยุดลองเอง — ลองเขียน sumAll(r) ที่หาผลรวมค่าทุกโหนด โดยใช้โครงเดียวกัน (เฉลย: ว่าง→0, ไม่งั้น→ r->data + sumAll(ซ้าย) + sumAll(ขวา))

C 
int height(struct Node *r) {
    if (r == NULL) return 0;
    int hl = height(r->left);
    int hr = height(r->right);
    return 1 + (hl > hr ? hl : hr);   // 1 + max
}

int countLeaf(struct Node *r) {
    if (r == NULL) return 0;
    if (r->left == NULL && r->right == NULL)
        return 1;                       // เป็นใบ
    return countLeaf(r->left) + countLeaf(r->right);
}

int countNode(struct Node *r) {
    if (r == NULL) return 0;
    return 1 + countNode(r->left) + countNode(r->right);
}

// เรียกใช้
printf("ความสูง   = %d\n", height(root));
printf("จำนวนใบ   = %d\n", countLeaf(root));
printf("จำนวนโหนด = %d\n", countNode(root));
PYTHON 
def height(r):
    if r is None:
        return 0
    return 1 + max(height(r.left), height(r.right))

def count_leaf(r):
    if r is None:
        return 0
    if r.left is None and r.right is None:
        return 1
    return count_leaf(r.left) + count_leaf(r.right)

def count_node(r):
    if r is None:
        return 0
    return 1 + count_node(r.left) + count_node(r.right)

print("ความสูง   =", height(root))
print("จำนวนใบ   =", count_leaf(root))
print("จำนวนโหนด =", count_node(root))

Hash Table แก้ชนด้วย Chaining

โจทย์: สร้าง Hash Table ขนาด 7 ช่อง ใช้ฟังก์ชันแฮช h(k) = k % 7 แก้ปัญหาการชนกัน (collision) ด้วยวิธี chaining (ต่อ linked list ในช่องเดียวกัน)
รองรับ insert และ search พร้อมบอกว่าค้นหาแล้วเทียบไปกี่ครั้ง
ตัวอย่างการทำงาน
ใส่: 15 22 8 29 3 h(15)=15%7=1 h(22)=22%7=1 h(8) = 8%7=1 h(29)=29%7=1 h(3) = 3%7=3 ตาราง: [0] → NULL [1] → 3ตัวชนกัน... 29 → 8 → 22 → 15 [2] → NULL [3] → 3 [4] → NULL [5] → NULL [6] → NULL search(8) → เจอ! เทียบ 2 ครั้ง search(20) → h=6 ช่องว่าง → ไม่เจอ
⚠️ ตัวอย่างนี้จงใจให้ชนกันเยอะ เพื่อให้เห็นว่าถ้าเลือกขนาดตารางหรือฟังก์ชันแฮชไม่ดี hash table จะกลายเป็น linked list ธรรมดา — ค้นหาจาก O(1) เหลือ O(n)
ทำไมต้องมี chaining?
เพราะ k % 7 ทำให้ค่าหลายตัวได้ช่องเดียวกัน (15, 22, 8, 29 ได้ช่อง 1 หมด) — เรียกว่า collision
chaining แก้ด้วยการให้แต่ละช่องเป็นหัวของ linked list ใครชนก็ต่อเข้าไปในโซ่
2 วิธีแก้ชนที่ต้องรู้จัก
วิธีทำยังไงข้อเสีย
Chainingต่อ list ในช่องใช้หน่วยความจำเพิ่ม
Linear Probingเลื่อนหาช่องว่างถัดไปเกิดกลุ่มติดกัน ช้าลง
💡 ทำไม insert แบบต่อหัว (ใส่ไว้หน้าสุด) ถึงนิยม? เพราะเร็ว O(1) ไม่ต้องเดินไปท้ายโซ่ — สังเกตในตัวอย่างว่า 29 ที่ใส่ทีหลังกลับอยู่หน้าสุด
⏸️ หยุดลองเอง — ถ้าเปลี่ยนขนาดตารางจาก 7 เป็น 10 ค่า 15 22 8 29 3 จะไปช่องไหนบ้าง? (เฉลย: 5, 2, 8, 9, 3 — กระจายดีกว่าเยอะ)

C 
#include <stdio.h>
#include <stdlib.h>
#define SIZE 7

struct Node { int key; struct Node *next; };

struct Node *table[SIZE] = {NULL};   // ทุกช่องเริ่มว่าง

int hashFn(int k) { return k % SIZE; }

void insertKey(int k) {
    int idx = hashFn(k);
    struct Node *n = malloc(sizeof(struct Node));
    n->key  = k;
    n->next = table[idx];      // ต่อหัว — O(1)
    table[idx] = n;
    printf("ใส่ %d ลงช่อง %d\n", k, idx);
}

int search(int k) {
    int idx = hashFn(k);
    int steps = 0;
    struct Node *cur = table[idx];
    while (cur != NULL) {
        steps++;
        if (cur->key == k) {
            printf("เจอ %d ที่ช่อง %d (เทียบ %d ครั้ง)\n",
                   k, idx, steps);
            return 1;
        }
        cur = cur->next;
    }
    printf("ไม่เจอ %d (เทียบ %d ครั้ง)\n", k, steps);
    return 0;
}

void show() {
    for (int i = 0; i < SIZE; i++) {
        printf("[%d]", i);
        for (struct Node *c = table[i]; c; c = c->next)
            printf(" -> %d", c->key);
        printf("\n");
    }
}

int main() {
    int keys[] = {15, 22, 8, 29, 3};
    for (int i = 0; i < 5; i++) insertKey(keys[i]);
    show();
    search(8);
    search(20);
    return 0;
}
PYTHON 
SIZE = 7

class Node:
    def __init__(self, key):
        self.key  = key
        self.next = None

table = [None] * SIZE

def hash_fn(k):
    return k % SIZE

def insert_key(k):
    idx = hash_fn(k)
    n = Node(k)
    n.next = table[idx]        # ต่อหัว
    table[idx] = n
    print(f"ใส่ {k} ลงช่อง {idx}")

def search(k):
    idx   = hash_fn(k)
    steps = 0
    cur   = table[idx]
    while cur is not None:
        steps += 1
        if cur.key == k:
            print(f"เจอ {k} ที่ช่อง {idx} (เทียบ {steps} ครั้ง)")
            return True
        cur = cur.next
    print(f"ไม่เจอ {k} (เทียบ {steps} ครั้ง)")
    return False

def show():
    for i in range(SIZE):
        row = []
        cur = table[i]
        while cur:
            row.append(str(cur.key))
            cur = cur.next
        print(f"[{i}]", " -> ".join(row) if row else "")

for k in [15, 22, 8, 29, 3]:
    insert_key(k)
show()
search(8)
search(20)
