แทรกโหนดให้ลิสต์ยังเรียงอยู่

โจทย์: มี linked list ที่เรียงจากน้อยไปมากอยู่แล้ว รับค่าใหม่เข้ามาทีละตัว ให้แทรกลงในตำแหน่งที่ถูกต้อง เพื่อให้ลิสต์ยังคงเรียงอยู่เสมอ
รับค่าไปเรื่อยๆ จนกว่าจะเจอ −1 แล้วพิมพ์ลิสต์ทั้งหมด
ตัวอย่างการทำงาน
ป้อน: 30 10 50 20 -1 ใส่ 30 → 30 ใส่ 10 → 10 30 (แทรกหัว) ใส่ 50 → 10 30 50 (แทรกท้าย) ใส่ 20 → 10 20 30 50 (แทรกกลาง) ผลลัพธ์: 10 20 30 50
3 กรณีที่ต้องแยก
① แทรกหัว — ค่าใหม่น้อยกว่าตัวแรก [10]→[30]→[50] ↑ head ต้องเปลี่ยน! ② แทรกกลาง — เจอตัวที่มากกว่า [10]→[20]→[30] ↑ ตัวใหม่ ③ แทรกท้าย — ไม่มีตัวไหนมากกว่า [10]→[30]→[50]→NULL ↑ ตัวใหม่
ทำไมข้อนี้ถึงเป็นข้อบังคับต้องทำเป็น?
เพราะมันรวมทุกทักษะพื้นฐานไว้ในข้อเดียว: สร้างโหนด · เดินลิสต์ · หาโหนดก่อนหน้า · แก้ pointer ตามลำดับที่ถูก · จัดการเคส head
ถ้าทำข้อนี้ได้ อีก 2 ข้อในชุดนี้จะง่ายลงมาก
⏸️ หยุดลองเอง — ลองเขียนบนกระดาษก่อน โดยตอบ 2 คำถามนี้ให้ได้: (1) จะรู้ได้ยังไงว่าต้องหยุดที่โหนดไหน? (2) ต้องเก็บ pointer ตัวไหนไว้บ้าง?
💡 ใบ้: การแทรกต้องรู้จัก "โหนดก่อนหน้า" เสมอ — แต่ linked list เดินย้อนกลับไม่ได้ ดังนั้นต้องหยุดให้ถูกจังหวะ ตอนที่ยังยืนอยู่บนโหนดก่อนหน้า

หยุดที่โหนด "ก่อนหน้า" ให้ถูกจังหวะ

STEP 1 — แตกปัญหา
  ① สร้างโหนดใหม่ n
  ② ถ้าลิสต์ว่าง หรือ ค่าใหม่ <= ตัวแรก
       → แทรกหัว: n->next = head; head = n
  ③ ไม่งั้น: เดินหา "โหนดก่อนหน้า"
       cur = head
       while (cur->next != NULL
              && cur->next->data < v)
           cur = cur->next
  ④ แทรกหลัง cur:
       n->next   = cur->next   ← ทำก่อน
       cur->next = n
เงื่อนไข while มี 2 ส่วน อ่านให้ขาด:
cur->next != NULL = ยังไม่ถึงท้ายลิสต์ (กันหลุด)
cur->next->data < v = ตัวถัดไปยังน้อยกว่า → เดินต่อได้
พอหลุดออกจากลูป cur คือโหนดก่อนหน้าพอดี
เดินด้วยมือ: แทรก 20 ใน 10→30→50
cur=[10] : next=[30], 30<20? ไม่ → หยุด cur คือ [10] ✔ n->next = [30] [10]->next = n ผล: 10 → 20 → 30 → 50

โค้ดเต็ม — C และ Python

C 
#include <stdio.h>
#include <stdlib.h>

struct Node { int data; struct Node *next; };

struct Node* newNode(int v) {
    struct Node *p = malloc(sizeof(struct Node));
    p->data = v; p->next = NULL;
    return p;
}

struct Node* insertSorted(struct Node *head, int v) {
    struct Node *n = newNode(v);

    // เคส ① แทรกหัว (รวมเคสลิสต์ว่างไว้ด้วย)
    if (head == NULL || v <= head->data) {
        n->next = head;
        return n;              // head ตัวใหม่
    }

    // เคส ②③ เดินหาโหนดก่อนหน้า
    struct Node *cur = head;
    while (cur->next != NULL && cur->next->data < v)
        cur = cur->next;

    n->next   = cur->next;     // ① ต่อตัวใหม่ก่อน
    cur->next = n;             // ② แล้วค่อยแก้ตัวเดิม
    return head;
}

int main() {
    struct Node *head = NULL;
    int v;
    while (scanf("%d", &v) == 1 && v != -1)
        head = insertSorted(head, v);   // ต้องรับค่ากลับ!

    for (struct Node *c = head; c != NULL; c = c->next)
        printf("%d ", c->data);
    printf("\n");
    return 0;
}
PYTHON 
class Node:
    def __init__(self, data):
        self.data = data
        self.next = None

def insert_sorted(head, v):
    n = Node(v)

    # เคส ① แทรกหัว
    if head is None or v <= head.data:
        n.next = head
        return n

    # เคส ②③ เดินหาโหนดก่อนหน้า
    cur = head
    while cur.next is not None and cur.next.data < v:
        cur = cur.next

    n.next   = cur.next
    cur.next = n
    return head

head = None
while True:
    v = int(input())
    if v == -1:
        break
    head = insert_sorted(head, v)

cur = head
while cur:
    print(cur.data, end=" ")
    cur = cur.next
print()

ลบโหนดที่มีค่าซ้ำทั้งหมด

โจทย์: มี linked list อยู่ ให้รับค่า x แล้วลบทุกโหนดที่มีค่าเท่ากับ x (ไม่ใช่แค่ตัวแรก) แล้วพิมพ์ลิสต์ที่เหลือ พร้อมบอกว่าลบไปกี่โหนด
ตัวอย่าง 1 — ค่าซ้ำอยู่กลาง
ลิสต์: 5 → 3 → 5 → 7 → 5 ลบ x = 5 ผลลัพธ์: 3 → 7 ลบไป 3 โหนด
ตัวอย่าง 2 — ค่าซ้ำติดกันที่หัว
ลิสต์: 5 → 5 → 5 → 9 ลบ x = 5 ผลลัพธ์: 9 ลบไป 3 โหนด
ตัวอย่าง 3 — ลบหมดทั้งลิสต์
ลิสต์: 5 → 5 ลบ x = 5 ผลลัพธ์: (ลิสต์ว่าง) ลบไป 2 โหนด
ตรงไหนที่ยากกว่าข้อ 1?
① ต้องลบหลายตัว ไม่ใช่ตัวเดียว
② ตัวที่ต้องลบอาจติดกัน — ถ้าเขียนแบบเลื่อน cur ทุกรอบจะข้ามตัวที่สอง
③ ตัวที่ต้องลบอาจอยู่ที่หัว และอาจติดกันหลายตัวที่หัวด้วย
บั๊กคลาสสิกที่ต้องเห็นก่อน
ลิสต์ 5→5→9 ลบ 5 ด้วยโค้ดผิด: cur=[5ตัวแรก] → ลบ → head=[5ตัวสอง] แล้ว cur = cur->next ← ผิด! cur ตอนนี้เป็น [9] แล้ว → [5ตัวสอง] ไม่โดนตรวจเลย ผลลัพธ์ผิด: 5 → 9 กฎ: ลบแล้วอย่าเลื่อน — ตรวจตำแหน่งเดิมซ้ำ
⏸️ หยุดลองเอง — เขียนโค้ดที่ผ่านทั้ง 3 ตัวอย่างข้างซ้าย โดยเฉพาะตัวอย่าง 3 ที่ลิสต์ต้องกลายเป็นว่าง (head = NULL)
โจทย์ที่ 2 · วิธีคิด + เฉลย

ลบหัวให้หมดก่อน แล้วค่อยลบตัวกลาง

แผน 2 ท่อน — คิดแยกกันจะง่ายมาก

ท่อนที่ 1: จัดการหัวให้จบก่อน
  while (head != NULL && head->data == x)
      tmp = head; head = head->next; free(tmp)
  ← ใช้ while ไม่ใช่ if เพราะหัวอาจซ้ำหลายตัว

ท่อนที่ 2: หลังจากนี้ head ปลอดภัยแน่นอน
  cur = head
  while (cur != NULL && cur->next != NULL)
      ถ้า cur->next->data == x
          ตัด cur->next ออก (ไม่เลื่อน cur)
      ไม่งั้น
          cur = cur->next
ทำไมท่อน 2 ถึงมองที่ cur->next ไม่ใช่ cur? เพราะการลบต้องใช้โหนดก่อนหน้า ถ้ายืนอยู่บนตัวที่จะลบ จะย้อนกลับไปหาตัวก่อนหน้าไม่ได้ — ยืนก่อนหนึ่งก้าวเสมอ
เดินด้วยมือ: 5→3→5→7→5 ลบ 5
ท่อน1: head=5 ตรง → ลบ → head=3 head=3 ไม่ตรง → หยุด ท่อน2: cur=3, next=5 ตรง → ลบ cur=3, next=7 ไม่ตรง → เลื่อน cur=7, next=5 ตรง → ลบ cur=7, next=NULL → จบ ผล: 3 → 7 (ลบไป 3)
C 
struct Node* deleteAll(struct Node *head, int x, int *cnt) {
    *cnt = 0;

    // ท่อน 1 — ลบที่หัวจนกว่าหัวจะไม่ใช่ x
    while (head != NULL && head->data == x) {
        struct Node *tmp = head;
        head = head->next;
        free(tmp);
        (*cnt)++;
    }

    // ท่อน 2 — ลบตัวกลาง/ตัวท้าย
    struct Node *cur = head;
    while (cur != NULL && cur->next != NULL) {
        if (cur->next->data == x) {
            struct Node *tmp = cur->next;
            cur->next = tmp->next;   // ข้ามตัวที่ลบ
            free(tmp);
            (*cnt)++;
            // ไม่เลื่อน cur! ตรวจตัวถัดไปที่เลื่อนมาแทน
        } else {
            cur = cur->next;
        }
    }
    return head;
}
PYTHON 
def delete_all(head, x):
    cnt = 0

    # ท่อน 1 — ลบที่หัว
    while head is not None and head.data == x:
        head = head.next
        cnt += 1

    # ท่อน 2 — ลบตัวกลาง/ตัวท้าย
    cur = head
    while cur is not None and cur.next is not None:
        if cur.next.data == x:
            cur.next = cur.next.next
            cnt += 1
        else:
            cur = cur.next

    return head, cnt

head, n = delete_all(head, 5)
print(f"ลบไป {n} โหนด")
ต่างกันตรง free(): C ต้องเก็บ tmp ไว้ก่อนแล้ว free ทีหลัง — ถ้า free ก่อนแล้วค่อยอ่าน tmp->next จะอ่านหน่วยความจำที่คืนไปแล้ว · Python ไม่ต้องทำอะไร ตัวเก็บขยะจัดการเอง

กลับด้านลิสต์ (Reverse)

โจทย์: มี linked list อยู่ ให้กลับลำดับทั้งลิสต์ โดยห้ามสร้างโหนดใหม่ และห้ามใช้อาร์เรย์ช่วย — ต้องกลับด้านด้วยการเปลี่ยนทิศลูกศรเท่านั้น
ตัวอย่างการทำงาน
ก่อน: 10 → 20 → 30 → NULL หลัง: 30 → 20 → 10 → NULL
แนวคิด: กลับลูกศรทีละอัน
เริ่ม: NULL [10]→[20]→[30]→NULL ↑ ↑ prev cur รอบ 1: ให้ 10 ชี้ NULL NULL←[10] [20]→[30]→NULL ↑ ↑ prev cur รอบ 2: ให้ 20 ชี้ 10 NULL←[10]←[20] [30]→NULL ↑ ↑ prev cur รอบ 3: ให้ 30 ชี้ 20 NULL←[10]←[20]←[30] NULL ↑ ↑ prev cur cur เป็น NULL → จบ head ตัวใหม่ = prev = [30]
ปัญหาที่ต้องแก้:
พอเราสั่ง cur->next = prev ลูกศรเดิมที่ชี้ไปข้างหน้าหายทันที — แล้วจะเดินต่อยังไง?

คำตอบ: ต้องจำตัวถัดไปไว้ก่อน ในตัวแปร nxt ก่อนจะกลับลูกศร
3 ตัวแปรที่ต้องมี
ตัวแปรหมายถึง
prevโหนดข้างหลัง (ที่กลับแล้ว) เริ่มที่ NULL
curโหนดที่กำลังกลับลูกศร เริ่มที่ head
nxtที่จองไว้ก่อนตัดลูกศร กันหลงทาง
ลำดับ 4 บรรทัดในลูป — ท่องให้ได้:
① nxt = cur->next (จำก่อน)
② cur->next = prev (กลับลูกศร)
③ prev = cur (เลื่อน prev)
④ cur = nxt (เลื่อน cur)
สลับลำดับเมื่อไหร่ = พังทันที
⏸️ หยุดลองเอง — ปิดสไลด์ถัดไป แล้วเขียนลูป 4 บรรทัดนี้จากความเข้าใจ (ไม่ใช่ท่องจำ) ให้ได้

C 
struct Node* reverse(struct Node *head) {
    struct Node *prev = NULL;
    struct Node *cur  = head;

    while (cur != NULL) {
        struct Node *nxt = cur->next;  // ① จำไว้ก่อน
        cur->next = prev;               // ② กลับลูกศร
        prev = cur;                     // ③ เลื่อน prev
        cur  = nxt;                     // ④ เลื่อน cur
    }
    return prev;   // prev คือ head ตัวใหม่
}

int main() {
    // ... สร้างลิสต์ 10 20 30 ...
    head = reverse(head);      // ต้องรับค่ากลับ
    for (struct Node *c = head; c; c = c->next)
        printf("%d ", c->data);
    return 0;
}
PYTHON 
def reverse(head):
    prev = None
    cur  = head
    while cur is not None:
        nxt = cur.next     # ① จำไว้ก่อน
        cur.next = prev    # ② กลับลูกศร
        prev = cur         # ③ เลื่อน prev
        cur  = nxt         # ④ เลื่อน cur
    return prev
