Array vs Linked List — ตัวเลขที่ต้องจำ

งานArrayLinked List
อ่านตัวที่ iO(1) ⚡O(n)
เพิ่มหัวO(n)O(1) ⚡
เพิ่มท้ายO(1)*O(n) หรือ O(1) ถ้ามี tail
ลบตรงกลางO(n)O(1)**
ค้นหาค่าO(n)O(n)
หน่วยความจำต่อเนื่อง ประหยัด+ pointer ทุกโหนด
* ถ้ายังไม่เต็ม  ** ถ้ารู้ตำแหน่งโหนดก่อนหน้าอยู่แล้ว (ถ้าต้องไล่หาก่อนก็ยัง O(n))
ทำไม array ลบตรงกลางถึงแพง
ลบ index 1 จาก [10, 20, 30, 40] [10][20][30][40] ↓ ลบ [10][ ][30][40] ← ← ← ต้องเลื่อนทุกตัวมาชิด [10][30][40] ยิ่งข้อมูลเยอะ ยิ่งเลื่อนนาน = O(n)
ทำไม linked list ถึงถูก
[10]→[20]→[30]→[40]→NULL ลบ 20 [10]───────↗ (แค่ย้ายลูกศรของ 10 ให้ชี้ไป 30 = 1 คำสั่ง) แต่กว่าจะ "ไปถึง" 10 ต้องไล่จาก head ทีละตัว ← นี่คือราคาที่จ่าย
💡 สรุปสั้น: Array เก่งเรื่องอ่าน · Linked List เก่งเรื่องแก้ — โจทย์บอกเองว่าจะทำอะไรบ่อย
โครงที่ต้องเขียนได้จากความจำ

Node — เทียบ C กับ Python

C 
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next;   // ต้องมี struct นำหน้า
};

// สร้างโหนดใหม่ — ท่องให้ได้ทั้งก้อน
struct Node* newNode(int v) {
    struct Node *p = malloc(sizeof(struct Node));
    p->data = v;
    p->next = NULL;
    return p;
}

int main() {
    struct Node *head = newNode(10);
    head->next = newNode(20);

    // เดินลิสต์ — ท่ามาตรฐาน
    struct Node *cur = head;
    while (cur != NULL) {
        printf("%d ", cur->data);
        cur = cur->next;
    }
    return 0;
}
PYTHON 
class Node:

head = Node(10)

เทคนิคที่ช่วยได้จริง

วาดกล่องกับลูกศรก่อน แล้วค่อยพิมพ์

กติกา: ห้ามพิมพ์โค้ด linked list ก่อนวาดภาพ "ก่อน" กับ "หลัง" ให้ครบ — 90% ของบั๊กมาจากการเรียงลำดับคำสั่งผิด ซึ่งภาพจะฟ้องทันที
ตัวอย่าง: แทรก n ระหว่าง a กับ b
ก่อน: [a] ──→ [b] [n] (ลอยอยู่) หลัง: [a] ──→ [n] ──→ [b] ลำดับคำสั่งที่ถูก: ① n->next = a->next; (n ชี้ b ก่อน) ② a->next = n; (แล้วค่อยให้ a ชี้ n)
ถ้าสลับลำดับจะพังยังไง
❌ ① a->next = n; ② n->next = a->next; พอทำ ① เสร็จ a->next กลายเป็น n แล้ว ② จะได้ n->next = n [a] ──→ [n] ⟲ (ชี้ตัวเอง) [b] หลุดหายไปเลย ผลลัพธ์: วนลูปไม่รู้จบ + ข้อมูลหาย
🔑 กฎเหล็ก: ตอนแทรก ให้ต่อลูกศรของโหนดใหม่ก่อนเสมอ แล้วค่อยแก้ลูกศรของโหนดเดิม — เพราะพอแก้ตัวเดิมไปแล้ว จะหาทางไปโหนดถัดไปไม่เจอ
📝 ในห้องสอบ: วาดลงกระดาษข้อสอบมุมขวาได้เลย ไม่มีใครหักคะแนน แต่ช่วยไม่ให้พลาดจุดนี้

4 เคสที่ต้องเทสต์ทุกครั้ง

เคสต้องเช็คว่า
ลิสต์ว่างhead == NULL แล้วโค้ดไม่ crash
มีตัวเดียวลบแล้ว head กลายเป็น NULL
ทำที่ตัวแรกต้องแก้ head ด้วย ไม่ใช่แค่ next
ทำที่ตัวสุดท้ายตัวใหม่สุดท้ายต้อง next = NULL
ทำไมเคส "ตัวแรก" ถึงพลาดกันเยอะ? เพราะโหนดอื่นๆ ถูกชี้โดย prev->next แต่โหนดแรกถูกชี้โดยตัวแปร head — คนละที่กัน จึงต้องเขียนโค้ดคนละแบบ
โครงป้องกันเคสขอบ (C)
struct Node* deleteVal(struct Node *head, int v) {
    if (head == NULL) return NULL;      // เคสว่าง

    if (head->data == v) {                // เคสตัวแรก
        struct Node *tmp = head;
        head = head->next;
        free(tmp);
        return head;
    }

    struct Node *cur = head;
    while (cur->next != NULL && cur->next->data != v)
        cur = cur->next;

    if (cur->next != NULL) {              // เจอ
        struct Node *tmp = cur->next;
        cur->next = tmp->next;
        free(tmp);
    }
    return head;
}
💡 สังเกตว่าฟังก์ชัน คืนค่า head กลับไป — เพราะ head อาจเปลี่ยน ถ้าไม่คืน ผู้เรียกจะยังถือ head เก่าที่ถูก free ไปแล้ว
