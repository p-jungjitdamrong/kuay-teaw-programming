#include <stdio.h>
#include <string.h>

int main() {
    char s[201];
    fgets(s, 201, stdin);

    char st[201];
    int top = -1;          // −1 = stack ว่าง
    int bad = 0;

    for (int i = 0; s[i] != '\0' && s[i] != '\n'; i++) {
        char c = s[i];

        if (c == '(' || c == '[' || c == '{') {
            st[++top] = c;                // push
        }
        else if (c == ')' || c == ']' || c == '}') {
            if (top == -1) {            // จุดเช็ค ①
                printf("ไม่สมดุล ที่ตำแหน่ง %d\n", i + 1);
                bad = 1; break;
            }
            char o = st[top];
            int ok = (o == '(' && c == ')')
                  || (o == '[' && c == ']')
                  || (o == '{' && c == '}');
            if (!ok) {                    // จุดเช็ค ②
                printf("ไม่สมดุล ที่ตำแหน่ง %d\n", i + 1);
                bad = 1; break;
            }
            top--;                        // pop
        }
    }

    if (!bad) {
        if (top == -1) printf("สมดุล\n");       // จุดเช็ค ③
        else          printf("ไม่สมดุล — มีวงเล็บเปิดค้าง\n");
    }
    return 0;
}

C 
#include <stdio.h>
#include <string.h>
#include <ctype.h>

int prec(char op) {
    if (op == '*' || op == '/') return 2;
    if (op == '+' || op == '-') return 1;
    return 0;                     // '(' ต่ำสุด
}

int main() {
    char s[201], post[201], st[201];
    int top = -1, k = 0;
    scanf("%s", s);

    // ---- ท่อน 1: แปลง infix → postfix ----
    for (int i = 0; s[i]; i++) {
        char c = s[i];
        if (isdigit(c)) {
            post[k++] = c;
        } else if (c == '(') {
            st[++top] = c;
        } else if (c == ')') {
            while (top >= 0 && st[top] != '(')
                post[k++] = st[top--];
            top--;                     // ทิ้ง '('
        } else {                       // + − * /
            while (top >= 0 && prec(st[top]) >= prec(c))
                post[k++] = st[top--];
            st[++top] = c;
        }
    }
    while (top >= 0) post[k++] = st[top--];
    post[k] = '\0';
    printf("postfix: %s\n", post);

    // ---- ท่อน 2: คำนวณ postfix ----
    int num[201], t2 = -1;
    for (int i = 0; post[i]; i++) {
        char c = post[i];
        if (isdigit(c)) {
            num[++t2] = c - '0';
        } else {
            int b = num[t2--];       // ตัวหลังออกก่อน!
            int a = num[t2--];
            int r = 0;
            if (c == '+') r = a + b;
            if (c == '-') r = a - b;
            if (c == '*') r = a * b;
            if (c == '/') r = a / b;
            num[++t2] = r;
        }
    }
    printf("ผลลัพธ์: %d\n", num[t2]);
    return 0;
}

Circular Queue — เต็มกับว่างต่างกันตรงไหน

โจทย์: สร้าง Circular Queue ขนาด 5 ช่อง ที่รองรับ enqueue / dequeue โดยใช้อาร์เรย์ธรรมดา และต้องวนกลับมาใช้ช่องที่ว่างแล้วได้ พร้อมตรวจสถานะเต็ม/ว่างให้ถูกต้อง
ทำไมต้องวน — ปัญหาของคิวธรรมดา
คิวธรรมดา ขนาด 5, enqueue 5 ตัว แล้ว dequeue 3 ตัว: [_][_][_][D][E] ↑front ↑rear ช่อง 0-2 ว่างแต่ใช้ไม่ได้ เพราะ rear เดินหน้าอย่างเดียว → เสียพื้นที่ไปเปล่าๆ Circular Queue แก้ด้วย: rear = (rear + 1) % SIZE front = (front + 1) % SIZE ← พอถึงท้ายก็วนกลับช่อง 0
ตัวอย่างการทำงาน (SIZE = 5)
enqueue A B C D → [A][B][C][D][_] f=0 r=3 dequeue x2 → [_][_][C][D][_] f=2 r=3 enqueue E F → [F][_][C][D][E] r=0 f=2 ← F วนกลับไปช่อง 0 แล้ว! enqueue G → คิวเต็ม ใส่ไม่ได้
ปัญหาคลาสสิกของข้อนี้:
ถ้าใช้แค่ front กับ rear เฉยๆ → เวลาคิวว่างกับคิวเต็ม ค่า front กับ rear จะอยู่ตำแหน่งเดียวกันพอดี แยกไม่ออก!
2 วิธีแก้ที่ใช้กันจริง
วิธีเต็มเมื่อว่างเมื่อ
นับจำนวน (count)count == SIZEcount == 0
เว้น 1 ช่อง(r+1)%N == fr == f
💡 แนะนำวิธี count สำหรับสอบ — เข้าใจง่ายกว่า ใช้ได้ครบทุกช่อง และไม่ต้องจำสูตร (r+1)%N ให้สับสน
ส่วนวิธี "เว้น 1 ช่อง" ต้องแลกด้วยการเสียไป 1 ช่องเสมอ (ขนาด 5 ใส่ได้จริง 4)

C 
#include <stdio.h>
#define SIZE 5

int q[SIZE];
int front = 0, rear = -1, cnt = 0;

int isFull()  { return cnt == SIZE; }
int isEmpty() { return cnt == 0; }

void enqueue(int v) {
    if (isFull()) {
        printf("คิวเต็ม ใส่ %d ไม่ได้\n", v);
        return;
    }
    rear = (rear + 1) % SIZE;    // หัวใจของ circular
    q[rear] = v;
    cnt++;
    printf("ใส่ %d ที่ช่อง %d (มี %d ตัว)\n", v, rear, cnt);
}

int dequeue() {
    if (isEmpty()) {
        printf("คิวว่าง\n");
        return -1;
    }
    int v = q[front];
    front = (front + 1) % SIZE;  // วนเหมือนกัน
    cnt--;
    printf("เอา %d ออก (เหลือ %d ตัว)\n", v, cnt);
    return v;
}

void show() {
    printf("คิวตอนนี้:");
    for (int i = 0; i < cnt; i++)
        printf(" %d", q[(front + i) % SIZE]);
    printf("\n");
}

int main() {
    enqueue(1); enqueue(2); enqueue(3); enqueue(4);
    dequeue(); dequeue();
    enqueue(5); enqueue(6);   // วนกลับช่อง 0
    show();
    return 0;
}
PYTHON 
SIZE = 5

class CircularQueue:
    def __init__(self, size=SIZE):
        self.size  = size
        self.q     = [None] * size
        self.front = 0
        self.rear  = -1
        self.cnt   = 0

    def is_full(self):  return self.cnt == self.size
    def is_empty(self): return self.cnt == 0

    def enqueue(self, v):
        if self.is_full():
            print(f"คิวเต็ม ใส่ {v} ไม่ได้")
            return
        self.rear = (self.rear + 1) % self.size
        self.q[self.rear] = v
        self.cnt += 1
        print(f"ใส่ {v} ที่ช่อง {self.rear} (มี {self.cnt} ตัว)")

    def dequeue(self):
        if self.is_empty():
            print("คิวว่าง")
            return None
        v = self.q[self.front]
        self.front = (self.front + 1) % self.size
        self.cnt -= 1
        print(f"เอา {v} ออก (เหลือ {self.cnt} ตัว)")
        return v

    def show(self):
        items = [self.q[(self.front + i) % self.size]
                 for i in range(self.cnt)]
        print("คิวตอนนี้:", *items)

cq = CircularQueue()
for v in [1, 2, 3, 4]: cq.enqueue(v)
cq.dequeue(); cq.dequeue()
cq.enqueue(5); cq.enqueue(6)
cq.show()
