บทที่ 1.5 · แนวคิด
Precedence คืออะไร? ทำอะไรก่อน?
เหมือนคณิตศาสตร์ที่ × ÷ มาก่อน + − ภาษา C มีกฎลำดับ operator ของตัวเอง ซับซ้อนกว่าที่คิด
คณิตศาสตร์ vs C — ตรงกัน?
/* 2 + 3 * 4 */ printf("%d", 2 + 3 * 4); // คณิต: 3*4=12 ก่อน → 2+12 = 14 ✓ // C: เหมือนกัน → 14 ✓ /* แต่พอมี operator หลายประเภท... */ int x = 5; printf("%d", x > 3 + 1); // + มาก่อน > → 3+1=4 ก่อน // แล้ว 5 > 4 = 1 (true) // ไม่ใช่ (5>3) + 1 = 2 !
❌ Bug ที่เจอบ่อยที่สุด
if (x & 0xFF == 0)
คิดว่า: (x & 0xFF) == 0
จริงๆ C ทำ: x & (0xFF == 0)
เพราะ == มี precedence สูงกว่า &

→ ใส่วงเล็บ: (x & 0xFF) == 0
🔑 กฎง่ายๆ จำไว้
สูงกว่า = ทำก่อน

บทที่ 1.5 · Associativity
Associativity — ซ้ายหรือขวาก่อน?
เมื่อ operator มี precedence เท่ากัน Associativity บอกว่าจะประเมินจากทิศทางไหน
L→R (Left-to-Right) — ส่วนใหญ่
ประเมินจากซ้ายไปขวา ตรงกับสัญชาตญาณ
ตัวอย่าง L→R — + และ -
/* 10 - 3 - 2 */ // L→R: (10-3)-2 = 7-2 = 5 ✓ // ไม่ใช่: 10-(3-2) = 10-1 = 9 printf("%d", 10 - 3 - 2); // 5 /* 12 / 4 / 3 */ // L→R: (12/4)/3 = 3/3 = 1 printf("%d", 12 / 4 / 3); // 1
R→L (Right-to-Left) — assignment & unary
assignment และ unary operators ประเมินจากขวาไปซ้าย
ตัวอย่าง R→L — assignment
/* a = b = c = 5 */ // R→L: c=5, b=5, a=5 ✓ int a, b, c; a = b = c = 5; // a=5, b=5, c=5 ทำแบบนี้ได้! /* !!x */ // R→L: !(!(x)) — แปลง x เป็น 0/1
กรณีที่น่าสนใจ — Chained comparison
/* ⚠️ ไม่ทำงานแบบคณิตศาสตร์ */ int x = 5; if (1 < x < 10) /* ดูเหมือนถูก */ printf("in range"); /* C ทำ: (1 < x) < 10 = (1 < 5) < 10 = 1 < 10 = true เสมอ! แม้ x=100 */
/* ✅ วิธีที่ถูกต้อง */ if (x > 1 && x < 10) printf("in range");
ผลลัพธ์ของ printf("%d", 2 * 3 + 4 * 5) คืออะไร?
และทำไมไม่ใช่ 2 * (3+4) * 5?
💡 * มี precedence สูงกว่า + ทั้งสองคู่จะคำนวณก่อน แล้วค่อย +
บทที่ 1.5 · Bug ที่เจอบ่อย
Bug จริงๆ จาก Precedence ผิด
Bug เหล่านี้ compile ผ่านหมด แต่ให้ผลผิด — เพราะ C ทำตามกฎ แต่กฎไม่ตรงกับที่เราคิด
Bug 1 — && กับ == ลำดับสับสน
/* อยากตรวจ: x==1 หรือ y==2 */ if (x == 1 || y == 2) /* ✅ ถูก */ /* ❌ พิมพ์ผิด! */ if (x == 1 | y == 2) /* | คือ Bitwise OR ไม่ใช่ Logical OR ผลอาจไม่ตรงเพราะ short-circuit ไม่ทำงาน */
Bug 2 — ++ กับ * สับสน
int arr[] = {10,20,30}; int *p = arr; printf("%d", *p++); /* * และ ++ มี precedence เท่ากัน (14) R→L: *(p++) — ใช้ค่า *p แล้วเลื่อน p พิมพ์ 10 แล้ว p ชี้ไปที่ arr[1] */ printf("%d", (*p)++); /* วงเล็บ: เพิ่มค่าที่ p ชี้ พิมพ์ 20 แล้ว arr[1]=21 */
Bug 3 — ternary กับ assignment
/* อยากได้: x = (a>b) ? a : b */ int x = 0; x = 5 > 3 ? 10 : 20; /* ✅ ทำงานถูก: x = (5>3)?10:20 = 10 เพราะ ?: precedence ต่ำกว่า > */ /* ❌ สับสน nested ternary */ x = a > b ? a : b > c ? b : c; /* R→L: a > b ? a : (b>c ? b : c) อ่านยากมาก ใส่วงเล็บดีกว่า! */
✅ กฎทองที่ต้องจำ
ถ้าไม่แน่ใจ — ใส่วงเล็บ

(a + b) * c ชัดกว่าa + b * c
(x & mask) == 0 ชัดกว่า x & mask == 0
(a > 0) && (b > 0) ชัดกว่า a > 0 && b > 0

วงเล็บไม่ทำให้ช้าลงเลย แต่ทำให้ code อ่านง่ายขึ้นมาก
บทที่ 1.5 · การใช้วงเล็บ
วงเล็บ — แก้ได้ทุกปัญหา
วงเล็บมี precedence สูงสุด (15) ใส่วงเล็บ = บอก C ให้ทำส่วนนั้นก่อนเสมอ
ก่อน-หลัง ใส่วงเล็บ
/* ไม่มีวงเล็บ — ผลอาจเซอร์ไพรส์ */ 2 + 3 * 4 // 14 5 > 3 + 1 // 5 > 4 = 1 (true) x & 0xFF == 0 // x & (0xFF==0) = x & 0 ! 1 << 2 + 1 // 1 << 3 = 8 /* ใส่วงเล็บ — ชัดเจน ถูกต้อง */ (2 + 3) * 4 // 20 (5 > 3) + 1 // 1 + 1 = 2 (x & 0xFF) == 0 // ถูกต้อง! (1 << 2) + 1 // 4 + 1 = 5
Real code — if ที่ซับซ้อน
/* ❌ อ่านยาก ต้องจำ precedence */ if (a > 0 && b > 0 || c == 0) printf("ok"); /* && ก่อน || → (a>0 && b>0) || c==0 แต่ผู้อ่านต้องรู้กฎก่อน! */
/* ✅ ชัดเจน ไม่ต้องจำ precedence */ if ((a > 0 && b > 0) || c == 0) printf("ok");
💡 เมื่อไหร่ต้องใส่วงเล็บ?
1. รวม arithmetic กับ comparison
  (a + b) > c

2. รวม bitwise กับ logical/comparison
  (x & mask) == 0

3. ternary ซ้อนกัน
  a ? b : (c ? d : e)

4. ทุกครั้งที่ไม่แน่ใจ 100%
บทที่ 1.5 · Think!
คิดก่อน เขียน
Trace ลำดับ operator ในหัว — ดูว่า C จะทำอะไรก่อน
1. ผลของ printf("%d", 3 + 4 * 2 - 1) คืออะไร?
💡 * ก่อน + และ − ทำตามลำดับ L→R หลังจากนั้น
2. if (x = 5) กับ if (x == 5) ต่างกันอย่างไร? ทั้งคู่ compile ผ่านไหม?
💡 = คือ assignment มีค่าเป็นค่าที่ assign (5 = true) — compile ผ่านแต่ทำงานผิด!
3. a = b = 10 ทำงานยังไง? ค่าของ a และ b คือเท่าไหร่?
💡 R→L: b=10 ก่อน แล้ว a = (ผลของ b=10) = 10
4. printf("%d", !0 + 1) พิมพ์อะไร?
! มี precedence สูงกว่า + ไหม?
💡 Unary ! (level 14) สูงกว่า + (level 12) → !0=1 ก่อน แล้ว 1+1
5. เขียนวงเล็บให้ชัดเจนใน:
a || b && c || d
💡 && (level 5) สูงกว่า || (level 4) → a || (b && c) || d
