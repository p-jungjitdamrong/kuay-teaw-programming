บทที่ 5 · พื้นฐาน
Memory Address — ทุกตัวแปรมี "บ้านเลขที่"
Memory คือแถวบ้านยาวๆ แต่ละบ้านมี address ตัวแปรทุกตัวที่ประกาศจะถูกจัดสรร address ใน memory
ดู address ด้วย & และ %p
int x = 42; float f = 3.14; char c = 'A'; /* & คือ address-of operator */ printf("x อยู่ที่: %p\n", &x); printf("f อยู่ที่: %p\n", &f); printf("c อยู่ที่: %p\n", &c); /* ตัวอย่าง output: x อยู่ที่: 0x7fff1000 f อยู่ที่: 0x7fff1004 c อยู่ที่: 0x7fff1008 */
💡 ทำไม scanf ต้องใช้ &
scanf("%d", &n) — ส่ง address ของ n ให้ scanf รู้ว่าจะเก็บค่าไว้ที่ไหนใน memory

ถ้าไม่ใส่ & = ส่งค่า n ไปแทน address → crash!

บทที่ 5 · Pointer
Pointer — ตัวแปรที่เก็บ Address
Pointer ไม่ได้เก็บ "ค่า" แต่เก็บ "ตำแหน่ง" ของตัวแปรอื่น — เหมือนบันทึกบ้านเลขที่ของเพื่อน
ประกาศ Pointer
int x = 42; int *p; /* p คือ pointer ชี้ไปที่ int */ p = &x; /* p เก็บ address ของ x */ printf("%p", p); /* address เช่น 0x1000 */ printf("%d", *p); /* 42 — ค่าที่ p ชี้ */ *p = 100; /* แก้ค่าผ่าน pointer */ printf("%d", x); /* 100 — x เปลี่ยนด้วย! */
* มีความหมาย 2 แบบ
ตอนประกาศ int *p;
หมายว่า "p เป็น pointer" — แค่บอก type

ตอนใช้งาน *p
หมายว่า "dereference" — เข้าไปดูค่าที่ p ชี้อยู่
MEMORY — p ชี้ไปที่ x

บทที่ 5 · Operators
& address-of · * dereference
สอง operator นี้เป็นหัวใจของ Pointer — ต้องแยกให้ออกทุกครั้งว่าใช้ในบริบทไหน
Operatorชื่อทำอะไร
&xAddress-ofให้ address ของตัวแปร x
*pDereferenceเข้าไปดู/แก้ค่าที่ p ชี้อยู่
int *pDeclarationประกาศว่า p เป็น pointer ชี้ไปที่ int
ตัวอย่างครบ
int a = 10, b = 20; int *p = &a; /* p ชี้ที่ a */ printf("%d\n", *p); /* 10 */ *p = 99; printf("%d\n", a); /* 99 */ p = &b; /* เปลี่ยน p ให้ชี้ที่ b */ printf("%d\n", *p); /* 20 */ *p = 55; printf("%d\n", b); /* 55 */
❌ Dangling Pointer — อันตรายมาก
int *p;
*p = 10; ← p ยังไม่ชี้ที่ไหน!

pointer ที่ไม่ได้ initialize จะชี้ที่ memory สุ่ม — เขียนทับข้อมูลสำคัญ crash หรือ security bug

✅ แก้: int *p = NULL; แล้วค่อย assign
⚠️ NULL pointer check
ก่อน dereference ตรวจก่อนเสมอ:

if (p != NULL) {
  printf("%d", *p);
}

dereference NULL = crash ทันที (segfault)
ถ้า int *p = &x แล้ว p++ — p จะชี้ไปที่ไหน?
💡 p++ ไม่ได้บวก 1 byte แต่บวก sizeof(int) = 4 bytes → ชี้ไปที่ตำแหน่งถัดไปใน memory
บทที่ 5 · Pass by Reference
Pass by Reference — แก้ตัวแปรนอก Function ได้
จากบทที่ 4 — pass by value แก้ข้างนอกไม่ได้ แต่ถ้าส่ง pointer (address) แทน function จะเข้าถึงตัวแปรจริงได้เลย
❌ Pass by Value — แก้ไม่ได้
void addTen(int n) { n += 10; /* แก้แค่ copy */ } int x = 5; addTen(x); printf("%d", x); /* ยังเป็น 5 */
✅ Pass by Reference — แก้ได้
void addTen(int *p) { *p += 10; /* แก้ตัวจริงผ่าน pointer */ } int x = 5; addTen(&x); /* ส่ง address ของ x */ printf("%d", x); /* 15 ✓ */
Use Case — swap สองตัวแปร
void swap(int *a, int *b) { int temp = *a; *a = *b; *b = temp; } int x = 10, y = 20; swap(&x, &y); printf("%d %d", x, y); /* 20 10 ✓ */
คืนค่าหลายตัวจาก Function
void minMax(int a[], int n, int *mn, int *mx) { int i; *mn = *mx = a[0]; for (i=1; i<n; i++) { if (a[i] < *mn) *mn = a[i]; if (a[i] > *mx) *mx = a[i]; } } int lo, hi; minMax(arr, 5, &lo, &hi);
บทที่ 5 · Pointer + Array
Pointer กับ Array — ความสัมพันธ์ที่ลึกมาก
ชื่อ Array ใน C คือ pointer ที่ชี้ไปยังช่องแรก — นี่คือเหตุผลที่ Array ส่งเข้า function แล้วแก้ค่าได้จริง
Array name = pointer ชี้ช่องแรก
int arr[] = {10, 20, 30}; int *p = arr; /* ไม่ต้องใช้ & */ /* เข้าถึงแบบ array */ printf("%d", arr[1]); /* 20 */ /* เข้าถึงแบบ pointer */ printf("%d", *(p+1)); /* 20 */ printf("%d", p[1]); /* 20 — ใช้ [] กับ pointer ได้! */ /* ทั้งสามแบบให้ผลเหมือนกัน */ /* arr[i] == *(arr+i) == *(p+i) == p[i] */
int arr[] = {10, 20, 30}
💡 arr[i] กับ *(arr+i) เหมือนกันทุกอย่าง
C แปลง arr[i] เป็น *(arr+i) โดยอัตโนมัติ ทั้งสองแบบ compile เป็น machine code เดียวกัน
บทที่ 5 · Pointer Arithmetic
Pointer Arithmetic — บวกลบ address
การบวกลบ pointer ไม่ได้บวก byte ธรรมดา — บวกตาม ขนาดของ type ที่ pointer ชี้อยู่
p++ เลื่อนกี่ byte?
int *pi; /* pi++ = +4 bytes */ char *pc; /* pc++ = +1 byte */ double*pd; /* pd++ = +8 bytes */ int arr[] = {10,20,30,40,50}; int *p = arr; printf("%d\n", *p); /* 10 */ p++; printf("%d\n", *p); /* 20 */ p += 2; printf("%d\n", *p); /* 40 */
วน loop ด้วย pointer
int arr[] = {1,2,3,4,5}; int *p; for (p = arr; p < arr+5; p++) printf("%d ", *p); // 1 2 3 4 5
⚠️ Pointer ลบกัน
int *p1 = &arr[0];
int *p2 = &arr[3];
p2 - p1 = 3 (ห่างกัน 3 elements)

ลบ pointer สองตัวจะได้จำนวน elements ไม่ใช่จำนวน byte
❌ อย่าทำ Pointer Out of Bounds
int arr[3]; int *p = arr;
p += 10; ← ชี้นอก array!
*p = 5; ← เขียนทับ memory สุ่ม

C ไม่มี bounds check — ระวัง!
บทที่ 5 · Advanced
Pointer ชนิดอื่น — const และ void
Pointer มีรูปแบบพิเศษที่ใช้บ่อยในโปรแกรมจริง — const pointer ป้องกันการแก้ค่า, void* รับได้ทุก type
const pointer — ป้องกันการแก้
/* ห้ามแก้ค่าที่ชี้อยู่ */ const int *p = &x; *p = 10; /* ❌ error */ p = &y; /* ✅ เปลี่ยนชี้ได้ */ /* ห้ามเปลี่ยนที่ชี้ */ int *const q = &x; *q = 10; /* ✅ แก้ค่าได้ */ q = &y; /* ❌ error */ /* ใช้บ่อยใน function parameter */ void print(const int *arr, int n) { /* guarantee ว่าไม่แก้ arr */ }
NULL pointer — กรณีไม่มีค่า
int *p = NULL; /* ยังไม่ชี้ที่ไหน */ /* ตรวจก่อน dereference เสมอ */ if (p != NULL) { printf("%d", *p); } else { printf("pointer is NULL\n"); }
💡 เชื่อมกับ scanf
scanf("%d", &n)

&n = pointer ชี้ไปที่ n
scanf รับ pointer แล้วใช้ dereference เขียนค่าลงไปที่ address นั้น

นี่คือเหตุผลที่ต้องใส่ & ใน scanf เสมอ
บทที่ 5 · Use Case
🏦 Use Case จริง — ระบบแลกเปลี่ยนเงิน
รวม Pointer + Pass by Reference + Array ในโปรแกรมเดียว — function คืนหลายค่าพร้อมกันผ่าน pointer
โจทย์ระบบธนาคาร
แปลงเงินบาทเป็นหลายสกุล
รับยอดเงินบาท แปลงเป็น USD, EUR, JPY พร้อมกันใน function เดียว คืนค่าผ่าน pointer 3 ตัว
ทำไมต้องใช้ Pointer
return ได้แค่ 1 ค่า
แต่ต้องการ 3 ค่า (USD, EUR, JPY)
→ ส่ง pointer 3 ตัวเข้า function
→ function เขียนค่าผ่าน *p โดยตรง
โปรแกรมแลกเงิน
#include <stdio.h> void convert(double thb, double *usd, double *eur, double *jpy) { *usd = thb / 35.5; *eur = thb / 38.2; *jpy = thb * 4.2; } int main() { double thb = 1000.0; double usd, eur, jpy; convert(thb, &usd, &eur, &jpy); printf("THB %.2f =\n", thb); printf(" USD %.2f\n", usd); printf(" EUR %.2f\n", eur); printf(" JPY %.2f\n", jpy); return 0; }
บทที่ 5 · ภาพรวม
Array + Function + Pointer — ครบทุก Concept
โปรแกรมนี้ใช้ทุกอย่างที่เรียนมา — Array เก็บข้อมูล, Function แยก logic, Pointer คืนหลายค่า
Pattern: Array + Function + Pointer
void analyze(
  int arr[], int n, // Array input
  int *min, int *max // Pointer output
)

เรียก: analyze(data, 5, &lo, &hi)
โปรแกรมวิเคราะห์อุณหภูมิ
void analyze(float arr[], int n, float *mn, float *mx, float *avg) { float sum = 0; int i; *mn = *mx = arr[0]; for (i=0; i<n; i++) { sum += arr[i]; if (arr[i] < *mn) *mn = arr[i]; if (arr[i] > *mx) *mx = arr[i]; } *avg = sum / n; } int main() { float temp[] = {32.5,35.1,29.8, 38.2,31.0}; float lo, hi, av; analyze(temp, 5, &lo, &hi, &av); printf("Min:%.1f Max:%.1f Avg:%.1f", lo, hi, av); }
บทที่ 5 · Think!
คิดก่อน เขียน
Trace ในหัว — pointer เป็น concept ที่ต้องคิดเป็นภาพ
1. ผลลัพธ์คืออะไร?
int x=5; int *p=&x; *p=10; printf("%d",x);
💡 *p=10 แก้ค่าที่ p ชี้อยู่ คือ x นั่นเอง
2. int *p = NULL; *p = 5; จะเกิดอะไร?
💡 NULL = address 0 — dereference NULL คือ segfault crash ทันที
3. int arr[]={1,2,3}; int *p=arr; p++; ตอนนี้ *p คือ?
💡 p เริ่มชี้ arr[0] พอ p++ จะชี้ arr[1] = 2
4. function swap ด้านล่างทำงานได้จริงไหม?
void swap(int *a, int *b) {
  int t=*a; *a=*b; *b=t;
}
แล้ว swap(&x, &y) จะ swap x กับ y ได้ไหม?
💡 ได้จริง! *a และ *b เข้าถึงตัวแปร x, y จริงๆ ผ่าน pointer
5. ทำไม scanf("%d", n) (ไม่มี &) ถึง crash?
💡 ส่งค่า n ไปแทน address — scanf จะพยายาม dereference ค่า n เหมือน address → crash
