บทที่ 3 · แนวคิด
ทำไมต้อง Loop?
ถ้าต้องพิมพ์เลข 1–100 เขียน printf 100 ครั้งได้ — แต่ถ้าเปลี่ยนเป็น 1–10000 จะทำยังไง?
❌ ไม่มี loop — เขียนซ้ำ
printf("%d\n", 1); printf("%d\n", 2); printf("%d\n", 3); /* ... ต้องเขียนอีก 97 บรรทัด */
✅ มี loop — เขียนครั้งเดียว
int i; for (i = 1; i <= 100; i++) { printf("%d\n", i); } /* 3 บรรทัด ทำงาน 100 ครั้ง */
🔑 Loop 3 แบบใน C
for — รู้จำนวนรอบล่วงหน้า
เช่น: วน 10 ครั้ง, วนทุก element

while — ไม่รู้จำนวนรอบ ตรวจก่อนทำ
เช่น: รับ input จนกว่า user จะพิมพ์ 0

do-while — ทำก่อน 1 ครั้ง แล้วค่อยตรวจ
เช่น: แสดง menu อย่างน้อย 1 ครั้งเสมอ
⚠️ Infinite Loop
ถ้าเงื่อนไขเป็น true ตลอด โปรแกรมจะวนไม่หยุด
while (1) { ... } — ต้องมี break ออก
มักเกิดจากลืมอัปเดต loop variable
บทที่ 3 · for loop
for loop — วนตามจำนวนที่รู้ล่วงหน้า
for loop มี 3 ส่วนใน header: init, condition, update — ทุกส่วนอยู่ที่เดียวทำให้อ่านง่าย
โครงสร้าง for loop
for (init; condition; update)

init — กำหนดค่าเริ่มต้น ทำครั้งเดียว
condition — ตรวจก่อนทุกรอบ ถ้า false หยุด
update — อัปเดตหลังจบแต่ละรอบ
1
ทำ init: i = 1ทำแค่ครั้งเดียวตอนเริ่ม
2
ตรวจ condition: i <= 5ถ้า false หยุดเลย
3
ทำ body แล้ว update: i++วนกลับไปข้อ 2
❌ Off-by-one error
i < 5 → รัน i=0,1,2,3,4 (5 รอบ)
i <= 5 → รัน i=0,1,2,3,4,5 (6 รอบ)

พิมพ์ 1–10: ใช้ i=1; i<=10 ไม่ใช่ i<10
ตัวอย่างครบทุกรูปแบบ
/* นับขึ้น 1 ถึง 5 */ for (i = 1; i <= 5; i++) printf("%d ", i); // 1 2 3 4 5 /* นับลง 5 ถึง 1 */ for (i = 5; i >= 1; i--) printf("%d ", i); // 5 4 3 2 1 /* นับทีละ 2 */ for (i = 0; i <= 10; i += 2) printf("%d ", i); // 0 2 4 6 8 10 /* sum 1 ถึง 100 */ int sum = 0; for (i = 1; i <= 100; i++) sum += i; printf("Sum=%d", sum); // 5050
บทที่ 3 · while loop
while loop — ไม่รู้จำนวนรอบ
ใช้เมื่อไม่รู้ล่วงหน้าว่าจะวนกี่รอบ — วนต่อเรื่อยๆ ตราบที่เงื่อนไขยังเป็น true
รูปแบบ while loop
while (condition) { /* body */ /* ต้องมีอะไรบางอย่างที่ทำให้ condition เปลี่ยนเป็น false มิฉะนั้น infinite loop! */ }
ตัวอย่าง — รับ input จนถูก
int n; printf("Enter positive number: "); scanf("%d", &n); while (n <= 0) { printf("Invalid! Try again: "); scanf("%d", &n); } printf("Valid: %d\n", n);
เทียบ for vs while — ผลเหมือนกัน
/* for — รู้จำนวน */ for (i=1; i<=5; i++) printf("%d ", i); /* while — เขียนเทียบเท่า */ i = 1; /* init */ while (i <= 5) { /* condition */ printf("%d ", i); i++; /* update — ต้องทำเอง! */ }
⚠️ ลืม update = Infinite Loop
i = 1;
while (i <= 5)
  printf("%d", i);
/* ลืม i++ — วนไม่หยุด! */

for loop ปลอดภัยกว่าเพราะ update อยู่ใน header ลืมยากกว่า
บทที่ 3 · do-while
do-while — ทำก่อน ตรวจทีหลัง
ต่างจาก while ตรงที่ body จะถูกทำ อย่างน้อย 1 ครั้งเสมอ ก่อนตรวจเงื่อนไข
รูปแบบ do-while
do { /* body ทำก่อน 1 ครั้ง */ } while (condition); /* ; ต้องมี! */
Use Case — Menu ที่แสดงก่อนเสมอ
int choice; do { printf("\n1.ฝาก 2.ถอน 3.ออก\n"); printf("เลือก: "); scanf("%d", &choice); if (choice == 1) deposit(); else if (choice == 2) withdraw(); } while (choice != 3); printf("Goodbye!\n");
เปรียบเทียบ 3 แบบ
for
ตรวจก่อน | รู้จำนวนล่วงหน้า
→ พิมพ์เลข, วน array

while
ตรวจก่อน | ไม่รู้จำนวน
→ รอ input ที่ถูกต้อง, อ่านไฟล์

do-while
ทำก่อน ตรวจทีหลัง | ทำ ≥ 1 ครั้ง
→ Menu system, ถามซ้ำถ้า input ผิด
ถ้าเงื่อนไขเป็น false ตั้งแต่แรก:
while(0) { printf("A"); } พิมพ์อะไร?
do { printf("B"); } while(0); พิมพ์อะไร?
💡 while ตรวจก่อน — do-while ทำก่อน 1 ครั้งเสมอ
บทที่ 3 · break / continue
break หยุด · continue ข้ามรอบ
ควบคุมการทำงาน loop ได้ละเอียดขึ้น — break ออกจาก loop ทันที, continue ข้ามรอบนี้ไปรอบถัดไป
break — ออกจาก loop ทันที
/* หาเลขแรกที่หาร 7 ลงตัว */ int i; for (i = 1; i <= 100; i++) { if (i % 7 == 0) { printf("Found: %d\n", i); break; /* ออกจาก loop ทันที */ } } // Found: 7
continue — ข้ามรอบนี้
/* พิมพ์เฉพาะเลขคี่ 1-10 */ for (i = 1; i <= 10; i++) { if (i % 2 == 0) continue; /* ข้ามเลขคู่ */ printf("%d ", i); } // 1 3 5 7 9
break ใน nested loop
break ออกจาก loop ชั้นในสุดเท่านั้น
ถ้าอยากออกทั้งสอง loop ต้องใช้ flag:

int found = 0;
for(...) {
  for(...) {
    if(cond) { found=1; break; }
  }
  if(found) break;
}
🏦 Use Case
ค้นหา transaction ผิดปกติ
วน loop ดูทุก transaction ถ้าเจอยอดเกิน limit ให้ break แจ้งเตือนทันที ไม่ต้องวนต่อ — ประหยัดเวลา
บทที่ 3 · Nested Loop
Nested Loop — loop ซ้อน loop
loop ข้างนอกรัน 1 รอบ → loop ข้างในรันครบทุกรอบ — ใช้กับตาราง 2 มิติ, pattern, และ matrix
ตัวอย่าง — ตารางสูตรคูณ
int i, j; for (i = 1; i <= 3; i++) { /* แถว */ for (j = 1; j <= 3; j++) { /* คอลัมน์ */ printf("%3d", i * j); } printf("\n"); }
1 2 3 2 4 6 3 6 9
รูปแบบสามเหลี่ยมดาว
for (i = 1; i <= 4; i++) { for (j = 1; j <= i; j++) /* ← j <= i */ printf("*"); printf("\n"); }
* ** *** ****
นับจำนวนรอบทั้งหมด
loop นอก n รอบ × loop ใน m รอบ
= ทำงาน n×m ครั้ง

เช่น 3×3 = 9 ครั้ง
100×100 = 10,000 ครั้ง

ระวัง nested loop ลึกๆ เพราะ
จำนวนรอบโตแบบ exponential
loop นอกรัน 3 รอบ, loop ในรัน i รอบ (i คือค่า loop นอก) รวมทั้งหมดกี่ครั้ง?
💡 1 + 2 + 3 = ? — รูปแบบนี้คือสามเหลี่ยมดาวที่เพิ่งเรียน
🏦 Use Case
ประมวลผลข้อมูล 2 มิติ
ระบบธนาคาร: 12 เดือน × หลาย branch — nested loop วนเดือน × branch คำนวณยอดรวมแต่ละ cell
บทที่ 3 · Patterns
Pattern ที่ใช้บ่อย — จำไว้ใช้ได้เลย
โจทย์ส่วนใหญ่เป็นการประยุกต์ pattern เหล่านี้ เข้าใจแล้วแก้โจทย์ได้ทันที
Pattern 1 — Accumulator (สะสม)
int sum = 0; /* เริ่มที่ 0 */ for (i = 1; i <= n; i++) sum += i; /* สะสมทุกรอบ */
Pattern 2 — Counter (นับ)
int count = 0; /* เริ่มที่ 0 */ for (i = 1; i <= n; i++) if (i % 2 == 0) count++; /* นับเฉพาะที่ผ่านเงื่อนไข */
Pattern 3 — Max/Min finder
int max = arr[0]; /* เริ่มจากตัวแรก */ for (i = 1; i < n; i++) if (arr[i] > max) max = arr[i]; /* อัปเดตถ้าเจอมากกว่า */
Pattern 4 — Input loop จนกว่าจะถูก
int n; do { printf("Enter 1-10: "); scanf("%d", &n); } while (n < 1 || n > 10);
Pattern 5 — Sentinel (หยุดเมื่อเจอค่าพิเศษ)
int val, sum = 0; printf("Enter (0 to stop): "); scanf("%d", &val); while (val != 0) { /* 0 คือ sentinel */ sum += val; scanf("%d", &val); } printf("Sum=%d\n", sum);
บทที่ 3 · Use Case
🏦 Use Case จริง — ระบบสรุปธุรกรรม
รวม for + while + break + accumulator ในโปรแกรมเดียว — จำลองระบบ mini bank statement
โจทย์ระบบธนาคาร
รับธุรกรรมและสรุปยอด
รับยอดธุรกรรมทีละรายการ (0=จบ)
คำนวณ: ยอดรวม, จำนวนรายการ, ค่าเฉลี่ย
แจ้งเตือนถ้ายอดใดเกิน 50,000

Pattern ที่ใช้: Sentinel + Accumulator + Counter + break
ใช้ทุก concept บทนี้
while — ไม่รู้จำนวนธุรกรรม
accumulator — สะสมยอดรวม
counter — นับจำนวนรายการ
sentinel (0) — หยุดรับข้อมูล
if + break — แจ้งเตือนทันที
โปรแกรมสรุปธุรกรรม
#include <stdio.h> int main() { int amount, count = 0; double total = 0, avg; int alert = 0; printf("Enter amount (0=done):\n"); scanf("%d", &amount); while (amount != 0) { /* sentinel */ total += amount; /* accumulate */ count++; /* count */ if (amount > 50000) { /* alert */ printf("⚠ Large txn: %d\n", amount); alert++; } scanf("%d", &amount); } if (count > 0) { avg = total / count; printf("\n=== Summary ===\n"); printf("Count : %d\n", count); printf("Total : %.2f\n", total); printf("Avg : %.2f\n", avg); printf("Alerts : %d\n", alert); } return 0; }
