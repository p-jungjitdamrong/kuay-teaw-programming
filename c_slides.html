FUNCTION — โครงสร้าง
Function 3 แบบ ที่พบบ่อย
แบ่งตามว่า "รับค่าไหม" และ "คืนค่าไหม"
① ไม่รับ ไม่คืน
void sayHello() { printf("Hello!\n"); } // เรียกใช้: sayHello();
② รับค่า ไม่คืน
void greet(char *name) { printf("Hi %s!\n", name); } // เรียกใช้: greet("Nont");
③ รับค่า และคืน
int add(int a, int b) { return a + b; } // เรียกใช้: int r = add(3, 4); // r = 7
⚠️
void แปลว่า "ไม่มี" — void function = ไม่คืนค่า, void parameter = ไม่รับค่า ถ้าลืม return ใน int function → compiler เตือน!
FUNCTION — สิ่งสำคัญ
C ส่งค่า"แบบ Copy"
เรียกว่า Pass by Value — function ได้รับ "สำเนา" ของค่า ไม่ใช่ตัวแปรจริง แก้ใน function ไม่กระทบตัวแปรนอก
🤔 THINK!
ถ้าอยากให้ function แก้ค่าตัวแปรข้างนอกได้จริงๆ — จะทำยังไง?
💡 Hint: ต้องส่ง "ตำแหน่ง" แทนการส่ง "ค่า" — นั่นคือเรื่องของ Pointer ที่จะเรียนต่อไป!
❌ แก้ใน function ไม่ได้
void tryChange(int n) { n = 99; // แก้แค่ "copy" } int main() { int x = 10; tryChange(x); printf("%d", x); // ยังเป็น 10! }
✅ Prototype ถูกต้อง
int add(int a, int b); // prototype int main() { printf("%d", add(3,4)); } int add(int a, int b) { return a + b; }

Pointer
ตัวแปรที่เก็บ "ตำแหน่ง" แทนค่า — เข้าใจ Pointer คือเข้าใจว่า C ทำงานกับ Memory ยังไง
Address ใน Memory
& และ * operators
Pass by Reference
Pointer + Array + Use Case
POINTER — พื้นฐาน
ตัวแปรทุกตัวมี "บ้านเลขที่"
Memory คือแถวบ้าน แต่ละบ้านมี address ตัวแปรทุกตัวอาศัยอยู่ที่ address ใดที่หนึ่ง Pointer คือตัวเก็บ "บ้านเลขที่" นั้น
&
Address-of operator
&x = "บ้านเลขที่ของ x" คือตำแหน่งใน memory
*
Dereference operator
*p = "เข้าไปในบ้านที่ p ชี้อยู่" แล้วดึงค่าออกมา
*p
ตอนประกาศ vs ตอนใช้งาน
int *p; แปลว่า "p เป็น pointer" ไม่ใช่ dereference
POINTER TO VARIABLE
int x = 42; int *p; // ประกาศ pointer p = &x; // p = address ของ x printf("%d", *p); // 42 (ค่าที่ p ชี้) *p = 100; // แก้ค่าผ่าน pointer printf("%d", x); // 100 (x เปลี่ยนตาม!)
POINTER — การประยุกต์ใช้
Pass by Reference — แก้ค่านอก function ได้
แทนที่จะส่งค่า → ส่ง address แทน function จะได้แก้ตัวแปรจริงๆ ได้
❌ Pass by Value (แก้ไม่ได้)
void addOne(int n) { n++; // แก้แค่ copy } int x = 5; addOne(x); printf("%d", x); // ยังเป็น 5
✅ Pass by Reference (แก้ได้)
void addOne(int *p) { (*p)++; // แก้ตัวจริง } int x = 5; addOne(&x); // ส่ง address! printf("%d", x); // x = 6 ✓
💡 เชื่อมกับที่รู้แล้ว
scanf ใช้ Pointer มาตลอด!
scanf("%d", &n) — ที่ต้องใส่ & เพราะ scanf ต้องรู้ว่าจะเก็บค่าที่ตำแหน่ง memory ไหน นั่นคือมันรับ pointer เบื้องหลังทุกครั้งที่เราใช้ scanf!
POINTER + ARRAY
Array ชื่อคือ Pointer
ใน C ชื่อ array คือ address ของช่องแรก ดังนั้น array กับ pointer จึงเชื่อมกันแน่น
ARRAY IN MEMORY
POINTER ARITHMETIC
int arr[] = {10, 20, 30}; int *p = arr; printf("%d", *p); // 10 (arr[0]) printf("%d", *(p+1)); // 20 (arr[1]) printf("%d", *(p+2)); // 30 (arr[2]) /* วน loop ด้วย pointer */ for (i = 0; i < 3; i++) { printf("%d ", *(p+i)); }
⚠️
Dangling Pointer! อย่าใช้ pointer ที่ยังไม่ได้กำหนดค่า — ชี้ไปที่ memory สุ่ม อาจ crash หรือให้ค่าผิด ควรตั้งค่า p = NULL ถ้ายังไม่พร้อมใช้

» "7 days" → ต้องการเก็บ 7 ค่า ใช้ int sales[7]
» "display all" → วน for loop พิมพ์ทุก index
» "highest day" → ต้องการ หาค่าสูงสุด + จำ index

int sales[7] เก็บข้อมูล 7 วัน

int sales[7] = {500,320,780,410,960,230,550}; int total = 0, maxVal = sales[0], maxIdx = 0, i; for (i = 0; i < 7; i++) { total += sales[i]; // สะสมรวม if (sales[i] > maxVal) { maxVal = sales[i]; maxIdx = i; } } printf("Best day: Day %d = %d baht\n", maxIdx+1, maxVal); printf("Total: %d baht\n", total);

float calcBMI(float w, float h)

<18.5 / 18.5–25 / 25–30 / >30

void convert(int thb, float *usd, float *eur)

*eur = thb / 38.0

void convert(int thb, float *usd, float *eur) { *usd = thb / 35.0; // แก้ค่า usd ใน main ผ่าน pointer *eur = thb / 38.0; // แก้ค่า eur ใน main ผ่าน pointer } int main() { float u, e; convert(1000, &u, &e); // ส่ง address ของ u และ e printf("USD: %.2f EUR: %.2f\n", u, e); return 0; }

A weather station records temperature readings for 7 days. Write a program that stores the readings in an array, then calls a single function analyzeTemp to find and return both the minimum and maximum temperature. After the call, the main program prints both values and decides whether to issue a "Heat Warning" if max > 38°C.
🔍 วิเคราะห์ทีละจุด

🛠 ผสมทั้ง 3 เทคนิค
float temp[7] — Array
void analyzeTemp(...) — Function
float *minT, *maxT — Pointer
if (*maxT > 38) — Warning
void analyzeTemp(float arr[], int size, float *minT, float *maxT) { *minT = *maxT = arr[0]; int i; for (i = 1; i < size; i++) { if (arr[i] < *minT) *minT = arr[i]; if (arr[i] > *maxT) *maxT = arr[i]; } } // เรียกใช้ใน main: float mn, mx; analyzeTemp(temp, 7, &mn, &mx); if (mx > 38) printf("⚠ Heat Warning!\n");

🏆CHALLENGE
A bank stores the last 10 transaction amounts for a customer. Write a function that finds both the largest deposit and the total balance, returning both values to main so the program can print "Low Balance Warning" if total < 1000.
💡 ต้องใช้อะไรบ้าง? → "10 transactions" = ? | "both values" = ? | "Low Balance" = ?
🏆 FINAL CHALLENGE — เฉลย
Bank Transaction วิเคราะห์ + เฉลย
❓ "10 transactions" = ?
ต้องเก็บ 10 ค่า
→ int txn[10]
ใช้ Array เก็บ + for loop วน
❓ "both values" = ?
คืน 2 ค่า — return ได้แค่ 1
→ int *maxDep, int *total
Pass by Reference ด้วย Pointer
❓ "Low Balance" = ?
ตรวจหลังเรียก function
→ if (total < 1000)
ใช้ *total ที่แก้จาก function
✅ FULL SOLUTION
#include <stdio.h> /* Array + Function + Pointer ครบ */ void analyze(int txn[], int size, int *maxDep, int *total) { *maxDep = txn[0]; *total = 0; int i; for (i = 0; i < size; i++) { *total += txn[i]; // สะสม balance if (txn[i] > *maxDep) *maxDep = txn[i]; // อัปเดต max } } int main() { int txn[10] = {200,-50,500,-30,150, -200,80,-100,300,-400}; int maxDep, total; analyze(txn, 10, &maxDep, &total); // ส่ง address! printf("Largest deposit : %d baht\n", maxDep); printf("Total balance : %d baht\n", total); if (total < 1000) printf("⚠ Low Balance Warning!\n"); return 0; }

ต้องการคืน 2 ค่าพร้อมกัน (maxDep และ total) — return ได้แค่ 1 จึงส่ง &maxDep, &total เป็น address function แก้ค่าผ่าน *maxDep และ *total โดยตรง
④ RESULT
กลับมาใน main ตัวแปร total ถูกแก้แล้ว ตรวจ if (total < 1000) แล้วพิมพ์ Warning output: Largest deposit: 500 | Total: 450 | ⚠ Low Balance Warning!
