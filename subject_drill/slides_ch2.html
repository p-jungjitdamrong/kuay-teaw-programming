หาค่ามัธยฐาน (Median)

โจทย์: รับตัวเลข n ตัว แล้วหาค่ามัธยฐาน (ค่ากลางหลังเรียงลำดับ)
• ถ้า n เป็นเลขคี่ → มัธยฐานคือตัวกลางตัวเดียว
• ถ้า n เป็นเลขคู่ → มัธยฐานคือค่าเฉลี่ยของสองตัวกลาง
ห้ามใช้ฟังก์ชัน sort สำเร็จรูป — ต้องเขียนการเรียงเอง
ตัวอย่าง 1 — จำนวนคี่
n = 5 ข้อมูล: 7 2 9 4 5 เรียงแล้ว: 2 4 5 7 9 มัธยฐาน = 5
ตัวอย่าง 2 — จำนวนคู่
n = 4 ข้อมูล: 8 3 1 6 เรียงแล้ว: 1 3 6 8 มัธยฐาน = (3+6)/2 = 4.50
หา pattern ของตำแหน่งกลาง
nindex (เริ่ม 0)ตัวกลาง
5 (คี่)0 1 2 3 4a[2] = a[n//2]
7 (คี่)0 1 2 3 4 5 6a[3] = a[n//2]
4 (คู่)0 1 2 3(a[1]+a[2])/2
6 (คู่)0 1 2 3 4 5(a[2]+a[3])/2
 
# สูตรที่ได้:
# คี่ → a[n//2]
# คู่ → (a[n//2 - 1] + a[n//2]) / 2
⏸️ หยุดลองเอง — โจทย์นี้มี 2 ส่วน: เรียงข้อมูล (ใช้ selection/bubble sort) + เลือกตัวกลาง (แยกเคสคี่/คู่) · ตารางให้สูตรมาแล้ว
โจทย์ที่ 1 · วิธีคิด

STEP 2–3: เรียงข้อมูลแล้วเลือกตัวกลาง

STEP 2 — แตกปัญหา
  ① รับ n และข้อมูล n ตัว
  ② เรียงข้อมูลจากน้อยไปมาก
  ③ เช็คว่า n คี่หรือคู่
  ④ เลือก/คำนวณตัวกลางตามเคส
  ⑤ แสดงผล

  ขั้น ② ซอยต่อ — Selection Sort:
    วน i จาก 0 ถึง n−2
      หา index ของตัวน้อยสุดใน a[i..n−1]
      สลับกับ a[i]
  ← สิ่งที่ทำซ้ำ 2 ชั้น = ลูปซ้อน
Selection Sort — ทำด้วยมือ
[7, 2, 9, 4, 5]
i=0: min ใน [7,2,9,4,5] = 2 (idx1) → สลับ
     [2, 7, 9, 4, 5]
i=1: min ใน [7,9,4,5] = 4 (idx3) → สลับ
     [2, 4, 9, 7, 5]
i=2: min ใน [9,7,5] = 5 (idx4) → สลับ
     [2, 4, 5, 7, 9]
i=3: min ใน [7,9] = 7 (ตัวเอง) → ไม่สลับ
     [2, 4, 5, 7, 9] ✓ เรียงเสร็จ
STEP 3 — ผังงาน
▱ รับ n และ a[0..n-1] │ ▭ i = 0 ▼ ┌─▶ ◇ i <= n-2 ──เท็จ──┐ │ ▭ min = i │ │ ▭ j = i+1 │ │ ▼ │ │ ┌▶ ◇ j <= n-1 ─เท็จ┐ │ │ │ ◇ a[j] < a[min] │ │ │ │ จริง▭ min = j │ │ │ └──── ▭ j++ │ │ │ ┌────────────┘ │ │ ▭ สลับ a[i] ↔ a[min] │ └──── ▭ i++ │ ┌──────────────┘ ◇ n % 2 == 1 จริง│ │เท็จ ▭ med=a[n/2] │ ▭ med=(a[n/2-1] │ +a[n/2])/2 └──┬──┘ ▽ แสดง med
เคสขอบ: n=1 → ลูป sort ไม่ทำงาน (i=0 > n−2=−1) และ n//2 = 0 → med = a[0] ✓ ถูกต้องอัตโนมัติ · n=0 → ต้องเช็คแยก (ไม่มีข้อมูลให้หาค่ากลาง)
โจทย์ที่ 1 · เฉลย

STEP 4: เขียนโค้ด — C และ Python

C 
#include <stdio.h>

int main() {
    int n, i, j, min, tmp;
    int a[100];
    double med;

    scanf("%d", &n);
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    /* Selection Sort */
    for (i = 0; i < n - 1; i++) {
        min = i;
        for (j = i + 1; j < n; j++)
            if (a[j] < a[min]) min = j;
        if (min != i) {          /* สลับ */
            tmp = a[i]; a[i] = a[min]; a[min] = tmp;
        }
    }

    if (n % 2 == 1)
        med = a[n/2];
    else
        med = (a[n/2 - 1] + a[n/2]) / 2.0;
    /*                             ↑ 2.0 สำคัญ! */

    printf("มัธยฐาน = %.2f\n", med);
    return 0;
}
PYTHON 
n = int(input())
a = []
for i in range(n):
    a.append(int(input()))

# Selection Sort
for i in range(n - 1):
    mn = i
    for j in range(i + 1, n):
        if a[j] < a[mn]:
            mn = j
    if mn != i:
        a[i], a[mn] = a[mn], a[i]    # สลับใน 1 บรรทัด!

if n % 2 == 1:
    med = a[n//2]
else:
    med = (a[n//2 - 1] + a[n//2]) / 2

print(f"มัธยฐาน = {med:.2f}")
# 2 จุดต่างที่สำคัญ:

# ① สลับค่า
#   C:      tmp = a; a = b; b = tmp;
#   Python: a, b = b, a      ← สั้นกว่ามาก

# ② หาร
#   C:      / 2.0 (ไม่งั้น int หาร int ปัดทิ้ง!)
#   Python: / 2   (ได้ float อยู่แล้ว)
#           แต่ index ต้องใช้ // (ต้องเป็น int)
กับดักที่เจ็บที่สุดของโจทย์นี้: ใน C ถ้าเขียน (a[1]+a[2]) / 2 จะได้ผลปัดทิ้ง (int หาร int) — ต้องเขียน / 2.0 · ส่วน Python ต้องใช้ // ตอนคำนวณ index เพราะ index ต้องเป็น int
โจทย์ที่ 2 · โจทย์ + ตัวอย่าง

ลบค่าซ้ำออกจากชุดข้อมูล

โจทย์: รับตัวเลข n ตัว แล้วแสดงเฉพาะค่าที่ไม่ซ้ำกัน โดยเรียงตามลำดับที่ปรากฏครั้งแรก
พร้อมบอกว่ามีค่าที่ไม่ซ้ำทั้งหมดกี่ตัว · ห้ามใช้ set()
ตัวอย่าง 1
n = 7 ข้อมูล: 3 5 3 8 5 1 8 ผลลัพธ์: 3 5 8 1 จำนวนค่าไม่ซ้ำ: 4
ตัวอย่าง 2 (เคสขอบ)
n = 4 ข้อมูล: 7 7 7 7 ผลลัพธ์: 7 จำนวนค่าไม่ซ้ำ: 1
ทำด้วยมือ — คิดยังไง
ข้อมูล: 3 5 3 8 5 1 8

หยิบ 3 → ยังไม่มีในผลลัพธ์ → เพิ่ม  [3]
หยิบ 5 → ยังไม่มี → เพิ่ม          [3,5]
หยิบ 3 → มีแล้ว! → ข้าม            [3,5]
หยิบ 8 → ยังไม่มี → เพิ่ม          [3,5,8]
หยิบ 5 → มีแล้ว → ข้าม             [3,5,8]
หยิบ 1 → ยังไม่มี → เพิ่ม          [3,5,8,1]
หยิบ 8 → มีแล้ว → ข้าม             [3,5,8,1]

สิ่งที่ทำซ้ำ:
  "เช็คว่าตัวนี้มีในผลลัพธ์แล้วหรือยัง"
  ← การเช็คนี้เองก็ต้องวนดูทุกตัวในผลลัพธ์
  → ลูปซ้อน 2 ชั้น
⏸️ หยุดลองเอง — จุดสำคัญคือ "เช็คว่ามีแล้วหรือยัง" ต้องใช้ลูปอีกชั้นหนึ่ง + ต้องมี flag บอกว่าเจอหรือไม่
โจทย์ที่ 2 · วิธีคิด + เฉลย

ใช้ flag บอกว่า "เจอแล้วหรือยัง"

STEP 2 — แตกปัญหา
  ① สร้าง array ผลลัพธ์ว่าง (count = 0)
  ② วนทุกตัวในข้อมูลเดิม (i)
       ③ ตั้ง flag found = 0
       ④ วนทุกตัวในผลลัพธ์ (j)
            ถ้าเจอค่าเท่ากัน → found = 1
       ⑤ ถ้า found == 0 → เพิ่มลงผลลัพธ์
  ⑥ แสดงผลลัพธ์ + count

  ← pattern "ตั้ง flag → วนเช็ค → ตัดสินใจ"
    ใช้ได้กับโจทย์ "มีอยู่แล้วไหม" ทุกข้อ
▭ count = 0 ▭ i = 0 ▼ ┌▶◇ i < n ──เท็จ──┐ │ ▭ found = 0 │ │ ▭ j = 0 │ │ ▼ │ │ ┌▶◇ j < count ─┐│ │ │ ◇ res[j]==a[i]││ │ │ จริง▭found=1 ││ │ └── ▭ j++ ││ │ ┌─────────┘│ │ ◇ found == 0 │ │ จริง▭ res[count]│ │ = a[i] │ │ ▭ count++ │ └──── ▭ i++ │ ┌──────────┘ ▽ แสดง res, count
C 
int n, i, j, count = 0, found;
int a[100], res[100];

scanf("%d", &n);
for (i = 0; i < n; i++) scanf("%d", &a[i]);

for (i = 0; i < n; i++) {
    found = 0;                    /* ตั้ง flag */
    for (j = 0; j < count; j++)
        if (res[j] == a[i]) {
            found = 1;
            break;               /* เจอแล้ว ไม่ต้องหาต่อ */
        }
    if (!found) {                 /* ยังไม่มี → เพิ่ม */
        res[count] = a[i];
        count++;
    }
}
for (i = 0; i < count; i++) printf("%d ", res[i]);
printf("\nจำนวนค่าไม่ซ้ำ: %d\n", count);
PYTHON 
n = int(input())
a = [int(input()) for _ in range(n)]

res = []
for x in a:
    found = False
    for y in res:
        if y == x:
            found = True
            break
    if not found:
        res.append(x)

print(*res)
print(f"จำนวนค่าไม่ซ้ำ: {len(res)}")

# Python เขียนสั้นได้ด้วย in
# for x in a:
#     if x not in res:      ← in เช็คให้เลย
#         res.append(x)
Pattern "flag" ใช้ได้ทุกโจทย์แนว "มีไหม/ใช่ไหม": ตั้ง flag เป็นค่าเริ่มต้น → วนเช็ค → เจอแล้วเปลี่ยน flag + break → ตัดสินใจจาก flag · โจทย์จำนวนเฉพาะก็ใช้ pattern นี้
โจทย์ที่ 3 · โจทย์ + ตัวอย่าง

หาคู่ที่รวมกันได้ผลลัพธ์ที่ต้องการ

โจทย์: รับตัวเลข n ตัว และค่าเป้าหมาย target
หาว่ามีคู่ของตัวเลข (2 ตัวที่ต่างตำแหน่งกัน) ที่บวกกันได้เท่ากับ target หรือไม่
ถ้ามี ให้แสดงทุกคู่ที่เป็นไปได้ (ไม่นับคู่ซ้ำ เช่น (2,7) กับ (7,2) นับเป็นคู่เดียว)
ตัวอย่าง 1
n = 5, target = 9 ข้อมูล: 2 7 11 4 5 พบคู่: (2, 7) (4, 5) รวม 2 คู่
ตัวอย่าง 2
n = 4, target = 100 ข้อมูล: 1 2 3 4 ไม่พบคู่ที่รวมกันได้ 100
คิดยังไงไม่ให้นับซ้ำ
ถ้าวนแบบนี้:
  for i in 0..n-1:
      for j in 0..n-1:
  → จะได้ (2,7) และ (7,2) ซ้ำกัน
    และยังจับคู่ตัวเองด้วย (2,2)!

ทางแก้: ให้ j เริ่มที่ i+1 เสมอ
  for i in 0..n-2:
      for j in i+1..n-1:
  → เทียบแต่ละคู่แค่ครั้งเดียว ✓
    และไม่จับคู่ตัวเองด้วย ✓
i=0: j = 1,2,3,4 (5-1 คู่) i=1: j = 2,3,4 (5-2 คู่) i=2: j = 3,4 i=3: j = 4 ← ครอบคลุมทุกคู่ ไม่ซ้ำ
⏸️ หยุดลองเอง — เทคนิค j = i+1 คือหัวใจของโจทย์ "จับคู่" ทั้งหมด · จำนวนคู่ทั้งหมด = n(n−1)/2 (สูตร Gauss จากวิชา Big-O!)
โจทย์ที่ 3 · วิธีคิด + เฉลย

ลูปซ้อนแบบ j = i+1

▱ รับ n, target, a[] │ ▭ found = 0 ▭ i = 0 ▼ ┌─▶ ◇ i <= n-2 ──เท็จ──┐ │ ▭ j = i + 1 │ │ ▼ │ │ ┌─▶ ◇ j <= n-1 ─เท็จ┐│ │ │ ◇ a[i]+a[j] ││ │ │ == target ││ │ │ จริง│ ││ │ │ ▽ แสดง (a[i],a[j])││ │ │ ▭ found++ ││ │ └──── ▭ j++ ││ │ ┌────────────┘│ └───── ▭ i++ │ ┌─────────────┘ ◇ found == 0 จริง▽ "ไม่พบคู่" เท็จ▽ "รวม found คู่"
C 
int n, target, i, j, found = 0;
int a[100];

scanf("%d %d", &n, &target);
for (i = 0; i < n; i++) scanf("%d", &a[i]);

for (i = 0; i < n - 1; i++) {
    for (j = i + 1; j < n; j++) {    /* j เริ่มที่ i+1! */
        if (a[i] + a[j] == target) {
            printf("พบคู่: (%d, %d)\n", a[i], a[j]);
            found++;
        }
    }
}
if (found == 0)
    printf("ไม่พบคู่ที่รวมกันได้ %d\n", target);
else
    printf("รวม %d คู่\n", found);
PYTHON 
n, target = map(int, input().split())
a = [int(input()) for _ in range(n)]

found = 0
for i in range(n - 1):
    for j in range(i + 1, n):      # j เริ่มที่ i+1
        if a[i] + a[j] == target:
            print(f"พบคู่: ({a[i]}, {a[j]})")
            found += 1

if found == 0:
    print(f"ไม่พบคู่ที่รวมกันได้ {target}")
else:
    print(f"รวม {found} คู่")
# ประสิทธิภาพ: O(n²)
# วิธีเร็วกว่า (O(n)) ใช้ dict เก็บ
# ค่าที่เคยเจอ แล้วเช็ค target−x
# ← ลองเขียนดูเป็นการบ้าน!
เทคนิค j = i+1 ใช้ได้กับโจทย์ตระกูล "จับคู่" ทั้งหมด: หาคู่ที่ผลรวมเท่ากับ X · หาคู่ที่ห่างกันมากสุด · เช็คว่ามีค่าซ้ำไหม — ทุกข้อใช้โครงเดียวกัน
