C 
#include <stdio.h>
#include <string.h>

int fee(int h) {                 // ค่าจอดแบบขั้นบันได
    int total = 0;
    if (h > 1) {                    // ชม. 2–3
        int n = (h < 3 ? h : 3) - 1;
        total += n * 20;
    }
    if (h > 3) total += (h - 3) * 30;   // ชม. 4 ขึ้นไป
    return total;
}

int main() {
    int N;
    scanf("%d", &N);

    char plate[101][32];       // อาร์เรย์ขนาน
    int  inTime[101];
    for (int i = 1; i <= N; i++) plate[i][0] = '\0';  // ว่าง

    char cmd[16], p[32];
    int t, income = 0;

    while (scanf("%s", cmd) == 1 && strcmp(cmd, "END") != 0) {
        scanf("%s %d", p, &t);

        if (strcmp(cmd, "IN") == 0) {
            int slot = -1;
            for (int i = 1; i <= N; i++)
                if (plate[i][0] == '\0') { slot = i; break; }

            if (slot == -1) {
                printf("%s ลานเต็ม ไม่มีช่องว่าง\n", p);
            } else {
                strcpy(plate[slot], p);
                inTime[slot] = t;
                printf("%s เข้าช่อง %d\n", p, slot);
            }
        }
        else if (strcmp(cmd, "OUT") == 0) {
            int slot = -1;
            for (int i = 1; i <= N; i++)
                if (strcmp(plate[i], p) == 0) { slot = i; break; }

            if (slot == -1) {
                printf("ไม่พบรถทะเบียน %s\n", p);
            } else {
                int h = t - inTime[slot];
                int money = fee(h);
                income += money;
                printf("%s ออกจากช่อง %d จอด %d ชม. = %d บาท\n",
                       p, slot, h, money);
                plate[slot][0] = '\0';      // คืนช่อง
            }
        }
    }

    printf("--------------------\n");
    printf("รายได้รวม = %d บาท\n", income);
    printf("ช่องที่ยังมีรถ:");
    for (int i = 1; i <= N; i++)
        if (plate[i][0] != '\0') printf(" %d", i);
    printf("\n");
    return 0;
}

def fee(h):
    total = 0
    if h > 1:
        total += (min(h, 3) - 1) * 20
    if h > 3:
        total += (h - 3) * 30
    return total

N = int(input())
slots = {i: None for i in range(1, N + 1)}   # ช่อง → (ทะเบียน, เวลาเข้า)
income = 0

while True:
    parts = input().split()
    if parts[0] == "END":
        break
    cmd, p, t = parts[0], parts[1], int(parts[2])

    if cmd == "IN":
        # ช่องว่างหมายเลขน้อยสุด
        empty = [i for i in slots if slots[i] is None]
        if not empty:
            print(f"{p} ลานเต็ม ไม่มีช่องว่าง")
        else:
            s = min(empty)
            slots[s] = (p, t)
            print(f"{p} เข้าช่อง {s}")

    elif cmd == "OUT":
        s = None
        for i, v in slots.items():
            if v is not None and v[0] == p:
                s = i
                break
        if s is None:
            print(f"ไม่พบรถทะเบียน {p}")
        else:
            h = t - slots[s][1]
            money = fee(h)
            income += money
            print(f"{p} ออกจากช่อง {s} จอด {h} ชม. = {money} บาท")
            slots[s] = None

print("-" * 20)
print(f"รายได้รวม = {income} บาท")
busy = [str(i) for i in slots if slots[i]]
print("ช่องที่ยังมีรถ:", " ".join(busy) if busy else "(ไม่มี)")

ระบบเรียกคิวร้านอาหาร — แบ่งโต๊ะตามจำนวนคน

โจทย์: ร้านมีโต๊ะ 3 ขนาด S (นั่งได้ 2), M (นั่งได้ 4), L (นั่งได้ 8) รับจำนวนโต๊ะแต่ละขนาดมาก่อน แล้วรับเหตุการณ์จนเจอ END
ADD <หมายเลขกลุ่ม> <จำนวนคน> — กลุ่มมาถึง
FREE <ขนาดโต๊ะ> — มีโต๊ะขนาดนั้นว่าง 1 ตัว
กติกาจัดโต๊ะ: ให้โต๊ะขนาดเล็กที่สุดที่นั่งพอ ถ้าขนาดนั้นเต็ม ให้เข้าคิวของขนาดนั้น (ไม่เลื่อนไปใช้โต๊ะใหญ่)
เมื่อมีโต๊ะว่าง ให้เรียกคิวหน้าสุดของขนาดนั้นเข้าทันที
ตัวอย่างการทำงาน
input: 1 1 1 ADD 101 2 ADD 102 3 ADD 103 2 ADD 104 6 ADD 105 9 FREE S FREE M END output: กลุ่ม 101 (2 คน) → โต๊ะ S กลุ่ม 102 (3 คน) → โต๊ะ M กลุ่ม 103 (2 คน) → คิว S (รอ 1) กลุ่ม 104 (6 คน) → โต๊ะ L กลุ่ม 105 (9 คน) → เกินความจุ รับไม่ได้ โต๊ะ S ว่าง → เรียกกลุ่ม 103 โต๊ะ M ว่าง → ไม่มีคิวรอ -------------------- คิวคงเหลือ S:- M:- L:-
หลักการของข้อนี้ ① เลือกขนาดโต๊ะ = ตารางค่าคงที่ ไม่ใช่ if ซ้อน
cap[] = {2, 4, 8} แล้ววนหาตัวแรกที่ คน <= cap[i]
→ เพิ่มขนาดโต๊ะใหม่ในอนาคตแค่เติมตัวเลข ไม่ต้องแก้ตรรกะ

② คิวแยกกัน 3 ชุด ไม่ใช่คิวเดียว
เพราะโจทย์บอกว่ารอเฉพาะขนาดของตัวเอง → ใช้คิว 2 มิติ q[3][MAX] พร้อม head[3], tail[3]

③ ตัวนับโต๊ะว่างต่อขนาด
free[i] ลดตอนจัดโต๊ะ เพิ่มตอน FREE — แต่ถ้ามีคิวรอ อย่าเพิ่ง +1 เพราะโต๊ะถูกใช้ต่อทันที

④ เคสปฏิเสธ
คนเกิน 8 = ไม่มีขนาดไหนรับได้ → ต้องแยกออกมาก่อนเข้าตรรกะจัดโต๊ะ
ADD <กลุ่ม> <คน> │ ▼ หา i ตัวแรกที่ คน <= cap[i] │ ┌───┴────────────┐ ไม่เจอ เจอ i │ │ ปฏิเสธ free[i] > 0 ? ┌─────┴─────┐ ใช่ ไม่ │ │ free[i]−− เข้าคิว i นั่งเลย
⏸️ ลองทำเองก่อน — เคสที่ต้องผ่าน: กลุ่ม 9 คน (ปฏิเสธ) · FREE ตอนไม่มีคิว (โต๊ะว่างเพิ่ม) · FREE ตอนมีคิว (โต๊ะว่างเท่าเดิม)

C 
#include <stdio.h>
#include <string.h>
#define MAXQ 1000

int main() {
    char name[3] = {'S', 'M', 'L'};
    int cap[3]  = {2, 4, 8};       // ความจุแต่ละขนาด
    int freeT[3];
    scanf("%d %d %d", &freeT[0], &freeT[1], &freeT[2]);

    int q[3][MAXQ];                // คิวแยก 3 ชุด
    int head[3] = {0,0,0}, tail[3] = {0,0,0};

    char cmd[16];
    while (scanf("%s", cmd) == 1 && strcmp(cmd, "END") != 0) {

        if (strcmp(cmd, "ADD") == 0) {
            int g, n;
            scanf("%d %d", &g, &n);

            int k = -1;                // ขนาดเล็กสุดที่นั่งพอ
            for (int i = 0; i < 3; i++)
                if (n <= cap[i]) { k = i; break; }

            if (k == -1) {
                printf("กลุ่ม %d (%d คน) → เกินความจุ รับไม่ได้\n", g, n);
            } else if (freeT[k] > 0) {
                freeT[k]--;
                printf("กลุ่ม %d (%d คน) → โต๊ะ %c\n", g, n, name[k]);
            } else {
                q[k][tail[k]++] = g;
                printf("กลุ่ม %d (%d คน) → คิว %c (รอ %d)\n",
                       g, n, name[k], tail[k] - head[k]);
            }
        }
        else if (strcmp(cmd, "FREE") == 0) {
            char s[4];
            scanf("%s", s);
            int k = (s[0] == 'S') ? 0 : (s[0] == 'M') ? 1 : 2;

            if (head[k] < tail[k]) {          // มีคิวรอ
                int g = q[k][head[k]++];
                printf("โต๊ะ %c ว่าง → เรียกกลุ่ม %d\n", name[k], g);
            } else {                          // ไม่มีคิว
                freeT[k]++;
                printf("โต๊ะ %c ว่าง → ไม่มีคิวรอ\n", name[k]);
            }
        }
    }

    printf("--------------------\n");
    printf("คิวคงเหลือ");
    for (int i = 0; i < 3; i++) {
        printf(" %c:", name[i]);
        if (head[i] == tail[i]) printf("-");
        for (int j = head[i]; j < tail[i]; j++) printf("%d ", q[i][j]);
    }
    printf("\n");
    return 0;
}
PYTHON 
from collections import deque

name = ['S', 'M', 'L']
cap  = [2, 4, 8]
idx  = {'S': 0, 'M': 1, 'L': 2}

free_t = list(map(int, input().split()))
queues = [deque(), deque(), deque()]     # คิวแยก 3 ชุด

while True:
    parts = input().split()
    if parts[0] == "END":
        break

    if parts[0] == "ADD":
        g, n = int(parts[1]), int(parts[2])

        k = None                          # ขนาดเล็กสุดที่นั่งพอ
        for i in range(3):
            if n <= cap[i]:
                k = i
                break

        if k is None:
            print(f"กลุ่ม {g} ({n} คน) → เกินความจุ รับไม่ได้")
        elif free_t[k] > 0:
            free_t[k] -= 1
            print(f"กลุ่ม {g} ({n} คน) → โต๊ะ {name[k]}")
        else:
            queues[k].append(g)
            print(f"กลุ่ม {g} ({n} คน) → คิว {name[k]} (รอ {len(queues[k])})")

    elif parts[0] == "FREE":
        k = idx[parts[1]]
        if queues[k]:                     # มีคิวรอ → ใช้โต๊ะต่อทันที
            g = queues[k].popleft()
            print(f"โต๊ะ {name[k]} ว่าง → เรียกกลุ่ม {g}")
        else:
            free_t[k] += 1
            print(f"โต๊ะ {name[k]} ว่าง → ไม่มีคิวรอ")

print("-" * 20)
parts = []
for i in range(3):
    body = " ".join(map(str, queues[i])) if queues[i] else "-"
    parts.append(f"{name[i]}:{body}")
print("คิวคงเหลือ", " ".join(parts))

        sold[i] = 0;
    }

    char cmd[16], s[64];
    int n, income = 0;

    while (scanf("%s", cmd) == 1 && strcmp(cmd, "END") != 0) {

        if (strcmp(cmd, "SELL") == 0) {
            scanf("%s %d", s, &n);
            int i = findItem(s);
            if (i == -1) { printf("ไม่มีสินค้า %s\n", s); continue; }

            int give = (n < qty[i]) ? n : qty[i];   // ขายเท่าที่มี
            qty[i]  -= give;
            sold[i] += give;
            income  += give * price[i];

            if (give < n)
                printf("ของไม่พอ! ขาย %s ได้แค่ %d ชิ้น = %d บาท\n",
                       s, give, give * price[i]);
            else
                printf("ขาย %s %d ชิ้น = %d บาท\n",
                       s, give, give * price[i]);
        }
        else if (strcmp(cmd, "ADD") == 0) {
            scanf("%s %d", s, &n);
            int i = findItem(s);
            if (i == -1) { printf("ไม่มีสินค้า %s\n", s); continue; }
            qty[i] += n;
            printf("เติม %s เป็น %d ชิ้น\n", s, qty[i]);
        }
        else if (strcmp(cmd, "REPORT") == 0) {
            printf("[REPORT] ใกล้หมด:");
            int found = 0;
            for (int i = 0; i < M; i++)
                if (qty[i] < 5) { printf(" %s(%d)", nameP[i], qty[i]); found = 1; }
            if (!found) printf(" (ไม่มี)");
            printf("\n");
        }
    }

    int best = 0;                 // จำ index ไม่ใช่จำค่า
    for (int i = 1; i < M; i++)
        if (sold[i] > sold[best]) best = i;

    printf("--------------------\n");
    printf("ยอดขายรวม = %d บาท\n", income);
    printf("ขายดีที่สุด = %s (%d ชิ้น)\n", nameP[best], sold[best]);
    return 0;
}
PYTHON 
M = int(input())
names, price, qty, sold = [], [], [], []

for _ in range(M):
    nm, p, q = input().split()
    names.append(nm)
    price.append(int(p))
    qty.append(int(q))
    sold.append(0)

def find_item(s):
    return names.index(s) if s in names else -1

income = 0
while True:
    parts = input().split()
    cmd = parts[0]
    if cmd == "END":
        break

    if cmd == "SELL":
        s, n = parts[1], int(parts[2])
        i = find_item(s)
        if i == -1:
            print(f"ไม่มีสินค้า {s}")
            continue

        give = min(n, qty[i])            # ขายเท่าที่มี
        qty[i]  -= give
        sold[i] += give
        money    = give * price[i]
        income  += money

        if give < n:
            print(f"ของไม่พอ! ขาย {s} ได้แค่ {give} ชิ้น = {money} บาท")
        else:
            print(f"ขาย {s} {give} ชิ้น = {money} บาท")

    elif cmd == "ADD":
        s, n = parts[1], int(parts[2])
        i = find_item(s)
        if i == -1:
            print(f"ไม่มีสินค้า {s}")
            continue
        qty[i] += n
        print(f"เติม {s} เป็น {qty[i]} ชิ้น")

    elif cmd == "REPORT":
        low = [f"{names[i]}({qty[i]})" for i in range(M) if qty[i] < 5]
        print("[REPORT] ใกล้หมด:", " ".join(low) if low else "(ไม่มี)")

best = 0
for i in range(1, M):
    if sold[i] > sold[best]:
        best = i

print("-" * 20)
print(f"ยอดขายรวม = {income} บาท")
print(f"ขายดีที่สุด = {names[best]} ({sold[best]} ชิ้น)")
จุดที่มักพลาด:
① หัก qty[i] -= n ตรงๆ → สต๊อกติดลบ ต้องใช้ give
② คิดเงินจาก n (ที่ขอ) แทน give (ที่ขายจริง) → ยอดเกิน
③ หาตัวขายดีโดยเก็บแค่ค่ามากสุด → รู้จำนวนแต่ไม่รู้ว่าสินค้าไหน
