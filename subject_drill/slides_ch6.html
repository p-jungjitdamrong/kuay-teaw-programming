ตู้กดบัตรคิว — จับจุดตรงไหน

ธนาคารแห่งหนึ่งเปิดให้บริการตั้งแต่เวลา 8.30 น. มีตู้กดบัตรคิวให้ลูกค้า บัตรมีหมายเลขเรียงกันเริ่มจาก 1 พนักงานจะเรียกลูกค้าตามลำดับที่มาถึง ธนาคารรองรับลูกค้าได้ไม่เกิน 100 คนต่อวัน
ให้เขียนโปรแกรมรับคำสั่งเป็นตัวเลขทีละบรรทัด
• 1 — ลูกค้ากดบัตร ให้แสดง Ticket X ถ้าครบ 100 ใบแล้วให้แสดง Sold out
• 2 — พนักงานเรียกคิว ให้แสดง Calling X ถ้าไม่มีลูกค้ารออยู่ให้แสดง No one waiting
• 0 — ปิดธนาคาร ให้แสดง Waiting: X โดย X คือจำนวนลูกค้าที่ยังไม่ได้รับบริการ แล้วจบโปรแกรม
ช่องคำตอบ
① รับตัวเลขทีละบรรทัด (1 / 2 / 0) · หยุดเมื่อเจอ 0
② พิมพ์พิมพ์ทันทีทุกคำสั่ง + ตอนจบพิมพ์จำนวนคนที่รอ
③ ซ้ำอ่านคำสั่ง 1 ตัว แล้วทำตามคำสั่ง
④ จำqueue ของหมายเลขบัตร + เลขบัตรใบถัดไป
⑤ เงื่อนไขไม่เกิน 100 → จองอาร์เรย์ 100 ช่อง
⑥ ขอบSold out (tail == 100) · No one waiting (head == tail)
C 
#include <stdio.h>

int main() {
    int q[100];                // ④ queue  ⑤ ไม่เกิน 100
    int head = 0, tail = 0;
    int next = 1;               // ④ เลขบัตรใบถัดไป
    int cmd;

    // ① ③ อ่านทีละคำสั่ง หยุดเมื่อเจอ 0
    while (scanf("%d", &cmd) == 1 && cmd != 0) {
        if (cmd == 1) {
            if (tail == 100) {                 // ⑥
                printf("Sold out\n");
            } else {
                q[tail++] = next;               // เข้าท้ายคิว
                printf("Ticket %d\n", next);  // ②
                next++;
            }
        } else if (cmd == 2) {
            if (head == tail) {                 // ⑥
                printf("No one waiting\n");
            } else {
                printf("Calling %d\n", q[head++]); // ออกหน้าคิว
            }
        }
    }
    printf("Waiting: %d\n", tail - head); // ②
    return 0;
}

• T c — พิมพ์ตัวอักษร c หนึ่งตัวต่อท้ายข้อความ
• U — ยกเลิกการพิมพ์ครั้งล่าสุด (Undo) ถ้าไม่มีอะไรให้ยกเลิก ไม่ต้องทำอะไร
เมื่อจบให้แสดงข้อความที่เหลืออยู่ ถ้าไม่เหลือตัวอักษรเลยให้แสดง (empty)
รับประกันว่าข้อความยาวไม่เกิน 50 ตัวอักษร
ช่องคำตอบ
① รับคำสั่งเป็นคำ (T / U / END) · T ตามด้วยตัวอักษร 1 ตัว · หยุดเมื่อเจอ END
② พิมพ์พิมพ์ครั้งเดียวตอนจบ (ระหว่างทางไม่พิมพ์)
③ ซ้ำอ่านคำสั่ง 1 คำ แล้วทำตาม
④ จำstack ของตัวอักษร — ตัวข้อความเองคือ stack
⑤ เงื่อนไขไม่เกิน 50 → จอง 51 ช่อง (เผื่อ \0)
⑥ ขอบundo ตอนว่าง → ข้าม · จบแล้วว่าง → (empty)
C 
#include <stdio.h>
#include <string.h>

int main() {
    char st[51];               // ④ stack  ⑤ 50 + '\0'
    int top = 0;
    char cmd[8], c;

    // ① ③ อ่านทีละคำ หยุดเมื่อเจอ END
    while (scanf("%s", cmd) == 1 && strcmp(cmd, "END") != 0) {
        if (strcmp(cmd, "T") == 0) {
            scanf(" %c", &c);         // เว้นวรรคหน้า %c ข้ามช่องว่าง
            st[top++] = c;           // push
        } else if (strcmp(cmd, "U") == 0) {
            if (top > 0) top--;      // ⑥ pop เฉพาะตอนไม่ว่าง
        }
    }

    st[top] = '\0';                // ปิดท้ายให้เป็นสตริง
    if (top == 0) printf("(empty)\n");   // ⑥
    else          printf("%s\n", st);  // ②
    return 0;
}
⚠️ ต่างจากโจทย์ 1 ตรงไหน? โจทย์ 1 หยุดด้วยตัวเลข (0) ข้อนี้หยุดด้วยคำ (END) → ต้องอ่านด้วย %s และเทียบด้วย strcmp ห้ามใช้ ==
ช่อง ① จึงต้องจดให้ละเอียดว่า "หยุดด้วยอะไร" เพราะเปลี่ยนวิธีเขียนลูปทั้งหมด

ร้านอาหารมีชั้นวางจานที่วางซ้อนกันได้สูงสุด 5 ใบ พนักงานล้างจานเสร็จจะวางไว้ด้านบนสุด และเมื่อจะใช้จะหยิบใบบนสุดออกไป จานแต่ละใบมีรหัสเป็นตัวเลข จานทุกใบของร้านเป็นเซรามิกสีขาว
รับจำนวนคำสั่ง N ก่อน แล้วรับคำสั่งอีก N บรรทัด
• W x — วางจานรหัส x ถ้าชั้นเต็มให้แสดง Full
• P — หยิบจาน ให้แสดง Take x ถ้าไม่มีจานให้แสดง Empty
เมื่อทำครบทุกคำสั่ง ให้แสดง Left: ตามด้วยรหัสจานที่เหลือบนชั้นเรียงจากบนลงล่าง
ช่องคำตอบ
① รับN ก่อน แล้วคำสั่ง N บรรทัด (W ตามด้วยเลข / P)
② พิมพ์ตอน P พิมพ์ทันที · จบแล้วพิมพ์จานที่เหลือ
③ ซ้ำทำ N รอบ รอบละ 1 คำสั่ง
④ จำstack ของรหัสจาน
⑤ เงื่อนไขสูงสุด 5 ใบ · พิมพ์บนลงล่าง = วนถอยหลัง
⑥ ขอบเต็ม (top == 5) · ว่าง (top == 0)
C 
#include <stdio.h>
#include <string.h>

int main() {
    int st[5];                  // ④ stack  ⑤ สูงสุด 5
    int top = 0;
    int n, x;
    char cmd[4];

    scanf("%d", &n);              // ① N มาก่อน
    for (int i = 0; i < n; i++) {  // ③ ทำ N รอบ
        scanf("%s", cmd);
        if (strcmp(cmd, "W") == 0) {
            scanf("%d", &x);
            if (top == 5) printf("Full\n");   // ⑥
            else          st[top++] = x;          // push
        } else if (strcmp(cmd, "P") == 0) {
            if (top == 0) printf("Empty\n");   // ⑥
            else          printf("Take %d\n", st[--top]);  // pop ②
        }
    }

    printf("Left:");
    for (int i = top - 1; i >= 0; i--)   // ⑤ บนลงล่าง
        printf(" %d", st[i]);
    printf("\n");
    return 0;
}
⚠️ กับดักของข้อนี้: ถ้าอ่านข้าม "เรียงจากบนลงล่าง" แล้ววน i = 0 → top−1 ตามความเคยชิน จะได้ 11 13 14 15 16 — ผลลัพธ์กลับด้าน ผิดทั้งบรรทัด
คำขยายท้ายประโยคแบบนี้คือช่อง ⑤ ที่คนอ่านข้ามบ่อยที่สุด

เครื่องพิมพ์ — จับจุดตรงไหน

สำนักงานแห่งหนึ่งใช้เครื่องพิมพ์รุ่น LaserJet ความเร็วสูง ซึ่งพิมพ์ได้นาทีละ 5 หน้า งานที่ส่งเข้ามาจะถูกพิมพ์ตามลำดับที่ส่งทีละงาน โดยเครื่องเริ่มพิมพ์งานแรกที่นาทีที่ 0
ให้รับจำนวนงาน N ตามด้วยจำนวนหน้าของแต่ละงาน (อย่างน้อย 1 หน้า) แล้วแสดงว่างานแต่ละงานพิมพ์เสร็จในนาทีที่เท่าไร ในรูปแบบ Job i done at minute t
ถ้างานใดใช้เวลาไม่ถึงนาทีเต็ม ให้นับเศษเป็น 1 นาทีเต็ม
ช่องคำตอบ
① รับN ก่อน แล้วตัวเลข N ตัว (จำนวนหน้า)
② พิมพ์พิมพ์ทันทีทุกงาน รูปแบบตามโจทย์
③ ซ้ำอ่านงาน 1 งาน → คำนวณ → พิมพ์
④ จำเวลาสะสม t ตัวเดียว — ไม่ต้องใช้ queue
⑤ เงื่อนไข5 หน้า/นาที · เศษปัดขึ้น · เริ่มที่ 0
⑥ ขอบอย่างน้อย 1 หน้า → ไม่ต้องกัน 0 · หน้าหารลงตัวพอดี (5, 10)
C 
#include <stdio.h>

int main() {
    int n, pages;
    int t = 0;                   // ④ เวลาสะสม  ⑤ เริ่มที่ 0

    scanf("%d", &n);              // ① N มาก่อน
    for (int i = 1; i <= n; i++) { // ③
        scanf("%d", &pages);
        t += (pages + 4) / 5;       // ⑤ ปัดขึ้น
        printf("Job %d done at minute %d\n", i, t);  // ②
    }
    return 0;
}

ถามช่อง ④ ให้ชัดว่า "ต้องเก็บไว้ก่อนแล้วค่อยใช้ทีหลังไหม" — ถ้าอ่านแล้วใช้ทันทีได้ ลำดับการอ่านก็คือคิวอยู่แล้ว
โจทย์ 1 ต้องใช้ queue เพราะคนกดบัตรไม่ได้ถูกเรียกทันที ต้องรอคำสั่ง 2 ก่อน
📌 สูตรปัดขึ้นของ int ที่ต้องจำ: (a + b − 1) / b
หาร 5 → (a + 4) / 5 · หาร 3 → (a + 2) / 3 · ไม่ต้อง include math.h

Palindrome — จับจุดตรงไหน

คำ palindrome คือคำที่อ่านจากหน้าไปหลังและจากหลังไปหน้าได้เหมือนกัน เช่น level, noon
ให้เขียนโปรแกรมตรวจว่าคำที่รับเข้ามาเป็น palindrome หรือไม่ โดยไม่สนใจตัวพิมพ์ใหญ่-เล็ก
โปรแกรมต้องใช้ stack และ queue อย่างละ 1 ชุด: นำตัวอักษรทุกตัวของคำใส่ทั้ง stack และ queue แล้วดึงออกมาเทียบกันทีละคู่
รับคำได้หลายคำ คำละ 1 บรรทัด จนกว่าจะพบคำว่า STOP แต่ละคำยาวไม่เกิน 30 ตัวอักษร ให้แสดง YES หรือ NO คำละ 1 บรรทัด
ช่องคำตอบ
① รับคำทีละคำ · หยุดเมื่อเจอ STOP
② พิมพ์YES / NO ทันทีหลังตรวจแต่ละคำ
③ ซ้ำ2 ชั้น: ซ้ำทีละคำ และในแต่ละคำซ้ำทีละตัวอักษร
④ จำstack + queue ของตัวอักษร · ผลการเทียบ ok
⑤ เงื่อนไขtolower ก่อนใส่ · ไม่เกิน 30 → จอง 31 ช่อง · ต้องใช้ stack+queue ตามที่โจทย์สั่ง
⑥ ขอบหลายคำ → รีเซ็ตทุกคำ · คำยาว 1 ตัวอักษร (ต้องได้ YES)
C 
#include <stdio.h>
#include <string.h>
#include <ctype.h>                 // tolower()

int main() {
    char w[31];                   // ⑤ 30 + '\0'
    char st[31], q[31];           // ④ stack + queue

    // ① ③ ชั้นนอก: ทีละคำ หยุดเมื่อเจอ STOP
    while (scanf("%s", w) == 1 && strcmp(w, "STOP") != 0) {
        int top = 0, head = 0, tail = 0;   // ⑥ รีเซ็ตทุกคำ!

        // ③ ชั้นใน: ใส่ทุกตัวอักษรลงทั้งคู่
        for (int i = 0; w[i] != '\0'; i++) {
            char c = tolower(w[i]);  // ⑤
            st[top++] = c;           // push
            q[tail++] = c;           // enqueue
        }

        // ดึงออกเทียบทีละคู่
        int ok = 1;
        while (top > 0) {
            if (st[--top] != q[head++]) {   // pop เทียบ dequeue
                ok = 0;
                break;
            }
        }
        printf("%s\n", ok ? "YES" : "NO");    // ②
    }
    return 0;
}
⚠️ ทำไมประกาศ top / head / tail ในลูป? เพื่อให้เริ่มจาก 0 ใหม่ทุกคำโดยอัตโนมัติ — ถ้าประกาศไว้ข้างนอก ต้องจำไปตั้งเป็น 0 เอง และนั่นคือจุดที่ลืมกันบ่อยที่สุด
💡 เจอคำว่า "รับได้หลายชุด / หลายคำ / หลายกรณีทดสอบ" ในโจทย์ → จดในช่อง ⑥ ทันทีว่า อะไรต้องเริ่มใหม่ทุกรอบ
