โจทย์ที่ 1 · Linked List

ระบบเพลย์ลิสต์เพลง

โจทย์: เขียนระบบจัดการเพลย์ลิสต์ รับคำสั่งจนเจอ END
ADD <ชื่อเพลง> — เพิ่มต่อท้ายเพลย์ลิสต์
INSERT <ชื่อเพลง> <ตำแหน่ง> — แทรกเพลงที่ตำแหน่งนั้น (เริ่มนับ 1)
DEL <ชื่อเพลง> — ลบเพลงนั้นออกจากเพลย์ลิสต์
NEXT — เลื่อนไปเพลงถัดไป (ถ้าถึงท้ายให้วนกลับเพลงแรก)
NOW — บอกว่ากำลังเล่นเพลงอะไร อยู่ลำดับที่เท่าไร
LIST — แสดงเพลย์ลิสต์ทั้งหมด
ข้อบังคับ: ห้ามใช้อาร์เรย์เก็บลำดับเพลง — ต้องใช้ linked list
ตัวอย่างการทำงาน
input: ADD เพลงA ADD เพลงB ADD เพลงC INSERT เพลงX 2 LIST NOW NEXT NEXT NOW DEL เพลงB LIST END output: เพิ่ม เพลงA (ทั้งหมด 1 เพลง) เพิ่ม เพลงB (ทั้งหมด 2 เพลง) เพิ่ม เพลงC (ทั้งหมด 3 เพลง) แทรก เพลงX ที่ลำดับ 2 เพลย์ลิสต์: 1.เพลงA 2.เพลงX 3.เพลงB 4.เพลงC กำลังเล่น: เพลงA (ลำดับ 1) ▶ เพลงX ▶ เพลงB กำลังเล่น: เพลงB (ลำดับ 3) ลบ เพลงB → เล่นต่อที่ เพลงC เพลย์ลิสต์: 1.เพลงA 2.เพลงX 3.เพลงC
ทำไมข้อนี้ถึงต้องใช้ Linked List เพราะงานหลักคือ "แทรกกลาง" กับ "ลบกลาง"
อาร์เรย์: แทรกที่ลำดับ 2 ต้องเลื่อนทุกเพลงข้างหลังไป 1 ช่อง = O(n) ทุกครั้ง
Linked List: แค่ย้ายลูกศร 2 เส้น = O(1) เมื่อรู้ตำแหน่งแล้ว

และเพราะมี "เพลงที่กำลังเล่น" ที่ต้องรอดจากการแก้ลิสต์
ถ้าใช้อาร์เรย์แล้วเก็บตำแหน่งเป็นตัวเลข พอลบเพลงข้างหน้า ตัวเลขจะชี้ผิดเพลงทันที
แต่ถ้าเก็บเป็น pointer ไปยังโหนด ต่อให้ลิสต์เปลี่ยนยังไง มันก็ยังชี้เพลงเดิมอยู่
หลักการที่ต้องใช้ ① เก็บ head + cur — cur คือโหนดที่กำลังเล่น ไม่ใช่ตัวเลขลำดับ
② แทรกที่ลำดับ k — เดินไปโหนดที่ k−1 แล้วต่อลูกศรตัวใหม่ก่อนเสมอ
③ ลบ — ต้องหาโหนดก่อนหน้าให้ได้ · ถ้าลบตัวที่กำลังเล่นอยู่ ให้เลื่อน cur ไปตัวถัดไปก่อนลบ
④ NEXT วนกลับ — ถ้า cur->next == NULL ให้ cur = head
⏸️ ลองทำเองก่อน — เคสที่ต้องผ่าน: แทรกที่ลำดับ 1 (ต้องเปลี่ยน head) · ลบเพลงแรก · ลบเพลงที่กำลังเล่นอยู่ · ลบจนเหลือเพลงเดียวแล้วกด NEXT

ระบบเพลย์ลิสต์ — C และ Python

C 
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct Song {
    char name[64];
    struct Song *next;
};

struct Song *head = NULL, *cur = NULL;

struct Song* newSong(char *s) {
    struct Song *p = malloc(sizeof(struct Song));
    strcpy(p->name, s);
    p->next = NULL;
    return p;
}

int count() {
    int n = 0;
    for (struct Song *c = head; c; c = c->next) n++;
    return n;
}

int posOf(struct Song *t) {         // ลำดับของโหนด (เริ่ม 1)
    int i = 1;
    for (struct Song *c = head; c; c = c->next, i++)
        if (c == t) return i;
    return 0;
}

void addEnd(char *s) {
    struct Song *n = newSong(s);
    if (!head) { head = cur = n; }
    else {
        struct Song *c = head;
        while (c->next) c = c->next;
        c->next = n;
    }
    printf("เพิ่ม %s (ทั้งหมด %d เพลง)\n", s, count());
}

void insertAt(char *s, int k) {
    struct Song *n = newSong(s);
    if (k <= 1 || !head) {           // แทรกหัว
        n->next = head;
        head = n;
        if (!cur) cur = n;
    } else {
        struct Song *c = head;
        for (int i = 1; i < k - 1 && c->next; i++)
            c = c->next;                // ไปโหนดที่ k−1
        n->next = c->next;              // ต่อตัวใหม่ก่อน
        c->next = n;
    }
    printf("แทรก %s ที่ลำดับ %d\n", s, k);
}

void delSong(char *s) {
    if (!head) { printf("เพลย์ลิสต์ว่าง\n"); return; }

    struct Song *target = NULL, *prev = NULL;
    for (struct Song *c = head; c; prev = c, c = c->next)
        if (strcmp(c->name, s) == 0) { target = c; break; }

    if (!target) { printf("ไม่พบเพลง %s\n", s); return; }

    if (cur == target)                  // เลื่อน cur ก่อนลบ
        cur = target->next ? target->next : head;

    if (prev) prev->next = target->next;
    else      head = target->next;

    if (cur == target) cur = head;      // เหลือเพลงเดียวแล้วลบ
    free(target);

    if (!head) printf("ลบ %s → เพลย์ลิสต์ว่าง\n", s);
    else       printf("ลบ %s → เล่นต่อที่ %s\n", s, cur->name);
}

int main() {
    char cmd[16], s[64];
    int k;
    while (scanf("%s", cmd) == 1 && strcmp(cmd, "END")) {
        if (!strcmp(cmd, "ADD"))    { scanf("%s", s); addEnd(s); }
        else if (!strcmp(cmd, "INSERT")) { scanf("%s %d", s, &k); insertAt(s, k); }
        else if (!strcmp(cmd, "DEL")) { scanf("%s", s); delSong(s); }
        else if (!strcmp(cmd, "NEXT")) {
            if (!cur) { printf("ไม่มีเพลง\n"); continue; }
            cur = cur->next ? cur->next : head;   // วนกลับ
            printf("▶ %s\n", cur->name);
        }
        else if (!strcmp(cmd, "NOW")) {
            if (!cur) printf("ไม่มีเพลง\n");
            else printf("กำลังเล่น: %s (ลำดับ %d)\n", cur->name, posOf(cur));
        }
        else if (!strcmp(cmd, "LIST")) {
            printf("เพลย์ลิสต์:");
            int i = 1;
            for (struct Song *c = head; c; c = c->next, i++)
                printf(" %d.%s", i, c->name);
            printf("\n");
        }
    }
    return 0;
}
PYTHON 
class Song:
    def __init__(self, name):
        self.name = name
        self.next = None

head = None
cur  = None

def count():
    n, c = 0, head
    while c:
        n += 1; c = c.next
    return n

def pos_of(t):
    i, c = 1, head
    while c:
        if c is t: return i
        i += 1; c = c.next
    return 0

while True:
    parts = input().split()
    cmd = parts[0]
    if cmd == "END":
        break

    if cmd == "ADD":
        n = Song(parts[1])
        if head is None:
            head = cur = n
        else:
            c = head
            while c.next: c = c.next
            c.next = n
        print(f"เพิ่ม {n.name} (ทั้งหมด {count()} เพลง)")

    elif cmd == "INSERT":
        name, k = parts[1], int(parts[2])
        n = Song(name)
        if k <= 1 or head is None:
            n.next = head
            head = n
            if cur is None: cur = n
        else:
            c, i = head, 1
            while i < k - 1 and c.next:
                c = c.next; i += 1
            n.next = c.next        # ต่อตัวใหม่ก่อน
            c.next = n
        print(f"แทรก {name} ที่ลำดับ {k}")

    elif cmd == "DEL":
        s = parts[1]
        target, prev, c = None, None, head
        while c:
            if c.name == s:
                target = c; break
            prev, c = c, c.next

        if target is None:
            print(f"ไม่พบเพลง {s}")
        else:
            if cur is target:
                cur = target.next if target.next else head
            if prev: prev.next = target.next
            else:    head = target.next
            if cur is target: cur = head
            if head is None:
                print(f"ลบ {s} → เพลย์ลิสต์ว่าง")
            else:
                print(f"ลบ {s} → เล่นต่อที่ {cur.name}")

    elif cmd == "NEXT":
        if cur is None:
            print("ไม่มีเพลง")
        else:
            cur = cur.next if cur.next else head
            print(f"▶ {cur.name}")

    elif cmd == "NOW":
        if cur is None: print("ไม่มีเพลง")
        else: print(f"กำลังเล่น: {cur.name} (ลำดับ {pos_of(cur)})")

    elif cmd == "LIST":
        out, c, i = [], head, 1
        while c:
            out.append(f"{i}.{c.name}")
            c = c.next; i += 1
        print("เพลย์ลิสต์:", " ".join(out))

ระบบสมาชิกสะสมแต้ม

โจทย์: ร้านมีสมาชิกได้ถึง 100,000 คน แต่ละคนมีรหัสสมาชิก (ตัวเลข) และแต้มสะสม
รับคำสั่งจนเจอ END
REG <รหัส> — สมัครสมาชิกใหม่ (แต้มเริ่ม 0) ถ้ามีอยู่แล้วให้แจ้ง
EARN <รหัส> <ยอดซื้อ> — ได้แต้ม = ยอดซื้อ ÷ 25 (ปัดลง)
USE <รหัส> <แต้ม> — ใช้แต้ม ถ้าไม่พอให้แจ้งและไม่หัก
CHECK <รหัส> — ดูแต้มคงเหลือ
ข้อบังคับ: การค้นหาสมาชิกต้องไม่วนทั้งฐานข้อมูล — ใช้ hash table แก้ชนแบบ chaining
ตัวอย่างการทำงาน
input: REG 10021 REG 30045 REG 10021 EARN 10021 500 EARN 30045 130 USE 10021 15 USE 30045 99 CHECK 10021 CHECK 99999 END output: สมัครสมาชิก 10021 สำเร็จ สมัครสมาชิก 30045 สำเร็จ รหัส 10021 มีอยู่แล้ว 10021 ซื้อ 500 → ได้ 20 แต้ม (รวม 20) 30045 ซื้อ 130 → ได้ 5 แต้ม (รวม 5) 10021 ใช้ 15 แต้ม (เหลือ 5) 30045 แต้มไม่พอ (มี 5 ขอใช้ 99) 10021 มี 5 แต้ม ไม่พบสมาชิก 99999
ทำไมข้อนี้ถึงต้องใช้ Hash เพราะทุกคำสั่งเริ่มด้วยการ "หาสมาชิกจากรหัส"
วนหาทีละคนในฐาน 100,000 คน = 100,000 ครั้งต่อ 1 คำสั่ง
Hash: รหัส % SIZE ได้ช่องทันที แล้วเดินเฉพาะโซ่ในช่องนั้น (ปกติ 1–2 ตัว)

ทำไมไม่ใช้อาร์เรย์ตรงๆ ที่ index = รหัส?
ทำได้ถ้ารหัสเป็น 0..99999 — แต่พอรหัสเป็น 13 หลักหรือมีตัวอักษร ต้องจองอาร์เรย์ใหญ่มหาศาล
hash แปลงรหัสอะไรก็ได้ให้เหลือช่วงเล็กๆ ที่จองไหว
หลักการที่ต้องใช้ ① h(k) = k % SIZE — เลือก SIZE เป็นจำนวนเฉพาะ (เช่น 1009) ให้กระจายดี
② chaining — แต่ละช่องเป็นหัวของ linked list · ชนกันก็ต่อหัวเข้าไป
③ ฟังก์ชัน find(k) — คำนวณช่อง → เดินโซ่ → คืนโหนดหรือ NULL · ทุกคำสั่งเรียกตัวนี้
④ REG ต้องเช็คซ้ำก่อน — เรียก find ก่อน ถ้าเจอแล้วห้ามใส่ซ้ำ
SIZE = 1009, h(k) = k % 1009 10021 % 1009 = 953 30045 % 1009 = 815 [815] → 30045(5 แต้ม) [953] → 10021(5 แต้ม) ช่องอื่น → NULL CHECK 99999 → 99999%1009 = 811 [811] ว่าง → ไม่พบ ทันที
⏸️ ลองทำเองก่อน — เคสที่ต้องผ่าน: สมัครซ้ำ · ใช้แต้มเกินที่มี · ทำคำสั่งกับรหัสที่ไม่เคยสมัคร · รหัส 2 ตัวที่ชนช่องเดียวกัน (เช่น 5 กับ 1014)

ระบบสมาชิกสะสมแต้ม — C และ Python

C 
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#define SIZE 1009          // จำนวนเฉพาะ กระจายดี

struct Member {
    int id, points;
    struct Member *next;    // โซ่ของ chaining
};

struct Member *table[SIZE] = {NULL};

int hashFn(int k) { return k % SIZE; }

struct Member* find(int k) {      // หัวใจของทั้งโปรแกรม
    struct Member *c = table[hashFn(k)];
    while (c) {
        if (c->id == k) return c;
        c = c->next;
    }
    return NULL;
}

void reg(int k) {
    if (find(k)) { printf("รหัส %d มีอยู่แล้ว\n", k); return; }
    int idx = hashFn(k);
    struct Member *m = malloc(sizeof(struct Member));
    m->id = k; m->points = 0;
    m->next = table[idx];       // ต่อหัว O(1)
    table[idx] = m;
    printf("สมัครสมาชิก %d สำเร็จ\n", k);
}

int main() {
    char cmd[16];
    int k, v;

    while (scanf("%s", cmd) == 1 && strcmp(cmd, "END")) {

        if (!strcmp(cmd, "REG")) {
            scanf("%d", &k);
            reg(k);
        }
        else if (!strcmp(cmd, "EARN")) {
            scanf("%d %d", &k, &v);
            struct Member *m = find(k);
            if (!m) { printf("ไม่พบสมาชิก %d\n", k); continue; }
            int got = v / 25;          // int หาร int = ปัดลงอยู่แล้ว
            m->points += got;
            printf("%d ซื้อ %d → ได้ %d แต้ม (รวม %d)\n",
                   k, v, got, m->points);
        }
        else if (!strcmp(cmd, "USE")) {
            scanf("%d %d", &k, &v);
            struct Member *m = find(k);
            if (!m) { printf("ไม่พบสมาชิก %d\n", k); continue; }
            if (m->points < v)
                printf("%d แต้มไม่พอ (มี %d ขอใช้ %d)\n", k, m->points, v);
            else {
                m->points -= v;
                printf("%d ใช้ %d แต้ม (เหลือ %d)\n", k, v, m->points);
            }
        }
        else if (!strcmp(cmd, "CHECK")) {
            scanf("%d", &k);
            struct Member *m = find(k);
            if (!m) printf("ไม่พบสมาชิก %d\n", k);
            else    printf("%d มี %d แต้ม\n", k, m->points);
        }
    }
    return 0;
}
PYTHON — เขียนเองแบบเดียวกับ C 
SIZE = 1009

class Member:
    def __init__(self, mid):
        self.id     = mid
        self.points = 0
        self.next   = None

table = [None] * SIZE

def hash_fn(k):
    return k % SIZE

def find(k):
    c = table[hash_fn(k)]
    while c:
        if c.id == k:
            return c
        c = c.next
    return None

while True:
    parts = input().split()
    cmd = parts[0]
    if cmd == "END":
        break
    k = int(parts[1])

    if cmd == "REG":
        if find(k):
            print(f"รหัส {k} มีอยู่แล้ว")
        else:
            idx = hash_fn(k)
            m = Member(k)
            m.next = table[idx]        # ต่อหัว
            table[idx] = m
            print(f"สมัครสมาชิก {k} สำเร็จ")
        continue

    m = find(k)
    if m is None:
        print(f"ไม่พบสมาชิก {k}")
        continue

    if cmd == "EARN":
        v = int(parts[2])
        got = v // 25                # // = ปัดลง
        m.points += got
        print(f"{k} ซื้อ {v} → ได้ {got} แต้ม (รวม {m.points})")

    elif cmd == "USE":
        v = int(parts[2])
        if m.points < v:
            print(f"{k} แต้มไม่พอ (มี {m.points} ขอใช้ {v})")
        else:
            m.points -= v
            print(f"{k} ใช้ {v} แต้ม (เหลือ {m.points})")

    elif cmd == "CHECK":
        print(f"{k} มี {m.points} แต้ม")

PUT <รหัสสินค้า> <ราคา> <จำนวน> — ใส่ลงตะกร้า
  · ถ้าสินค้านั้นมีอยู่ในตะกร้าแล้ว ให้บวกจำนวนรวมกัน ไม่ใช่เพิ่มรายการใหม่
  · ถ้ายังไม่มี ให้ต่อท้ายรายการ ตามลำดับที่ใส่
REMOVE <รหัสสินค้า> — เอาออกจากตะกร้าทั้งรายการ
SHOW — แสดงรายการตามลำดับที่ใส่ พร้อมยอดรวม
ข้อบังคับ: เช็คว่ามีสินค้าอยู่แล้วหรือไม่ ต้องใช้ hash · เก็บลำดับรายการต้องใช้ linked list
ตัวอย่างการทำงาน
input: PUT 501 120 2 PUT 302 45 1 PUT 501 120 3 SHOW REMOVE 302 PUT 777 90 1 SHOW END output: ใส่ 501 x2 ใส่ 302 x1 501 มีอยู่แล้ว → รวมเป็น x5 ตะกร้า: 501 ราคา 120 x5 = 600 302 ราคา 45 x1 = 45 รวม 645 บาท เอา 302 ออกจากตะกร้า ใส่ 777 x1 ตะกร้า: 501 ราคา 120 x5 = 600 777 ราคา 90 x1 = 90 รวม 690 บาท
ทำไมต้องใช้ทั้งสองอย่าง โจทย์บังคับ 2 เรื่องที่ขัดกัน:
① ต้องรู้เร็วว่ามีสินค้านี้แล้วหรือยัง → งานของ hash
② ต้องแสดงตามลำดับที่ใส่ และลบกลางได้ → งานของ linked list

hash อย่างเดียวไม่จำลำดับ · linked list อย่างเดียวต้องวนหาทุกครั้ง
คำตอบคือใช้คู่กัน: เก็บโหนดไว้ในลิสต์ตามลำดับ แล้วให้ hash เก็บตัวชี้ไปยังโหนดเดียวกันนั้น
→ โหนดมีชุดเดียว แต่เข้าถึงได้ 2 ทาง
Linked list (ลำดับที่ใส่): head → [501] → [302] → [777] → NULL Hash table (ค้นหาเร็ว): [ ... ] [302 %S] ─────┐ [501 %S] ──┐ │ [777 %S] ─┐│ │ ││ │ ▼▼ ▼ ชี้ไปโหนดเดียวกันในลิสต์ PUT 501 → hash บอกทันทีว่ามี → ไปบวกจำนวนที่โหนดนั้นเลย REMOVE → hash หาโหนดเจอ → ตัดออกจากลิสต์ + ลบจาก hash
หลักการที่ต้องใช้ ① โหนดเดียว 2 ลิงก์ — next ใช้เรียงลำดับ · hnext ใช้ต่อโซ่ในช่อง hash
② PUT = find ก่อนเสมอ · เจอ → บวกจำนวน · ไม่เจอ → สร้างโหนด แล้วใส่ทั้ง 2 ที่
③ REMOVE = ต้องลบทั้ง 2 ที่ ไม่งั้น hash จะชี้โหนดที่ไม่อยู่ในลิสต์แล้ว
④ SHOW = เดินตาม next อย่างเดียว ไม่ยุ่งกับ hash

ระบบตะกร้าสินค้า — C และ Python

C 
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#define SIZE 101

struct Item {
    int id, price, qty;
    struct Item *next;    // ลำดับในตะกร้า
    struct Item *hnext;   // โซ่ใน hash
};

struct Item *head = NULL, *tailP = NULL;
struct Item *table[SIZE] = {NULL};

int hashFn(int k) { return k % SIZE; }

struct Item* find(int k) {
    struct Item *c = table[hashFn(k)];
    while (c) {
        if (c->id == k) return c;
        c = c->hnext;
    }
    return NULL;
}

void put(int id, int price, int qty) {
    struct Item *m = find(id);
    if (m) {                          // มีแล้ว → บวกจำนวน
        m->qty += qty;
        printf("%d มีอยู่แล้ว → รวมเป็น x%d\n", id, m->qty);
        return;
    }
    m = malloc(sizeof(struct Item));
    m->id = id; m->price = price; m->qty = qty;
    m->next = NULL;

    if (!head) head = tailP = m;      // ต่อท้ายลิสต์
    else { tailP->next = m; tailP = m; }

    int idx = hashFn(id);             // ใส่ลง hash ด้วย
    m->hnext = table[idx];
    table[idx] = m;

    printf("ใส่ %d x%d\n", id, qty);
}

void removeItem(int id) {
    struct Item *m = find(id);
    if (!m) { printf("ไม่มี %d ในตะกร้า\n", id); return; }

    // ① ตัดออกจากลิสต์ลำดับ
    if (head == m) head = m->next;
    else {
        struct Item *c = head;
        while (c->next != m) c = c->next;
        c->next = m->next;
        if (tailP == m) tailP = c;
    }
    if (!head) tailP = NULL;

    // ② ตัดออกจากโซ่ hash ด้วย — ห้ามลืม!
    int idx = hashFn(id);
    if (table[idx] == m) table[idx] = m->hnext;
    else {
        struct Item *c = table[idx];
        while (c->hnext != m) c = c->hnext;
        c->hnext = m->hnext;
    }

    free(m);
    printf("เอา %d ออกจากตะกร้า\n", id);
}

int main() {
    char cmd[16];
    int a, b, c;
    while (scanf("%s", cmd) == 1 && strcmp(cmd, "END")) {
        if (!strcmp(cmd, "PUT")) {
            scanf("%d %d %d", &a, &b, &c);
            put(a, b, c);
        }
        else if (!strcmp(cmd, "REMOVE")) {
            scanf("%d", &a);
            removeItem(a);
        }
        else if (!strcmp(cmd, "SHOW")) {
            printf("ตะกร้า:\n");
            int total = 0;
            for (struct Item *p = head; p; p = p->next) {
                int sub = p->price * p->qty;
                total += sub;
                printf("  %d ราคา %d x%d = %d\n",
                       p->id, p->price, p->qty, sub);
            }
            printf("  รวม %d บาท\n", total);
        }
    }
    return 0;
}
PYTHON 
SIZE = 101

class Item:
    def __init__(self, iid, price, qty):
        self.id    = iid
        self.price = price
        self.qty   = qty
        self.next  = None     # ลำดับในตะกร้า
        self.hnext = None     # โซ่ใน hash

head = tail = None
table = [None] * SIZE

def hash_fn(k):
    return k % SIZE

def find(k):
    c = table[hash_fn(k)]
    while c:
        if c.id == k: return c
        c = c.hnext
    return None

while True:
    parts = input().split()
    cmd = parts[0]
    if cmd == "END":
        break

    if cmd == "PUT":
        iid, price, qty = map(int, parts[1:4])
        m = find(iid)
        if m:
            m.qty += qty
            print(f"{iid} มีอยู่แล้ว → รวมเป็น x{m.qty}")
        else:
            m = Item(iid, price, qty)
            if head is None:
                head = tail = m
            else:
                tail.next = m
                tail = m
            idx = hash_fn(iid)            # ใส่ลง hash ด้วย
            m.hnext = table[idx]
            table[idx] = m
            print(f"ใส่ {iid} x{qty}")

    elif cmd == "REMOVE":
        iid = int(parts[1])
        m = find(iid)
        if m is None:
            print(f"ไม่มี {iid} ในตะกร้า")
            continue

        # ① ตัดออกจากลิสต์ลำดับ
        if head is m:
            head = m.next
        else:
            c = head
            while c.next is not m:
                c = c.next
            c.next = m.next
            if tail is m:
                tail = c
        if head is None:
            tail = None

        # ② ตัดออกจากโซ่ hash ด้วย
        idx = hash_fn(iid)
        if table[idx] is m:
            table[idx] = m.hnext
        else:
            c = table[idx]
            while c.hnext is not m:
                c = c.hnext
            c.hnext = m.hnext

        print(f"เอา {iid} ออกจากตะกร้า")

    elif cmd == "SHOW":
        print("ตะกร้า:")
        total, p = 0, head
        while p:
            sub = p.price * p.qty
            total += sub
            print(f"  {p.id} ราคา {p.price} x{p.qty} = {sub}")
            p = p.next
        print(f"  รวม {total} บาท")
⚠️ บั๊กอันดับ 1 ของข้อนี้: REMOVE แล้วลบออกจากลิสต์อย่างเดียว ลืมลบออกจาก hash
→ PUT รหัสเดิมอีกครั้ง find จะยัง "เจอ" โหนดที่ถูกลบไปแล้ว แล้วไปบวกจำนวนใส่โหนดที่ไม่อยู่ในตะกร้า
ผลคือ SHOW ไม่เห็นสินค้านั้นเลย ทั้งที่ระบบบอกว่าใส่แล้ว — หาสาเหตุยากมาก
จุดที่มักพลาด:
① ลบตัวท้ายแล้วไม่อัปเดต tail → PUT ครั้งถัดไปต่อผิดที่
② สับสน next กับ hnext ตอนเดินโซ่ — เดิน hash ต้องใช้ hnext เท่านั้น
③ PUT ของที่มีอยู่แล้วแต่ราคาต่างกัน — โจทย์นี้ให้ยึดราคาเดิม ไม่ต้องแก้
