C 
#include <stdio.h>
#include <string.h>

int main() {
    char s[101];
    fgets(s, 101, stdin);      // อ่านทั้งบรรทัด (มีเว้นวรรคได้)
    int n = strlen(s);
    if (n > 0 && s[n-1] == '\n') s[--n] = '\0'; // ตัด \n ท้าย

    int hasUp = 0, hasLow = 0, hasDigit = 0, hasSpace = 0;
    for (int i = 0; i < n; i++) {
        char c = s[i];
        if (c >= 'A' && c <= 'Z') hasUp    = 1;
        if (c >= 'a' && c <= 'z') hasLow   = 1;
        if (c >= '0' && c <= '9') hasDigit = 1;
        if (c == ' ')             hasSpace = 1;
    }

    if (n >= 8 && hasUp && hasLow && hasDigit && !hasSpace) {
        printf("ผ่าน\n");
    } else {
        printf("ไม่ผ่าน\n");
        if (n < 8)   printf(" - สั้นกว่า 8 ตัว\n");
        if (!hasUp)    printf(" - ไม่มีตัวพิมพ์ใหญ่\n");
        if (!hasLow)   printf(" - ไม่มีตัวพิมพ์เล็ก\n");
        if (!hasDigit) printf(" - ไม่มีตัวเลข\n");
        if (hasSpace)  printf(" - มีช่องว่าง\n");
    }
    return 0;
}

C 
#include <stdio.h>

int main() {
    int u;
    scanf("%d", &u);

    double cost = 0.0;
    int t1 = (u < 150) ? u : 150;      // min(u,150)
    cost += t1 * 3.0;

    if (u > 150) {
        int hi = (u < 400) ? u : 400;  // min(u,400)
        cost += (hi - 150) * 4.0;
    }
    if (u > 400) {
        cost += (u - 400) * 5.0;
    }

    double vat   = cost * 0.07;
    double total = cost + vat;
    printf("ค่าไฟ = %.2f\n", cost);
    printf("VAT   = %.2f\n", vat);
    printf("รวม   = %.2f\n", total);
    return 0;
}

ระบบจัดคิวร้านอาหาร

โจทย์: ร้านมีโต๊ะว่าง T โต๊ะ (1 โต๊ะ = 1 กลุ่ม) มีลูกค้ามาต่อคิว n กลุ่ม เรียงตามลำดับที่มาถึง
• กลุ่มที่มาถึงตอนมีโต๊ะว่าง → ได้นั่งทันที (โต๊ะว่างลด 1)
• กลุ่มที่มาตอนโต๊ะเต็ม → ต้องรอ เก็บไว้ในคิว
• เมื่อกลุ่มไหนกินเสร็จ (ป้อนค่า 0) → โต๊ะว่างเพิ่ม 1 และเรียกคิวหน้าสุดเข้าทันที
สุดท้ายรายงานว่าใครได้นั่งบ้าง และเหลือใครรออยู่
รูปแบบข้อมูลเข้า
บรรทัดที่ 1: T (จำนวนโต๊ะ)
บรรทัดถัดไป: เหตุการณ์ทีละบรรทัด
  เลขบวก k = กลุ่มหมายเลข k มาถึง
  0         = มีโต๊ะหนึ่งกินเสร็จ
  -1        = จบข้อมูล
ตัวอย่างการทำงาน
T = 2 เหตุการณ์: 1 2 3 4 0 0 -1 กลุ่ม 1 → ได้โต๊ะ (ว่างเหลือ 1) กลุ่ม 2 → ได้โต๊ะ (ว่างเหลือ 0) กลุ่ม 3 → เต็ม เข้าคิว [3] กลุ่ม 4 → เต็ม เข้าคิว [3,4] เสร็จ 1 โต๊ะ → เรียกกลุ่ม 3 เข้า เสร็จ 1 โต๊ะ → เรียกกลุ่ม 4 เข้า ได้นั่ง: 1 2 3 4 ยังรอ : (ไม่มี)
แก่นของข้อนี้ = คิว (Queue)
คิว = เข้าท้าย ออกหน้า (FIFO)
  First In, First Out
  = มาก่อน ได้ก่อน — ตรงกับ
    ความยุติธรรมของร้านอาหารพอดี

เก็บด้วยอาร์เรย์ + 2 ตัวชี้:
  head = ตำแหน่งคนหน้าสุด
  tail = ตำแหน่งที่จะใส่คนใหม่

  เข้าคิว : q[tail] = k; tail++
  ออกคิว : k = q[head]; head++
  คิวว่าง : head == tail
  จำนวนที่รอ : tail − head
head tail ↓ ↓ q = [ _, 3, 4, _, _ ] ออก เข้า
⏸️ หยุดลองเอง — โจทย์นี้ยาว แต่แก่นมีแค่ 2 การกระทำ: คนมา (มีที่ว่างไหม?) กับ คนออก (มีคนรอไหม?) ที่เหลือคือรายละเอียด

  ① รับ T, ตั้ง free = T
  ② วนอ่านเหตุการณ์จนเจอ −1
  ③ ถ้าเป็นเลขบวก k (คนมา):
       ถ้า free > 0 → นั่งเลย, free−−
       ไม่งั้น      → เข้าคิว
  ④ ถ้าเป็น 0 (คนออก):
       ถ้าคิวไม่ว่าง → ดึงหน้าสุดเข้านั่ง
       ไม่งั้น       → free++
  ⑤ จบ: พิมพ์คนที่ได้นั่ง และคิวที่เหลือ

C 
#include <stdio.h>

int main() {
    int T;
    scanf("%d", &T);
    int free_tb = T;

    int q[1000], head = 0, tail = 0;   // คิว
    int seated[1000], sc = 0;           // คนที่ได้นั่ง

    int e;
    while (scanf("%d", &e) == 1 && e != -1) {
        if (e > 0) {                      // คนมา
            if (free_tb > 0) {
                free_tb--;
                seated[sc++] = e;
                printf("กลุ่ม %d ได้โต๊ะ (ว่างเหลือ %d)\n", e, free_tb);
            } else {
                q[tail++] = e;
                printf("กลุ่ม %d เต็ม เข้าคิว (รอ %d)\n", e, tail - head);
            }
        } else if (e == 0) {               // คนออก
            if (head < tail) {              // คิวไม่ว่าง
                int k = q[head++];
                seated[sc++] = k;
                printf("เรียกกลุ่ม %d เข้า\n", k);
            } else {
                free_tb++;
                printf("โต๊ะว่างเพิ่ม (ว่าง %d)\n", free_tb);
            }
        }
    }

    printf("ได้นั่ง:");
    for (int i = 0; i < sc; i++) printf(" %d", seated[i]);
    printf("\nยังรอ :");
    if (head == tail) printf(" (ไม่มี)");
    for (int i = head; i < tail; i++) printf(" %d", q[i]);
    printf("\n");
    return 0;
}

T = int(input())
free_tb = T
q = []          # คิว: เข้าท้าย ออกหน้า
seated = []

while True:
    e = int(input())
    if e == -1:
        break
    if e > 0:                       # คนมา
        if free_tb > 0:
            free_tb -= 1
            seated.append(e)
            print(f"กลุ่ม {e} ได้โต๊ะ (ว่างเหลือ {free_tb})")
        else:
            q.append(e)
            print(f"กลุ่ม {e} เต็ม เข้าคิว (รอ {len(q)})")
    elif e == 0:                    # คนออก
        if q:
            k = q.pop(0)             # ดึงหน้าสุด = FIFO
            seated.append(k)
            print(f"เรียกกลุ่ม {k} เข้า")
        else:
            free_tb += 1
            print(f"โต๊ะว่างเพิ่ม (ว่าง {free_tb})")

print("ได้นั่ง:", *seated)
