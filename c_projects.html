#01 Easy ✓ เสร็จแล้ว
🏦 ระบบคำนวณดอกเบี้ยเงินฝากธรรมดา
📋 Requirement
ธนาคาร KTB ต้องการโปรแกรมช่วยพนักงานคำนวณดอกเบี้ยเงินฝากให้ลูกค้า โปรแกรมจะรับ ยอดเงินต้น (บาท) และ อัตราดอกเบี้ยต่อปี (%) และ จำนวนปีที่ฝาก จากนั้นคำนวณและแสดงผล ดอกเบี้ยที่ได้รับ และ ยอดรวมเมื่อครบกำหนด โดยตัวเลขทางการเงินต้องมีทศนิยม 2 ตำแหน่งเสมอ
double scanf / printf %.2f arithmetic
INPUT
เงินต้น: 50000 อัตรา (%/ปี): 3.5 จำนวนปี: 3
OUTPUT
ดอกเบี้ย : 5250.00 บาท ยอดรวม : 55250.00 บาท
1ใช้ double เก็บยอดเงิน เพราะเงินมีทศนิยมและตัวเลขใหญ่
2สูตร: ดอกเบี้ย = เงินต้น × (rate/100) × ปี
3ใช้ %lf ใน scanf สำหรับ double และ %.2f ใน printf
#include <stdio.h>
int main() {
    double principal, rate, years;
    double interest, total;

    printf("เงินต้น: ");        scanf("%lf", &principal);
    printf("อัตรา (%%/ปี): ");   scanf("%lf", &rate);
    printf("จำนวนปี: ");       scanf("%lf", &years);

    interest = principal * (rate / 100.0) * years;
    total    = principal + interest;

    printf("ดอกเบี้ย : %.2f บาท\n", interest);
    printf("ยอดรวม  : %.2f บาท\n", total);
    return 0;
}
อธิบาย: ใช้ double เพราะเงินมีทศนิยมและอาจตัวใหญ่ — rate/100.0 แปลง % เป็นทศนิยม — %.2f แสดงผล 2 ตำแหน่งเสมอ
#02Easy ✓ เสร็จแล้ว
🎓 ระบบตรวจผลสอบและแสดงเกรด
📋 Requirement
มหาวิทยาลัยต้องการโปรแกรมรับ คะแนนสอบ (0–100) แล้วแสดง เกรด A–F ตามเกณฑ์มาตรฐาน (A=90+, B=80+, C=70+, D=60+, F=ต่ำกว่า 60) ถ้าคะแนนไม่อยู่ในช่วง 0–100 ให้แสดง "คะแนนไม่ถูกต้อง" และหยุดทันที โปรแกรมต้องตรวจค่าผิดปกติก่อนเสมอ
int if / else if Guard Clause || operator
INPUT
คะแนน: 85
OUTPUT
เกรด: B ผลการเรียน: ผ่าน
1ตรวจ invalid ก่อนเสมอด้วย if (score < 0 || score > 100) แล้ว return ออก
2ใช้ if-else if จากสูงลงต่ำ: >= 90, >= 80, >= 70, >= 60, else
3เพิ่ม ternary สำหรับ ผ่าน/ตก: score >= 60 ? "ผ่าน" : "ตก"
#include <stdio.h>
int main() {
    int score;
    printf("คะแนน: "); scanf("%d", &score);

    if (score < 0 || score > 100) {
        printf("คะแนนไม่ถูกต้อง\n");
        return 0;  /* Guard clause: ออกทันที */
    }
    char grade;
    if      (score >= 90) grade = 'A';
    else if (score >= 80) grade = 'B';
    else if (score >= 70) grade = 'C';
    else if (score >= 60) grade = 'D';
    else                   grade = 'F';

    printf("เกรด: %c\n", grade);
    printf("ผลการเรียน: %s\n",
           (score >= 60) ? "ผ่าน" : "ตก");
    return 0;
}
อธิบาย: Guard clause ตรวจ invalid แล้ว return ออกก่อน — if-else if เรียงจากสูงลงต่ำ — ternary สำหรับ ผ่าน/ตก บรรทัดเดียว
⬤ Medium — บทที่ 2–3
#03Medium ✓ เสร็จแล้ว
🔐 ระบบตรวจความแข็งแกร่งของรหัสผ่าน
📋 Requirement
ระบบ Internet Banking ของ KTB ต้องตรวจรหัสผ่านก่อนยอมรับ รับ 3 ข้อมูล: ความยาว (ตัวเลข), จำนวน digit ในรหัส, และ มีตัวพิมพ์ใหญ่ไหม (1=มี 0=ไม่มี) แล้วตัดสินว่า: ความยาว < 8 → "อ่อนมาก" | ความยาว 8–11 และ digit ≥ 2 → "พอใช้" | ความยาว ≥ 12 และ digit ≥ 3 และมีตัวพิมพ์ใหญ่ → "แข็งแกร่ง" | กรณีอื่น → "ปานกลาง" พร้อมพิมพ์คำแนะนำสั้นๆ ทุกกรณี
int if-else if chain && operator Guard clause
INPUT
ความยาว: 13 จำนวน digit: 4 มีตัวพิมพ์ใหญ่ (1/0): 1
OUTPUT
ระดับ: แข็งแกร่ง 💪 คำแนะนำ: รหัสผ่านผ่านมาตรฐาน
1ตรวจ len < 8 ก่อนเลย (Guard Clause) → return ทันที
2ตรวจ "แข็งแกร่ง" ก่อน "พอใช้" เพราะ condition ครอบคลุมกว่า
3ใช้ len >= 12 && digits >= 3 && hasUpper เชื่อม 3 เงื่อนไข
4else สุดท้ายรับ "ปานกลาง" ทุกกรณีที่เหลือ
#include <stdio.h>
int main() {
    int len, digits, hasUpper;
    printf("ความยาว: ");            scanf("%d", &len);
    printf("จำนวน digit: ");          scanf("%d", &digits);
    printf("มีตัวพิมพ์ใหญ่ (1/0): ");  scanf("%d", &hasUpper);

    if (len < 8) {
        printf("ระดับ: อ่อนมาก ❌\n");
        printf("คำแนะนำ: ต้องยาวอย่างน้อย 8 ตัวอักษร\n");
    } else if (len >= 12 && digits >= 3 && hasUpper) {
        printf("ระดับ: แข็งแกร่ง 💪\n");
        printf("คำแนะนำ: รหัสผ่านผ่านมาตรฐาน\n");
    } else if (len >= 8 && digits >= 2) {
        printf("ระดับ: พอใช้ 👍\n");
        printf("คำแนะนำ: เพิ่มความยาวและตัวพิมพ์ใหญ่\n");
    } else {
        printf("ระดับ: ปานกลาง ⚠\n");
        printf("คำแนะนำ: เพิ่มตัวเลขและตัวพิมพ์ใหญ่\n");
    }
    return 0;
}
อธิบาย: ตรวจ "แข็งแกร่ง" ก่อน "พอใช้" เพราะถ้าสลับ รหัส len=13 จะถูกจับที่ else if แรก (len>=8) แล้วได้ "พอใช้" แทน — ลำดับ if-else if สำคัญมาก
#04Medium ✓ เสร็จแล้ว
📊 ระบบวิเคราะห์ยอดขายรายสัปดาห์
📋 Requirement
ร้านก๋วยเตี๋ยวต้องการโปรแกรมรับ ยอดขายรายวัน 7 วัน จากผู้ใช้ทีละวัน แล้วคำนวณและแสดง: ยอดรวมทั้งสัปดาห์, ค่าเฉลี่ยต่อวัน, วันที่ขายดีที่สุด (บอกเป็นชื่อวัน เช่น "จันทร์") และ วันที่ขายน้อยที่สุด ถ้าวันไหนยอดขายเกิน 5000 บาท ให้แสดง ⭐ หลังยอดวันนั้น
int array[7] for loop Max/Min pattern if ใน loop
INPUT
จ: 3200 อ: 4100 พ: 5500 พฤ: 2900 ศ: 6200 ส: 7800 อา: 4300
OUTPUT
รวม : 34000 บาท เฉลี่ย: 4857.14 บาท ดีสุด : เสาร์ (7800) ⭐ น้อยสุด: พฤหัส (2900)
1ประกาศ int sales[7] และ char *days[] = {"จันทร์","อังคาร",...}
2for loop รับค่าด้วย scanf เข้า sales[i] ทีละวัน
3loop ที่สอง: sum += sales[i], หา maxIdx, minIdx
4if (sales[i] > 5000) printf("⭐") ใน loop แสดงผล
#include <stdio.h>
int main() {
    int sales[7], i;
    char *days[] = {"จันทร์","อังคาร","พุธ",
                    "พฤหัส","ศุกร์","เสาร์","อาทิตย์"};
    for (i=0; i<7; i++) {
        printf("%s: ", days[i]);
        scanf("%d", &sales[i]);
    }
    int sum=0, maxI=0, minI=0;
    for (i=0; i<7; i++) {
        sum += sales[i];
        if (sales[i] > sales[maxI]) maxI = i;
        if (sales[i] < sales[minI]) minI = i;
    }
    printf("รวม   : %d บาท\n", sum);
    printf("เฉลี่ย: %.2f บาท\n", (float)sum/7);
    printf("ดีสุด : %s (%d)%s\n", days[maxI],
           sales[maxI], sales[maxI]>5000?"⭐":"");
    printf("น้อยสุด: %s (%d)\n", days[minI], sales[minI]);
    return 0;
}
อธิบาย: char *days[] คือ array ของ string — เก็บ index ของ max/min แทนค่า เพื่อจะได้รู้ชื่อวันด้วย
#05Medium ✓ เสร็จแล้ว
🎰 เกมทายตัวเลข (High-Low)
📋 Requirement
เขียนเกมทายตัวเลข โปรแกรม กำหนดเลขลับ 1–100 ไว้ล่วงหน้า (ใช้ตัวเลขใดก็ได้ใน code) แล้วให้ผู้เล่น ทายซ้ำๆ จนกว่าจะถูก แต่ละครั้งที่ทายผิดให้บอกว่า "สูงเกินไป" หรือ "ต่ำเกินไป" เมื่อทายถูกให้แสดง จำนวนครั้งที่ใช้ทาย และถ้าทายได้ภายใน 7 ครั้งให้แสดง "เก่งมาก!"
while loop do-while หรือ while(1) counter if-else chain
SESSION
ทาย: 50 → ต่ำเกินไป ทาย: 75 → สูงเกินไป ทาย: 63 → ถูก!
OUTPUT
ถูกต้อง! ใช้ 3 ครั้ง เก่งมาก!
1กำหนด int secret = 63; ไว้ใน code
2ใช้ while(1) วนรับ input แล้ว break เมื่อทายถูก
3ประกาศ int attempts = 0; แล้ว attempts++ ทุกรอบ
4if (attempts <= 7) printf("เก่งมาก!");
#include <stdio.h>
int main() {
    int secret = 63, guess, attempts = 0;
    printf("ทายตัวเลข 1-100!\n");
    while (1) {
        printf("ทาย: "); scanf("%d", &guess);
        attempts++;
        if      (guess < secret) printf("ต่ำเกินไป!\n");
        else if (guess > secret) printf("สูงเกินไป!\n");
        else { break; }
    }
    printf("ถูกต้อง! ใช้ %d ครั้ง\n", attempts);
    if (attempts <= 7) printf("เก่งมาก!\n");
    return 0;
}
อธิบาย: while(1) วนไม่หยุด — break ออกเมื่อ guess == secret — attempts นับทุกรอบก่อน break
⬤ Medium-Hard — บทที่ 3–4
#06Medium-Hard ✓ เสร็จแล้ว
🏫 ระบบจัดการคะแนนนักเรียนทั้งห้อง
📋 Requirement
ครูต้องการโปรแกรมจัดการคะแนนนักเรียน 30 คน โปรแกรมรับคะแนน (0–100) ทีละคนเก็บใน array จากนั้น เรียก function แยก เพื่อ: (1) คำนวณค่าเฉลี่ย, (2) หาคะแนนสูงสุดและต่ำสุด, (3) นับจำนวนนักเรียนที่ได้เกรด A (90+), และ (4) พิมพ์ใบรายงานสรุป ทุก function ต้องมี prototype และรับ array เป็น parameter
int array[30] function + prototype for loop pass array
Functions ที่ต้องมี
float calcAvg(int[], int) int findMax(int[], int) int findMin(int[], int) int countA (int[], int) void printReport(int[], int)
OUTPUT
ค่าเฉลี่ย : 72.30 สูงสุด : 98 ต่ำสุด : 45 เกรด A : 4 คน
1เขียน prototype ทั้งหมดก่อน main
2รับค่าใน main ด้วย for loop + scanf(&scores[i])
3แต่ละ function รับ int arr[], int n และวน loop หาค่า
4countA: นับจำนวน element ที่ >= 90
#include <stdio.h>
/* Prototypes */
float calcAvg(int a[], int n);
int   findMax(int a[], int n);
int   findMin(int a[], int n);
int   countA (int a[], int n);

int main() {
    int scores[30], i;
    for (i=0; i<30; i++) {
        printf("คะแนนคนที่ %d: ", i+1);
        scanf("%d", &scores[i]);
    }
    printf("\n=== รายงาน ===\n");
    printf("ค่าเฉลี่ย  : %.2f\n", calcAvg(scores,30));
    printf("สูงสุด    : %d\n",   findMax(scores,30));
    printf("ต่ำสุด    : %d\n",   findMin(scores,30));
    printf("เกรด A   : %d คน\n", countA (scores,30));
    return 0;
}
float calcAvg(int a[], int n) {
    int i, s=0;
    for(i=0;i<n;i++) s+=a[i];
    return (float)s/n;
}
int findMax(int a[], int n) {
    int i, m=a[0];
    for(i=1;i<n;i++) if(a[i]>m) m=a[i];
    return m;
}
int findMin(int a[], int n) {
    int i, m=a[0];
    for(i=1;i<n;i++) if(a[i]<m) m=a[i];
    return m;
}
int countA(int a[], int n) {
    int i, c=0;
    for(i=0;i<n;i++) if(a[i]>=90) c++;
    return c;
}
อธิบาย: Prototype บน main ทำให้ compiler รู้ function ก่อน — (float)s/n หาร int ด้วย float ได้ทศนิยม
#07Medium-Hard ✓ เสร็จแล้ว
📅 ระบบบันทึกการเข้าเรียน 5 วิชา 5 คน
📋 Requirement
อาจารย์ต้องการโปรแกรมบันทึกการเข้าเรียน นักเรียน 5 คน × 5 วิชา เก็บใน 2D array (1=มา, 0=ขาด) รับข้อมูลจากผู้ใช้ทีละช่อง จากนั้นเรียก function แยก: (1) นับจำนวนครั้งที่แต่ละคนมาเรียน และพิมพ์ว่า "ผ่าน" (มา ≥ 4 วิชา) หรือ "ตก", (2) หาวิชาที่มีคนมาน้อยที่สุด คืนหมายเลขวิชานั้น ทุก function ต้องมี prototype และรับ 2D array เป็น parameter
int att[5][5] nested loop function + prototype 2D array param [][5]
Functions ที่ต้องมี
void checkAttend(int a[][5], int n); int worstSubject(int a[][5], int n);
OUTPUT
คนที่ 1: มา 4 วิชา → ผ่าน คนที่ 2: มา 3 วิชา → ตก ... วิชาที่คนมาน้อยสุด: วิชา 3
1รับค่า nested loop: for i → for j → scanf(&att[i][j])
2checkAttend: loop แถว (คน), sum ทีละคอลัมน์ → if sum>=4 "ผ่าน"
3worstSubject: loop คอลัมน์ (วิชา), sum แต่ละคอลัมน์, จำ minIdx
42D array parameter ต้องระบุขนาดคอลัมน์: int a[][5]
#include <stdio.h>
void checkAttend (int a[][5], int n);
int  worstSubject(int a[][5], int n);

int main() {
    int att[5][5], i, j;
    printf("กรอก (1=มา 0=ขาด):\n");
    for (i=0; i<5; i++) {
        printf("คนที่ %d (5 วิชา): ", i+1);
        for (j=0; j<5; j++)
            scanf("%d", &att[i][j]);
    }
    checkAttend(att, 5);
    printf("วิชาที่คนมาน้อยสุด: วิชา %d\n",
           worstSubject(att, 5) + 1);
    return 0;
}

void checkAttend(int a[][5], int n) {
    int i, j, sum;
    for (i=0; i<n; i++) {
        sum = 0;
        for (j=0; j<5; j++) sum += a[i][j];
        printf("คนที่ %d: มา %d วิชา → %s\n",
               i+1, sum, sum>=4 ? "ผ่าน" : "ตก");
    }
}

int worstSubject(int a[][5], int n) {
    int j, i, sum, minSum=n+1, minIdx=0;
    for (j=0; j<5; j++) {
        sum = 0;
        for (i=0; i<n; i++) sum += a[i][j];
        if (sum < minSum) { minSum=sum; minIdx=j; }
    }
    return minIdx;
}
อธิบาย: checkAttend วน loop แถว (คน) sum คอลัมน์ — worstSubject วน loop คอลัมน์ (วิชา) sum แถว — 2D array parameter ต้องเขียน [][5] ระบุคอลัมน์เสมอ
⬤ Hard — บทที่ 4–5 (Array + Function + Pointer)
#08Hard ✓ เสร็จแล้ว
💱 ระบบแลกเงินหลายสกุล
📋 Requirement
ธนาคารต้องการโปรแกรมแปลงสกุลเงิน รับ จำนวนเงินบาท จากผู้ใช้ แล้วเรียก function เดียว convertAll ที่รับ THB เข้าไป และ คืนค่า USD, EUR, JPY, GBP พร้อมกันผ่าน pointer 4 ตัว (อัตรา: USD=35.5, EUR=38.2, JPY=0.24, GBP=45.1) จากนั้น main แสดงผลทุกสกุล ถ้า THB น้อยกว่า 100 บาทให้แจ้ง "ยอดต่ำกว่าขั้นต่ำ" และไม่แปลง
double * Pass by Reference multiple return via pointer Guard clause
Function Signature
void convertAll( double thb, double *usd, double *eur, double *jpy, double *gbp );
OUTPUT (1000 บาท)
USD: 28.17 EUR: 26.18 JPY: 4166.67 GBP: 22.17
1ตรวจ thb < 100 ใน main ก่อนเรียก function
2ประกาศ double usd, eur, jpy, gbp; ใน main
3เรียก convertAll(thb, &usd, &eur, &jpy, &gbp);
4ใน function: *usd = thb / 35.5; ทำทุกสกุล
#include <stdio.h>
void convertAll(double thb, double *usd,
               double *eur, double *jpy, double *gbp);
int main() {
    double thb, usd, eur, jpy, gbp;
    printf("จำนวนเงินบาท: "); scanf("%lf", &thb);
    if (thb < 100) {
        printf("ยอดต่ำกว่าขั้นต่ำ\n");
        return 0;
    }
    convertAll(thb, &usd, &eur, &jpy, &gbp);
    printf("USD: %.2f\nEUR: %.2f\nJPY: %.2f\nGBP: %.2f\n",

void analyzeTemp(float a[], int n,
                float *mn, float *mx, float *av) {
    float sum=0; int i;
    *mn = *mx = a[0];
    for(i=0;i<n;i++) {
        sum+=a[i];
        if(a[i]<*mn) *mn=a[i];
        if(a[i]>*mx) *mx=a[i];
    }
    *av = sum/n;
}

int  calcBalance(int a[], int n);
void findTopTxn(int a[], int n, int*md, int*mw);

int main() {
    int txn[50], count=0, val;
    printf("กรอกธุรกรรม (0=จบ):\n");
    while(1) {
        scanf("%d",&val);
        if(val==0||count>=50) break;
        txn[count++]=val;
    }
    int bal=calcBalance(txn,count);
    int md=0,mw=0,deps=0,withs=0;
    findTopTxn(txn,count,&md,&mw);
    countTxn  (txn,count,&deps,&withs);
    printf("ยอดคงเหลือ: %d บาท\n", bal);
    printf("ฝากสูงสุด : %d บาท\n", md);
    printf("ถอนมากสุด : %d บาท\n", mw);
    printf("รายการฝาก : %d ครั้ง\n", deps);
    printf("รายการถอน : %d ครั้ง\n", withs);

    return 0;

int calcBalance(int a[], int n) {
    int i,s=0; for(i=0;i<n;i++) s+=a[i]; return s;
}
void findTopTxn(int a[], int n, int*md, int*mw) {
    int i;
    for(i=0;i<n;i++) {
        if(a[i]>0 && a[i]>*md) *md=a[i];
        if(a[i]<0 && -a[i]>*mw) *mw=-a[i];
    }
}
void countTxn(int a[], int n, int*d, int*w) {
    int i;
    for(i=0;i<n;i++) {
        if(a[i]>0) (*d)++;
        else        (*w)++;
    }
}
