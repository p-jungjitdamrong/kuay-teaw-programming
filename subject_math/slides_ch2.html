ทำไมคอมพิวเตอร์ใช้ฐาน 2?

เพราะ hardware แยกแยะได้แค่ 2 สถานะ: มีไฟ (1) กับ ไม่มีไฟ (0) — เชื่อถือได้ที่สุด

โลกกายภาพ
transistor:  เปิด/ปิด        → 1/0
แรงดันไฟ:    สูง/ต่ำ          → 1/0
จานแม่เหล็ก:  N/S             → 1/0
SSD cell:    มีประจุ/ไม่มี     → 1/0
แผ่น CD:     สะท้อน/ไม่สะท้อน → 1/0

/* ถ้าใช้ฐาน 10 ต้องแยก
   แรงดัน 10 ระดับ → ผิดพลาดง่าย
   2 ระดับ → ชัดเจน ทนทาน noise */
ทุกอย่างคือ bit
ตัวเลข 65        = 01000001
ตัวอักษร 'A'     = 01000001  ← เหมือนกัน!
สี แดงสด        = 11111111 00000000 00000000
เสียง (1 sample) = 0110100101010011
รูปภาพ 1 pixel   = 24 bits

/* ความหมายขึ้นกับ "การตีความ"
   bits เดียวกัน ตีความต่างกัน
   ได้ผลต่างกัน — นี่คือ type
   ในภาษา C นั่นเอง! */
int x = 65 กับ char c = 'A' คือ bits เดียวกันเป๊ะ — ต่างแค่วิธีตีความ นี่คือเหตุผลที่ C แปลง char↔int ได้ตรงๆ

บทที่ 2 · แปลงฐาน

แปลง 10 → 2 : สองวิธี เลือกที่ถนัด

วิธี 1: หารสองเก็บเศษ (มาตรฐาน)
/* 45 → ฐาน 2 */
45 ÷ 2 = 22  เศษ 1  ↑ อ่านจากล่าง
22 ÷ 2 = 11  เศษ 0  │ ขึ้นบน
11 ÷ 2 = 5   เศษ 1  │
 5 ÷ 2 = 2   เศษ 1  │
 2 ÷ 2 = 1   เศษ 0  │
 1 ÷ 2 = 0   เศษ 1  │

อ่านเศษจากล่างขึ้นบน:
45 = 101101₂
วิธี 2: หักลบค่าหลัก (เร็วกว่าถ้าคล่อง)
/* 45 → ฐาน 2
   ไล่จาก 128 64 32 16 8 4 2 1 */

128 > 45         → 0
 64 > 45         → 0
 32 ≤ 45 หัก เหลือ 13 → 1
 16 > 13         → 0
  8 ≤ 13 หัก เหลือ 5  → 1
  4 ≤ 5  หัก เหลือ 1  → 1
  2 > 1          → 0
  1 ≤ 1  หัก เหลือ 0  → 1

45 = 00101101₂ = 101101₂ ✓ ตรงกัน!
ตรวจคำตอบเสมอ: แปลงกลับ 32+8+4+1 = 45 ✓ — แปลงไปกลับได้ = เข้าใจจริง

บทที่ 2 · หน่วย

bit → byte → KB — ทำไม 1KB = 1024?

ลำดับหน่วย
1 bit  = 0 หรือ 1 (เล็กสุด)
1 byte = 8 bits   = เก็บ 0..255
1 KB   = 1024 bytes = 2¹⁰
1 MB   = 1024 KB    = 2²⁰ bytes
1 GB   = 1024 MB    = 2³⁰ bytes
1 TB   = 1024 GB    = 2⁴⁰ bytes

/* ทำไม 1024 ไม่ใช่ 1000?
   เพราะ 2¹⁰ = 1024 ใกล้ 1000
   และ address memory เป็นฐาน 2
   → ใช้กำลังของ 2 ลงตัวพอดี */
n bits เก็บอะไรได้บ้าง
bitsค่าที่เก็บได้ใช้กับ
12 ค่าbool
8256 ค่าchar, สี 1 channel
1665,536short, เสียง CD
324.29 × 10⁹int, IPv4
641.8 × 10¹⁹long long, pointer
 
/* สูตร: n bits → 2ⁿ ค่า
   ช่วง unsigned: 0 .. 2ⁿ−1 */
เกร็ด: HDD ขายแบบ 1TB = 10¹² bytes (ฐาน 10) แต่ OS นับฐาน 2 → เห็นแค่ 931 GB — ไม่ได้โดนโกง แค่คนละหน่วย!
บทที่ 2 · โค้ด C

โค้ด C — แปลงฐานด้วยตัวเอง

#include <stdio.h>

/* 10 → 2 : หารสองเก็บเศษ (เก็บใส่ array แล้วพิมพ์ย้อน) */
void print_binary(unsigned int n) {
    int bits[32], count = 0;
    if (n == 0) { printf("0"); return; }
    while (n > 0) {
        bits[count++] = n % 2;   /* เศษ = bit */
        n /= 2;                  /* หารต่อ */
    }
    for (int i = count-1; i >= 0; i--)  /* อ่านย้อน! */
        printf("%d", bits[i]);
}

/* 2 → 10 : บวกค่าประจำหลัก */
int binary_to_dec(const char *bin) {
    int result = 0;
    for (int i = 0; bin[i] != '\0'; i++) {
        result = result * 2 + (bin[i] - '0');  /* เลื่อนหลัก + บวก */
    }
    return result;
}

int main() {
    print_binary(45);                        /* 101101 */
    printf("\n%d\n", binary_to_dec("101101")); /* 45 */
    printf("%x\n", 181);   /* b5  — %x พิมพ์ hex ได้เลย */
    printf("%d\n", 0xB5);  /* 181 — เขียน hex ในโค้ดได้เลย */
}
บทที่ 2 · เศษส่วนฐาน 2

Bonus: ทำไม 0.1 + 0.2 ≠ 0.3 ในคอมพิวเตอร์?

เพราะ 0.1 เขียนเป็นฐาน 2 ได้ไม่ลงตัว — เหมือน 1/3 เขียนฐาน 10 ได้ไม่ลงตัว (0.3333...)

เศษส่วนฐาน 2
หลักหลังจุดฐาน 2:
0.5   = 2⁻¹ = 0.1₂
0.25  = 2⁻² = 0.01₂
0.75  = 0.5+0.25 = 0.11₂
0.625 = 0.5+0.125 = 0.101₂

/* แต่ 0.1 (หนึ่งส่วนสิบ)? */
0.1₁₀ = 0.0001100110011...₂
        ← ซ้ำ 0011 ไม่รู้จบ!

/* คอมเก็บได้จำกัด bits
   → ต้องตัด → ค่าเพี้ยนนิดๆ */
ผลในโค้ดจริง
double a = 0.1 + 0.2;
printf("%.17f\n", a);
/* 0.30000000000000004  ← เพี้ยน! */

if (a == 0.3)  /* false!! */

/* วิธีถูก: เทียบด้วย epsilon */
#include <math.h>
if (fabs(a - 0.3) < 1e-9)  /* true ✓ */

/* เงิน: ห้ามใช้ float/double!
   ใช้ int เก็บเป็นสตางค์แทน
   129.99 บาท → เก็บ 12999 */
ระบบธนาคารเก็บเงินเป็น integer สตางค์เสมอ — float สะสม error จนเงินหายได้จริง
