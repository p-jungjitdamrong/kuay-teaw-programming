Mean (ค่าเฉลี่ย) — บวกหมด หารด้วยจำนวน

สูตร + ตัวอย่าง
x̄ = (x₁ + x₂ + ... + xₙ) / n

/* คะแนนสอบ 5 คน */
{70, 80, 60, 90, 75}

Mean = (70+80+60+90+75) / 5
     = 375 / 5
     = 75

/* จุดอ่อน: โดน outlier ลาก */
{70, 80, 60, 90, 0}   ← คนหนึ่งขาดสอบ
Mean = 300/5 = 60     ← ต่ำผิดจริง!
โค้ด C
double mean(double arr[], int n) {
    double sum = 0;
    for (int i = 0; i < n; i++)
        sum += arr[i];
    return sum / n;
}
/* Time O(n), Space O(1) */

int main() {
    double scores[] = {70,80,60,90,75};
    printf("mean = %.2f\n",
           mean(scores, 5));
    /* mean = 75.00 */
}
ใช้ Mean เมื่อ: ข้อมูลกระจายปกติ ไม่มี outlier รุนแรง — เช่น ส่วนสูงคน, คะแนนสอบทั่วไป

Median (มัธยฐาน) — เรียงก่อน เอาตัวกลาง

วิธีหา
/* 1. sort ข้อมูลก่อน (ใช้วิชา Sorting!)
   2. n คี่ → ตัวกลางพอดี
      n คู่ → เฉลี่ยสองตัวกลาง */

{12, 15, 11, 13, 950}  n=5 (คี่)
sort → {11, 12, 13, 15, 950}
                ↑
Median = arr[n/2] = arr[2] = 13

{10, 20, 30, 40}  n=4 (คู่)
Median = (arr[1]+arr[2])/2
       = (20+30)/2 = 25

/* 950 ไม่มีผลต่อ median เลย!
   → ทนต่อ outlier */
โค้ด C
/* ต้อง sort ก่อน! */
double median(double arr[], int n) {
    quick_sort(arr, 0, n-1); /* O(n log n) */

    if (n % 2 == 1)
        return arr[n/2];
    return (arr[n/2 - 1] + arr[n/2]) / 2.0;
}

/* เห็นไหม — วิชา Sorting ใช้ตรงนี้!
   หา median ต้อง sort ก่อน
   → O(n log n)

   (มี Quickselect หา median
   ได้ O(n) โดยไม่ sort เต็ม) */
ใช้ Median เมื่อ: มี outlier — เงินเดือน (CEO ลากค่าเฉลี่ย), ราคาบ้าน, response time

Mode (ฐานนิยม) — ค่าที่เจอบ่อยสุด

ตัวอย่าง
{2, 3, 3, 5, 3, 7, 2}

นับ: 2→2ครั้ง, 3→3ครั้ง, 5→1, 7→1
Mode = 3  (เจอบ่อยสุด)

/* ใช้กับข้อมูล category ได้
   ซึ่ง mean/median ทำไม่ได้: */
สีเสื้อขายดี: {แดง,แดง,ดำ,ขาว,แดง}
Mode = แดง ← mean หาไม่ได้!

/* ใช้ใน CS:
   - หา request ที่เจอบ่อยสุด (caching!)
   - หา error code ที่เกิดบ่อยสุด
   - Majority element problem */
โค้ด C (ใช้ counting)
/* ค่าเป็น int ช่วง 0..k */
int mode(int arr[], int n, int k) {
    int count[k+1];
    for (int i=0; i<=k; i++) count[i]=0;

    for (int i=0; i<n; i++)
        count[arr[i]]++;   /* นับ */

    int best = 0;
    for (int v=1; v<=k; v++)
        if (count[v] > count[best])
            best = v;
    return best;
}
/* Time O(n+k), Space O(k)
   — เหมือน Counting Sort เลย! */
Mean = แบ่งเท่ากัน · Median = คนตรงกลาง · Mode = เสียงส่วนใหญ่ — สามค่านี้เล่าเรื่องต่างกัน

Variance และ SD — วัด "ความแกว่ง"

ค่ากลางเท่ากัน ไม่ได้แปลว่าข้อมูลเหมือนกัน — ต้องดูการกระจายด้วย

สองเซิร์ฟเวอร์ mean เท่ากัน
Server A: {14, 15, 15, 16, 15}  mean=15
Server B: {5, 25, 10, 20, 15}   mean=15

/* A นิ่ง — B แกว่งหนัก!
   ถ้าดูแค่ mean จะไม่รู้เลย */

/* Variance σ² =
   เฉลี่ยของ (ระยะห่างจาก mean)² */

A: ((-1)²+0²+0²+1²+0²)/5 = 0.4
B: ((-10)²+10²+(-5)²+5²+0²)/5 = 50

SD = √Variance
A: SD = 0.63 ms  ← เสถียร
B: SD = 7.07 ms  ← แกว่ง 11 เท่า!
โค้ด C
#include <math.h>

double variance(double a[], int n) {
    double m = mean(a, n);
    double sum = 0;
    for (int i=0; i<n; i++)
        sum += (a[i]-m) * (a[i]-m);
    return sum / n;
}

double std_dev(double a[], int n) {
    return sqrt(variance(a, n));
}
/* ทำไมยกกำลังสอง?
   ① กันค่าลบหักล้างค่าบวก
   ② โทษ outlier แรงขึ้น */
SD ต่ำ = ข้อมูลเกาะกลุ่มใกล้ mean · SD สูง = กระจายกว้าง — หน่วยเดียวกับข้อมูลเดิม (variance หน่วยยกกำลังสอง)
บทที่ 0 · Percentile

Percentile — p50, p95, p99 ที่เจอในงานจริง

p95 = ค่าที่ 95% ของข้อมูลต่ำกว่า — มาตรฐานวัด latency ของทุกบริษัท tech

ความหมาย
p50 = median (ครึ่งหนึ่งเร็วกว่านี้)
p95 = 95% ของ requests เร็วกว่านี้
p99 = 99% ของ requests เร็วกว่านี้

/* ตัวอย่าง SLA จริง:
   "p99 latency < 200ms"
   = ใน 100 requests
     อย่างน้อย 99 ต้องเร็วกว่า 200ms */

/* ทำไมไม่ใช้ max?
   max โดน outlier ตัวเดียวพัง
   p99 ตัด 1% ที่แย่สุดออก
   → เสถียรกว่า วัดจริงได้ */
โค้ด C — หา percentile
/* sort ก่อนเสมอ! */
double percentile(double a[], int n,
                  double p) {
    quick_sort(a, 0, n-1);
    int idx = (int)(p / 100.0 * n);
    if (idx >= n) idx = n - 1;
    return a[idx];
}

/* latency 10 ตัว (sorted):
{11,12,12,13,13,14,14,15,15,950}

p50 = a[5]  = 14 ms
p90 = a[9]  = 950 ms  ← จับ outlier ได้!
*/
อีกครั้ง — percentile ต้อง sort ก่อน นี่คือเหตุผลที่เรียน Sorting มาก่อนบทนี้

บทที่ 0 · Use Case

🏦 วิเคราะห์ Transaction แบบครบเครื่อง

#include <stdio.h>
#include <math.h>

typedef struct {
    double mean, median, sd, p95;
} Report;

Report analyze(double amounts[], int n) {
    Report r;
    r.mean = mean(amounts, n);              /* O(n) */
    quick_sort(amounts, 0, n-1);           /* O(n log n) */
    r.median = (n % 2) ? amounts[n/2]
             : (amounts[n/2-1] + amounts[n/2]) / 2;
    r.sd  = std_dev(amounts, n);            /* O(n) */
    r.p95 = amounts[(int)(0.95 * n)];      /* sorted แล้ว O(1) */
    return r;
}

int main() {
    double txn[] = {120, 85, 200, 95, 110, 105, 98, 15000, 130, 90};
    Report r = analyze(txn, 10);
    /* mean=1603 (โดน 15000 ลาก!) median=107 sd≈4467
       → mean >> median = มี outlier
       → 15000 อาจเป็น fraud! แจ้งเตือนระบบ */
}
Fraud detection เบื้องต้น: transaction ที่ห่างจาก mean เกิน 3×SD = ผิดปกติ ควรตรวจสอบ
