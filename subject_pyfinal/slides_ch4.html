ติวบทที่ 4 · def

สร้างฟังก์ชันเอง — def

รูปแบบ
def ชื่อฟังก์ชัน(parameter):
    คำสั่ง
    return ค่าที่ส่งกลับ

# ตัวอย่าง
def add(a, b):
    result = a + b
    return result

# เรียกใช้
x = add(3, 5)
print(x)          # 8
ไม่มี return ก็ได้
def sayhi(name):
    print("สวัสดี", name)

sayhi("สมชาย")     # สวัสดี สมชาย

x = sayhi("สมศรี")
print(x)          # None ← ไม่มี return = คืน None
⚠️ ต้องนิยามก่อนเรียก!
# ❌ ผิด — เรียกก่อนนิยาม

def add(a, b):
    return a + b

# ✓ ถูก — นิยามก่อน แล้วค่อยเรียก

ติวบทที่ 4 · ศัพท์

Parameter vs Argument — ต่างกันตรงไหน

def add(a, b):      # a, b = Parameter
    return a + b       #      (ตัวรับ ตอนนิยาม)

add(3, 5)               # 3, 5 = Argument
                        #        (ค่าจริง ตอนเรียก)
นิยาม เรียกใช้ def add(a, b) ←→ add(3, 5) └─┬─┘ └─┬─┘ Parameter Argument (ตัวแปรรับค่า) (ค่าที่ส่งไป)
# จำนวนต้องตรงกัน!
add(3)          # ❌ TypeError: missing 1 argument
add(3,5,7)      # ❌ TypeError: too many arguments
ส่งค่าได้ 2 แบบ
def info(name, age):
    print(name, age)

# ① Positional — เรียงตามลำดับ
info("สมชาย", 20)      # สมชาย 20
info(20, "สมชาย")      # 20 สมชาย ← สลับผิด!

# ② Keyword — ระบุชื่อ (ลำดับไม่สำคัญ)
info(name="สมชาย", age=20)
info(age=20, name="สมชาย")  # ✓ เหมือนกัน
Default argument
def greet(name, msg="สวัสดี"):
    print(msg, name)

greet("สมชาย")              # สวัสดี สมชาย
greet("สมศรี", "หวัดดี")     # หวัดดี สมศรี

# ⚠️ ตัวมี default ต้องอยู่หลังสุด
def f(a=1, b):   # ❌ SyntaxError
ศัพท์ที่ออกสอบ: Parameter = ตัวแปรใน def (ตัวรับ) · Argument = ค่าจริงตอนเรียก · และตัวที่มี default value ต้องอยู่หลังสุดเสมอ
ติวบทที่ 4 · return

return — คืนค่าได้หลายตัว

คืนค่าเดียว
def square(x):
    return x ** 2

print(square(4))      # 16
คืนหลายค่า — Python ทำได้!
def calc(a, b):
    return a+b, a-b, a*b   # คั่นด้วย ,

s, d, m = calc(10, 3)
print(s, d, m)         # 13 7 30

# หรือรับเป็นก้อนเดียว (tuple)
r = calc(10, 3)
print(r)              # (13, 7, 30)
print(r[0])           # 13
โจทย์ที่ใช้บ่อย
def min_max(nums):
    mn = nums[0]
    mx = nums[0]
    for x in nums:
        if x < mn: mn = x
        if x > mx: mx = x
    return mn, mx        # คืน 2 ค่า

lo, hi = min_max([5,2,9,1])
print(lo, hi)          # 1 9
ฟังก์ชันเรียกฟังก์ชันได้
def area(r):
    return 3.14159 * r ** 2

def total_area(r1, r2):
    return area(r1) + area(r2)   # เรียกตัวอื่น

print(total_area(1, 2))       # 15.7…
จุดเด่นของ Python: คืนหลายค่าได้ด้วย return a, b, c แล้วรับด้วย x, y, z = f() — ภาษา C ทำแบบนี้ไม่ได้ (ต้องใช้ pointer หรือ struct)
ติวบทที่ 4 · Variable Lifetime

local vs global — หัวข้อที่สไลด์ตั้งชื่อไว้เลย

Local — เกิดและตายในฟังก์ชัน
def f():
    x = 10          # local variable
    print(x)        # 10 ✓

f()
print(x)            # ❌ NameError!
#                  x ตายไปแล้วเมื่อจบฟังก์ชัน
Global — ใช้ได้ทั้งโปรแกรม
y = 100               # global variable

def g():
    print(y)        # 100 ✓ อ่านได้

g()
print(y)            # 100 ✓
⚠️ อ่านได้ แต่แก้ไม่ได้!
y = 100

def g():
    y = 50          # สร้าง local ใหม่! ไม่ได้แก้ global
    print(y)        # 50

g()
print(y)            # 100 ← ไม่เปลี่ยน!
อยากแก้ global จริงๆ → ใช้ global
y = 100

def g():
    global y        # ประกาศว่าจะใช้ตัวข้างนอก
    y = 50
    print(y)        # 50

g()
print(y)            # 50 ← เปลี่ยนแล้ว ✓
กฎ 3 ข้อของ scope: ① ตัวแปรในฟังก์ชันเป็น local ใช้นอกฟังก์ชันไม่ได้ ② ตัวแปรนอกฟังก์ชันเป็น global ฟังก์ชันอ่านได้ ③ แต่ถ้ากำหนดค่าในฟังก์ชัน จะสร้าง local ใหม่ ไม่กระทบ global — เว้นแต่ประกาศ global
ติวบทที่ 4 · ไล่ scope

ข้อสอบสาย scope — ไล่ให้เป็น

ตัวอย่างที่ 1
x = 10

def f(x):
    x = x + 5
    print("ใน f:", x)

f(x)
print("นอก f:", x)
ใน f: 15 นอก f: 10
# x ใน f เป็น parameter (local)
# แก้แล้วไม่กระทบ x ข้างนอก
ตัวอย่างที่ 2 — list ต่างออกไป!
a = [1, 2, 3]

def f(lst):
    lst.append(4)      # แก้ตัวเดิม!

f(a)
print(a)             # [1,2,3,4] ← เปลี่ยน!
# แต่ถ้ากำหนดค่าใหม่ทั้งก้อน
def g(lst):
    lst = [9, 9]      # ชี้ไป list ใหม่

g(a)
print(a)             # [1,2,3,4] ← ไม่เปลี่ยน
# สรุป:
# ตัวเลข/string → ส่งค่า แก้ไม่กระทบ
# list/dict     → ส่ง reference
#                 แก้สมาชิก = กระทบตัวจริง
#                 แต่กำหนดใหม่ทั้งก้อน = ไม่กระทบ
จุดที่แยกคนได้คนตก: ส่ง ตัวเลข/string เข้าฟังก์ชัน แก้แล้วข้างนอกไม่เปลี่ยน · แต่ส่ง list/dict แล้วใช้ append()/แก้สมาชิก ข้างนอกเปลี่ยนด้วย!
ติวบทที่ 4 · โจทย์แนวสอบ

เขียนโปรแกรมด้วยฟังก์ชัน — โครงที่ควรใช้

# โจทย์: โปรแกรมคำนวณเกรดนักเรียน แยกเป็นฟังก์ชัน

def get_grade(score):              # ① ฟังก์ชันคำนวณเกรด
    if score >= 80:   return "A"
    elif score >= 70: return "B"
    elif score >= 60: return "C"
    else:              return "F"

def average(scores):              // ② ฟังก์ชันหาค่าเฉลี่ย
    if len(scores) == 0:
        return 0                    # กันหารศูนย์
    return sum(scores) / len(scores)

def summary(scores):              # ③ ฟังก์ชันสรุป (คืน 3 ค่า)
    return average(scores), max(scores), min(scores)

# ── โปรแกรมหลัก ──
scores = []
n = int(input("จำนวนนักเรียน: "))
for i in range(n):
    s = int(input(f"คะแนนคนที่ {i+1}: "))
    scores.append(s)
    print(f"  → เกรด {get_grade(s)}")      # เรียกใช้ฟังก์ชัน

avg, mx, mn = summary(scores)                # รับ 3 ค่าพร้อมกัน
print(f"\nเฉลี่ย {avg:.2f}  สูงสุด {mx}  ต่ำสุด {mn}")
print(f"เกรดเฉลี่ย: {get_grade(avg)}")

"""
จำนวนนักเรียน: 3
คะแนนคนที่ 1: 85  → เกรด A
คะแนนคนที่ 2: 72  → เกรด B
คะแนนคนที่ 3: 55  → เกรด F

เฉลี่ย 70.67  สูงสุด 85  ต่ำสุด 55
เกรดเฉลี่ย: B
"""
หลักออกแบบฟังก์ชัน: 1 ฟังก์ชัน = 1 หน้าที่ · ฟังก์ชันควร return ค่ากลับมาให้ผู้เรียกตัดสินใจ ดีกว่า print ในตัวมันเอง (นำไปใช้ต่อได้)

โค้ดนี้พิมพ์อะไร?

def f(x):
    print(x * 2)

y = f(5)
print(y)
A 10
B 10 แล้ว 10
C 10 แล้ว None
D Error
f(5) → print(5*2) พิมพ์ 10 · ฟังก์ชันไม่มี return → คืน None
y = None → print(y) พิมพ์ None

⚠️ print() ในฟังก์ชัน ≠ return — ถ้าอยากเอาค่าไปใช้ต่อ ต้อง return
ติวบทที่ 4 · ลองข้อสอบ

โค้ดนี้พิมพ์อะไร?

count = 0

def add():
    count = count + 1

add()
print(count)
A 1
B 0
C UnboundLocalError
D None
พอ Python เห็น count = ... ในฟังก์ชัน → ถือว่า count เป็น local ทั้งฟังก์ชัน
แต่ฝั่งขวา count + 1 ต้องอ่าน count ซึ่ง local ยังไม่มีค่า → UnboundLocalError

แก้:  def add():
          global count
          count = count + 1
