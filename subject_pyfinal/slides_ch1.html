ติวบทที่ 1 · Slice

Slice [start:stop] — ตัดเป็นช่วง

รูปแบบ
a = [10, 20, 30, 40, 50]
#     0    1    2    3    4

a[1:4]     # [20,30,40]  ← ไม่รวม index 4!
a[0:3]     # [10,20,30]
a[:3]      # [10,20,30]  เว้นหน้า = เริ่มที่ 0
a[2:]      # [30,40,50]  เว้นหลัง = ถึงท้าย
a[:]       # [10,20,30,40,50] ทั้งหมด
a[-2:]     # [40,50]  2 ตัวท้าย
# มี step ด้วย [start:stop:step]
a[::2]     # [10,30,50]  เว้นตัว
a[::-1]    # [50,40,30,20,10] กลับด้าน!
กฎเดียวที่ต้องจำ
# a[start:stop]
#   รวม start
#   ไม่รวม stop
# → จำนวนสมาชิก = stop − start

a[1:4]  # ได้ 4−1 = 3 ตัว

ติวบทที่ 1 · Methods

List Methods ที่ต้องรู้จัก

Methodทำอะไรตัวอย่าง (a = [1,2,3])
append(x)เพิ่มต่อท้ายa.append(4) → [1,2,3,4]
insert(i, x)แทรกที่ตำแหน่ง ia.insert(0,9) → [9,1,2,3]
remove(x)ลบตัวที่มีค่า x (ตัวแรก)a.remove(2) → [1,3]
pop(i)ลบตามตำแหน่ง + คืนค่าa.pop(0) → คืน 1, a=[2,3]
pop()ลบตัวท้าย + คืนค่าa.pop() → คืน 3, a=[1,2]

reverse()กลับลำดับa.reverse() → [3,2,1]
index(x)หาตำแหน่งของค่า xa.index(2) → 1
count(x)นับว่ามีค่า x กี่ตัว[1,2,2].count(2) → 2

ติวบทที่ 1 · จุดหลอก

sort() vs sorted() + method คืนค่าอะไร

sort() — แก้ list เดิม คืน None
a = [3, 1, 2]
a.sort()
print(a)          # [1,2,3] ✓

# ❌ ผิดที่คนพลาดมาก
b = a.sort()
print(b)          # None !!
# เพราะ sort() ไม่คืน list
# มันแก้ตัวเดิมแล้วคืน None
sorted() — สร้างใหม่ ไม่แตะเดิม
a = [3, 1, 2]
b = sorted(a)
print(a)   # [3,1,2] เดิมไม่เปลี่ยน
print(b)   # [1,2,3] ตัวใหม่
เรียงมากไปน้อย
a.sort(reverse=True)   # [3,2,1]
sorted(a, reverse=True)  # [3,2,1]
method ที่คืน None (แก้ตัวเดิม)
append() insert() remove()
sort()   reverse() clear()
# ทั้งหมดคืน None → ห้ามเอาไปรับค่า!

# method ที่คืนค่าจริง
pop()      # คืนตัวที่ถูกลบ
index()    # คืนตำแหน่ง
count()    # คืนจำนวน
len() sum() max() min()  # built-in
กฎจำง่าย: method ที่แก้ list เดิม (append/sort/remove...) คืน None — เขียน a = a.sort() แล้ว a จะกลายเป็น None ทันที (bug ที่หาไม่เจอ)
ติวบทที่ 1 · วน list

วน list 2 วิธี + built-in ที่ช่วยได้

① วนค่าตรงๆ (นิยม)
scores = [85, 90, 78]

for s in scores:
    print(s)
# 85 / 90 / 78
② วนด้วย index (เมื่อต้องรู้ตำแหน่ง)
for i in range(len(scores)):
    print(i, scores[i])
# 0 85 / 1 90 / 2 78

# ต้องแก้ค่าใน list → ต้องใช้แบบนี้
for i in range(len(scores)):
    scores[i] += 5      ✓ แก้ได้

for s in scores:
    s += 5              ❌ ไม่เปลี่ยน list!
built-in ที่ใช้กับ list
a = [85, 90, 78, 92]

len(a)     # 4    จำนวนสมาชิก
sum(a)     # 345  ผลรวม
max(a)     # 92   มากสุด
min(a)     # 78   น้อยสุด
sum(a)/len(a)  # 86.25 เฉลี่ย

90 in a    # True  มีไหม
99 in a    # False
โจทย์คลาสสิก — เขียนเองไม่ใช้ built-in
total = 0
mx = a[0]           # เริ่มที่ตัวแรก!
for x in a:
    total += x
    if x > mx:
        mx = x
print(total, mx, total/len(a))
จุดหลอก: วนแบบ for s in scores: s += 5 ไม่เปลี่ยน list เพราะ s เป็นแค่สำเนาค่า — ถ้าต้องแก้ค่าใน list ต้องวนด้วย index แล้วเขียน scores[i] += 5
ติวบทที่ 1 · โจทย์แนวสอบ

โจทย์ที่ออกบ่อย — เขียนให้ได้ทั้ง 3 ข้อ

① รับคะแนน n คน แล้วสรุป
n = int(input("จำนวนคน: "))
scores = []                      # list ว่าง

for i in range(n):
    s = int(input(f"คะแนนคนที่ {i+1}: "))
    scores.append(s)          # เพิ่มเข้า list

print("รวม   =", sum(scores))
print("เฉลี่ย =", sum(scores)/len(scores))
print("สูงสุด =", max(scores))
print("ต่ำสุด =", min(scores))
② นับเฉพาะที่เข้าเงื่อนไข
scores = [85, 45, 90, 60, 30]
passed = 0
for s in scores:
    if s >= 50:
        passed += 1
print(f"ผ่าน {passed}/{len(scores)} คน")
# ผ่าน 3/5 คน
③ แยกเป็น 2 list
nums = [1,2,3,4,5,6]
even = []
odd  = []
for x in nums:
    if x % 2 == 0:
        even.append(x)
    else:
        odd.append(x)
print(even, odd)
# [2,4,6] [1,3,5]
