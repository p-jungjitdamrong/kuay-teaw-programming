Dictionary — เก็บเป็นคู่ key : value

นิยามตามสไลด์
# "ดิกท์ (Dictionary) ตัวแปรชุดที่เก็บข้อมูล
#  ในรูปแบบคู่ของ key และ value โดยที่
#  • key ใช้เป็นดัชนี (Index) ในการเข้าถึงข้อมูล
#  • value คือค่าของข้อมูล"

score = {'A': 97, 'B': 86, 'C': 68}
#        keys:  'A'  'B'  'C'
#        values: 97   86   68
รูปแบบการสร้าง
ชื่อตัวแปร = { key1: value1, key2: value2 }

score = {"Jim": 'A', "Kim": 'B'}
empty = {}                    # dict ว่าง

# key เป็นได้ทั้ง int, float, string, bool
# value เป็นอะไรก็ได้
d = {1: "หนึ่ง", "a": [1,2], True: 3.5}
เข้าถึงด้วย key ไม่ใช่ index!
score = {'A': 97, 'B': 86, 'C': 68}

score['A']      # 97 ✓
score[0]        # ❌ KeyError! ไม่มี key 0
score['Z']      # ❌ KeyError!

# ปลอดภัยกว่า — ใช้ get()
score.get('Z')          # None (ไม่ error)
score.get('Z', 0)       # 0 (ค่า default)

# เช็คก่อนใช้
if 'A' in score:
    print(score['A'])
# ⚠️ กฎสำคัญ: key ต้องไม่ซ้ำกัน
d = {'a': 1, 'a': 2}
print(d)     # {'a': 2} ← ตัวหลังทับตัวแรก
ต่างจาก List ตรงไหน: List เข้าถึงด้วย index ตัวเลข 0,1,2 · Dict เข้าถึงด้วย key ที่เรากำหนดเอง · และ dict ไม่มีลำดับให้ slice — d[0:2] ใช้ไม่ได้
ติวบทที่ 2 · เพิ่ม/แก้/ลบ

Updating & Removing Dictionary Items

เพิ่ม / แก้ — เขียนเหมือนกัน!
score = {'A': 97, 'B': 86}

# เพิ่ม (key ยังไม่มี)
score['C'] = 68
# {'A':97, 'B':86, 'C':68}

# แก้ (key มีอยู่แล้ว)
score['A'] = 100
# {'A':100, 'B':86, 'C':68}

# ← syntax เดียวกัน! Python ดูให้เองว่า
#   key มีอยู่แล้วหรือยัง

# รวม dict
score.update({'D': 55, 'E': 70})
ลบ 3 วิธี
del score['B']        # ลบ key 'B'

score.pop('C')         # ลบ + คืนค่า 68

score.clear()          # ล้างทั้งหมด → {}

# ลบ key ที่ไม่มี → KeyError
del score['Z']        # ❌
score.pop('Z', None)   # ✓ ไม่ error
method ที่ควรรู้
keys()คืน key ทั้งหมด
values()คืน value ทั้งหมด
items()คืนคู่ (key, value)
get(k, d)อ่านค่าแบบไม่ error
len(d)จำนวนคู่ทั้งหมด
จุดที่ต่างจาก list ชัดๆ: list ต้องใช้ append() เพิ่มสมาชิก แต่ dict แค่ d['new'] = value ก็เพิ่มได้เลย — และคำสั่งเดียวกันนี้ใช้แก้ค่าเดิมด้วย

Looping and Dictionaries — 3 แบบ

score = {'A': 97, 'B': 86, 'C': 68}

# ① วน key (ค่าเริ่มต้น)
for k in score:
    print(k, score[k])
A 97 B 86 C 68
# เท่ากับเขียน for k in score.keys():

# ② วน value อย่างเดียว
for v in score.values():
    print(v)      # 97 / 86 / 68
# ③ วนทั้งคู่ด้วย items() — นิยมสุด
for k, v in score.items():
    print(f"{k} ได้ {v} คะแนน")
A ได้ 97 คะแนน B ได้ 86 คะแนน C ได้ 68 คะแนน
โจทย์คลาสสิก — นับความถี่
text = "banana"
count = {}
for ch in text:
    if ch in count:
        count[ch] += 1
    else:
        count[ch] = 1
print(count)
# {'b':1, 'a':3, 'n':2}
โจทย์นับความถี่คือข้อสอบยอดฮิตของ dict — โครงคือ "ถ้ามี key แล้ว +1 ถ้ายังไม่มีตั้งเป็น 1" · เขียนสั้นได้ด้วย count[ch] = count.get(ch, 0) + 1

ติวบทที่ 2 · String Methods

String Methods ที่ต้องรู้จัก

Methodทำอะไรตัวอย่าง (s = " Hello World ")
upper()เป็นตัวใหญ่หมด"abc".upper() → "ABC"
lower()เป็นตัวเล็กหมด"ABC".lower() → "abc"
strip()ตัดช่องว่างหัว-ท้ายs.strip() → "Hello World"
split(x)แยกเป็น list"a,b,c".split(",") → ['a','b','c']
replace(a,b)แทนที่ข้อความ"aaa".replace("a","b") → "bbb"
find(x)หาตำแหน่ง (ไม่เจอ = −1)"Hello".find("l") → 2
count(x)นับจำนวนครั้ง"Hello".count("l") → 2
join(list)รวม list เป็น string",".join(['a','b']) → "a,b"
isdigit()เป็นตัวเลขล้วนไหม"123".isdigit() → True
isalpha()เป็นตัวอักษรล้วนไหม"abc".isalpha() → True
คู่หูที่ใช้บ่อยที่สุด: split() แยก string → list · join() รวม list → string · strip() ตัด \n ตอนอ่านไฟล์ (บท 5!)

String Compare และการวนอ่านทีละตัว

เปรียบเทียบตามลำดับ ASCII
"abc" == "abc"     # True
"abc" == "ABC"     # False ← case-sensitive!
"a" < "b"          # True  (a=97 < b=98)
"A" < "a"          # True  (A=65 < a=97)
"apple" < "banana" # True  เทียบทีละตัว

# เทียบแบบไม่สนตัวพิมพ์
s1.lower() == s2.lower()
เช็คว่ามีข้อความย่อยไหม
"ell" in "Hello"     # True
"xyz" in "Hello"     # False

s = "Hello"

for ch in s:
    print(ch, end=' ')   # H e l l o

# ② วนด้วย index

โจทย์รวม dict + string

# โจทย์: รับชื่อ-คะแนนนักเรียน แล้วสรุปผล

students = {}                                # dict ว่าง

n = int(input("จำนวนนักเรียน: "))
for i in range(n):
    line = input("ชื่อ,คะแนน: ")          # เช่น "สมชาย,85"
    name, score = line.split(",")          # แยกด้วย , → list 2 ตัว
    students[name.strip()] = int(score)    # key=ชื่อ value=คะแนน

# แสดงทั้งหมด
print("\n── รายชื่อ ──")
for name, sc in students.items():        // วนทั้งคู่
    status = "ผ่าน" if sc >= 50 else "ตก"
    print(f"{name:<12}{sc:>4}  {status}")

# สรุป — ใช้ values() กับ built-in
scores = list(students.values())
print(f"\nเฉลี่ย {sum(scores)/len(scores):.2f}")
print(f"สูงสุด {max(scores)}  ต่ำสุด {min(scores)}")

# หาว่าใครได้คะแนนสูงสุด
for name, sc in students.items():
    if sc == max(scores):
        print("คะแนนสูงสุด:", name)

"""
จำนวนนักเรียน: 3
ชื่อ,คะแนน: สมชาย,85
ชื่อ,คะแนน: สมศรี,92
ชื่อ,คะแนน: สมหญิง,45

── รายชื่อ ──
สมชาย          85  ผ่าน
สมศรี          92  ผ่าน
สมหญิง         45  ตก

เฉลี่ย 74.00
สูงสุด 92  ต่ำสุด 45
คะแนนสูงสุด: สมศรี
"""
3 เทคนิคในโจทย์นี้: ① split(",") แยกข้อความเป็นหลายค่า ② list(d.values()) แปลง values เป็น list เพื่อใช้ sum/max/min ③ items() วนทั้ง key และ value พร้อมกัน

String เป็น immutable — อ่าน s[0] ได้ แต่เขียนทับไม่ได้

d = {'x':1, 'y':2} แล้ว d['z'] ได้อะไร?

A None
B 0
C KeyError
D IndexError
d['z']            ❌ KeyError: 'z'
