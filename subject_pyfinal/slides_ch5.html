open() รับ 2 อย่าง — ชื่อไฟล์ + โหมด

ชื่อตัวแปร = open( "ชื่อไฟล์" , "โหมด" )

f = open("data.txt", "r")     # เปิดอ่าน
โหมดชื่อเต็มทำอะไรถ้าไฟล์ยังไม่มีของเดิม

"w"Writeเขียนสร้างให้ใหม่ลบทิ้งหมด!
"a"Appendเขียนต่อท้ายสร้างให้ใหม่เก็บไว้

write() — เขียนได้เฉพาะ string

เขียนทับ (w) vs เขียนต่อ (a)
# "w" — ล้างของเดิมแล้วเขียนใหม่
f = open("log.txt", "w")
f.write("บรรทัดที่ 1\n")
f.write("บรรทัดที่ 2\n")
f.close()
บรรทัดที่ 1 บรรทัดที่ 2
# "a" — ต่อท้ายของเดิม
f = open("log.txt", "a")
f.write("บรรทัดที่ 3\n")
f.close()
บรรทัดที่ 1 บรรทัดที่ 2 บรรทัดที่ 3
2 จุดพลาดที่เจอบ่อย
# ❌ ลืม \n — ทุกอย่างติดกันหมด
f.write("A")
f.write("B")
f.write("C")
# ไฟล์ได้: ABC   (บรรทัดเดียว)
# write ไม่ขึ้นบรรทัดให้เอง! ต่างจาก print

# ❌ เขียนตัวเลขตรงๆ → TypeError
f.write(100)          # Error!
f.write(str(100))     # ✓ แปลงก่อน
f.write(f"{score}\n")  # ✓ f-string
writelines() — เขียน list ทีเดียว
names = ["Somchai\n", "Somsri\n"]
f = open("out.txt", "w")
f.writelines(names)
f.close()
# ก็ไม่เติม \n ให้เหมือนกัน
# ต้องใส่มาในแต่ละตัวเอง
เทียบให้จำ: print() ขึ้นบรรทัดใหม่ให้อัตโนมัติ · write() ไม่ขึ้นให้ ต้องใส่ \n เอง — และ write รับ string เท่านั้น ตัวเลขต้อง str() ก่อน
ติวบทที่ 5 · with

with — เปิดแล้วปิดให้เอง (วิธีที่ควรใช้)

แบบธรรมดา — ต้องปิดเอง
f = open("data.txt", "r")
data = f.read()
f.close()          # ← ลืมบ่อย!

# ถ้าโค้ดตรงกลางพัง (error)
# → close() ไม่ถูกเรียก
# → ไฟล์ค้างเปิด
แบบ with — ปิดอัตโนมัติ
with open("data.txt", "r") as f:
    data = f.read()
    print(data)
# ออกจากบล็อก → ปิดให้เอง
# แม้เกิด error ก็ยังปิด ✓
ใช้จริง — อ่านและเขียน
# อ่านทีละบรรทัด
with open("scores.txt", "r") as f:
    for line in f:
        print(line.strip())

# เขียน
with open("out.txt", "w") as f:
    for i in range(1, 6):
        f.write(f"บรรทัดที่ {i}\n")

# เปิด 2 ไฟล์พร้อมกัน (copy)
with open("in.txt") as src, \
     open("out.txt", "w") as dst:
    dst.write(src.read())
โครงที่ควรใช้ตลอด: with open(ชื่อไฟล์, โหมด) as f: แล้วเขียนงานไว้ในบล็อก — ไม่ต้องเขียน close() เลย และปลอดภัยกว่า

ปัญหา: เปิดไฟล์ที่ไม่มี → พัง

import os

if os.path.exists("data.txt"):
    with open("data.txt") as f:

    print("ไม่พบไฟล์")

try:
    with open("data.txt") as f:

    print("ไม่พบไฟล์")

ติวบทที่ 5 · Use Case

🏦 โปรแกรมบันทึกธุรกรรม — เขียนแล้วอ่านกลับ

import os

FILE = "transactions.txt"

# ── ① บันทึกธุรกรรมใหม่ (ต่อท้าย ไม่ลบของเดิม) ──
def add_txn(name, amount):
    with open(FILE, "a") as f:          # "a" = append!
        f.write(f"{name},{amount}\n")   # คั่นด้วย , (รูปแบบ CSV)

# ── ② อ่านทั้งหมดมาสรุป ──
def summary():
    if not os.path.exists(FILE):            # เช็คก่อนเสมอ
        print("ยังไม่มีข้อมูล")
        return

    total = 0
    count = 0
    with open(FILE, "r") as f:
        for line in f:                      # วนทีละบรรทัด
            line = line.strip()               # ตัด \n ทิ้ง
            if line == "":                    # ข้ามบรรทัดว่าง
                continue
            name, amount = line.split(",")    # แยกด้วย ,
            amount = float(amount)             # แปลงกลับเป็นตัวเลข
            print(f"  {name:<10}{amount:>10,.2f}")
            total += amount
            count += 1

    print(f"  รวม {count} รายการ = {total:,.2f} บาท")

# ── ใช้งาน ──
add_txn("สมชาย", 1200)
add_txn("สมศรี", 850.50)
add_txn("สมหญิง", 3000)
summary()

"""
ไฟล์ transactions.txt ที่ได้:
สมชาย,1200
สมศรี,850.5
สมหญิง,3000

ผลลัพธ์:
  สมชาย        1,200.00
  สมศรี          850.50
  สมหญิง       3,000.00
  รวม 3 รายการ = 5,050.50 บาท
"""

ติวบทที่ 5 · รับค่าจากผู้ใช้

Entry — get() อ่านค่า, delete() ล้างช่อง

โปรแกรมทักทาย (ครบวงจร)
import tkinter as tk

win = tk.Tk()
win.title("ทักทาย")
win.geometry("280x150")

def greet():
    name = ent.get()          # อ่านค่าจากช่อง
    if name == "":
        result.config(text="กรุณากรอกชื่อ")
    else:
        result.config(text=f"สวัสดี {name}!")

tk.Label(win, text="ชื่อ:").pack()
ent = tk.Entry(win, width=25)
ent.pack()

tk.Button(win, text="ทักทาย",
          command=greet).pack(pady=8)

result = tk.Label(win, text="",
                  font=("Arial", 14))
result.pack()

win.mainloop()
┌─ ทักทาย ─────── ─ □ ✕ ┐ │ ชื่อ: │ │ [ สมชาย ] │ │ │ │ [ ทักทาย ] │ │ │ │ สวัสดี สมชาย! │ └───────────────────────┘
method ของ Entry ที่ใช้บ่อย
ent.get()              # อ่านค่า (ได้ string!)
ent.delete(0, tk.END)  # ล้างช่อง
ent.insert(0, "ค่าเริ่มต้น")

# get() คืน string เสมอ!
# ต้องแปลงก่อนคำนวณ
n = int(ent.get())      # ← ต้องแปลง
Label เปลี่ยนข้อความด้วย config()
result.config(text="ข้อความใหม่")
# หรือ result["text"] = "ข้อความใหม่"
ent.get() คืน string เสมอ เหมือน input() — จะคำนวณต้อง int() หรือ float() ก่อน และควรดักด้วย try-except เผื่อผู้ใช้พิมพ์ตัวอักษร
ติวบทที่ 5 · Use Case

🏦 เครื่องคิดดอกเบี้ยเงินฝาก — รวมทุกอย่าง

import tkinter as tk

win = tk.Tk()
win.title("คำนวณดอกเบี้ย")
win.geometry("340x220")

def calculate():
    try:                                        # ดัก error (บทที่แล้ว!)
        money = float(ent_money.get())          # get() ได้ string → แปลง
        rate  = float(ent_rate.get())
        years = int(ent_year.get())
    except ValueError:
        result.config(text="⚠️ กรุณากรอกตัวเลข", fg="red")
        return

    if money <= 0 or years <= 0:            # ตรวจความสมเหตุสมผล
        result.config(text="⚠️ ค่าต้องมากกว่า 0", fg="red")
        return

    total = money
    for i in range(years):                     # ลูปทบต้นทีละปี
        total = total * (1 + rate / 100)

    interest = total - money
    result.config(
        text=f"เงินรวม {total:,.2f} บาท\nดอกเบี้ย {interest:,.2f} บาท",
        fg="green")

# ── จัดหน้าด้วย grid (ตาราง 2 คอลัมน์) ──
tk.Label(win, text="เงินต้น (บาท):").grid(row=0, column=0, padx=8, pady=5, sticky="e")
ent_money = tk.Entry(win, width=15)
ent_money.grid(row=0, column=1)

tk.Label(win, text="ดอกเบี้ย (%/ปี):").grid(row=1, column=0, padx=8, pady=5, sticky="e")
ent_rate = tk.Entry(win, width=15)
ent_rate.grid(row=1, column=1)

tk.Label(win, text="จำนวนปี:").grid(row=2, column=0, padx=8, pady=5, sticky="e")
ent_year = tk.Entry(win, width=15)
ent_year.grid(row=2, column=1)

tk.Button(win, text="คำนวณ", command=calculate, width=12) \
  .grid(row=3, column=0, columnspan=2, pady=10)   # กินพื้นที่ 2 คอลัมน์

result = tk.Label(win, text="", font=("Arial", 12))
result.grid(row=4, column=0, columnspan=2)

win.mainloop()

# เงินต้น 10000 · ดอกเบี้ย 2.5% · 3 ปี
# → เงินรวม 10,768.91 บาท / ดอกเบี้ย 768.91 บาท
