Array ใน Java: ประกาศ ≠ สร้าง — ต้อง new!

3 ขั้นตอน (สไลด์อาจารย์เน้นตรงนี้)
// ① ประกาศ — "ยังไม่มีอาเรย์เกิดขึ้นจริง"
//   (คำพูดในสไลด์ตรงๆ!)
int table[];
int[] table2;   // เขียนได้สองแบบ

// ② สร้างด้วย new — ตอนนี้ค่อยมีจริง
table = new int[5];

// ③ ใช้งาน — index เริ่ม 0
table[0] = 1;

// หรือรวบขั้น ② กับค่าเริ่มต้น:
int monthDays[] = {31,28,31,30,...};
ต่างจาก C ยังไง (ข้อเทียบภาษา)
/* C:   int a[5];  ← ได้พื้นที่ทันที
   Java: int[] a;   ← ได้แค่ตัวแปรอ้างอิง
         a = new int[5]; ← จึงมีของจริง

   เพราะ array ใน Java เป็น object!
   (สร้างด้วย new เหมือน object อื่น)

   ใช้ก่อน new → NullPointerException
   ← ข้อ "โค้ดนี้ error เพราะอะไร"
     ชอบเอาจุดนี้มาออก */

int[] a;
a[0] = 5;   // ❌ ยังไม่ new!
ประโยคจากสไลด์ที่เป็นข้อสอบได้เลย: "หลังจากการประกาศ ไม่มีการสร้างอาเรย์เกิดขึ้นจริง" — array เกิดจริงตอน new หรือใส่ {ค่าเริ่มต้น} เท่านั้น
ติวบทที่ 2 · จุดหลอกที่ 2

length ไม่มีวงเล็บ vs length() มีวงเล็บ

Array → .length (field)
int[] arr = {1, 2, 3};
for (int i = 0; i < arr.length; i++)
    System.out.println(arr[i]);
// arr.length = 3  ← ไม่มีวงเล็บ!

// 2 มิติ:
// arr.length     = จำนวนแถว
// arr[i].length  = จำนวนคอลัมน์แถว i
String → .length() (method)
String s = "Dot saw I was Tod";
int len = s.length();   // มีวงเล็บ!

/* ทำไมต่างกัน?
   array.length เป็น field พิเศษ
   String.length() เป็น method

   เขียนสลับ = compile error
   arr.length()  ❌
   s.length      ❌

   ← ข้อ "บรรทัดใด compile ผ่าน"
     เอาคู่นี้มาหลอกทุกปี */
ท่องสั้นๆ: array เปลือย (length) · String ใส่รองเท้า (length()) — และ charAt(i) ของ String มีวงเล็บเสมอเพราะเป็น method เหมือนกัน
ติวบทที่ 2 · foreach

foreach — syntax ตามสไลด์เป๊ะๆ

1 มิติ
// for (type var : array)
int[] arr = {1, 2, 2, 3};
for (int eachArr : arr) {
    System.out.println(eachArr);
}

/* อ่านว่า "สำหรับแต่ละ int ใน arr"
   ใช้เมื่อ: อ่านทุกตัวเรียงลำดับ
   ไม่ใช้เมื่อ: ต้องรู้ index หรือ
   ต้องแก้ค่าใน array */
2 มิติ — ซ้อนสองชั้น
int[][] arr2D = { {1,2,3}, {4,5,6} };

for (int[] arr1D : arr2D) {     // ทีละแถว
    for (int elem : arr1D) {    // ทีละตัว
        System.out.print(elem);
    }
}

/* จุดสอบ: ชั้นนอกรับ int[]
   (ทั้งแถว) ไม่ใช่ int!
   for (int x : arr2D) ❌ */
ข้อเติม syntax: foreach 2 มิติ ชั้นนอกต้องเป็น int[] — เพราะสมาชิกของ array 2 มิติคือ "array 1 มิติ" ทั้งแถว ไม่ใช่ตัวเลขเดี่ยว
ติวบทที่ 2 · String

String — สร้าง 2 แบบ + methods หลัก

สร้าง (ตามสไลด์)
// แบบ literal
String greeting = "Hello world !";

// แบบ char array → String
char[] helloArray =
    {'h','e','l','l','o','.'};
String helloString =
    new String(helloArray);
methods ที่สไลด์ไล่ให้
s.length()        ความยาว
s.charAt(8)       ตัวอักษรตำแหน่ง 8
s.substring(2,5)  ตัดช่วง [2,5)
s.equals(t)       เทียบ "เนื้อหา"
s.indexOf("ab")   หาตำแหน่ง
s.replace(a, b)   แทนที่ (คืนตัวใหม่!)
s.split(" ")      แยกเป็น array

/* "Strings are immutable"
   — ชื่อตัวอย่างในสไลด์บอกใบ้เอง!
   ทุก method คืน String ใหม่
   ตัวเดิมไม่เคยเปลี่ยน */
เก็งจากสไลด์: ตัวอย่าง charAt ในสไลด์ใช้ s = "Strings are immutable", charAt(8) → นับจาก 0: S-t-r-i-n-g-s-␣-a → ได้ 'a' — ฝึกนับ index จาก 0 ให้ชิน โจทย์ charAt ให้คะแนนฟรีถ้านับเป็น

String immutable — bug เงียบที่ออกข้อสอบบ่อยสุด

โค้ดหลอก
String s = "sad sad";
s.replace("sad", "ok");
System.out.println(s);
// พิมพ์ "sad sad" — ไม่เปลี่ยน!

// replace คืน String ใหม่
// แต่ไม่มีใครรับ → ถูกทิ้ง

// แก้ให้ถูก: รับค่ากลับ
s = s.replace("sad", "ok");
// ตอนนี้ "ok ok" ✓
โยงถึงบท 3 (== vs equals)
String s1 = new String("abc");
String s2 = new String("abc");

s1.equals(s2)   // true  (เนื้อหาเท่า)
s1 == s2        // false (คนละวัตถุ!)

/* == เทียบ "reference ชี้ตัวเดียวกันไหม"
   equals เทียบ "เนื้อหาเหมือนกันไหม"
   ← สไลด์บท 3 มีตัวอย่างนี้ตรงๆ
   จะเจาะเต็มๆ บทหน้า */
