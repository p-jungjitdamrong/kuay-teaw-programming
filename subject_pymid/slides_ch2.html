① 5 / 2 = 2.5 — Python 3 หารได้ทศนิยมเสมอ ไม่ปัดลง (ต่างจาก C!) · อยากได้จำนวนเต็มใช้ //

ติวบทที่ 2 · เจาะจุดหลอก

ทำไม −5 % 2 = 1 (ไม่ใช่ −1)

หลักการ: Python ยึด "ตัวหาร"
# สูตร: a % b มีเครื่องหมายตาม b

 5 % 2  = 1     # b บวก → ผลบวก
-5 % 2  = 1     # b บวก → ผลบวก
 5 % -2 = -1    # b ลบ → ผลลบ
-5 % -2 = -1    # b ลบ → ผลลบ
คิดยังไงให้ได้ 1
# Python: a // b ปัดลงเสมอ (floor)
-5 // 2 = -3      (ไม่ใช่ -2!)

# แล้วใช้สูตร a = (a//b)*b + (a%b)
-5 = (-3)*2 + r
-5 = -6 + r
 r = 1          ✓
ตัวหารและตัวหารลงตัว
7 // 2   = 3      # หารปัดลง (int)
7 / 2    = 3.5    # หารปกติ (float)
7 % 2    = 1      # เศษ

-7 // 2  = -4     ⚠️ ปัดลง ไม่ใช่ -3
-7 % 2   = 1      ⚠️

2 ** 3   = 8      # ยกกำลัง
2 ** 0.5 = 1.414  # รากที่สอง
# การใช้ % ที่ออกสอบบ่อย:
# เช็คเลขคู่/คี่
if n % 2 == 0: print("คู่")
# หารลงตัวไหม
if n % 5 == 0: print("หาร 5 ลงตัว")
# แยกหลัก
n = 123
print(n % 10)     # 3  หลักหน่วย
print(n // 10 % 10) # 2  หลักสิบ
ท่องสั้นๆ: % ผลลัพธ์เครื่องหมายตามตัวหาร · // ปัดลงเสมอ (−7//2 = −4 ไม่ใช่ −3) — สองข้อนี้คือจุดที่คนคุ้นภาษา C ตอบผิด
ติวบทที่ 2 · ลำดับ

ลำดับตัวดำเนินการ — ตามสไลด์อาจารย์

ลำดับ (สูงไปต่ำ)
1.  ( )           วงเล็บ
2.  **            ยกกำลัง
3.  *  /  //  %   คูณ หาร มอด
4.  +  -          บวก ลบ

Note จากสไลด์: ถ้าลำดับเท่ากัน
คำนวณจากซ้ายไปขวา
ตัวอย่างที่อาจารย์ให้
Width = 2*5+(6*2)/5

# ① วงเล็บก่อน: (6*2) = 12
# ② คูณ/หาร ซ้าย→ขวา:
#    2*5 = 10
#    12/5 = 2.4
# ③ บวก: 10 + 2.4 = 12.4
อีกตัวอย่าง
High = (6+5)+102%2

# ① วงเล็บ: (6+5) = 11
# ② % : 102 % 2 = 0
# ③ + : 11 + 0 = 11
ฝึกเอง — ไล่ทีละขั้น
10 + 6 * 2 - 4 / 2
# ① 6*2 = 12    (คูณก่อน)
# ② 4/2 = 2.0   (หาร ซ้ายไปขวา)
# ③ 10+12 = 22
# ④ 22-2.0 = 20.0    ← float!

2 ** 3 ** 2
# ** คำนวณ "ขวาไปซ้าย" (ข้อยกเว้น!)
# 3**2 = 9 ก่อน → 2**9 = 512
# ไม่ใช่ (2**3)**2 = 64
เทคนิคทำข้อสอบคำนวณ: เขียนผลลัพธ์กำกับใต้แต่ละขั้น ห้ามคิดในหัวรวดเดียว · และระวังชนิดข้อมูล — มี / เมื่อไหร่ผลลัพธ์เป็น float ทันที (20.0 ไม่ใช่ 20)
ติวบทที่ 2 · ตัวดำเนินการย่อ

+= −= *= /= %= — ตามสไลด์อาจารย์

ความหมาย
x += 1    ➔  x = x + 1
x += 5    ➔  x = x + 5
x -= 9    ➔  x = x - 9
x -= -9   ➔  x = x - (-9)  # = x + 9
x *= 7    ➔  x = x * 7
x /= 5    ➔  x = x / 5
x %= 5    ➔  x = x % 5
x %= x    ➔  x = x % x     # = 0 เสมอ
ตัวอย่างในสไลด์
n, m = 3, 4      # กำหนดหลายตัวพร้อมกัน
n *= m
print(n)          # 12
ไล่ trace ให้เป็น
x = 10
x += 5      # x = 15
x -= 3      # x = 12
x *= 2      # x = 24
x /= 4      # x = 6.0  ← กลายเป็น float!
x %= 4      # x = 2.0
print(x)     # 2.0
# ⚠️ Python ไม่มี ++ และ --
x++        # ❌ SyntaxError
x += 1     # ✓ ต้องเขียนแบบนี้

# กำหนดค่าหลายตัวพร้อมกัน
a, b = 3, 4
a, b = b, a      # สลับค่า! a=4 b=3
x = y = z = 0    # ให้ค่าเดียวกันหมด
2 จุดหลอก: ① Python ไม่มี ++ / -- (ต่างจาก C) ต้องใช้ += 1 ② /= ทำให้ตัวแปรกลายเป็น float แม้หารลงตัว (24/4 = 6.0)
ติวบทที่ 2 · Type Conversion

แปลงชนิดข้อมูล 2 รูปแบบ (ตามสไลด์)

① Implicit — Python แปลงให้เอง
# "Python automatically converts"

a = 10        # int
b = 3.5       # float
c = a + b     # int + float
print(c)      # 13.5
print(type(c)) # <class 'float'>

# กฎ: แปลงไปทาง "ชนิดที่ใหญ่กว่า"
# เพื่อไม่ให้ข้อมูลหาย
#   int → float → complex

5 / 2       # 2.5 (int/int ได้ float!)
True + 1    # 2   (bool→int, True=1)
False + 10  # 10  (False=0)
② Explicit — เราสั่งแปลงเอง
int("10")       # 10
int(3.9)        # 3   ← ตัดทิ้ง ไม่ปัด!
int(-3.9)       # -3  ← ตัดเข้าหา 0
float("3.5")    # 3.5
float(10)       # 10.0
str(100)        # "100"
bool(0)         # False
bool(5)         # True
# ⚠️ แปลงไม่ได้ → ValueError
int("abc")      # ❌ ValueError
int("3.5")      # ❌ ValueError!
float("3.5")    # ✓ 3.5
int(float("3.5")) # ✓ 3 (แปลง 2 ชั้น)
จุดหลอกที่ออกบ่อย: int(3.9) ได้ 3 ไม่ใช่ 4 (ตัดทิ้ง ไม่ใช่ปัด — ถ้าอยากปัดใช้ round()) · และ int("3.5") error ต้องผ่าน float ก่อน
ติวบทที่ 2 · Relational

ตัวดำเนินการเปรียบเทียบ — ผลลัพธ์เป็น bool

ตัวดำเนินการความหมายตัวอย่าง
==เท่ากับ5 == 5 → True
!=ไม่เท่ากับ5 != 3 → True
>มากกว่า5 > 3 → True
<น้อยกว่า5 < 3 → False
>=มากกว่าหรือเท่า5 >= 5 → True
<=น้อยกว่าหรือเท่า3 <= 5 → True
⚠️ = กับ == คนละตัว!
x = 5      # กำหนดค่า (assignment)
x == 5     # เปรียบเทียบ → True

if x = 5:   # ❌ SyntaxError!
if x == 5:  # ✓
# เปรียบเทียบ string ได้ (ตามลำดับ ASCII)
"abc" == "abc"   # True
"a" < "b"        # True (a=97, b=98)
"A" < "a"        # True (A=65, a=97)
"apple" < "banana" # True

# เชื่อมต่อกันได้ (Python พิเศษ!)
1 < x < 10        # ✓ ใช้ได้เลย
# ภาษาอื่นต้องเขียน x>1 and x<10
เก็งข้อสอบ: ให้นิพจน์เปรียบเทียบมาแล้วถามว่า True หรือ False · ระวัง = (กำหนดค่า) กับ == (เปรียบเทียบ) ซึ่งเป็นข้อผิดพลาดคลาสสิก

ใช้จริงในเงื่อนไข
# ตรวจช่วงคะแนน
if score >= 0 and score <= 100:
    print("คะแนนถูกต้อง")

# หรือเขียนสั้นแบบ Python
if 0 <= score <= 100:
    print("คะแนนถูกต้อง")

# ตรวจหลายเงื่อนไข
if age >= 18 and has_id:
    print("เข้าได้")
if day == "เสาร์" or day == "อาทิตย์":
    print("วันหยุด")
ลำดับรวมทั้งหมด (สูง→ต่ำ): ( ) → ** → * / // % → + − → เปรียบเทียบ → not → and → or · เลขคำนวณก่อน เปรียบเทียบทีหลัง ตรรกะท้ายสุด

ข้อ 2

a += 3       # a = 8
a *= 2       # a = 16
a %= 5       # a = 1
print(a)     # 1
ข้อ 3 — ผสมชนิดข้อมูล
print(10 + 3 * 2)          # 16
print((10 + 3) * 2)        # 26
print(10 / 2 + 3)          # 8.0 ← float
print(int(10 / 2) + 3)     # 8   ← int

x = 7

print(x > 10 or x % 7 == 0) # True

ติวบทที่ 2 · ลองข้อสอบ

ค่าของ -5 % 2 คือ?

A -1
B 1
C 0
D Error
Python: a % b มีเครื่องหมายตามตัวหาร (b)
  -5 % 2  = 1     (b=2 บวก → ผลบวก)
   5 % -2 = -1    (b=-2 ลบ → ผลลบ)

พิสูจน์: -5 // 2 = -3 (ปัดลง) → -5 = (-3)×2 + r → r = 1 ✓
ติวบทที่ 2 · ลองข้อสอบ

ค่าของ 2 * 5 + (6 * 2) / 5 คือ?

A 12
B 12.0
C 12.4
D 4.4
① วงเล็บก่อน:   (6 * 2) = 12
② คูณ/หาร ซ้าย→ขวา:  2 * 5 = 10  ·  12 / 5 = 2.4
③ บวก:          10 + 2.4 = 12.4

⚠️ มี / อยู่ในนิพจน์ → ผลลัพธ์เป็น float แน่นอน
