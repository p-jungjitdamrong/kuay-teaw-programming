บทที่ 4 · Array + Loop
Array + Loop — คู่หูที่แยกกันไม่ออก
Loop ทำให้เข้าถึง array ทุกช่องโดยไม่ต้องเขียนชื่อซ้ำ — ใช้ i เป็น index
Pattern พื้นฐาน — วนทุก element
int scores[] = {85,90,72,88,95}; int n = 5, i; /* พิมพ์ทุกตัว */ for (i = 0; i < n; i++) printf("%d ", scores[i]); /* หาผลรวม */ int sum = 0; for (i = 0; i < n; i++) sum += scores[i]; printf("Avg: %.1f", (float)sum/n); /* หาค่าสูงสุด */ int max = scores[0]; for (i = 1; i < n; i++) if (scores[i] > max) max = scores[i]; printf("Max: %d", max); /* 95 */
รับค่าเข้า array ด้วย scanf
int arr[5], i; for (i = 0; i < 5; i++) { printf("arr[%d]: ", i); scanf("%d", &arr[i]); /* &arr[i] */ }
⚠️ sizeof กับ Array
int arr[5];
sizeof(arr) = 20 bytes (5×4)
sizeof(arr)/sizeof(arr[0]) = 5

วิธีนี้หา size ได้โดยไม่ต้อง hardcode แต่
ใช้ไม่ได้เมื่อส่ง array เข้า function!
ถ้า int arr[5] = {0} แล้ว loop วน i = 0; i < 5; i++ พิมพ์ arr[i] — output คืออะไร?
💡 {0} initialize ทุกช่องเป็น 0 ไม่ใช่แค่ช่องแรก

✅ Readable — ชื่อ function บอกว่าทำอะไร
✅ Debug ง่าย — หา bug แยกแต่ละ function
✅ แก้ที่เดียว — เปลี่ยน logic แก้ใน function เดียว
บทที่ 4 · Prototype
Function Prototype — แจ้งให้ C รู้ก่อน
C อ่าน code จากบนลงล่าง ถ้า function อยู่หลัง main ต้องประกาศ prototype ก่อน ไม่งั้น compile error
❌ ไม่มี prototype — error
int main() { int r = add(3, 4); /* error! */ return 0; } int add(int a, int b) { /* ช้าไป */ return a + b; }
✅ มี prototype — ถูกต้อง
/* prototype ก่อน main */ int add(int a, int b); /* แค่ประกาศ */ int main() { int r = add(3, 4); /* ✓ */ return 0; } int add(int a, int b) { /* ตัวจริง */ return a + b; }
ทางเลือก — เขียน function ก่อน main
อีกวิธีคือเขียน function ทั้งหมดไว้เหนือ main เลย ไม่ต้องมี prototype แต่ถ้า function เยอะอ่านยาก

ในโปรเจคจริง มักใช้ .h header file เก็บ prototype ทั้งหมด
⚠️ Prototype vs Definition
Prototype — แจ้ง type เท่านั้น ลง ; จบ
int add(int a, int b);

Definition — ตัว function จริง มี body {}
int add(int a, int b) { return a+b; }
บทที่ 4 · Pass by Value
Pass by Value — C ส่งแค่ "สำเนา"
เมื่อส่งตัวแปรเข้า function C จะส่งแค่ค่า (copy) ไม่ใช่ตัวแปรจริง แก้ใน function ไม่กระทบตัวแปรนอก
❌ คาดว่า x จะเปลี่ยน แต่ไม่เปลี่ยน
void double_it(int n) { n = n * 2; /* แก้แค่ copy! */ } int main() { int x = 5; double_it(x); printf("%d", x); /* ยังเป็น 5! */ }
✅ ใช้ return value แทน
int double_it(int n) { return n * 2; /* คืนค่ากลับ */ } int main() { int x = 5; x = double_it(x); /* รับค่ากลับ */ printf("%d", x); /* 10 ✓ */ }
💡 แต่ Array ต่างออกไป!
เมื่อส่ง array เข้า function C ส่ง address ของช่องแรก (ไม่ใช่ copy) ทำให้แก้ค่าใน function ได้จริง

นี่คือเหตุผลว่าทำไม array ถึงแก้ได้ใน function แต่ int ธรรมดาแก้ไม่ได้
function นี้ swap a กับ b ได้จริงไหม?
void swap(int a, int b) {
  int t=a; a=b; b=t;
}
ถ้าไม่ได้ — ต้องทำยังไง?
💡 Pass by value → ไม่ได้ ต้องใช้ pointer ซึ่งเรียนในบทที่ 5
บทที่ 4 · Array in Function
ส่ง Array เข้า Function
Array ส่งแบบพิเศษ — C ส่ง address มาให้ ไม่ copy ทั้ง array function จึงเข้าถึงข้อมูลตัวจริงได้ ต้องส่ง size มาด้วยเสมอ
รูปแบบการรับ array
/* ทั้งสองแบบนี้เหมือนกัน */ void printArr(int arr[], int size) void printArr(int *arr, int size) /* ตัวอย่าง */ void printArr(int arr[], int size) { int i; for (i = 0; i < size; i++) printf("%d ", arr[i]); printf("\n"); } float calcAvg(int arr[], int size) { int i, sum = 0; for (i = 0; i < size; i++) sum += arr[i]; return (float)sum / size; }
เรียกใช้
int main() { int scores[] = {85,90,72,88,95}; int n = 5; printArr(scores, n); printf("Avg: %.1f\n", calcAvg(scores, n)); return 0; }
⚠️ ต้องส่ง size เสมอ
ใน function sizeof(arr) ไม่ให้ขนาด array จริงๆ แล้ว — ให้ขนาดของ pointer แทน (4 หรือ 8 bytes)

จึงต้องส่ง size เป็น parameter เสมอ
✅ Array แก้ค่าใน Function ได้จริง
เพราะ C ส่ง address มา function จึงเข้าถึงข้อมูลตัวจริง ถ้า function แก้ arr[i] ค่าใน main จะเปลี่ยนด้วย

บทที่ 4 · Scope
Variable Scope — ตัวแปรมองเห็นได้แค่ไหน
ตัวแปรใน C มองเห็นได้เฉพาะใน block {} ที่ประกาศไว้ — เข้าใจ scope ช่วยหลีกเลี่ยง bug ที่หายากมาก
Local vs Global scope
int globalVar = 100; /* global */ void funcA() { int x = 10; /* local */ printf("%d", x); /* ✓ */ printf("%d", globalVar); /* ✓ */ } void funcB() { printf("%d", x); /* ❌ error! */ printf("%d", globalVar); /* ✓ */ } int main() { int i; for (i=0; i<3; i++) { int temp = i*2; /* local to loop */ } /* temp ไม่มีที่นี่แล้ว */ }
⚠️ Global Variable — ระวัง!
Global variable ใช้ง่ายแต่อันตราย — ทุก function แก้ค่าได้ ทำให้หา bug ยาก

ควรใช้ local variable + parameter ส่งค่าแทน เว้นแต่จำเป็นจริงๆ
✅ Best Practice
ประกาศตัวแปร ใกล้ที่ใช้มากที่สุด

ใช้ function parameter แทน global

ชื่อตัวแปรใน function ต่างกันได้ ไม่ conflict
เพราะแต่ละ function มี scope แยกกัน
ถ้า 2 function ประกาศ int x; เหมือนกัน จะ conflict กันไหม?
💡 Local variable อยู่คนละ scope — ไม่ conflict เลย
บทที่ 4 · Think!
คิดก่อน เขียน
Trace ในหัว และวิเคราะห์ก่อนตอบ
1. int arr[3] = {1, 2, 3};
arr[3] = 99; — จะเกิดอะไรขึ้น?
💡 Index 0–2 เท่านั้น [3] คือนอก array C ไม่ stop แต่ memory เสียหาย
2. function นี้ทำงานถูกไหม?
int sum(int arr[], int n) {
  int i, s=0;
  for(i=0;i<n;i++) s+=arr[i];
  return s;
}
💡 ถูกต้อง — รับ array + size คำนวณผลรวม return กลับ
3. ถ้าเรียก printArr(scores, 5) แล้ว function แก้ arr[0] = 999 — ค่า scores[0] ใน main จะเปลี่ยนไหม?
💡 Array ส่ง address มา แก้ใน function = แก้ตัวจริง ต่างจาก int ธรรมดา
4. prototype ต่อไปนี้ถูกไหม?
float calcAvg(int arr[], int size);
💡 ถูกต้อง — มี return type, ชื่อ function, parameter types และ ;
5. จะหา index ของค่าสูงสุดใน array ได้ยังไง? (ไม่ใช่แค่ค่า แต่ต้องการ index)
💡 เก็บ maxIdx = 0 ด้วย ถ้าเจอค่าใหม่ที่มากกว่า อัปเดตทั้ง maxVal และ maxIdx
