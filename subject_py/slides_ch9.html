บทที่ 9 · ทำไมต้องมี

ข้อมูลในตัวแปรหายเมื่อปิดโปรแกรม

ปัญหา
# โปรแกรมบันทึกคะแนน
scores = []
while True:
    s = input("คะแนน (0=จบ): ")
    if s == "0": break
    scores.append(int(s))

print(scores)   # [85, 90, 78]

# ปิดโปรแกรม → ข้อมูลหายหมด!
# เปิดใหม่ต้องพิมพ์ใหม่ทั้งหมด 😱
ทางแก้: เก็บลงไฟล์
# ข้อมูลอยู่ในดิสก์ → เปิดใหม่ก็ยังอยู่
f = open("scores.txt", "w")
f.write("85\n90\n78\n")
f.close()
3 ขั้นตอนของงานไฟล์ — ท่องไว้
① เปิดไฟล์ open() │ ▼ ② อ่าน / เขียน read() / write() │ ▼ ③ ปิดไฟล์ close()
/* งานที่ทำกับไฟล์ได้ (4 อย่าง):
   • สร้าง  (creating)
   • อ่าน   (reading)
   • เขียน  (writing)
   • ลบ     (deleting)

   ⚠️ ลืม close() = ข้อมูลอาจ
   เขียนไม่ครบ หรือไฟล์ถูกล็อกค้าง
   → มีวิธีที่ปิดให้อัตโนมัติ (with)

คำศัพท์: ข้อมูลในตัวแปรอยู่ใน RAM (หายเมื่อปิด) · ข้อมูลในไฟล์อยู่ใน ดิสก์ (อยู่ถาวร) — เรียกว่า persistent data
บทที่ 9 · open()

open() รับ 2 อย่าง — ชื่อไฟล์ + โหมด

ชื่อตัวแปร = open( "ชื่อไฟล์" , "โหมด" )

f = open("data.txt", "r")     # เปิดอ่าน
โหมดชื่อเต็มทำอะไรถ้าไฟล์ยังไม่มีของเดิม
"r"Readอ่านอย่างเดียว (ค่าเริ่มต้น)Error!—
"w"Writeเขียนสร้างให้ใหม่ลบทิ้งหมด!
"a"Appendเขียนต่อท้ายสร้างให้ใหม่เก็บไว้
"x"Createสร้างไฟล์ใหม่สร้างให้Error ถ้ามีแล้ว
 
# โหมดเสริม (ต่อท้ายได้)
# "t" = text (ค่าเริ่มต้น)
# "b" = binary (รูป เสียง)
open("a.txt", "rt")   # = "r"
open("pic.jpg", "rb")  # อ่านไฟล์รูป
# ⚠️ กับดักที่เจ็บที่สุด:
f = open("สำคัญมาก.txt", "w")
# ← ไฟล์ที่มีข้อมูลอยู่ ถูกล้างทันที
#   ตั้งแต่บรรทัด open เลย!
#   ยังไม่ทันเขียนอะไรก็หายแล้ว

# อยากเขียนเพิ่มโดยไม่ลบ → ใช้ "a"
วิธีจำ: read อ่าน · write เขียนทับ · append เขียนต่อ · exclusive สร้างใหม่เท่านั้น — ข้อสอบชอบถาม "โหมดใดลบข้อมูลเดิม" คำตอบคือ w
บทที่ 9 · อ่านไฟล์

อ่านได้ 4 วิธี — เลือกให้ถูกงาน

สมมติไฟล์ data.txt มี 3 บรรทัด: Somchai / Somsri / Somying

① read() — ทั้งไฟล์เป็น string เดียว
f = open("data.txt", "r")
print(f.read())
f.close()
Somchai Somsri Somying
② readline() — ทีละบรรทัด
f = open("data.txt", "r")
print(f.readline())   # Somchai
print(f.readline())   # Somsri
f.close()
# เรียกครั้งถัดไปได้บรรทัดถัดไป
# (ตัวชี้ตำแหน่งเลื่อนไปเอง)
③ readlines() — ได้ list
f = open("data.txt", "r")
lines = f.readlines()
f.close()
print(lines)
['Somchai\n', 'Somsri\n', 'Somying']
# มี \n ติดมาด้วย! ลบด้วย strip()
for line in lines:
    print(line.strip())
④ วนด้วย for — นิยมที่สุด
f = open("data.txt", "r")
for line in f:          # วนได้เลย!
    print(line.strip())
f.close()
# ประหยัดหน่วยความจำสุด
# (ไม่โหลดทั้งไฟล์เข้า RAM)
อย่าลืม .strip() — ทุกบรรทัดในไฟล์มี \n ติดท้าย ถ้าไม่ลบจะได้บรรทัดว่างแทรก หรือเทียบ string ไม่ตรง
บทที่ 9 · เขียนไฟล์

write() — เขียนได้เฉพาะ string

เขียนทับ (w) vs เขียนต่อ (a)
# "w" — ล้างของเดิมแล้วเขียนใหม่
f = open("log.txt", "w")
f.write("บรรทัดที่ 1\n")
f.write("บรรทัดที่ 2\n")
f.close()
บรรทัดที่ 1 บรรทัดที่ 2
# "a" — ต่อท้ายของเดิม
f = open("log.txt", "a")
f.write("บรรทัดที่ 3\n")
f.close()
บรรทัดที่ 1 บรรทัดที่ 2 บรรทัดที่ 3
2 จุดพลาดที่เจอบ่อย
# ❌ ลืม \n — ทุกอย่างติดกันหมด
f.write("A")
f.write("B")
f.write("C")
# ไฟล์ได้: ABC   (บรรทัดเดียว)
# write ไม่ขึ้นบรรทัดให้เอง! ต่างจาก print

# ❌ เขียนตัวเลขตรงๆ → TypeError
f.write(100)          # Error!
f.write(str(100))     # ✓ แปลงก่อน
f.write(f"{score}\n")  # ✓ f-string
writelines() — เขียน list ทีเดียว
names = ["Somchai\n", "Somsri\n"]
f = open("out.txt", "w")
f.writelines(names)
f.close()
# ก็ไม่เติม \n ให้เหมือนกัน
# ต้องใส่มาในแต่ละตัวเอง
เทียบให้จำ: print() ขึ้นบรรทัดใหม่ให้อัตโนมัติ · write() ไม่ขึ้นให้ ต้องใส่ \n เอง — และ write รับ string เท่านั้น ตัวเลขต้อง str() ก่อน
บทที่ 9 · with

with — เปิดแล้วปิดให้เอง (วิธีที่ควรใช้)

แบบธรรมดา — ต้องปิดเอง
f = open("data.txt", "r")
data = f.read()
f.close()          # ← ลืมบ่อย!

# ถ้าโค้ดตรงกลางพัง (error)
# → close() ไม่ถูกเรียก
# → ไฟล์ค้างเปิด
แบบ with — ปิดอัตโนมัติ
with open("data.txt", "r") as f:
    data = f.read()
    print(data)
# ออกจากบล็อก → ปิดให้เอง
# แม้เกิด error ก็ยังปิด ✓
ใช้จริง — อ่านและเขียน
# อ่านทีละบรรทัด
with open("scores.txt", "r") as f:
    for line in f:
        print(line.strip())

# เขียน
with open("out.txt", "w") as f:
    for i in range(1, 6):
        f.write(f"บรรทัดที่ {i}\n")

# เปิด 2 ไฟล์พร้อมกัน (copy)
with open("in.txt") as src, \
     open("out.txt", "w") as dst:
    dst.write(src.read())
โครงที่ควรใช้ตลอด: with open(ชื่อไฟล์, โหมด) as f: แล้วเขียนงานไว้ในบล็อก — ไม่ต้องเขียน close() เลย และปลอดภัยกว่า
บทที่ 9 · ตรวจสอบ & ลบ

เช็คว่ามีไฟล์ไหม + ลบไฟล์ (ใช้ os)

ปัญหา: เปิดไฟล์ที่ไม่มี → พัง
f = open("ไม่มีจริง.txt", "r")
# FileNotFoundError: [Errno 2]
# No such file or directory
ทางแก้ 1: เช็คก่อนเปิด
import os

if os.path.exists("data.txt"):
    with open("data.txt") as f:
        print(f.read())
else:
    print("ไม่พบไฟล์")
ทางแก้ 2: ดักด้วย try
try:
    with open("data.txt") as f:
        print(f.read())
except FileNotFoundError:
    print("ไม่พบไฟล์")
ลบไฟล์
import os

# ลบตรงๆ (พังถ้าไม่มีไฟล์)
os.remove("old.txt")

# ลบแบบปลอดภัย
if os.path.exists("old.txt"):
    os.remove("old.txt")
    print("ลบแล้ว")
else:
    print("ไม่มีไฟล์นี้")
# คำสั่ง os อื่นที่ควรรู้จัก
os.path.exists("f.txt")   # มีไหม
os.rename("a.txt","b.txt") # เปลี่ยนชื่อ
os.mkdir("folder")        # สร้างโฟลเดอร์
os.rmdir("folder")        # ลบโฟลเดอร์
os.listdir(".")           # ดูไฟล์ในโฟลเดอร์
ต้อง import os ก่อนเสมอ — คำสั่งจัดการไฟล์/โฟลเดอร์อยู่ในโมดูล os ไม่ใช่คำสั่งพื้นฐานของ Python
บทที่ 9 · Use Case

🏦 โปรแกรมบันทึกธุรกรรม — เขียนแล้วอ่านกลับ

import os

FILE = "transactions.txt"

# ── ① บันทึกธุรกรรมใหม่ (ต่อท้าย ไม่ลบของเดิม) ──
def add_txn(name, amount):
    with open(FILE, "a") as f:          # "a" = append!
        f.write(f"{name},{amount}\n")   # คั่นด้วย , (รูปแบบ CSV)

# ── ② อ่านทั้งหมดมาสรุป ──
def summary():
    if not os.path.exists(FILE):            # เช็คก่อนเสมอ
        print("ยังไม่มีข้อมูล")
        return

    total = 0
    count = 0
    with open(FILE, "r") as f:
        for line in f:                      # วนทีละบรรทัด
            line = line.strip()               # ตัด \n ทิ้ง
            if line == "":                    # ข้ามบรรทัดว่าง
                continue
            name, amount = line.split(",")    # แยกด้วย ,
            amount = float(amount)             # แปลงกลับเป็นตัวเลข
            print(f"  {name:<10}{amount:>10,.2f}")
            total += amount
            count += 1

    print(f"  รวม {count} รายการ = {total:,.2f} บาท")

# ── ใช้งาน ──
add_txn("สมชาย", 1200)
add_txn("สมศรี", 850.50)
add_txn("สมหญิง", 3000)
summary()

"""
ไฟล์ transactions.txt ที่ได้:
สมชาย,1200
สมศรี,850.5
สมหญิง,3000

ผลลัพธ์:
  สมชาย        1,200.00
  สมศรี          850.50
  สมหญิง       3,000.00
  รวม 3 รายการ = 5,050.50 บาท
"""
รูปแบบ CSV (คั่นด้วยจุลภาค) เป็นวิธีเก็บข้อมูลหลายฟิลด์ที่ง่ายที่สุด — เขียนด้วย f"{a},{b}\n" อ่านกลับด้วย line.strip().split(",") · เปิดด้วย Excel ได้ด้วย

บทที่ 9 · ทดสอบ

ต้องการเขียนข้อมูลต่อท้ายไฟล์เดิมโดยไม่ลบของเก่า ใช้โหมดใด?

A open(f, "r")
B open(f, "w")
C open(f, "a")
D open(f, "x")
r = อ่านอย่างเดียว (ไฟล์ต้องมีอยู่)
w = เขียน — ล้างของเดิมทิ้งหมด
a = เขียนต่อท้าย — เก็บของเดิมไว้ ✓
x = สร้างใหม่เท่านั้น (error ถ้ามีแล้ว)
