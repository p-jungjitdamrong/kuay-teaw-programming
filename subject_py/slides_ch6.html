บทที่ 7 · Inheritance

Inheritance — class ลูกสืบทอดจาก class พ่อแม่

ใส่ชื่อ parent class ในวงเล็บ — ใช้ super() เรียก constructor พ่อแม่

class BankAccount:            # Parent / Base class
    def __init__(self, owner):
        self.owner = owner
        self._balance = 0

    def deposit(self, amount):
        self._balance += amount

    def __str__(self):
        return f"{self.owner}: {self._balance:,} บาท"


class SavingsAccount(BankAccount):  # Child — สืบทอด BankAccount
    def __init__(self, owner, rate=0.015):
        super().__init__(owner)   # เรียก __init__ ของ parent
        self.rate = rate

    def add_interest(self):   # method ใหม่ที่ parent ไม่มี
        interest = self._balance * self.rate
        self._balance += interest
        return interest


acc = SavingsAccount("สมชาย")
acc.deposit(100_000)   # สืบทอดจาก BankAccount
acc.add_interest()     # method ของ SavingsAccount เอง
print(acc)             # สืบทอด __str__ ด้วย
บทที่ 7 · Override

Method Overriding — เขียนทับ method ของ parent

ถ้า child class นิยาม method ชื่อเดิมจาก parent — จะใช้ method ของ child แทน

class PremiumAccount(BankAccount):
    def __init__(self, owner, credit_limit=100_000):
        super().__init__(owner)
        self.credit_limit = credit_limit

    def withdraw(self, amount):
        # Override — เพิ่ม credit facility
        available = self._balance + self.credit_limit
        if amount > available:
            raise ValueError("เกิน credit limit")
        self._balance -= amount

    def __str__(self):
        # Override __str__ ให้แสดง credit ด้วย
        base = super().__str__()   # เรียก parent __str__
        return f"{base} | เครดิต {self.credit_limit:,}"

prem = PremiumAccount("วิชัย")
prem.deposit(50_000)
prem.withdraw(120_000)   # ถอนได้ เพราะมี credit
print(prem)
super() ใน Python 3 ไม่ต้องส่ง class และ self — เขียนแค่ super() เลย (Python 2 ต้องส่ง)
บทที่ 7 · Polymorphism

Polymorphism — เรียก method เดิม ผลต่างกัน

เรียก method ชื่อเดิมบน object ต่าง type — แต่ละ object ทำงานตาม class ของตัวเอง

class FixedDeposit(BankAccount):
    def withdraw(self, amount):
        raise PermissionError("ฝากประจำถอนไม่ได้ก่อนครบกำหนด")

    def __str__(self):
        return f"[ฝากประจำ] {self.owner}: {self._balance:,} บาท"


# Polymorphism — ลูปเดียว จัดการได้ทุก type
accounts = [
    SavingsAccount("สมชาย"),
    PremiumAccount("สมหญิง"),
    FixedDeposit("วิชัย"),
]

for acc in accounts:
    acc.deposit(10_000)   # method เดิม ทำงานเหมือนกัน
    print(acc)             # __str__ แต่ละ type ต่างกัน

# isinstance() ตรวจ type
print(isinstance(accounts[0], BankAccount))   # True — SavingsAccount IS-A BankAccount
print(isinstance(accounts[0], SavingsAccount)) # True
บทที่ 7 · Duck Typing

Duck Typing — "ถ้ามันเดินเหมือนเป็ด และร้องเหมือนเป็ด มันคือเป็ด"

Python ไม่สนว่า object เป็น type อะไร — สนแค่ว่ามี method ที่ต้องการหรือเปล่า (ต่างจาก Java ที่ต้อง implement interface)

Java — ต้อง implement Interface
interface Printable {
    void print();
}
class Report implements Printable {
    public void print() { ... }
}
Python — ไม่ต้อง declare อะไรเลย
class Report:
    def generate(self):
        return "report data"

class PDF:
    def generate(self):
        return "pdf bytes"
# Duck typing — ทั้งสอง class ไม่มีความสัมพันธ์กันเลย
# แต่ทำงานด้วยกันได้เพราะมี method generate()
def export(generator):
    data = generator.generate()   # ไม่สนว่า type อะไร
    print(f"Exported: {data}")

export(Report())   # ✅ มี generate()
export(PDF())     # ✅ มี generate() เหมือนกัน
Duck Typing ทำให้ Python ยืดหยุ่นมาก — ไม่ต้อง inherit จาก class เดียวกันเพื่อทำงานร่วมกัน
บทที่ 7 · ABC

Abstract Class — บังคับให้ subclass เขียน method

ถ้าต้องการบังคับ subclass ทุกตัวให้มี method บางตัว — ใช้ ABC + @abstractmethod

from abc import ABC, abstractmethod

class Account(ABC):   # Abstract Base Class — สร้าง instance ตรงๆ ไม่ได้
    def __init__(self, owner):
        self.owner = owner
        self._balance = 0

    @abstractmethod
    def withdraw(self, amount):   # subclass ต้อง override นี้
        pass

    def deposit(self, amount):    # method ปกติ — สืบทอดได้
        self._balance += amount


class Savings(Account):
    def withdraw(self, amount):   # ต้องเขียน ไม่งั้น TypeError
        if amount > self._balance:
            raise ValueError("ยอดไม่พอ")
        self._balance -= amount

# Account()  # ❌ TypeError: Can't instantiate abstract class
s = Savings("สมชาย")   # ✅ Savings implement ครบแล้ว
Abstract Class คือสัญญา — บอกว่า subclass ทุกตัวต้อง implement method พวกนี้
บทที่ 7 · @dataclass

@dataclass — Data Class อัตโนมัติ

Python 3.7+ มี @dataclass ที่สร้าง __init__, __str__, __repr__, __eq__ ให้อัตโนมัติ

แบบปกติ — เขียนเยอะ
class Transaction:
    def __init__(self, txn_id,
                 amount, type):
        self.txn_id = txn_id
        self.amount = amount
        self.type = type

    def __repr__(self):
        return (
            f"Transaction("
            f"{self.txn_id},"
            f"{self.amount})"
        )
@dataclass — กระชับกว่า
from dataclasses import dataclass

@dataclass
class Transaction:
    txn_id: str
    amount: float
    type: str
    # __init__, __repr__,
    # __eq__ สร้างให้อัตโนมัติ!
t1 = Transaction("TXN001", 5000.0, "deposit")
t2 = Transaction("TXN001", 5000.0, "deposit")
print(t1)          # Transaction(txn_id='TXN001', amount=5000.0, type='deposit')
print(t1 == t2)    # True — @dataclass สร้าง __eq__ ให้

# frozen=True → immutable เหมือน tuple
@dataclass(frozen=True)
class Config:
    host: str = "localhost"
    port: int = 5432

บทที่ 7 · Banking Use Case

🏦 ระบบบัญชีธนาคาร OOP แบบสมบูรณ์

from abc import ABC, abstractmethod
from dataclasses import dataclass, field

@dataclass
class Txn:
    type: str
    amount: float

class Account(ABC):
    def __init__(self, owner):
        self.owner = owner
        self._balance = 0.0
        self._history: list[Txn] = []

    def deposit(self, amt):
        self._balance += amt
        self._history.append(Txn("D", amt))
        return self

    @abstractmethod
    def withdraw(self, amt): pass

    @property
    def balance(self): return self._balance

class Savings(Account):
    def withdraw(self, amt):
        if amt > self._balance: raise ValueError("ยอดไม่พอ")
        self._balance -= amt
        self._history.append(Txn("W", amt))
        return self

    def __str__(self):
        return f"[ออมทรัพย์] {self.owner}: {self._balance:,.2f}"

acc = Savings("สมชาย")
acc.deposit(50_000).deposit(10_000).withdraw(8_000)
print(acc)  # [ออมทรัพย์] สมชาย: 52,000.00
