โครงหน้าต่าง — 4 บรรทัดท่องจำ
import tkinter as tk

win = tk.Tk()              # ① สร้างหน้าต่าง
win.title("โปรแกรมแรก")   # ② ตั้งชื่อ (ไม่บังคับ)
win.geometry("300x200")  # ③ ขนาด กว้างxสูง

# ... วาง widget ตรงนี้ ...

win.mainloop()           # ④ เริ่มรอ event — ท้ายสุดเสมอ!

บทที่ 10 · Event Handling

กดปุ่มแล้วทำงาน — command=

โครง 3 ขั้น
import tkinter as tk

win = tk.Tk()
win.geometry("250x120")

# ① เขียนฟังก์ชันที่จะทำงาน
def say_hi():
    lbl.config(text="สวัสดีครับ!")

lbl = tk.Label(win, text="ยังไม่ได้กด")
lbl.pack()

# ② ผูกฟังก์ชันกับปุ่มด้วย command
btn = tk.Button(win, text="กดสิ",
                 command=say_hi)
btn.pack()

win.mainloop()    # ③ รอ event
⚠️ จุดพลาดอันดับ 1
# ❌ ใส่วงเล็บ = เรียกทันที!
tk.Button(win, command=say_hi())
#                        ↑↑
# ฟังก์ชันทำงานตอนสร้างปุ่มเลย
# แล้วปุ่มก็กดไม่ได้อะไร

# ✓ ไม่ใส่วงเล็บ = ส่ง "ตัวฟังก์ชัน" ไป
tk.Button(win, command=say_hi)
ส่ง argument ด้วย lambda
def greet(name):
    print("สวัสดี", name)

# ❌ command=greet("สมชาย") → ทำงานทันที
# ✓ ห่อด้วย lambda
tk.Button(win, text="ทัก",
    command=lambda: greet("สมชาย"))
จำสั้นๆ: command=ชื่อฟังก์ชัน (ไม่มีวงเล็บ) — ใส่วงเล็บเมื่อไหร่ = สั่งให้ทำงานทันที ไม่ใช่รอกด · ถ้าต้องส่ง argument ให้ห่อด้วย lambda:
บทที่ 10 · รับค่าจากผู้ใช้

Entry — get() อ่านค่า, delete() ล้างช่อง

โปรแกรมทักทาย (ครบวงจร)
import tkinter as tk

win = tk.Tk()
win.title("ทักทาย")
win.geometry("280x150")

def greet():
    name = ent.get()          # อ่านค่าจากช่อง
    if name == "":
        result.config(text="กรุณากรอกชื่อ")
    else:
        result.config(text=f"สวัสดี {name}!")

tk.Label(win, text="ชื่อ:").pack()
ent = tk.Entry(win, width=25)
ent.pack()

tk.Button(win, text="ทักทาย",
          command=greet).pack(pady=8)

result = tk.Label(win, text="",
                  font=("Arial", 14))
result.pack()

win.mainloop()
┌─ ทักทาย ─────── ─ □ ✕ ┐ │ ชื่อ: │ │ [ สมชาย ] │ │ │ │ [ ทักทาย ] │ │ │ │ สวัสดี สมชาย! │ └───────────────────────┘
method ของ Entry ที่ใช้บ่อย
ent.get()              # อ่านค่า (ได้ string!)
ent.delete(0, tk.END)  # ล้างช่อง
ent.insert(0, "ค่าเริ่มต้น")

# get() คืน string เสมอ!
# ต้องแปลงก่อนคำนวณ
n = int(ent.get())      # ← ต้องแปลง
Label เปลี่ยนข้อความด้วย config()
result.config(text="ข้อความใหม่")
# หรือ result["text"] = "ข้อความใหม่"
ent.get() คืน string เสมอ เหมือน input() — จะคำนวณต้อง int() หรือ float() ก่อน และควรดักด้วย try-except เผื่อผู้ใช้พิมพ์ตัวอักษร
บทที่ 10 · Use Case

🏦 เครื่องคิดดอกเบี้ยเงินฝาก — รวมทุกอย่าง

import tkinter as tk

win = tk.Tk()
win.title("คำนวณดอกเบี้ย")
win.geometry("340x220")

def calculate():
    try:                                        # ดัก error (บทที่แล้ว!)
        money = float(ent_money.get())          # get() ได้ string → แปลง
        rate  = float(ent_rate.get())
        years = int(ent_year.get())
    except ValueError:
        result.config(text="⚠️ กรุณากรอกตัวเลข", fg="red")
        return

    if money <= 0 or years <= 0:            # ตรวจความสมเหตุสมผล
        result.config(text="⚠️ ค่าต้องมากกว่า 0", fg="red")
        return

    total = money
    for i in range(years):                     # ลูปทบต้นทีละปี
        total = total * (1 + rate / 100)

    interest = total - money
    result.config(
        text=f"เงินรวม {total:,.2f} บาท\nดอกเบี้ย {interest:,.2f} บาท",
        fg="green")

# ── จัดหน้าด้วย grid (ตาราง 2 คอลัมน์) ──
tk.Label(win, text="เงินต้น (บาท):").grid(row=0, column=0, padx=8, pady=5, sticky="e")
ent_money = tk.Entry(win, width=15)
ent_money.grid(row=0, column=1)

tk.Label(win, text="ดอกเบี้ย (%/ปี):").grid(row=1, column=0, padx=8, pady=5, sticky="e")
ent_rate = tk.Entry(win, width=15)
ent_rate.grid(row=1, column=1)

tk.Label(win, text="จำนวนปี:").grid(row=2, column=0, padx=8, pady=5, sticky="e")
ent_year = tk.Entry(win, width=15)
ent_year.grid(row=2, column=1)

tk.Button(win, text="คำนวณ", command=calculate, width=12) \
  .grid(row=3, column=0, columnspan=2, pady=10)   # กินพื้นที่ 2 คอลัมน์

result = tk.Label(win, text="", font=("Arial", 12))
result.grid(row=4, column=0, columnspan=2)

win.mainloop()

# เงินต้น 10000 · ดอกเบี้ย 2.5% · 3 ปี
# → เงินรวม 10,768.91 บาท / ดอกเบี้ย 768.91 บาท
