บทที่ 2 · Conditional

if / elif / else

Python ไม่มี switch (Python 3.10+ มี match) — ใช้ elif แทน else if

C
if (score >= 80) {
    grade = 'A';
} else if (score >= 70) {
    grade = 'B';
} else if (score >= 60) {
    grade = 'C';
} else {
    grade = 'F';
}
Python
if score >= 80:
    grade = "A"
elif score >= 70:
    grade = "B"
elif score >= 60:
    grade = "C"
else:
    grade = "F"
# Ternary expression — Python style
label = "ผ่าน" if score >= 50 else "ไม่ผ่าน"

# Truthy/Falsy — Python เช็ค condition โดยไม่ต้อง == 0
if balance:        # True ถ้า balance ไม่ใช่ 0
    print("มียอด")
if not name:      # True ถ้า name เป็น "" หรือ None
    print("ไม่มีชื่อ")
บทที่ 2 · for loop

for loop — วนซ้ำแบบ Pythonic

Python for วน บน iterable โดยตรง — ไม่ต้องใช้ index เหมือน C

# วนบน list โดยตรง (ไม่ต้องใช้ i)
amounts = [1000, 2500, 500]
for amt in amounts:
    print(f"โอน {amt:,} บาท")

# range(start, stop, step) — เหมือน for(i=0;i<n;i++)
for i in range(5):          # 0,1,2,3,4
    print(i)
for i in range(1, 6):       # 1,2,3,4,5
    print(i)
for i in range(10, 0, -2):  # 10,8,6,4,2
    print(i)

# enumerate — ได้ทั้ง index และค่า
for i, amt in enumerate(amounts, start=1):
    print(f"ธุรกรรม #{i}: {amt:,} บาท")

# zip — วนสอง list พร้อมกัน
names = ["สมชาย", "สมหญิง"]
bals = [5000, 12000]
for name, bal in zip(names, bals):
    print(f"{name}: {bal:,} บาท")
บทที่ 2 · while

while, break, continue — และ else บน loop

Python มี else ต่อท้าย loop — ทำงานเมื่อ loop จบโดยไม่ถูก break

# while loop พื้นฐาน
retry = 0
while retry < 3:
    pin = input("ใส่ PIN: ")
    if pin == "1234":
        print("เข้าใช้งานได้")
        break           # ออกจาก loop
    retry += 1
    print(f"PIN ผิด ({retry}/3)")
else:                   # else block: ทำงานเมื่อ loop จบโดยไม่ break
    print("บัญชีถูกระงับ")

# continue — ข้ามรอบนี้
for txn in transactions:
    if txn['amount'] <= 0:
        continue   # ข้าม transaction ที่ไม่ valid
    process(txn)
else บน loop คือ feature ที่ C ไม่มี — ใช้แทนการตั้ง flag เพื่อเช็คว่า loop จบแบบปกติหรือถูก break
บทที่ 2 · Comprehension

List Comprehension — หัวใจของ Pythonic Code

แทน for loop + append ด้วยบรรทัดเดียว — อ่านง่ายกว่า เร็วกว่า

แบบ C/loop ปกติ
# สร้าง list ยกกำลัง 2
squares = []
for i in range(1, 6):
    squares.append(i ** 2)
# [1, 4, 9, 16, 25]
List Comprehension
# บรรทัดเดียว!
squares = [i**2 for i in range(1, 6)]
# [1, 4, 9, 16, 25]
# Syntax: [expression for item in iterable if condition]

# กรองเฉพาะธุรกรรมที่ > 10,000
big_txns = [t for t in transactions if t > 10_000]

# แปลงค่า — บวกภาษีมูลค่าเพิ่ม 7%
with_vat = [round(p * 1.07, 2) for p in prices]

# Dict Comprehension
accounts = {"สมชาย": 5000, "สมหญิง": 12000}
rich = {k: v for k, v in accounts.items() if v > 8000}
# {"สมหญิง": 12000}
บทที่ 2 · match

match/case — switch ของ Python (3.10+)

Python 3.10 เพิ่ม match ซึ่งทรงพลังกว่า switch/case ของ C มาก

C — switch/case
switch (txn_type) {
    case 1: deposit(); break;
    case 2: withdraw(); break;
    default: error();
}
Python 3.10 — match/case
match txn_type:
    case "deposit":
        deposit()
    case "withdraw":
        withdraw()
    case _:         # default
        error()
# match กับ tuple — ทรงพลังกว่า switch มาก
match (action, amount):
    case ("deposit", n) if n > 0:
        print(f"ฝาก {n} บาท")
    case ("withdraw", n) if 0 < n <= balance:
        print(f"ถอน {n} บาท")
    case _:
        print("รายการไม่ถูกต้อง")
ถ้า Python version ต่ำกว่า 3.10 ใช้ if/elif แทน — เช็ค version ด้วย python --version
บทที่ 2 · Truthy/Falsy

Truthy & Falsy — Python ถือว่าอะไรเป็น True/False

ใน Python ทุกค่ามี "ความเป็น True/False" — ใช้ใน if โดยไม่ต้องเปรียบเทียบตรงๆ

❌ Falsy — ถือเป็น False
False
0         # int 0
0.0       # float 0
""        # string ว่าง
[]        # list ว่าง
{}        # dict ว่าง
None      # None
✅ Truthy — ถือเป็น True
True
1, -1, 42  # non-zero int
0.001      # non-zero float
"a"        # string ไม่ว่าง
[0]        # list ไม่ว่าง (แม้ค่าข้างในเป็น 0)
{"a":1}    # dict ไม่ว่าง
# ใช้ Truthy/Falsy ตรวจสอบ
transactions = []
if not transactions:
    print("ไม่มีประวัติธุรกรรม")  # list ว่าง = Falsy

balance = 0
if not balance:
    print("ยอดเงินเป็น 0")  # 0 = Falsy
บทที่ 2 · Banking Use Case

🏦 วิเคราะห์ธุรกรรมด้วย Comprehension

โปรแกรมธนาคารที่ใช้ loop + comprehension + f-string

transactions = [
    {"type": "deposit",  "amount": 50_000},
    {"type": "withdraw", "amount": 12_000},
    {"type": "withdraw", "amount": 3_500},
    {"type": "deposit",  "amount": 8_000},
]

# Comprehension กรองและคำนวณ
deposits  = [t['amount'] for t in transactions if t['type'] == 'deposit']
withdraws = [t['amount'] for t in transactions if t['type'] == 'withdraw']

balance = sum(deposits) - sum(withdraws)

print(f"ฝากรวม  : {sum(deposits):>10,} บาท")
print(f"ถอนรวม  : {sum(withdraws):>10,} บาท")
print(f"คงเหลือ : {balance:>10,} บาท")

# แจ้งเตือนถ้า withdraw ครั้งไหน > 10,000
big = [t['amount'] for t in transactions
       if t['type'] == 'withdraw' and t['amount'] > 10_000]
if big:
    print(f"⚠️ พบถอนเงินจำนวนมาก: {big}")
