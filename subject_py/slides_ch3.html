บทที่ 3 · def

def — นิยาม function

ใน C ต้อง declare return type และ parameter type — Python ไม่ต้อง และ return ได้หลายค่าพร้อมกัน

C — function
double calc_interest(double p,
                        double r,
                        int y) {
    return p * r * y;
}
Python — function
def calc_interest(p, r, y):
    return p * r * y
# Return หลายค่าพร้อมกัน — C ทำไม่ได้!
def deposit_summary(principal, rate, years):
    interest = principal * rate * years
    total = principal + interest
    return interest, total   # return tuple

earned, final = deposit_summary(100_000, 0.025, 3)
print(f"ดอกเบี้ย {earned:,} บาท | รวม {final:,} บาท")

# ไม่มี return → คืน None
def greet(name):
    print(f"สวัสดี {name}")   # คืน None
บทที่ 3 · Arguments

Default Arguments & Keyword Arguments

Default args ทำให้ไม่ต้องใส่ทุก argument — Keyword args เรียกโดยใส่ชื่อ parameter ได้เลย

# Default arguments — ต้องอยู่ท้าย parameter list
def open_account(name, acc_type="ออมทรัพย์", initial=500):
    print(f"เปิดบัญชี {acc_type} ชื่อ {name} เงินเริ่มต้น {initial:,}")

open_account("สมชาย")                          # ใช้ default ทั้งคู่
open_account("สมหญิง", "ฝากประจำ")            # เปลี่ยน acc_type
open_account("วิชัย", initial=10_000)          # Keyword — ข้ามได้!

# Keyword arguments — เรียกโดยระบุชื่อ parameter
def transfer(from_acc, to_acc, amount, note=""):
    print(f"โอน {amount:,} จาก {from_acc} → {to_acc} | {note}")

# ลำดับไม่สำคัญเมื่อใช้ keyword
transfer(amount=5_000, to_acc="A002", from_acc="A001", note="ค่าเช่า")
⚠️ อย่าใช้ mutable object (list/dict) เป็น default arg — ใช้ None แทนแล้วตั้งค่าใน body
บทที่ 3 · *args **kwargs

*args และ **kwargs — รับ argument ไม่จำกัดจำนวน

*args รับ positional args เป็น tuple — **kwargs รับ keyword args เป็น dict

# *args — ไม่รู้ว่าจะส่งกี่ค่า
def total_amount(*amounts):
    print(f"รายการ: {amounts}")   # amounts เป็น tuple
    return sum(amounts)

total_amount(100, 200, 300)          # ส่ง 3 ค่า
total_amount(500, 1000)             # ส่ง 2 ค่า — ก็ได้!

# **kwargs — keyword arguments ที่ไม่รู้ชื่อล่วงหน้า
def create_customer(**info):
    print(f"สร้างลูกค้า:")
    for key, val in info.items():
        print(f"  {key}: {val}")

create_customer(name="สมชาย", age=30, tier="Gold")

# ใช้ทั้งสองพร้อมกัน
def log(action, *targets, **options):
    print(f"[{action}] {targets} {options}")

log("transfer", "A001", "A002", amount=5000, note="ค่าเช่า")
บทที่ 3 · lambda

lambda — function แบบไม่มีชื่อ (Anonymous Function)

lambda สร้าง function จากบรรทัดเดียว — ใช้เมื่อต้องการ function แบบ throw-away

# Syntax: lambda parameters: expression
square = lambda x: x ** 2
print(square(5))    # 25

add_vat = lambda price: price * 1.07
print(add_vat(100))  # 107.0

# ใช้จริง: sort list of dicts ตาม key
customers = [
    {"name": "วิชัย",  "balance": 5_000},
    {"name": "สมชาย", "balance": 50_000},
    {"name": "มานี",  "balance": 12_000},
]
# เรียงตาม balance มากสุดก่อน
sorted_customers = sorted(customers, key=lambda c: c["balance"], reverse=True)
for c in sorted_customers:
    print(f"{c['name']:8} {c['balance']:>10,} บาท")
lambda ใช้กับ key= ของ sorted(), min(), max() บ่อยมาก — ทำให้ไม่ต้องสร้าง function แยก
บทที่ 3 · map/filter

map() และ filter() — Functional Programming

แปลงและกรอง iterable โดยไม่ต้องเขียน loop — มักใช้คู่กับ lambda

amounts = [1000, 5000, 200, 15000, 800]

# map(function, iterable) — ทำกับทุกตัว
with_vat = list(map(lambda x: x * 1.07, amounts))
# [1070.0, 5350.0, 214.0, 16050.0, 856.0]

# filter(function, iterable) — เก็บเฉพาะที่ function คืน True
big = list(filter(lambda x: x > 1000, amounts))
# [5000, 15000]

# Pythonic: ใช้ comprehension แทน map/filter ก็ได้
with_vat = [x * 1.07 for x in amounts]        # แทน map
big      = [x for x in amounts if x > 1000]  # แทน filter

# sorted() + lambda
top3 = sorted(amounts, reverse=True)[:3]  # 3 ค่ามากสุด
print(f"Top 3: {top3}")
บทที่ 3 · Scope

Scope — LEGB Rule

Python หา variable ตาม LEGB: Local → Enclosing → Global → Built-in

balance = 100_000   # Global

def show_balance():
    print(balance)    # อ่าน global ได้

def withdraw(amount):
    balance = 50_000  # Local — ไม่แตะ global!
    balance -= amount
    print(f"local balance: {balance}")

withdraw(1000)
print(balance)    # global ยังเป็น 100,000

# global keyword — ถ้าต้องการแก้ global จริงๆ
def deposit(amount):
    global balance
    balance += amount

deposit(5_000)
print(balance)    # 105,000
หลีกเลี่ยง global variable — ส่งผ่าน parameter และ return ค่ากลับดีกว่า
บทที่ 3 · Banking Use Case

🏦 ระบบคำนวณดอกเบี้ยหลายรูปแบบ

ใช้ default args และ lambda เพื่อสร้าง function ที่ยืดหยุ่น

def calc_interest(principal, rate=0.025, years=1, compound=False):
    """คำนวณดอกเบี้ยแบบ simple หรือ compound"""
    if compound:
        total = principal * (1 + rate) ** years
    else:
        total = principal * (1 + rate * years)
    return total - principal, total

# เรียกหลายแบบ
print(calc_interest(100_000))                        # ใช้ default ทั้งหมด
print(calc_interest(100_000, years=3))               # 3 ปี simple
print(calc_interest(100_000, years=3, compound=True)) # compound

# lambda สร้าง preset function
fixed_deposit  = lambda p, y: calc_interest(p, rate=0.035, years=y, compound=True)
savings        = lambda p: calc_interest(p, rate=0.015)

products = ["ออมทรัพย์", "ฝากประจำ 3 ปี"]
funcs    = [lambda p: savings(p), lambda p: fixed_deposit(p, 3)]
for prod, fn in zip(products, funcs):
    earned, total = fn(100_000)
    print(f"{prod}: ดอกเบี้ย {earned:,.2f} บาท")
