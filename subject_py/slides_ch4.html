บทที่ 5 · list

list — Dynamic Array ของ Python

list ขยายขนาดได้เอง เก็บได้ทุก type ปะปนกัน — ไม่ต้องกำหนดขนาดล่วงหน้าเหมือน array C

txns = [1000, 5000, 200]   # สร้าง

# เพิ่ม/ลบ
txns.append(3000)          # เพิ่มท้าย: [1000,5000,200,3000]
txns.insert(1, 9999)       # แทรกที่ index 1
txns.remove(200)           # ลบค่า 200 ออก (ตัวแรก)
txns.pop()                 # ดึงค่าสุดท้ายออก
txns.pop(0)               # ดึงค่า index 0 ออก

# Query
len(txns)                  # ความยาว
txns.index(5000)          # หา index ของค่า
5000 in txns              # True/False
txns.count(1000)          # นับจำนวนที่ซ้ำ

# Sort
txns.sort()                # in-place ascending
txns.sort(reverse=True)   # in-place descending
sorted_txns = sorted(txns) # คืน list ใหม่ (ไม่แก้ original)

# Slice [start:end:step]
txns[1:3]    # index 1,2
txns[::-1]   # reverse ทั้ง list
บทที่ 5 · tuple

tuple — Immutable Sequence

tuple คือ list ที่เปลี่ยนแปลงไม่ได้หลัง create — เร็วกว่า list และใช้เป็น key ของ dict ได้

# สร้าง tuple
account = ("A001", "สมชาย", 50_000)
point = (13.76, 100.50)   # lat, lng — ไม่ควรเปลี่ยน

# Unpack
acc_id, name, balance = account
print(f"{name} มียอด {balance:,} บาท")

# tuple เหมาะสำหรับ return หลายค่า
def get_account(acc_id):
    return "สมชาย", 50_000   # return tuple (ไม่ต้องใส่วงเล็บ)

name, bal = get_account("A001")

# tuple เป็น key ของ dict ได้ (list ไม่ได้!)
branches = {
    (13.76, 100.50): "สาขาสีลม",
    (13.80, 100.55): "สาขาสุขุมวิท",
}

# แปลงกลับไปมา
t = tuple([1, 2, 3])   # list → tuple
l = list((1, 2, 3))   # tuple → list
tuple เร็วกว่า list ~10-15% — ใช้เมื่อข้อมูลไม่ควรเปลี่ยน เช่น coordinates, config values
บทที่ 5 · dict

dict — Key-Value Store, O(1) Lookup

dict คือ Hash Map ที่ Python สร้างให้ built-in — lookup เร็ว O(1) ไม่ว่าจะมีกี่ key

account = {
    "id": "A001",
    "name": "สมชาย",
    "balance": 50_000,
    "tier": "Gold",
}

# Access
account["name"]                    # "สมชาย"
account.get("email", "ไม่มี")      # "ไม่มี" (ไม่ error ถ้า key ไม่มี)

# แก้ไข/เพิ่ม
account["balance"] += 5_000
account["email"] = "somchai@email.com"  # เพิ่ม key ใหม่

# ลบ
del account["tier"]
account.pop("email", None)           # ลบโดยไม่ error ถ้าไม่มี

# วน
for key, val in account.items():
    print(f"  {key}: {val}")

# เช็ค key
"name" in account    # True

# update หลาย key พร้อมกัน
account.update({"tier": "Platinum", "score": 850})
บทที่ 5 · set

set — Unique Values, O(1) Membership Test

set เก็บค่าไม่ซ้ำ ค้นหาเร็ว O(1) — รองรับ set operations เหมือนในคณิตศาสตร์

# กรองค่าซ้ำออกด้วย set
raw = ["A001", "A002", "A001", "A003", "A002"]
unique = set(raw)    # {"A001","A002","A003"}

# เพิ่ม/ลบ
unique.add("A004")
unique.discard("A999")   # ลบโดยไม่ error ถ้าไม่มี

# Membership — O(1)
"A001" in unique    # True — เร็วกว่า list มาก!

# Set Operations
ktb_customers = {"A001", "A002", "A003"}
scb_customers = {"A002", "A003", "A004"}

both     = ktb_customers & scb_customers   # intersection: {"A002","A003"}
either   = ktb_customers | scb_customers   # union: {"A001".."A004"}
ktb_only = ktb_customers - scb_customers   # difference: {"A001"}

print(f"ลูกค้าที่ใช้ทั้งสองธนาคาร: {both}")
ตรวจสอบ membership ใน list = O(n) — ใน set = O(1) ถ้า list ใหญ่มาก ควรแปลงเป็น set ก่อน
บทที่ 5 · Nested

Nested Collections — ซ้อนกันได้

list of dict, dict of list — ใช้แทน struct array ของ C ได้เลย

# List of dicts — แทน array of struct ใน C
customers = [
    {"id": "A001", "name": "สมชาย", "balance": 50_000},
    {"id": "A002", "name": "สมหญิง", "balance": 12_000},
    {"id": "A003", "name": "วิชัย",  "balance": 120_000},
]

# ค้นหาด้วย comprehension
rich = [c for c in customers if c["balance"] > 50_000]
total = sum(c["balance"] for c in customers)

# Dict of lists — เก็บหลาย transaction ต่อ account
history = {
    "A001": [1000, -500, 2000],
    "A002": [5000, -1000],
}
history["A001"].append(3000)   # เพิ่ม transaction ให้ A001

# defaultdict — ไม่ต้อง init list ก่อน
from collections import defaultdict
history2 = defaultdict(list)
history2["A003"].append(7000)   # ไม่ต้อง history["A003"] = [] ก่อน
บทที่ 5 · Banking Use Case

🏦 ระบบ Mini Bank ด้วย dict + list

ไม่ต้องใช้ struct หรือ Node — dict + list ทำได้เลย

bank = {}   # dict เก็บ account

def create_account(acc_id, name):
    bank[acc_id] = {"name": name, "balance": 0, "txns": []}

def deposit(acc_id, amount):
    if acc_id not in bank:
        return "ไม่พบบัญชี"
    bank[acc_id]["balance"] += amount
    bank[acc_id]["txns"].append((+amount, "ฝาก"))

def withdraw(acc_id, amount):
    acc = bank.get(acc_id)
    if not acc:
        return "ไม่พบบัญชี"
    if acc["balance"] < amount:
        return "ยอดไม่พอ"
    acc["balance"] -= amount
    acc["txns"].append((-amount, "ถอน"))

create_account("A001", "สมชาย")
deposit("A001", 50_000)
withdraw("A001", 12_000)
print(f"ยอด: {bank['A001']['balance']:,} บาท")  # 38,000

Think!
มีรายชื่อลูกค้า 1,000,000 คน ต้องการตรวจว่า "A12345" อยู่ในรายชื่อหรือเปล่า

ถ้าเก็บเป็น list — ใช้เวลา O(n) = ตรวจ 1 ล้านครั้ง
ถ้าเก็บเป็น set — ใช้เวลา O(1) = ตรวจ 1 ครั้ง

แต่ set ไม่มีลำดับและ duplicate ไม่ได้ — ถ้า use case ต้องการลำดับและ membership test เร็ว จะใช้อะไร?
